// QueryUtils.setFieldValue for the PostgreSQL writer, including uploaded
// binary content and GeoJSON geometry values.
private Z8Primary z8_application_query_parse_value(Z8Object field, JSONValue value, bool acceptBinary = false) {
    auto type = z8_field_type_id(field);
    // QueryUtils passes JsonObject.getString to string-like fields; arrays and
    // objects become their JSON text. The web client sends FileField context
    // values as arrays, including [] before any file has been attached.
    bool structured = value.type == JSONType.array || value.type == JSONType.object;
    if (structured && !["string"w, "text"w, "file"w, "attachments"w, "geometry"w].canFind(type))
        throw new Exception("invalid query field value");
    auto text = z8_json_value_text(value);
    Z8Primary boxed;
    if (type == "string"w || type == "text"w || type == "file"w || type == "attachments"w) boxed = z8_primary_box(text);
    else if (type == "binary"w) {
        if (!acceptBinary) throw new Exception("binary value requires a file upload");
        auto index = z8_int_parse(text);
        auto files = z8_nonnull(z8_application_require_request().files);
        if (index < 0 || index >= files.z8_size()) throw new Exception("invalid uploaded file index");
        boxed = z8_primary_box(z8_file_to_binary(z8_nonnull(files.z8_get(index))));
    }
    else if (type == "geometry"w) boxed = z8_primary_box(z8_geometry_fromGeoJson(text, z8_optional_require(z8_field_srs(field))));
    else if (!text.length) boxed = z8_field_default_value(field);
    else switch (type) {
        case "int"w: boxed = z8_primary_box(z8_int_parse(text)); break;
        case "float"w: boxed = z8_primary_box(z8_decimal_parse(text)); break;
        case "boolean"w: boxed = z8_primary_box(z8_bool_parse(text)); break;
        case "guid"w: boxed = z8_primary_box(z8_guid_parse(text)); break;
        case "date"w: case "datetime"w: boxed = z8_primary_box(z8_date_parse(text)); break;
        case "datespan"w: boxed = z8_primary_box(z8_datespan_parse(text)); break;
        default: throw new Exception("unsupported query write field type");
    }
    return boxed;
}
private void z8_application_query_set_value(Z8Object field, JSONValue value, bool acceptBinary = false) {
    z8_field_set(field, z8_application_query_parse_value(field, value, acceptBinary));
}

private void z8_application_query_write_affected(Z8Object query, Z8Object[] selected, Z8Guid[] ids,
        Z8JsonWriterState writer, Z8Privileges privileges, Z8StringMap params, Z8Object context) {
    auto key = z8_nonnull(z8_query_primary_key(query));
    foreach (field; z8_query_declared_fields(query, true)) z8_field_reset(field);
    auto contextFields = z8_application_query_context_values(query, privileges, params, true, context);
    auto filter = new Z8SqlBool(Z8SqlBoolKind.inVector);
    filter.vectorValue = new Z8SqlFieldToken(key);
    foreach (id; ids) filter.vectorItems ~= new Z8SqlBoxedConstant(z8_primary_box(id));
    Z8QueryReadRequest read;
    read.fields = selected; read.where = filter; read.byRecordIds = true;
    z8_query_prepare_read(query, read);
    read.queryScope = z8_query_scope(query);
    auto saved = new Z8QuerySavedState(query, contextFields ~ selected);
    scope(exit) saved.restore();
    z8_query_open(query, read, true, true);
    z8_application_query_write_rows(query, writer, selected);
}

// UpdateAction may create a missing linked record using the owner's ID, then
// attach it to the owner in the same transaction. Values from the surrounding
// form are resolved against the context query, as in the original server.
private void z8_application_query_update_linked(Z8Object context, Z8Object query, Z8Object link,
        JSONValue records, Z8Privileges privileges, Z8StringMap params) {
    auto owner = z8_nonnull(link.z8_get_owner());
    if (!z8_native_is_instance(z8_table_name, owner) || !z8_native_is_instance(z8_link_name, link))
        throw new Exception("query update requires a writable table link");
    auto ownerKey = z8_nonnull(z8_query_primary_key(owner));
    auto targetKey = z8_nonnull(z8_query_primary_key(query));
    auto ownerTableKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, owner);
    auto targetTableKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, query);
    auto linkKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, link);
    auto selected = z8_application_query_fields(owner, privileges, params);
    auto connection = z8_connection_current_backend();
    if (connection is null) throw new Exception("no current database connection");
    Z8Guid[] ids;
    connection.beginTransaction();
    scope(failure) connection.rollback();
    foreach (record; records.array) {
        if (record.type != JSONType.object) throw new Exception("invalid query update record");
        Z8Guid readKey(Z8Object field) {
            auto member = field.z8_id().to!string in record.object;
            if (member is null || member.type == JSONType.null_ || (member.type == JSONType.string && !member.str.length)) return z8_guid_zero();
            if (member.type != JSONType.string) throw new Exception("invalid query update record id");
            return z8_guid_parse(member.str.to!wstring);
        }
        auto ownerId = readKey(ownerKey), targetId = readKey(targetKey);
        if (ownerId is null || ownerId == z8_guid_zero()) throw new Exception("query update requires an owner record id");
        auto create = targetId is null || targetId == z8_guid_zero();
        if (create && (!privileges.getTableAccess(ownerTableKey).write || !privileges.getTableAccess(targetTableKey).create ||
                !privileges.getFieldAccess(ownerTableKey, linkKey).write))
            throw new Exception("query update link access denied");
        foreach (name, value; record.object) {
            auto field = z8_query_try_find_field_id(context, name.to!wstring);
            if (field is null || field is ownerKey || field is targetKey) continue;
            auto fieldOwner = field.z8_get_owner();
            if (fieldOwner is query || (create && fieldOwner is owner)) {
                auto tableKey = fieldOwner is query ? targetTableKey : ownerTableKey;
                auto fieldKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, field);
                if (!privileges.getFieldAccess(tableKey, fieldKey).write) throw new Exception("query update field access denied");
            } else if (!z8_application_query_field_readable(context, field, privileges))
                throw new Exception("query update field access denied");
            z8_application_query_set_value(field, value, true);
        }
        if (create) {
            targetId = ownerId;
            z8_field_set(targetKey, z8_primary_box(targetId));
            z8_query_on_new(query);
            if (z8_query_user_events_enabled()) query.z8_query_virtual_onCreateAction(targetId);
            z8_query_insert_id(query, targetId);
            z8_field_set(link, z8_primary_box(targetId));
            z8_query_update_id(owner, ownerId);
        } else {
            if (z8_query_user_events_enabled()) query.z8_query_virtual_onUpdateAction(targetId);
            z8_query_update_id(query, targetId);
        }
        ids ~= ownerId;
    }
    connection.commit();
    z8_application_query_write_affected(owner, selected, ids, z8_application_require_request().writer, privileges, params, context);
}

void z8_application_dispatch_query_update(wstring className) {
    z8_application_dispatch_query_update_target(className, null);
}
private void z8_application_dispatch_query_update_target(wstring className, Z8Object query) {
    auto request = z8_application_require_request();
    auto session = z8_nonnull(request.session);
    auto params = z8_nonnull(request.parameters);
    if (request.writer.scopes.length != 2) throw new Exception("response object is not open");
    if (params.z8_get("action"w) != "update"w) throw new Exception("unsupported query action");
    if (!z8_application_may_execute(session, className, "update"w)) throw new Exception("request access denied");
    auto keys = params.z8_keys();
    for (long i; i < keys.z8_size(); ++i) {
        auto name = keys.z8_get(i);
        if (name is null) throw new Exception("invalid query update parameter");
    }
    if (query is null) query = z8_object_new_instance_native(z8_string_required(className));
    if (!z8_native_is_instance("org.zenframework.z8.server.base.table.Table"w, query))
        throw new Exception("query update requires a Table");
    auto privileges = z8_nonnull(session.user).privileges;
    if (privileges is null) throw new Exception("query update access denied");
    auto context = query;
    query = z8_application_action_query(context, privileges, params.z8_get("query"w), params.z8_get("link"w));
    auto queryKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, query);
    if (!privileges.getTableAccess(queryKey).write) throw new Exception("query update access denied");
    auto key = z8_query_primary_key(query);
    if (key is null) throw new Exception("query has no primary key");
    auto selected = z8_application_query_fields(query, privileges, params);
    z8_application_query_context_values(query, privileges, params, false, context);
    auto json = params.z8_get("data"w);
    if (json is null || !json.length) throw new Exception("query update requires records");
    auto records = z8_json_parse(json.to!string);
    if (records.type != JSONType.array || !records.array.length) throw new Exception("query update requires records");
    auto linkId = params.z8_get("link"w);
    if (linkId !is null && linkId.length) {
        auto link = z8_application_context_field(context, privileges, linkId);
        auto ownerKey = z8_query_primary_key(z8_nonnull(link.z8_get_owner()));
        if (ownerKey !is null) {
            bool hasOwner;
            foreach (record; records.array)
                if (record.type == JSONType.object && (ownerKey.z8_id().to!string in record.object) !is null) hasOwner = true;
            if (hasOwner) { z8_application_query_update_linked(context, query, link, records, privileges, params); return; }
        }
    }
    auto connection = z8_connection_current_backend();
    if (connection is null) throw new Exception("no current database connection");
    Z8Guid[] ids;
    connection.beginTransaction();
    scope(failure) connection.rollback();
    foreach (record; records.array) {
        if (record.type != JSONType.object) throw new Exception("invalid query update record");
        auto member = key.z8_id().to!string in record.object;
        if (member is null || member.type != JSONType.string || !member.str.length) throw new Exception("query update requires a record id");
        auto id = z8_guid_parse(member.str.to!wstring);
        if (id is null) throw new Exception("invalid query update record id");
        foreach (name, value; record.object) {
            auto field = z8_query_try_find_field_id(query, name.to!wstring);
            if (field is null) continue;
            if (field is key) continue;
            if (field.z8_get_owner() !is query) {
                // UpdateAction also sets values of linked fields from the
                // client, but only persists fields owned by the target query.
                // Selectors send both the link id and the displayed value.
                if (!z8_application_query_field_readable(query, field, privileges))
                    throw new Exception("query update field access denied");
                z8_application_query_set_value(field, value, true);
                continue;
            }
            auto fieldKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, field);
            if (!privileges.getFieldAccess(queryKey, fieldKey).write) throw new Exception("query update field access denied");
            z8_application_query_set_value(field, value, true);
        }
        if (z8_query_user_events_enabled()) query.z8_query_virtual_onUpdateAction(id);
        z8_query_update_id(query, id);
        ids ~= id;
    }
    connection.commit();
    z8_application_query_write_affected(query, selected, ids, request.writer, privileges, params, context);
}
