// Native adapter for org.zenframework.z8.server.base.json.parser.JsonObject.
private enum z8_json_object_name = "org.zenframework.z8.server.base.json.parser.JsonObject"w;
private class Z8JsonObjectState {
    JSONValue value;
    Z8Decimal[string] decimals;
    Object[string] children;
    this(JSONValue value) { this.value = value; }
}
private JSONValue z8_json_empty_object() {
    JSONValue[string] fields;
    return JSONValue(fields);
}
private JSONValue z8_json_object_read(wstring source) {
    import std.string : strip;
    auto text = z8_string_required(source).to!string;
    if (!text.strip().length) return z8_json_empty_object();
    auto result = z8_json_parse(text);
    if (result.type != JSONType.object) throw new Exception("JSON object expected");
    return result;
}
private Z8JsonObjectState z8_json_object_state(Z8Object self) {
    self = z8_nonnull(self);
    if (!z8_native_is_instance(z8_json_object_name, self)) throw new Exception("invalid native JSON object");
    auto state = cast(Z8JsonObjectState)self.z8_get_native_state(z8_json_object_name);
    if (state is null) {
        state = new Z8JsonObjectState(z8_json_empty_object());
        self.z8_set_native_state(z8_json_object_name, state);
    }
    return state;
}
private Z8Object z8_json_object_parse(wstring source) {
    if (source is null) return null;
    auto value = z8_json_object_read(source);
    auto result = z8_native_new(z8_json_object_name);
    z8_nonnull(result).z8_set_native_state(z8_json_object_name, new Z8JsonObjectState(value));
    return result;
}
private void z8_json_object_assign(Z8Object self, wstring source) {
    z8_json_object_state(self);
    auto value = z8_json_object_read(source);
    self.z8_set_native_state(z8_json_object_name, new Z8JsonObjectState(value));
}
private JSONValue z8_json_object_get(Z8Object self, wstring name) {
    auto state = z8_json_object_state(self);
    auto value = z8_string_required(name).to!string in state.value.object;
    return value is null ? JSONValue.init : *value;
}
private wstring z8_json_quote(wstring value) {
    enum string hex = "0123456789abcdef";
    wstring result = "\""w;
    wchar previous;
    foreach (wchar current; value) {
        switch (current) {
            case '\\': result ~= "\\\\"w; break;
            case '"': result ~= "\\\""w; break;
            case '/': if (previous == '<') result ~= '\\'; result ~= current; break;
            case '\b': result ~= "\\b"w; break;
            case '\t': result ~= "\\t"w; break;
            case '\n': result ~= "\\n"w; break;
            case '\f': result ~= "\\f"w; break;
            case '\r': result ~= "\\r"w; break;
            default:
                if (current < 0x20 || (current >= 0x80 && current < 0xa0) ||
                    (current >= 0x2000 && current < 0x2100)) {
                    result ~= "\\u"w;
                    foreach_reverse (shift; [0, 4, 8, 12]) result ~= cast(wchar)hex[(current >> shift) & 15];
                } else result ~= current;
        }
        previous = current;
    }
    return result ~ "\""w;
}
private string z8_json_number_text(string value) {
    if (value.indexOf('.') > 0 && value.indexOf('e') < 0 && value.indexOf('E') < 0) {
        while (value.length && value[$ - 1] == '0') value = value[0 .. $ - 1];
        if (value.length && value[$ - 1] == '.') value = value[0 .. $ - 1];
    }
    return value;
}
private wstring z8_json_render(JSONValue value) {
    final switch (value.type) {
        case JSONType.null_: return "null"w;
        case JSONType.string: return z8_json_quote(value.str.to!wstring);
        case JSONType.integer: return value.integer.to!wstring;
        case JSONType.uinteger: return value.uinteger.to!wstring;
        case JSONType.float_: return z8_json_number_text(z8_java_double_string(value.floating)).to!wstring;
        case JSONType.true_: return "true"w;
        case JSONType.false_: return "false"w;
        case JSONType.array:
            wstring result = "["w;
            foreach (i, entry; value.array) {
                if (i) result ~= ","w;
                result ~= z8_json_render(entry);
            }
            return result ~ "]"w;
        case JSONType.object:
            wstring result = "{"w;
            bool first = true;
            foreach (key, entry; value.object) {
                if (!first) result ~= ","w;
                first = false;
                result ~= z8_json_quote(key.to!wstring) ~ ":"w ~ z8_json_render(entry);
            }
            return result ~ "}"w;
    }
}
private wstring z8_json_object_render_state(Z8JsonObjectState state) {
    wstring result = "{"w;
    bool first = true;
    foreach (key, value; state.value.object) {
        if (!first) result ~= ","w;
        first = false;
        auto decimalValue = key in state.decimals;
        auto child = key in state.children;
        result ~= z8_json_quote(key.to!wstring) ~ ":"w ~
            (child !is null ? z8_json_render_state(*child) : decimalValue is null ? z8_json_render(value) : z8_json_number_text(z8_json_bigdecimal_text(*decimalValue)).to!wstring);
    }
    return result ~ "}"w;
}
private wstring z8_json_object_to_string(Z8Object self) {
    return z8_string_new(z8_json_object_render_state(z8_json_object_state(self)));
}
private wstring z8_json_value_text(JSONValue value) {
    final switch (value.type) {
        case JSONType.null_: return ""w;
        case JSONType.string: return value.str.length ? value.str.to!wstring : " "w[0 .. 0];
        case JSONType.integer: return value.integer.to!wstring;
        case JSONType.uinteger: return value.uinteger.to!wstring;
        case JSONType.float_: return z8_java_double_string(value.floating).to!wstring;
        case JSONType.true_: return "true"w;
        case JSONType.false_: return "false"w;
        case JSONType.array: case JSONType.object: return z8_json_render(value);
    }
}
private wstring z8_json_object_get_string(Z8Object self, wstring name) {
    auto state = z8_json_object_state(self);
    auto key = z8_string_required(name).to!string;
    auto child = key in state.children;
    if (child !is null) return z8_string_new(z8_json_render_state(*child));
    auto decimalValue = key in state.decimals;
    if (decimalValue !is null) return z8_string_new(z8_json_bigdecimal_text(*decimalValue).to!wstring);
    return z8_string_new(z8_json_value_text(z8_json_object_get(self, name)));
}
// BigDecimal.toString uses scientific notation when its adjusted exponent
// is below -6 or its scale is negative; toPlainString is used otherwise.
private string z8_json_bigdecimal_text(Z8Decimal value) {
    auto digits = value.coefficient.to!string;
    string sign;
    if (digits.length && digits[0] == '-') { sign = "-"; digits = digits[1 .. $]; }
    auto exponent = cast(long)digits.length - 1 - value.scale;
    if (value.scale >= 0 && exponent >= -6) return value.z8_string().to!string;
    auto mantissa = digits[0 .. 1] ~ (digits.length > 1 ? "." ~ digits[1 .. $] : "");
    return sign ~ mantissa ~ "E" ~ (exponent >= 0 ? "+" : "") ~ exponent.to!string;
}
private long z8_json_bigdecimal_long(Z8Decimal value) {
    auto number = value.coefficient;
    if (value.scale > 0) {
        if (cast(long)value.scale >= cast(long)z8_precision(number)) return 0;
        number /= z8_pow10(cast(uint)value.scale);
    } else if (value.scale < 0) {
        if (value.scale <= -64) return 0;
        number *= z8_pow10(cast(uint)-value.scale);
    }
    auto modulus = BigInt("18446744073709551616");
    number %= modulus;
    if (number < 0) number += modulus;
    return cast(long)number.to!string.to!ulong;
}
private Z8Optional!bool z8_json_object_has(Z8Object self, wstring name) {
    auto state = z8_json_object_state(self);
    return z8_optional_wrap((z8_string_required(name).to!string in state.value.object) !is null);
}
private Z8Optional!bool z8_json_object_empty(Z8Object self) {
    return z8_optional_wrap(z8_json_object_state(self).value.object.length == 0);
}
private Z8Array!wstring z8_json_object_names(Z8Object self) {
    auto result = new Z8Array!wstring();
    foreach (name; z8_json_object_state(self).value.object.byKey)
        result.z8_add(z8_string_new(name.to!wstring));
    return result;
}
private Z8Optional!bool z8_json_object_get_bool(Z8Object self, wstring name) {
    auto value = z8_json_object_get(self, name);
    if (value.type == JSONType.null_) throw new Exception("null JSON boolean");
    return z8_optional_wrap(value.type == JSONType.true_);
}
private Z8Optional!long z8_json_object_get_int(Z8Object self, wstring name) {
    import std.math : isNaN;
    auto state = z8_json_object_state(self);
    auto decimalValue = z8_string_required(name).to!string in state.decimals;
    if (decimalValue !is null) return z8_optional_wrap(z8_json_bigdecimal_long(*decimalValue));
    auto value = z8_json_object_get(self, name);
    switch (value.type) {
        case JSONType.integer: return z8_optional_wrap(value.integer);
        case JSONType.uinteger: return z8_optional_wrap(cast(long)value.uinteger);
        case JSONType.float_:
            auto number = value.floating;
            if (isNaN(number)) return z8_optional_wrap(0L);
            if (number >= cast(double)long.max) return z8_optional_wrap(long.max);
            if (number <= cast(double)long.min) return z8_optional_wrap(long.min);
            return z8_optional_wrap(cast(long)number);
        case JSONType.string:
            if (value.str.length && (value.str[$ - 1] == 'L' || value.str[$ - 1] == 'l'))
                throw new Exception("invalid JSON integer");
            return z8_optional_wrap(z8_int_parse(value.str.to!wstring));
        default: throw new Exception("invalid JSON integer");
    }
}
private Z8Decimal z8_json_value_decimal(JSONValue value) {
    import std.string : strip;
    double number;
    switch (value.type) {
        case JSONType.integer: number = cast(double)value.integer; break;
        case JSONType.uinteger: number = cast(double)value.uinteger; break;
        case JSONType.float_: number = value.floating; break;
        case JSONType.string: number = z8_literal_parse_double(value.str.strip()); break;
        default: throw new Exception("invalid JSON decimal");
    }
    return z8_math_result(number);
}
private Z8Optional!Z8Decimal z8_json_object_get_decimal(Z8Object self, wstring name) {
    auto state = z8_json_object_state(self);
    auto decimalValue = z8_string_required(name).to!string in state.decimals;
    if (decimalValue !is null)
        return z8_optional_wrap(z8_math_result(z8_literal_parse_double(z8_json_bigdecimal_text(*decimalValue))));
    return z8_optional_wrap(z8_json_value_decimal(z8_json_object_get(self, name)));
}
private Z8Guid z8_json_object_get_guid(Z8Object self, wstring name) {
    auto state = z8_json_object_state(self);
    auto decimalValue = z8_string_required(name).to!string in state.decimals;
    if (decimalValue !is null) return z8_guid_parse(z8_json_bigdecimal_text(*decimalValue).to!wstring);
    auto value = z8_json_object_get(self, name);
    return value.type == JSONType.null_ ? null : z8_guid_parse(z8_json_value_text(value));
}
private Z8Date z8_json_object_get_date(Z8Object self, wstring name) {
    return z8_date_parse(z8_json_object_get_string(self, name));
}
private struct Z8JsonPrimaryValue {
    JSONValue value;
    bool hasDecimal;
    Z8Decimal decimalValue;
}
private Z8JsonPrimaryValue z8_json_from_primary(Z8Primary value) {
    Z8JsonPrimaryValue result;
    if (value !is null) final switch (value.kind) {
        case Z8PrimaryKind.integer: result.value = JSONValue(value.integerValue); break;
        case Z8PrimaryKind.boolean: result.value = JSONValue(value.booleanValue); break;
        case Z8PrimaryKind.text: result.value = JSONValue(z8_string_or_empty(value.textValue).to!string); break;
        case Z8PrimaryKind.guid: result.value = JSONValue(z8_guid_to_string(value.guidValue).to!string); break;
        case Z8PrimaryKind.datespan: result.value = JSONValue(z8_datespan_milliseconds(value.datespanValue)); break;
        case Z8PrimaryKind.date:
            auto ticks = z8_date_ticks(value.dateValue);
            result.value = JSONValue((ticks == z8_date_ticks(z8_date_min) || ticks == z8_date_ticks(z8_date_max) ?
                                   ""w : z8_date_toString(value.dateValue)).to!string);
            break;
        case Z8PrimaryKind.decimal:
            if (!value.decimalValue.nan) { result.value = JSONValue(0L); result.hasDecimal = true; result.decimalValue = value.decimalValue; }
            break;
        case Z8PrimaryKind.binary: throw new Exception("binary JSON value is unsupported");
        case Z8PrimaryKind.geometry: case Z8PrimaryKind.file: case Z8PrimaryKind.user:
            throw new Exception("JSON value type is not supported yet");
    }
    return result;
}
private Z8Object z8_json_object_set_primary(Z8Object self, wstring name, Z8Primary value) {
    auto state = z8_json_object_state(self);
    auto key = z8_string_required(name).to!string;
    auto converted = z8_json_from_primary(value);
    state.decimals.remove(key);
    state.children.remove(key);
    if (converted.hasDecimal) state.decimals[key] = converted.decimalValue;
    state.value.object[key] = converted.value;
    return self;
}
private Z8Object z8_json_object_remove(Z8Object self, wstring name) {
    auto state = z8_json_object_state(self);
    auto key = z8_string_required(name).to!string;
    state.decimals.remove(key);
    state.children.remove(key);
    state.value.object.remove(key);
    return self;
}
private Z8Object z8_json_object_get_nested(Z8Object self, wstring name, JSONType expected) {
    auto state = z8_json_object_state(self);
    auto key = z8_string_required(name).to!string;
    auto value = key in state.value.object;
    if (value is null || value.type == JSONType.null_) return null;
    if (value.type != expected) throw new Exception("invalid nested JSON value");
    auto child = key in state.children;
    if (child is null) {
        state.children[key] = z8_json_new_state(*value);
        child = key in state.children;
    }
    return z8_json_wrap_state(*child);
}
private Z8Object z8_json_object_get_array(Z8Object self, wstring name) { return z8_json_object_get_nested(self, name, JSONType.array); }
private Z8Object z8_json_object_get_object(Z8Object self, wstring name) { return z8_json_object_get_nested(self, name, JSONType.object); }
private Z8Object z8_json_object_set_nested(Z8Object self, wstring name, Z8Object value) {
    auto state = z8_json_object_state(self);
    auto key = z8_string_required(name).to!string;
    auto child = z8_json_state_from_object(value);
    state.value.object[key] = z8_json_state_value(child);
    state.decimals.remove(key);
    state.children[key] = child;
    return self;
}
void z8_bind_builtin_json_object() {
    enum prefix = z8_json_object_name ~ "."w;
    z8_native_bind(prefix ~ "m_parse"w, &z8_json_object_parse);
    z8_native_bind(prefix ~ "z8cast_string"w, &z8_json_object_to_string);
    z8_native_bind(prefix ~ "m_toString"w, &z8_json_object_to_string);
    z8_native_bind(prefix ~ "z8op_assign"w, &z8_json_object_assign);
    z8_native_bind(prefix ~ "m_getString"w, &z8_json_object_get_string);
    z8_native_bind(prefix ~ "m_has"w, &z8_json_object_has);
    z8_native_bind(prefix ~ "m_isEmpty"w, &z8_json_object_empty);
    z8_native_bind(prefix ~ "m_getNames"w, &z8_json_object_names);
    z8_native_bind(prefix ~ "m_getBool"w, &z8_json_object_get_bool);
    z8_native_bind(prefix ~ "m_getInt"w, &z8_json_object_get_int);
    z8_native_bind(prefix ~ "m_getDecimal"w, &z8_json_object_get_decimal);
    z8_native_bind(prefix ~ "m_getGuid"w, &z8_json_object_get_guid);
    z8_native_bind(prefix ~ "m_getDate"w, &z8_json_object_get_date);
    z8_native_bind(prefix ~ "m_getJsonArray"w, &z8_json_object_get_array);
    z8_native_bind(prefix ~ "m_getJsonObject"w, &z8_json_object_get_object);
    z8_native_bind(prefix ~ "m_set"w, &z8_json_object_set_nested);
    z8_native_bind(prefix ~ "m_put"w, &z8_json_object_set_nested);
    z8_native_bind(prefix ~ "m_set"w, &z8_json_object_set_primary);
    z8_native_bind(prefix ~ "m_put"w, &z8_json_object_set_primary);
    z8_native_bind(prefix ~ "m_remove"w, &z8_json_object_remove);
}
shared static this() { z8_bind_builtin_json_object(); }
