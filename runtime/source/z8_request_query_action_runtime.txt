// CommandAction adapter for query contexts and typed parameters.
private Z8Object z8_action_find_member(Z8Object query, wstring id) {
    import std.array : split;
    id = z8_string_required(id);
    auto own = query.z8_id();
    if (id.length < own.length || id[0 .. own.length] != own) throw new Exception("query member not found");
    auto path = id[own.length .. $];
    if (own.length && path.length) {
        if (path[0] != '.') throw new Exception("query member not found");
        path = path[1 .. $];
    }
    auto current = query;
    foreach (part; path.split("."w)) {
        if (!part.length) throw new Exception("query member not found");
        Z8Object found;
        foreach (member; z8_object_members(current)) if (member.z8_index() == part) { found = member; break; }
        if (found is null) throw new Exception("query member not found");
        current = found;
    }
    return current;
}
private Z8Object z8_action_find(Z8Object query, wstring id) {
    auto current = z8_action_find_member(query, id);
    if (!z8_native_is_instance(z8_action_name, current)) throw new Exception("action not found");
    return current;
}
private Z8Array!Z8Guid z8_action_guid_list(wstring value) {
    auto result = new Z8Array!Z8Guid();
    if (value is null || !value.length) return result;
    auto json = value[0] == '[' ? value : value[0] == '"' ? "["w ~ value ~ "]"w : "[\""w ~ value ~ "\"]"w;
    auto parsed = z8_json_parse(json.to!string);
    if (parsed.type != JSONType.array) throw new Exception("action record ids must be an array");
    foreach (item; parsed.array) {
        if (item.type != JSONType.string || !item.str.length) throw new Exception("invalid action record id");
        auto id = z8_guid_parse(item.str.to!wstring);
        if (id is null) throw new Exception("invalid action record id");
        result.z8_add(id);
    }
    return result;
}
void z8_application_dispatch_query_action(wstring className) {
    z8_application_dispatch_query_action_target(className, null);
}
private void z8_application_dispatch_query_action_target(wstring className, Z8Object query) {
    auto request = z8_application_require_request();
    auto session = z8_nonnull(request.session);
    auto params = z8_nonnull(request.parameters);
    if (request.writer.scopes.length != 2) throw new Exception("response object is not open");
    if (params.z8_get("action"w) != "action"w) throw new Exception("unsupported query action");
    if (!z8_application_may_execute(session, className, "action"w)) throw new Exception("request access denied");
    auto keys = params.z8_keys();
    for (long i; i < keys.z8_size(); ++i) {
        auto name = keys.z8_get(i);
        if (name is null) throw new Exception("invalid action parameter");
        if (name == "request"w || name == "session"w || name == "server"w || name == "ip"w || name == "action"w ||
            name == "name"w || name == "records"w || name == "selection"w || name == "parameters"w || name == "recordId"w ||
            name == "query"w || name == "link"w) continue;
        if (params.z8_get(name).length) throw new Exception("unsupported action parameter: " ~ name.to!string);
    }
    if (query is null) query = z8_object_new_instance_native(z8_string_required(className));
    if (!z8_native_is_instance("org.zenframework.z8.server.base.query.Query"w, query)) throw new Exception("action requires a Query");
    auto context = query;
    auto linkId = params.z8_get("link"w), queryId = params.z8_get("query"w);
    if (linkId !is null && linkId.length) {
        auto link = z8_action_find_member(context, linkId);
        if (!z8_native_is_instance("org.zenframework.z8.server.base.table.value.Link"w, link)) throw new Exception("action link not found");
        query = z8_nonnull(z8_link_query(link));
    } else if (queryId !is null && queryId.length) {
        query = z8_action_find_member(context, queryId);
        if (!z8_native_is_instance("org.zenframework.z8.server.base.query.Query"w, query)) throw new Exception("action query not found");
    }
    auto action = z8_action_find(context, params.z8_get("name"w));
    auto state = z8_action_state(action);
    auto parameterValues = params.z8_get("parameters"w);
    if (parameterValues !is null && parameterValues.length) {
        auto parsed = z8_json_parse(parameterValues.to!string);
        if (parsed.type != JSONType.array) throw new Exception("action parameters must be an array");
        foreach (entry; parsed.array) {
            if (entry.type != JSONType.object) throw new Exception("invalid action parameter");
            auto id = "id" in entry.object;
            if (id is null || id.type != JSONType.string) throw new Exception("action parameter requires an id");
            Z8Object parameter;
            if (state.parameters !is null && state.parameters.view() !is null) {
                auto candidates = state.parameters.view().z8_erased_data();
                for (long i; i < candidates.z8_size(); ++i) {
                    auto candidate = cast(Z8Object)candidates.z8_get(i);
                    if (z8_parameter_state(candidate).text == id.str.to!wstring) { parameter = candidate; break; }
                }
            }
            if (parameter is null) throw new Exception("unknown action parameter: " ~ id.str);
            auto value = "value" in entry.object;
            if (value !is null && (value.type == JSONType.array || value.type == JSONType.object)) throw new Exception("invalid action parameter value");
            z8_parameter_parse(parameter, value is null ? ""w : z8_json_value_text(*value));
        }
    }
    auto records = z8_action_guid_list(params.z8_get("records"w));
    auto selected = z8_action_guid_list(params.z8_get("selection"w));
    z8_action_guid_list(params.z8_get("recordId"w));
    auto connection = z8_optional_require(state.useTransaction) ? z8_connection_current_backend() : null;
    if (z8_optional_require(state.useTransaction) && connection is null) throw new Exception("no current database connection");
    if (connection !is null) connection.beginTransaction();
    scope(failure) if (connection !is null) connection.rollback();
    action.z8_action_virtual_execute_many(records, context, selected, query);
    if (connection !is null) connection.commit();
}
