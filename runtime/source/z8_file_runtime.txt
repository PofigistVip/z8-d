// File metadata, path helpers and filesystem operations.
private __gshared string z8_file_working_root, z8_file_storage_root;
void z8_file_set_roots(string working, string storage = "") {
    import std.file : getcwd;
    import std.path : absolutePath, buildPath, isAbsolute;
    working = absolutePath(working.length ? working : getcwd());
    if (!storage.length) storage = buildPath(working, "storage");
    else if (!isAbsolute(storage)) storage = buildPath(working, storage);
    z8_file_working_root = working;
    z8_file_storage_root = absolutePath(storage);
}
shared static this() {
    import std.process : environment;
    z8_file_set_roots(environment.get("Z8_WORKING_PATH", ""), environment.get("Z8_STORAGE_PATH", ""));
}
class Z8File {
    Z8Guid id, user;
    wstring name = ""w, path = ""w, author = ""w;
    Z8Date time;
    Z8Optional!long size;
    Z8StringMap details;
    this() {
        id = z8_guid_zero(); user = z8_guid_zero();
        time = z8_date_now(); size = z8_optional_wrap(0L);
        details = new Z8StringMap();
    }
    override bool opEquals(Object other) {
        auto right = cast(Z8File)other;
        return right !is null && id !is null && id == right.id;
    }
}

wstring z8_file_name(wstring value) {
    value = z8_string_required(value);
    auto start = z8_binary_name_start(value);
    return value[start .. $];
}
wstring z8_file_baseName(wstring value) {
    value = z8_string_required(value);
    auto start = z8_binary_name_start(value);
    return value[start .. z8_binary_extension_start(value, start)];
}
wstring z8_file_extension(wstring value) {
    value = z8_string_required(value);
    auto dot = z8_binary_extension_start(value, z8_binary_name_start(value));
    return dot == value.length ? ""w : value[dot + 1 .. $];
}
void z8_file_assign(Z8File value, wstring path) {
    import std.path : baseName;
    z8_nonnull(value).path = z8_string_required(path);
    value.name = baseName(path.to!string).to!wstring;
}
Z8File z8_file_from_path(wstring path) { auto result = new Z8File(); z8_file_assign(result, path); return result; }
wstring z8_file_path(Z8File value) { return z8_nonnull(value).path; }
long z8_file_hashCode(Z8File value) { z8_nonnull(value); return value.id is null ? 0 : z8_guid_hash(value.id); }
wstring z8_file_to_string(Z8File value) {
    import std.json : JSONValue;
    z8_nonnull(value);
    JSONValue jsonString(wstring text) { return text is null ? JSONValue.init : JSONValue(text.to!string); }
    JSONValue[string] fields;
    fields["name"] = jsonString(value.name);
    if (value.time is null) fields["time"] = JSONValue.init;
    else {
        auto ticks = z8_date_ticks(value.time);
        fields["time"] = jsonString(ticks == z8_date_ticks(z8_date_min) || ticks == z8_date_ticks(z8_date_max) ? ""w : z8_date_toString(value.time));
    }
    fields["size"] = value.size.present ? JSONValue(value.size.value) : JSONValue.init;
    fields["path"] = jsonString(value.path);
    fields["id"] = value.id is null ? JSONValue.init : jsonString(z8_guid_to_string(value.id));
    fields["user"] = value.user is null ? JSONValue.init : jsonString(z8_guid_to_string(value.user));
    fields["author"] = jsonString(value.author);
    JSONValue[string] details;
    if (value.details !is null) {
        foreach (key, item; value.details.entries) if (item !is null) details[key.to!string] = jsonString(item);
        if (value.details.hasNullKey && value.details.nullKeyValue !is null) throw new Exception("null file detail key");
    }
    fields["details"] = JSONValue(details);
    return JSONValue(fields).toString().to!wstring;
}
Z8Array!Z8File z8_file_parse(wstring input) {
    import std.json : JSONValue, JSONType, parseJSON;
    auto text = z8_string_required(input).to!string;
    if (!text.length) text = "[]";
    else if (text[0] != '[') text = "[" ~ text ~ "]";
    auto json = z8_json_parse(text);
    wstring field(JSONValue[string] values, string key, wstring fallback = ""w) {
        auto p = key in values;
        if (p is null || p.type == JSONType.null_) return fallback;
        if (p.type == JSONType.string) return p.str.to!wstring;
        if (p.type == JSONType.integer) return p.integer.to!wstring;
        if (p.type == JSONType.uinteger) return p.uinteger.to!wstring;
        if (p.type == JSONType.true_) return "true"w;
        if (p.type == JSONType.false_) return "false"w;
        throw new Exception("invalid file metadata field");
    }
    auto result = new Z8Array!Z8File();
    foreach (item; json.array) {
        auto values = item.object;
        auto value = new Z8File();
        value.path = "file" in values ? field(values, "file") : field(values, "path", null);
        value.name = field(values, "name");
        auto size = field(values, "size");
        if (!size.length) throw new Exception("invalid file size");
        value.size = z8_optional_wrap(size.to!long);
        value.id = z8_guid_parse(field(values, "id"));
        value.user = z8_guid_parse(field(values, "user"));
        value.author = field(values, "author");
        auto time = field(values, "time");
        value.time = time.canFind('/') ? z8_date_parse(time, "D/M/y H:m:s"w) : z8_date_parse(time);
        if (auto details = "details" in values) {
            foreach (key, unused; details.object) value.details.z8_add(key.to!wstring, field(details.object, key));
        }
        result.z8_add(value);
    }
    return result;
}
wstring z8_file_toJson(Z8Array!Z8File values) {
    z8_nonnull(values);
    wstring result = "["w;
    foreach (i; 0 .. values.z8_size()) {
        if (i) result ~= ","w;
        result ~= z8_file_to_string(values.z8_get(i));
    }
    return result ~ "]"w;
}
bool z8_file_compare(string op)(Z8File left, Z8File right) {
    import std.path : absolutePath;
    auto a = absolutePath(z8_file_system_path(left));
    if (right is null) return op == "!=";
    auto b = absolutePath(z8_file_system_path(right));
    static if (op == "==") return a == b;
    else static if (op == "!=") return a != b;
    else static assert(0, "unsupported file comparison");
}
wstring z8_file_fileName(Z8File value) {
    z8_nonnull(value);
    auto path = z8_string_required(value.path);
    return z8_file_name(path.length ? path : value.name);
}
wstring z8_file_baseName_instance(Z8File value) {
    z8_nonnull(value);
    auto path = z8_string_required(value.path);
    return z8_file_baseName(path.length ? path : value.name);
}
wstring z8_file_extension_instance(Z8File value) {
    import std.uni : toLower;
    import std.utf : encode;
    z8_nonnull(value);
    auto path = z8_string_required(value.path);
    auto extension = z8_file_extension(path.length ? path : value.name);
    wchar[] lower;
    foreach (dchar c; extension) encode(lower, toLower(c));
    return lower.length ? lower.idup : ""w;
}
wstring z8_file_folder(Z8File value) {
    import std.string : replace;
    auto path = z8_string_required(z8_nonnull(value).path).to!string.replace("\\", "/").to!wstring;
    return path[0 .. z8_binary_name_start(path)];
}

private string z8_file_system_path(Z8File value) {
    import std.path : buildPath, isAbsolute;
    import std.string : startsWith;
    auto path = z8_string_required(z8_nonnull(value).path).to!string;
    if (isAbsolute(path)) return path;
    if (path.startsWith("storage")) {
        auto suffix = path["storage".length .. $];
        version (Windows) {
            while (suffix.length && (suffix[0] == '/' || suffix[0] == '\\')) suffix = suffix[1 .. $];
        } else {
            while (suffix.length && suffix[0] == '/') suffix = suffix[1 .. $];
        }
        return suffix.length ? buildPath(z8_file_storage_root, suffix) : z8_file_storage_root;
    }
    return path.length ? buildPath(z8_file_working_root, path) : z8_file_working_root;
}
bool z8_file_exists(Z8File value) {
    import std.file : exists;
    return exists(z8_file_system_path(value));
}
bool z8_file_isFolder(Z8File value) {
    import std.file : exists, isDir;
    auto path = z8_file_system_path(value);
    return exists(path) && isDir(path);
}
bool z8_file_isEmpty(Z8File value) {
    import std.file : exists, isDir, getSize, dirEntries, SpanMode;
    import std.range : empty;
    auto path = z8_file_system_path(value);
    if (!exists(path)) return true;
    return isDir(path) ? dirEntries(path, SpanMode.shallow).empty : getSize(path) == 0;
}
wstring z8_file_read(Z8File value) {
    return z8_file_read(value, z8_encoding_by_id(0));
}
wstring z8_file_read(Z8File value, Z8Encoding encoding) {
    import std.file : read;
    return z8_encoding_decode(cast(ubyte[])read(z8_file_system_path(value)), encoding);
}
Z8Binary z8_file_to_binary(Z8File value) {
    import std.file : read;
    return new Z8Binary(cast(ubyte[])read(z8_file_system_path(value)));
}
Z8Date z8_file_lastModified(Z8File value) {
    import std.file : exists, timeLastModified;
    auto path = z8_file_system_path(value);
    if (!exists(path)) return z8_date_from_ticks(0);
    auto t = timeLastModified(path);
    t.timezone = z8_date_timezone();
    return new Z8Date(t);
}
private Z8File z8_file_from_disk(string path) {
    import std.file : exists, isFile, getSize;
    import std.path : absolutePath, relativePath, dirSeparator;
    auto full = absolutePath(path);
    auto working = z8_file_working_root;
    auto prefix = working ~ dirSeparator;
    import std.string : startsWith;
    auto stored = full.startsWith(prefix) ? relativePath(full, working) : full;
    auto result = z8_file_from_path(stored.to!wstring);
    result.size = z8_optional_wrap(exists(full) && isFile(full) ? cast(long)getSize(full) : 0L);
    result.time = z8_file_lastModified(result);
    return result;
}
private __gshared string[] z8_temp_paths;
private __gshared Object z8_temp_lock;
shared static this() { z8_temp_lock = new Object(); }
shared static ~this() {
    import std.file : exists, isFile, isSymlink, remove;
    foreach (path; z8_temp_paths) {
        try { if (exists(path) && isFile(path) && !isSymlink(path)) remove(path); }
        catch (Exception) { }
    }
}
Z8File z8_file_createTempFile(wstring prefix, wstring extension) {
    import std.file : mkdirRecurse;
    import std.path : buildPath;
    auto start = z8_string_required(prefix).to!string;
    auto suffix = z8_string_required(extension).to!string;
    if (suffix.length && suffix[0] != '.') suffix = "." ~ suffix;
    auto folder = buildPath(z8_file_working_root, "files", "temp");
    mkdirRecurse(folder);
    auto stamp = z8_date_toString(z8_date_now(), "Y-MM-dd HH-mm-ss"w).to!string;
    auto path = buildPath(folder, start ~ stamp ~ suffix);
    synchronized (z8_temp_lock) z8_temp_paths ~= path;
    return z8_file_from_disk(path);
}
Z8File z8_file_parent(Z8File value) {
    import std.path : absolutePath, dirName;
    auto parent = dirName(absolutePath(z8_file_system_path(value)));
    if (!parent.length) throw new Exception("file has no parent");
    return z8_file_from_disk(parent);
}
Z8Array!Z8File z8_file_listFiles(Z8File value) {
    import std.file : exists, isDir, dirEntries, SpanMode;
    auto result = new Z8Array!Z8File();
    auto path = z8_file_system_path(value);
    if (!exists(path) || !isDir(path)) return result;
    foreach (entry; dirEntries(path, SpanMode.shallow)) result.z8_add(z8_file_from_disk(entry.name));
    return result;
}
private Z8File z8_file_write_bytes(Z8File value, const(ubyte)[] contents, bool append) {
    import std.file : mkdirRecurse, getSize;
    import std.path : dirName;
    import std.stdio : File;
    string path;
    if (!z8_string_required(z8_nonnull(value).path).length) {
        auto temporary = z8_file_createTempFile(""w, "txt"w);
        value.id = temporary.id; value.name = temporary.name; value.path = temporary.path;
        value.time = temporary.time; value.size = temporary.size; value.user = temporary.user;
        value.author = temporary.author; value.details = temporary.details;
        path = z8_file_system_path(value);
    } else path = z8_file_system_path(value);
    auto parent = dirName(path);
    if (parent.length) mkdirRecurse(parent);
    auto output = File(path, append ? "ab" : "wb");
    output.rawWrite(contents);
    output.close();
    value.size = z8_optional_wrap(cast(long)getSize(path));
    return value;
}
Z8File z8_file_write(Z8File value, wstring contents) {
    return z8_file_write(value, contents, z8_encoding_by_id(0));
}
Z8File z8_file_append(Z8File value, wstring contents) {
    return z8_file_append(value, contents, z8_encoding_by_id(0));
}
Z8File z8_file_write(Z8File value, wstring contents, Z8Encoding encoding) {
    return z8_file_write_bytes(value, z8_encoding_encode(contents, encoding), false);
}
Z8File z8_file_append(Z8File value, wstring contents, Z8Encoding encoding) {
    return z8_file_write_bytes(value, z8_encoding_encode(contents, encoding), true);
}
Z8File z8_file_write(Z8File value, Z8Binary contents) {
    z8_nonnull(contents);
    auto result = z8_file_write_bytes(value, contents.data[contents.position .. $], false);
    contents.position = contents.data.length;
    return result;
}
Z8File z8_file_append(Z8File value, Z8Binary contents) {
    z8_nonnull(contents);
    auto result = z8_file_write_bytes(value, contents.data[contents.position .. $], true);
    contents.position = contents.data.length;
    return result;
}
void z8_file_delete(Z8File value, bool recursively = false) {
    import std.file : exists, isDir, remove, rmdir, rmdirRecurse;
    auto path = z8_file_system_path(value);
    if (!exists(path)) return;
    if (isDir(path)) {
        if (recursively) rmdirRecurse(path);
        else rmdir(path);
    } else remove(path);
}
