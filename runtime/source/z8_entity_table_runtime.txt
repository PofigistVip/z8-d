    auto result = z8_entity_table_fields(self);

    if (dependencies !is null) for (long i; i < dependencies.z8_size(); ++i) {

        add(z8_nonnull(z8_query_primary_key(dependency)));
    }
    auto key = z8_nonnull(z8_query_primary_key(self));
    add(key);
    return result;
}

private wstring z8_entity_table_relative_id(Z8Object self, Z8Object field) {

private Z8Object z8_entity_table_init_fields(Z8Object self, Z8Object entity, Z8Optional!bool changedOnly) {
    auto only = z8_optional_require(changedOnly);

private void z8_entity_table_init_dependencies(Z8Object self, Z8Object registry, Z8Object[] dependencies) {
    import std.algorithm.searching : startsWith;
    auto parent = z8_nonnull(self).z8_id();
    foreach (dependency; dependencies) {
        auto id = dependency.z8_id();
        if (id != (parent.length ? parent ~ "."w ~ dependency.z8_index() : dependency.z8_index())) continue;
        auto entityId = z8_query_record_id(dependency);
        if (entityId is null || entityId == z8_guid_zero() || z8_entity_registry_find(registry, entityId) !is null) continue;
        auto values = z8_entity_table_current_values(dependency);
        auto entity = z8_nonnull(dependency.z8_entity_table_virtual_new_values(values));
        if (!z8_optional_require(z8_entity_initialized(entity))) z8_entity_init_map(entity, values);
        z8_entity_state(entity).persistent = true;
        z8_entity_register_to(entity, registry);
        Z8Object[] nested;
        foreach (candidate; dependencies)
            if (candidate.z8_id().startsWith(id ~ "."w)) nested ~= candidate;
        z8_entity_table_init_dependencies(dependency, registry, nested);
    }
}
private Z8Object z8_entity_table_get_entity(Z8Object self) {
    auto id = z8_query_record_id(self);
    auto registry = z8_entity_table_registry(self);
    auto entity = z8_entity_registry_find(registry, id);
    if (entity !is null) return entity;
    auto values = z8_entity_table_current_values(self);
    entity = z8_nonnull(self.z8_entity_table_virtual_new_values(values));
    if (!z8_optional_require(z8_entity_initialized(entity))) z8_entity_init_map(entity, values);
    z8_entity_state(entity).persistent = true;
    z8_entity_register_to(entity, registry);
    z8_entity_table_init_dependencies(self, registry, z8_entity_table_state(self).dependencies);
    entity.z8_set_attribute("entityTable"w, self.z8_class_id());
    self.z8_entity_table_virtual_afterRead(entity);
    return entity;
}
