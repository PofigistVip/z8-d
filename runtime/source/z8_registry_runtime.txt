// Host-populated subset of AbstractRuntime's application registry.
// Keys use table names for tables and class IDs for requests, as in Java.
private class Z8ControlSumState { int value; }
private int z8_control_abs(int value) { return value == int.min ? value : value < 0 ? -value : value; }
private int z8_control_string_hash(wstring value) {
    uint hash;
    foreach (wchar code; z8_string_required(value)) hash = 31u * hash + code;
    return cast(int)hash;
}
private wstring z8_control_field_sql_type(Z8Object field) {
    if (z8_native_is_instance("org.zenframework.z8.server.base.table.value.Expression"w, field)) return "null"w;
    switch (z8_field_type_id(field)) {
        case "int"w: case "date"w: case "datetime"w: case "datespan"w: return "bigint"w;
        case "boolean"w: return "smallint"w;
        case "float"w: return "numeric"w;
        case "string"w: return "character varying"w;
        case "guid"w: return "uuid"w;
        case "binary"w: case "text"w: case "file"w: case "attachments"w: return "bytea"w;
        case "geometry"w: return "geometry(Geometry, "w ~ z8_optional_require(z8_field_srs(field)).to!wstring ~ ")"w;
        default: throw new Exception("unsupported control sum field type");
    }
}
// OBJECT caches nonzero sums; Table and Field override its calculation.
int z8_object_control_sum(Z8Object value) {
    value = z8_nonnull(value);
    enum stateName = "org.zenframework.z8.server.runtime.OBJECT.controlSum"w;
    auto state = cast(Z8ControlSumState)value.z8_get_native_state(stateName);
    if (state is null) { state = new Z8ControlSumState(); value.z8_set_native_state(stateName, state); }
    if (state.value != 0) return state.value;
    if (z8_native_is_instance("org.zenframework.z8.server.base.table.Table"w, value)) {
        uint sum;
        foreach (field; z8_query_declared_fields(value, true)) sum += cast(uint)z8_object_control_sum(field);
        state.value = z8_control_abs(z8_control_string_hash((cast(int)sum).to!wstring));
    } else if (z8_native_is_instance("org.zenframework.z8.server.base.table.value.Field"w, value)) {
        auto name = z8_native_call!wstring("org.zenframework.z8.server.runtime.OBJECT.m_name"w, value);
        state.value = z8_control_abs(z8_control_string_hash(name ~ " "w ~ z8_control_field_sql_type(value)));
    } else state.value = z8_control_abs(z8_control_string_hash(value.z8_class_id()));
    return state.value;
}
struct Z8RuntimeFieldRegistration {
    wstring index, name, displayName, description, type;
    long ordinal;
    Z8Guid key;
}
struct Z8RuntimeRegistration {
    wstring classId, name, displayName, description;
    // A bare [job] is an explicitly disabled schedule; no attribute is absent.
    wstring cron;
    bool hasSchedule;
    Z8Guid key;
    Z8RuntimeFieldRegistration[] fields;
    private TypeInfo_Class runtimeType;
}
struct Z8VersionRecord {
    Z8Guid id;
    wstring classId, name, displayName, controlSum;
}
struct Z8RuntimeVersion { wstring number; Z8VersionRecord[] details; }
private bool z8_is_system_tools(Z8Object value) {
    import std.string : lastIndexOf;
    enum className = "org.zenframework.z8.server.base.table.system.SystemTools";
    for (auto type = value.classinfo; type !is null; type = type.base) {
        auto dot = type.name.lastIndexOf('.');
        auto symbol = dot < 0 ? type.name : type.name[dot + 1 .. $];
        if (z8_object_class_symbol_name(symbol, dot < 0 ? "" : type.name[0 .. dot]) == className) return true;
    }
    return false;
}
private wstring z8_version_format(int sum) {
    auto number = z8_control_abs(sum).to!wstring;
    while (number.length < 10) number = "0"w ~ number;
    return number[0 .. 1] ~ "."w ~ number[1 .. 4] ~ "."w ~ number[4 .. 7] ~ "."w ~ number[7 .. $];
}
// The original control-object list combines tables, jobs and entries by class
// ID, excludes SystemTools entries, and sorts by Java's UTF-16 string order.
Z8RuntimeVersion z8_runtime_version(Z8RuntimeRegistry registry, Z8Object[] additionalTables = null) {
    import std.algorithm.sorting : sort;
    registry = z8_nonnull(registry);
    Z8Object[wstring] objects;
    foreach (entry; registry.tables() ~ registry.jobs()) objects[entry.classId] = z8_object_new_instance(entry.classId);
    foreach (table; additionalTables) {
        if (!z8_native_is_instance("org.zenframework.z8.server.base.table.Table"w, z8_nonnull(table)))
            throw new Exception("runtime version requires a Table");
        auto classId = table.z8_class_id();
        objects[classId] = z8_object_new_instance(classId);
    }
    foreach (entry; registry.entries()) {
        auto object = z8_object_new_instance(entry.classId);
        if (!z8_is_system_tools(object)) objects[entry.classId] = object;
    }
    auto names = objects.keys;
    sort(names);
    Z8RuntimeVersion result;
    uint sum;
    foreach (classId; names) {
        auto object = objects[classId];
        auto control = z8_object_control_sum(object);
        sum += cast(uint)control;
        result.details ~= Z8VersionRecord(z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, object),
            classId, z8_object_name(object), z8_object_display_name(object), control.to!wstring);
    }
    result.number = z8_version_format(cast(int)sum);
    result.details = [Z8VersionRecord(z8_guid_create_name("system.version"w), "Version"w, "Version"w,
        "Database version"w, result.number)] ~ result.details;
    return result;
}
wstring z8_runtime_version_details_json(Z8RuntimeVersion revision) {
    auto writer = new Z8JsonWriterState();
    writer.start('[');
    foreach (record; revision.details) {
        writer.start('{');
        z8_monitor_property(writer, "recordId"w, record.id);
        z8_monitor_property(writer, "className"w, record.classId);
        z8_monitor_property(writer, "name"w, record.name);
        z8_monitor_property(writer, "displayName"w, record.displayName);
        z8_monitor_property(writer, "controlSum"w, record.controlSum);
        writer.finish('}');
    }
    writer.finish(']');
    return writer.stream;
}
private struct Z8VersionText { bool present; wstring text; }
private Z8VersionText z8_version_json_text(JSONValue record, string name) {
    if (record.type != JSONType.object) throw new Exception("version detail must be an object");
    auto value = name in record.object;
    if (value is null || value.type == JSONType.null_) return Z8VersionText.init;
    return Z8VersionText(true, z8_json_value_text(*value));
}
private JSONValue z8_version_json_value(Z8VersionText value) {
    return value.present ? JSONValue(value.text.to!string) : JSONValue.init;
}
// Implements Version.getVersionDiff, retaining removed records and appending
// new ones in their current order, including any original extra properties.
wstring z8_runtime_version_diff(wstring savedDetails, Z8RuntimeVersion current) {
    auto old = z8_json_array_read(savedDetails);
    auto newer = z8_json_array_read(z8_runtime_version_details_json(current));
    bool[] used; used.length = newer.array.length;
    auto rows = old.array.dup;
    foreach (ref record; rows) {
        auto className = z8_version_json_text(record, "className");
        if (!className.present) throw new Exception("version detail has no class name");
        foreach (index, candidate; newer.array) {
            if (used[index] || z8_version_json_text(candidate, "className").text != className.text) continue;
            used[index] = true;
            auto name = z8_version_json_text(candidate, "name");
            auto displayName = z8_version_json_text(candidate, "displayName");
            auto control = z8_version_json_text(candidate, "controlSum");
            auto previous = z8_version_json_text(record, "controlSum");
            record.object["newName"] = z8_version_json_value(name);
            record.object["newDisplayName"] = z8_version_json_value(displayName);
            record.object["newControlSum"] = z8_version_json_value(control);
            record.object["changed"] = JSONValue(previous.present != control.present ||
                (previous.present && previous.text != control.text));
            break;
        }
    }
    foreach (index, candidate; newer.array) {
        if (used[index]) continue;
        auto name = z8_version_json_text(candidate, "name");
        auto displayName = z8_version_json_text(candidate, "displayName");
        auto control = z8_version_json_text(candidate, "controlSum");
        candidate.object["newName"] = z8_version_json_value(name);
        candidate.object["name"] = JSONValue("");
        candidate.object["newDisplayName"] = z8_version_json_value(displayName);
        candidate.object["displayName"] = JSONValue("");
        candidate.object["newControlSum"] = z8_version_json_value(control);
        candidate.object["controlSum"] = JSONValue("");
        candidate.object["changed"] = JSONValue(true);
        rows ~= candidate;
    }
    return z8_json_render(JSONValue(rows));
}
private Z8Object z8_runtime_version_view_data() {
    auto registry = z8_runtime_get_registry();
    if (registry is null) throw new Exception("application registry is not configured");
    auto provider = cast(IZ8RuntimeVersionSource)z8_connection_current_backend();
    auto current = provider is null ? z8_runtime_version(registry) : provider.currentRuntimeVersion();
    auto saved = z8_settings_string_default(z8_guid_create_name("system.version.details"w), "[]"w);
    return z8_json_array_parse(z8_runtime_version_diff(saved, current));
}
private Z8File z8_runtime_version_export() {
    import std.string : replace;
    enum wstring[] fields = ["recordId"w, "className"w, "name"w, "newName"w, "displayName"w,
        "newDisplayName"w, "controlSum"w, "newControlSum"w, "changed"w];
    auto rows = z8_json_array_state(z8_runtime_version_view_data()).value.array;
    wstring csv;
    if (rows.length) {
        foreach (i, field; fields) {
            if (i) csv ~= ";"w;
            csv ~= "\""w ~ field.replace("\""w, "\"\""w) ~ "\""w;
        }
        foreach (row; rows) {
            csv ~= "\n"w;
            foreach (i, field; fields) {
                if (i) csv ~= ";"w;
                auto value = z8_version_json_text(row, field.to!string);
                csv ~= "\""w ~ (value.present ? value.text.replace("\""w, "\"\""w) : ""w) ~ "\""w;
            }
        }
    }
    auto bytes = z8_encoding_encode(csv, z8_encoding_by_id(0));
    auto file = new Z8File(); file.id = null; file.name = "runtime.csv"w;
    file.path = z8_temporary_archive_store(bytes, file.name);
    file.size = z8_optional_wrap(cast(long)bytes.length);
    return file;
}
shared static this() {
    z8_native_bind("org.zenframework.z8.server.base.table.system.view.RuntimeVersionData.m_read"w, &z8_runtime_version_view_data);
    z8_native_bind("org.zenframework.z8.server.base.table.system.view.RuntimeVersionData.m_export"w, &z8_runtime_version_export);
}
class Z8RuntimeRegistry {
    private Z8RuntimeRegistration[] tableEntries, requestEntries, jobEntries, entryEntries, executableEntries, namedEntries;
    private Z8Object[wstring] namedSingletons;
    private Object mutex;
    this() { mutex = new Object(); }
    private Z8RuntimeRegistration descriptor(Z8Object value, bool table) {
        value = z8_nonnull(value);
        if (table && !z8_native_is_instance("org.zenframework.z8.server.base.table.Table"w, value))
            throw new Exception("runtime table registration requires a Table");
        auto classId = value.z8_class_id();
        auto name = z8_native_call!wstring("org.zenframework.z8.server.runtime.OBJECT.m_name"w, value);
        if (classId is null || !classId.length || name is null || !name.length) throw new Exception("runtime class ID and name are required");
        auto displayName = z8_native_call!wstring("org.zenframework.z8.server.runtime.OBJECT.m_displayName"w, value);
        Z8RuntimeRegistration entry;
        entry.classId = classId.idup; entry.name = name.idup; entry.displayName = displayName.idup;
        entry.description = value.z8_get_attribute("description"w);
        entry.runtimeType = value.classinfo;
        entry.key = z8_guid_create_name(table ? name : classId);
        if (table) foreach (field; z8_query_declared_fields(value, true)) {
            Z8RuntimeFieldRegistration item;
            item.index = field.z8_index();
            item.name = z8_native_call!wstring("org.zenframework.z8.server.runtime.OBJECT.m_name"w, field);
            item.displayName = field.z8_get_attribute("displayName"w);
            if (item.displayName is null || !item.displayName.length) item.displayName = item.name;
            item.description = field.z8_get_attribute("description"w);
            item.type = z8_field_type_id(field);
            if (item.type == "string"w) item.type ~= "("w ~ z8_optional_require(z8_field_state(field).length).to!wstring ~ ")"w;
            item.ordinal = field.z8_get_ordinal();
            item.key = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, field);
            foreach (existing; entry.fields) if (existing.key == item.key) throw new Exception("duplicate runtime field key");
            entry.fields ~= item;
        }
        return entry;
    }
    private void add(ref Z8RuntimeRegistration[] entries, Z8RuntimeRegistration entry, bool table, bool named = false) {
        synchronized (mutex) {
            foreach (ref existing; entries) {
                if (existing.classId == entry.classId && existing.name == entry.name && existing.key == entry.key) { existing = entry; return; }
                if ((table || named) && entry.runtimeType !is null && existing.runtimeType !is null) {
                    // A more specialized table supersedes its named base,
                    // regardless of the order in which startup code sees them.
                    if (entry.runtimeType.isBaseOf(existing.runtimeType)) return;
                    if (existing.name == entry.name && existing.runtimeType.isBaseOf(entry.runtimeType)) { existing = entry; return; }
                }
                if (existing.classId == entry.classId || existing.key == entry.key || ((table || named) && existing.name == entry.name))
                    throw new Exception("conflicting runtime registration");
            }
            entries ~= entry;
        }
    }
    void registerTable(Z8Object table) { add(tableEntries, descriptor(table, true), true); }
    void registerRequest(Z8Object request) { add(requestEntries, descriptor(request, false), false); }
    void registerJob(Z8Object job) {
        if (!z8_native_is_instance(z8_executable_name, job)) throw new Exception("runtime job registration requires an Executable");
        auto entry = descriptor(job, false);
        auto cron = job.z8_get_attribute("job"w);
        entry.hasSchedule = cron !is null;
        if (entry.hasSchedule) entry.cron = z8_string_new(cron);
        add(jobEntries, entry, false);
    }
    void registerEntry(Z8Object entry) { add(entryEntries, descriptor(entry, false), false); }
    void registerExecutable(Z8Object executable) {
        if (!z8_native_is_instance(z8_executable_name, executable)) throw new Exception("runtime executable registration requires an Executable");
        add(executableEntries, descriptor(executable, false), false, true);
    }
    // Descriptors are lazy: native classes need not have their runtime adapters
    // initialized until an application actually asks for an instance.
    void registerNamedType(wstring name, wstring classId, TypeInfo_Class runtimeType) {
        name = z8_string_required(name); classId = z8_string_required(classId);
        if (!name.length || !classId.length || runtimeType is null || !typeid(Z8Object).isBaseOf(runtimeType))
            throw new Exception("invalid named class registration");
        Z8RuntimeRegistration entry;
        entry.name = name.idup; entry.classId = classId.idup; entry.key = z8_guid_create_name(classId); entry.runtimeType = runtimeType;
        synchronized (mutex) {
            foreach (ref existing; namedEntries) {
                if (entry.runtimeType.isBaseOf(existing.runtimeType)) return;
                if (existing.name == entry.name) {
                    existing = entry; namedSingletons.remove(name); return;
                }
            }
            namedEntries ~= entry;
        }
    }
    private Z8RuntimeRegistration[] snapshot(ref Z8RuntimeRegistration[] entries) {
        Z8RuntimeRegistration[] result;
        synchronized (mutex) {
            foreach (entry; entries) {
                entry.key = new Z8Guid(entry.key.majorBits, entry.key.minorBits);
                entry.fields = entry.fields.dup;
                foreach (ref field; entry.fields) field.key = new Z8Guid(field.key.majorBits, field.key.minorBits);
                result ~= entry;
            }
        }
        return result;
    }
    Z8RuntimeRegistration[] tables() { return snapshot(tableEntries); }
    Z8RuntimeRegistration[] requests() { return snapshot(requestEntries); }
    Z8RuntimeRegistration[] jobs() { return snapshot(jobEntries); }
    Z8RuntimeRegistration[] entries() { return snapshot(entryEntries); }
    Z8RuntimeRegistration[] executables() { return snapshot(executableEntries); }
    Z8RuntimeRegistration[] named() { return snapshot(namedEntries); }
    Z8Guid[] tableKeys() { Z8Guid[] keys; foreach (item; tables()) keys ~= item.key; return keys; }
    Z8Guid[] requestKeys() { Z8Guid[] keys; foreach (item; requests()) keys ~= item.key; return keys; }
    Z8Guid[] jobKeys() { Z8Guid[] keys; foreach (item; jobs()) keys ~= item.key; return keys; }
    Z8Guid[] entryKeys() { Z8Guid[] keys; foreach (item; entries()) keys ~= item.key; return keys; }
    Z8Object newTableByName(wstring name) {
        name = z8_string_required(name);
        foreach (item; tables()) if (item.name == name) return z8_object_new_instance(item.classId);
        return null;
    }
    Z8Object newTableByClass(wstring classId) {
        classId = z8_string_required(classId);
        foreach (item; tables()) if (item.classId == classId) return z8_object_new_instance(item.classId);
        return null;
    }
    Z8Object newRequestByClass(wstring classId) {
        classId = z8_string_required(classId);
        foreach (item; requests()) if (item.classId == classId) return z8_object_new_instance(item.classId);
        return null;
    }
    Z8Object newJobByClass(wstring classId) {
        classId = z8_string_required(classId);
        foreach (item; jobs()) if (item.classId == classId) return z8_object_new_instance(item.classId);
        return null;
    }
    Z8Object newEntryByClass(wstring classId) {
        classId = z8_string_required(classId);
        foreach (item; entries()) if (item.classId == classId) return z8_object_new_instance(item.classId);
        return null;
    }
    Z8Object newExecutableByName(wstring name) {
        name = z8_string_required(name);
        foreach (item; executables()) if (item.name == name) return z8_object_new_instance(item.classId);
        return null;
    }
    Z8Object newNamedByName(wstring name) {
        name = z8_string_required(name);
        foreach (item; named()) if (item.name == name) return z8_object_new_instance(item.classId);
        return null;
    }
    Z8Object namedSingleton(wstring name) {
        name = z8_string_required(name);
        synchronized (mutex) {
            auto existing = name in namedSingletons;
            if (existing !is null) return *existing;
            foreach (item; namedEntries) if (item.name == name) {
                auto value = z8_object_new_instance(item.classId);
                namedSingletons[name] = value;
                return value;
            }
        }
        return null;
    }
    Z8Object[] newTables() {
        Z8Object[] result;
        foreach (item; tables()) result ~= z8_object_new_instance(item.classId);
        return result;
    }
}
private __gshared Object z8_runtime_registry_lock;
private __gshared Z8RuntimeRegistry z8_runtime_registry;
shared static this() { z8_runtime_registry_lock = new Object(); }
Z8RuntimeRegistry z8_runtime_set_registry(Z8RuntimeRegistry value) {
    synchronized (z8_runtime_registry_lock) {
        auto previous = z8_runtime_registry; z8_runtime_registry = value; return previous;
    }
}
Z8RuntimeRegistry z8_runtime_get_registry() {
    synchronized (z8_runtime_registry_lock) return z8_runtime_registry;
}
