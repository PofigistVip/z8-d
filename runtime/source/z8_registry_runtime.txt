// Host-populated subset of AbstractRuntime's table and request registry.
// Keys use table names for tables and class IDs for requests, as in Java.
struct Z8RuntimeFieldRegistration {
    wstring index, name, displayName, description, type;
    long ordinal;
    Z8Guid key;
}
struct Z8RuntimeRegistration {
    wstring classId, name, displayName, description;
    Z8Guid key;
    Z8RuntimeFieldRegistration[] fields;
    private TypeInfo_Class runtimeType;
}
class Z8RuntimeRegistry {
    private Z8RuntimeRegistration[] tableEntries, requestEntries;
    private Object mutex;
    this() { mutex = new Object(); }
    private Z8RuntimeRegistration descriptor(Z8Object value, bool table) {
        value = z8_nonnull(value);
        if (table && !z8_native_is_instance("org.zenframework.z8.server.base.table.Table"w, value))
            throw new Exception("runtime table registration requires a Table");
        auto classId = value.z8_class_id();
        auto name = z8_native_call!wstring("org.zenframework.z8.server.runtime.OBJECT.m_name"w, value);
        if (classId is null || !classId.length || name is null || !name.length) throw new Exception("runtime class ID and name are required");
        auto displayName = z8_native_call!wstring("org.zenframework.z8.server.runtime.OBJECT.m_displayName"w, value);
        Z8RuntimeRegistration entry;
        entry.classId = classId.idup; entry.name = name.idup; entry.displayName = displayName.idup;
        entry.description = value.z8_get_attribute("description"w);
        entry.runtimeType = value.classinfo;
        entry.key = z8_guid_create_name(table ? name : classId);
        if (table) foreach (field; z8_query_declared_fields(value, true)) {
            Z8RuntimeFieldRegistration item;
            item.index = field.z8_index();
            item.name = z8_native_call!wstring("org.zenframework.z8.server.runtime.OBJECT.m_name"w, field);
            item.displayName = field.z8_get_attribute("displayName"w);
            if (item.displayName is null || !item.displayName.length) item.displayName = item.name;
            item.description = field.z8_get_attribute("description"w);
            item.type = z8_field_type_id(field);
            if (item.type == "string"w) item.type ~= "("w ~ z8_optional_require(z8_field_state(field).length).to!wstring ~ ")"w;
            item.ordinal = field.z8_get_ordinal();
            item.key = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, field);
            foreach (existing; entry.fields) if (existing.key == item.key) throw new Exception("duplicate runtime field key");
            entry.fields ~= item;
        }
        return entry;
    }
    private void add(ref Z8RuntimeRegistration[] entries, Z8RuntimeRegistration entry, bool table) {
        synchronized (mutex) {
            foreach (ref existing; entries) {
                if (existing.classId == entry.classId && existing.name == entry.name && existing.key == entry.key) { existing = entry; return; }
                if (table && entry.runtimeType !is null && existing.runtimeType !is null) {
                    // A more specialized table supersedes its named base,
                    // regardless of the order in which startup code sees them.
                    if (entry.runtimeType.isBaseOf(existing.runtimeType)) return;
                    if (existing.name == entry.name && existing.runtimeType.isBaseOf(entry.runtimeType)) { existing = entry; return; }
                }
                if (existing.classId == entry.classId || existing.key == entry.key || (table && existing.name == entry.name))
                    throw new Exception("conflicting runtime registration");
            }
            entries ~= entry;
        }
    }
    void registerTable(Z8Object table) { add(tableEntries, descriptor(table, true), true); }
    void registerRequest(Z8Object request) { add(requestEntries, descriptor(request, false), false); }
    private Z8RuntimeRegistration[] snapshot(bool table) {
        Z8RuntimeRegistration[] result;
        synchronized (mutex) {
            auto entries = table ? tableEntries : requestEntries;
            foreach (entry; entries) {
                entry.key = new Z8Guid(entry.key.majorBits, entry.key.minorBits);
                entry.fields = entry.fields.dup;
                foreach (ref field; entry.fields) field.key = new Z8Guid(field.key.majorBits, field.key.minorBits);
                result ~= entry;
            }
        }
        return result;
    }
    Z8RuntimeRegistration[] tables() { return snapshot(true); }
    Z8RuntimeRegistration[] requests() { return snapshot(false); }
    Z8Guid[] tableKeys() { Z8Guid[] keys; foreach (item; tables()) keys ~= item.key; return keys; }
    Z8Guid[] requestKeys() { Z8Guid[] keys; foreach (item; requests()) keys ~= item.key; return keys; }
    Z8Object newTableByName(wstring name) {
        name = z8_string_required(name);
        foreach (item; tables()) if (item.name == name) return z8_object_new_instance(item.classId);
        return null;
    }
    Z8Object newTableByClass(wstring classId) {
        classId = z8_string_required(classId);
        foreach (item; tables()) if (item.classId == classId) return z8_object_new_instance(item.classId);
        return null;
    }
    Z8Object newRequestByClass(wstring classId) {
        classId = z8_string_required(classId);
        foreach (item; requests()) if (item.classId == classId) return z8_object_new_instance(item.classId);
        return null;
    }
    Z8Object[] newTables() {
        Z8Object[] result;
        foreach (item; tables()) result ~= z8_object_new_instance(item.classId);
        return result;
    }
}
private __gshared Object z8_runtime_registry_lock;
private __gshared Z8RuntimeRegistry z8_runtime_registry;
shared static this() { z8_runtime_registry_lock = new Object(); }
Z8RuntimeRegistry z8_runtime_set_registry(Z8RuntimeRegistry value) {
    synchronized (z8_runtime_registry_lock) {
        auto previous = z8_runtime_registry; z8_runtime_registry = value; return previous;
    }
}
Z8RuntimeRegistry z8_runtime_get_registry() {
    synchronized (z8_runtime_registry_lock) return z8_runtime_registry;
}
