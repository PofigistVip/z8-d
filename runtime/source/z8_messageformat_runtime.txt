// Basic MessageFormat placeholders with the Russian locale's number formatting.
private wstring z8_message_number(wstring value, size_t minFraction = 0, bool grouping = true, size_t minInteger = 1, size_t groupSize = 3) {
    auto dot = value.length;
    foreach (i, wchar c; value) if (c == '.') { dot = i; break; }
    auto integer = value[0 .. dot];
    auto start = integer.length && integer[0] == '-' ? 1 : 0;
    wstring padded = start ? "-"w : ""w;
    if (integer.length - start < minInteger) foreach (_; 0 .. minInteger - (integer.length - start)) padded ~= '0';
    padded ~= integer[start .. $];
    integer = padded;
    wstring result;
    foreach (i, wchar c; integer) {
        if (grouping && i > start && (integer.length - i) % groupSize == 0) result ~= cast(wchar)0x00A0;
        result ~= c;
    }
    if (dot < value.length) {
        auto end = value.length;
        while (end > dot + 1 + minFraction && value[end - 1] == '0') --end;
        if (end > dot + 1) result ~= ","w ~ value[dot + 1 .. end];
    }
    if (minInteger == 0 && result.length >= 2 && result[0 .. 2] == "0,"w) result = result[1 .. $];
    else if (minInteger == 0 && result.length >= 3 && result[0 .. 3] == "-0,"w) result = "-"w ~ result[2 .. $];
    return result.length ? result : ""w;
}
private struct Z8NumberPattern {
    wstring prefix, suffix;
    int digits;
    size_t minFraction, minInteger, groupSize;
    bool grouping, decimalAlways;
    int multiplier = 1;
}
private Z8NumberPattern z8_message_number_pattern(wstring pattern) {
    Z8NumberPattern result;
    size_t first = pattern.length, last;
    bool quoted;
    for (size_t i; i < pattern.length; ++i) {
        auto c = pattern[i];
        if (c == '\'') {
            if (i + 1 < pattern.length && pattern[i + 1] == '\'') { ++i; continue; }
            quoted = !quoted;
        } else if (!quoted && (c == '#' || c == '0')) {
            if (first == pattern.length) first = i;
            last = i + 1;
        }
    }
    if (first == pattern.length || quoted) throw new Exception("unsupported number pattern");
    if (first && pattern[first - 1] == '.') --first;
    if (last < pattern.length && pattern[last] == '.') { result.decimalAlways = true; ++last; }
    bool fraction;
    size_t groupDigits;
    foreach (c; pattern[first .. last]) {
        if (c == '.') { if (fraction) throw new Exception("unsupported number pattern"); fraction = true; }
        else if (c == ',' && !fraction) { result.grouping = true; groupDigits = 0; }
        else if (c == '0' || c == '#') {
            if (fraction) { ++result.digits; if (c == '0') ++result.minFraction; }
            else { ++groupDigits; if (c == '0') ++result.minInteger; }
        } else throw new Exception("unsupported number pattern");
    }
    if (result.digits > 1000 || (result.grouping && !groupDigits)) throw new Exception("unsupported number pattern");
    if (pattern[first] == '.' && !result.minFraction) result.minFraction = 1;
    result.groupSize = result.grouping ? groupDigits : 3;
    wstring affix(wstring part) {
        wstring output;
        bool inQuote;
        for (size_t i; i < part.length; ++i) {
            auto c = part[i];
            if (c == '\'') {
                if (i + 1 < part.length && part[i + 1] == '\'') { output ~= '\''; ++i; }
                else inQuote = !inQuote;
            } else if (!inQuote && (c == '%' || c == '‰')) {
                if (result.multiplier != 1) throw new Exception("invalid number pattern multiplier");
                result.multiplier = c == '%' ? 100 : 1000;
                output ~= c;
            } else if (!inQuote && c == '¤') {
                if (i + 1 < part.length && part[i + 1] == '¤') { output ~= "RUB"w; ++i; }
                else output ~= "руб."w;
            } else output ~= c;
        }
        if (inQuote) throw new Exception("invalid number pattern quote");
        return output;
    }
    result.prefix = affix(pattern[0 .. first]);
    result.suffix = affix(pattern[last .. $]);
    return result;
}
private wstring z8_message_formatted_number(Z8Primary value, wstring style) {
    import std.string : strip;
    z8_nonnull(value);
    Z8Decimal number;
    switch (value.kind) {
        case Z8PrimaryKind.integer: number = z8_decimal_from_int(value.integerValue); break;
        case Z8PrimaryKind.decimal: number = value.decimalValue; break;
        case Z8PrimaryKind.datespan: number = z8_decimal_from_int(z8_datespan_milliseconds(value.datespanValue)); break;
        default: throw new Exception("MessageFormat argument is not numeric");
    }
    if (number.nan) return "null"w;
    style = style.strip();
    int digits = 3;
    size_t minFraction, minInteger = 1;
    bool grouping = true, percent, currency, decimalAlways;
    size_t groupSize = 3;
    wstring prefix, suffix;
    bool negative = number.coefficient < 0;
    if (style == "integer"w) digits = 0;
    else if (style == "percent"w) { digits = 0; percent = true; number = number * z8_decimal_from_int(100); }
    else if (style == "currency"w) { digits = 2; currency = true; }
    else if (style.length) {
        size_t separator = style.length;
        bool quoted;
        for (size_t i; i < style.length; ++i) {
            if (style[i] == '\'') {
                if (i + 1 < style.length && style[i + 1] == '\'') { ++i; continue; }
                quoted = !quoted;
            } else if (style[i] == ';' && !quoted) { separator = i; break; }
        }
        auto positivePattern = z8_message_number_pattern(style[0 .. separator]);
        digits = positivePattern.digits; minFraction = positivePattern.minFraction;
        minInteger = positivePattern.minInteger; grouping = positivePattern.grouping; groupSize = positivePattern.groupSize;
        decimalAlways = positivePattern.decimalAlways;
        auto selected = positivePattern;
        if (negative && separator < style.length) selected = z8_message_number_pattern(style[separator + 1 .. $]);
        prefix = (negative && separator == style.length ? "-"w : ""w) ~ selected.prefix;
        suffix = selected.suffix;
        if (positivePattern.multiplier != 1) number = number * z8_decimal_from_int(positivePattern.multiplier);
        if (negative) number = -number;
    }
    auto rounded = number.z8_round(digits, 6);
    auto formatted = z8_message_number(rounded.z8_string(), minFraction, grouping, minInteger, groupSize);
    if (decimalAlways && !formatted.canFind(',')) formatted ~= ',';
    if (rounded.coefficient == 0 && number.coefficient < 0) formatted = "-"w ~ formatted;
    if (percent) formatted ~= "%"w;
    if (currency) formatted ~= " руб."w;
    return prefix ~ formatted ~ suffix;
}
private wstring z8_message_value(Z8Primary value) {
    z8_nonnull(value);
    final switch (value.kind) {
        case Z8PrimaryKind.integer: return z8_message_number(value.integerValue.to!wstring);
        case Z8PrimaryKind.decimal:
            if (value.decimalValue.nan) return "null"w;
            auto rounded = value.decimalValue.z8_round(3, 6);
            auto formatted = z8_message_number(rounded.z8_string());
            return rounded.coefficient == 0 && value.decimalValue.coefficient < 0 ? "-"w ~ formatted : formatted;
        case Z8PrimaryKind.datespan: return z8_message_number(z8_datespan_milliseconds(value.datespanValue).to!wstring);
        case Z8PrimaryKind.text: return z8_string_or_empty(value.textValue);
        case Z8PrimaryKind.boolean: return value.booleanValue ? "true"w : "false"w;
        case Z8PrimaryKind.guid: return z8_guid_to_string(value.guidValue);
        case Z8PrimaryKind.geometry: return z8_geometry_to_string(value.geometryValue);
        case Z8PrimaryKind.file: throw new Exception("uploaded file MessageFormat value is unsupported");
        case Z8PrimaryKind.user: throw new Exception("user primary MessageFormat value is unsupported");
        case Z8PrimaryKind.binary: case Z8PrimaryKind.date: throw new Exception("unsupported MessageFormat value");
    }
}
private wstring z8_message_formatted_date(Z8Primary value, wstring style, bool timeOnly) {
    import std.string : strip;
    z8_nonnull(value);
    long ticks;
    switch (value.kind) {
        case Z8PrimaryKind.integer: ticks = value.integerValue; break;
        case Z8PrimaryKind.datespan: ticks = z8_datespan_milliseconds(value.datespanValue); break;
        case Z8PrimaryKind.decimal:
            if (value.decimalValue.nan) return "null"w;
            ticks = value.decimalValue.z8_round(0, 1).coefficient.to!long;
            break;
        default: throw new Exception("MessageFormat date argument is not numeric");
    }
    style = style.strip();
    if (timeOnly) {
        if (style == ""w || style == "medium"w) style = "H:mm:ss"w;
        else if (style == "short"w) style = "H:mm"w;
        else if (style == "long"w) style = "H:mm:ss z"w;
        else if (style == "full"w) style = "H:mm:ss z"w;
    } else {
        if (style == ""w || style == "medium"w) style = "dd.MM.yyyy"w;
        else if (style == "short"w) style = "dd.MM.yy"w;
        else if (style == "long"w) style = "d MMMM y 'г'."w;
        else if (style == "full"w) style = "d MMMM y 'г'."w;
    }
    return z8_date_toString(z8_date_from_ticks(ticks), style);
}
private wstring z8_message_choice(wstring style, Z8Primary value, Z8Array!Z8Primary values) {
    import std.string : strip;
    z8_nonnull(value);
    double number;
    switch (value.kind) {
        case Z8PrimaryKind.integer: number = cast(double)value.integerValue; break;
        case Z8PrimaryKind.decimal:
            if (value.decimalValue.nan) return "null"w;
            number = value.decimalValue.z8_as_double(); break;
        case Z8PrimaryKind.datespan: number = cast(double)z8_datespan_milliseconds(value.datespanValue); break;
        default: throw new Exception("MessageFormat choice argument is not numeric");
    }
    wstring selected;
    bool first = true;
    size_t start;
    while (start <= style.length) {
        size_t separator = start;
        while (separator < style.length && style[separator] != '#' && style[separator] != '<' && style[separator] != '≤') ++separator;
        if (separator == style.length) throw new Exception("invalid MessageFormat choice");
        auto limitText = style[start .. separator].strip();
        double limit = limitText == "∞"w ? double.infinity : limitText == "-∞"w ? -double.infinity : limitText.to!double;
        auto end = separator + 1;
        size_t depth;
        bool quoted;
        while (end < style.length) {
            auto c = style[end];
            if (c == '\'') {
                if (end + 1 < style.length && style[end + 1] == '\'') { end += 2; continue; }
                quoted = !quoted;
            } else if (!quoted && c == '{') ++depth;
            else if (!quoted && c == '}' && depth) --depth;
            else if (!quoted && c == '|' && !depth) break;
            ++end;
        }
        if (first || (style[separator] == '<' ? number > limit : number >= limit)) selected = style[separator + 1 .. end];
        first = false;
        if (end == style.length) break;
        start = end + 1;
    }
    wstring unquoted;
    size_t i;
    while (i < selected.length) {
        auto c = selected[i++];
        if (c == '\'') {
            if (i < selected.length && selected[i] == '\'') { unquoted ~= '\''; ++i; }
        } else unquoted ~= c;
    }
    return unquoted.canFind('{') ? z8_string_format(unquoted, values) : unquoted;
}
wstring z8_string_format(wstring pattern, Z8Array!Z8Primary values) {
    pattern = z8_string_required(pattern);
    z8_nonnull(values);
    // string.z8_format extracts getValue() for every argument before parsing
    // the pattern. The original primary implementation rejects user subclasses.
    foreach (i; 0 .. values.z8_size()) {
        auto value = z8_nonnull(values.z8_get(i));
        if (value.kind == Z8PrimaryKind.user) throw new Exception("user primary MessageFormat value is unsupported");
    }
    wstring result;
    bool quoted;
    size_t i;
    while (i < pattern.length) {
        auto c = pattern[i++];
        if (c == '\'') {
            if (i < pattern.length && pattern[i] == '\'') { result ~= '\''; ++i; }
            else quoted = !quoted;
        } else if (!quoted && c == '{') {
            size_t number;
            auto begin = i;
            while (i < pattern.length && pattern[i] >= '0' && pattern[i] <= '9') {
                if (number > 1000000) throw new Exception("invalid MessageFormat index");
                number = number * 10 + pattern[i++] - '0';
            }
            if (i == begin || i == pattern.length) throw new Exception("unsupported MessageFormat pattern");
            bool numeric, choice, dateFormat, timeFormat;
            wstring style;
            if (pattern[i] == ',') {
                import std.string : strip;
                auto kindStart = ++i;
                while (i < pattern.length && pattern[i] != ',' && pattern[i] != '}') ++i;
                if (i == pattern.length) throw new Exception("unsupported MessageFormat pattern");
                auto kind = pattern[kindStart .. i].strip();
                numeric = kind == "number"w; choice = kind == "choice"w;
                dateFormat = kind == "date"w; timeFormat = kind == "time"w;
                if (!numeric && !choice && !dateFormat && !timeFormat) throw new Exception("unsupported MessageFormat pattern");
                if (pattern[i] == ',') {
                    auto styleStart = ++i;
                    size_t depth;
                    bool styleQuoted;
                    while (i < pattern.length) {
                        auto current = pattern[i];
                        if (current == '\'') {
                            if (i + 1 < pattern.length && pattern[i + 1] == '\'') { i += 2; continue; }
                            styleQuoted = !styleQuoted;
                        } else if (!styleQuoted && current == '{') ++depth;
                        else if (!styleQuoted && current == '}') { if (!depth) break; --depth; }
                        ++i;
                    }
                    if (i == pattern.length) throw new Exception("unsupported MessageFormat pattern");
                    style = pattern[styleStart .. i];
                } else if (choice) throw new Exception("missing MessageFormat choice pattern");
            }
            if (pattern[i] != '}') throw new Exception("unsupported MessageFormat pattern");
            ++i;
            if (number >= cast(size_t)values.z8_size()) result ~= "{"w ~ number.to!wstring ~ "}"w;
            else {
                auto value = values.z8_get(cast(long)number);
                // getValue() returns Java null for NaN decimals, empty geometry
                // and files without an uploaded FileItem. MessageFormat formats
                // null as "null" even when a numeric or date style is present.
                bool javaNull = (value.kind == Z8PrimaryKind.decimal && value.decimalValue.nan) ||
                    (value.kind == Z8PrimaryKind.geometry && value.geometryValue.rawHex is null && value.geometryValue.shape == -1) ||
                    (value.kind == Z8PrimaryKind.file && !z8_file_has_upload(value.fileValue));
                result ~= javaNull ? "null"w : numeric ? z8_message_formatted_number(value, style) :
                    choice ? z8_message_choice(style, value, values) :
                    (dateFormat || timeFormat) ? z8_message_formatted_date(value, style, timeFormat) : z8_message_value(value);
            }
        } else result ~= c;
    }
    return result.length ? result : ""w;
}
