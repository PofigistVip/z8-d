private Z8Decimal z8_math_result(double value) {
    import std.math : isNaN, isFinite;
    if (isNaN(value)) return z8_decimal_nan();
    if (!isFinite(value)) throw new Exception("non-finite decimal result");
    return z8_decimal_literal(format("%.17g", value));
}
private Z8Decimal z8_math_unary(string name)(double x) {
    import std.math : sin, cos, tan, asin, acos, atan, sinh, cosh, tanh, exp, log, log10, sqrt, cbrt, fabs, PI;
    static if (name == "abs") return z8_math_result(fabs(x));
    else static if (name == "sin") return z8_math_result(sin(x));
    else static if (name == "cos") return z8_math_result(cos(x));
    else static if (name == "tan") return z8_math_result(tan(x));
    else static if (name == "asin") return z8_math_result(asin(x));
    else static if (name == "acos") return z8_math_result(acos(x));
    else static if (name == "atan") return z8_math_result(atan(x));
    else static if (name == "sinh") return z8_math_result(sinh(x));
    else static if (name == "cosh") return z8_math_result(cosh(x));
    else static if (name == "tanh") return z8_math_result(tanh(x));
    else static if (name == "toRadians") return z8_math_result(x / 180.0 * cast(double)PI);
    else static if (name == "toDegrees") return z8_math_result(x * 180.0 / cast(double)PI);
    else static if (name == "exp") return z8_math_result(exp(x));
    else static if (name == "ln") return z8_math_result(log(x));
    else static if (name == "log10") return z8_math_result(log10(x));
    else static if (name == "sqrt") return z8_math_result(sqrt(x));
    else static if (name == "cbrt") return z8_math_result(cast(double)cbrt(x));
    else static assert(0, "unsupported math method");
}
Z8Decimal z8_decimal_math(string name)(Z8Decimal value) if (name != "power") {
    return value.nan ? z8_decimal_nan() : z8_math_unary!name(value.z8_as_double());
}
Z8Decimal z8_decimal_math(string name)(Z8Decimal value, Z8Decimal power) if (name == "power") {
    import std.math : pow;
    if (value.nan || power.nan) return z8_decimal_nan();
    return z8_math_result(pow(value.z8_as_double(), power.z8_as_double()));
}

Z8Decimal z8_int_power(long x, Z8Decimal power) {
    import std.math : pow;
    if (power.nan) throw new Exception("null decimal value");
    return z8_math_result(pow(cast(double)x, power.z8_as_double()));
}
