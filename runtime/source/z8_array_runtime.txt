// User-defined BL enums compile to singleton objects; their declaration order
// is supplied by generated code, so separately compiled enums remain usable.
Z8Array!T z8_enum_array_sort(T)(Z8Array!T array, T[] order) if (is(T == class)) {
    import std.algorithm.sorting : sort;
    import std.algorithm.mutation : SwapStrategy;
    array = z8_nonnull(array);
    if (array.z8_size() > 1) {
        struct Entry { T value; size_t ordinal; }
        Entry[] entries;
        foreach (i; 0 .. array.z8_size()) {
            auto value = z8_nonnull(array.z8_get(i));
            size_t ordinal; bool found;
            foreach (index, candidate; order) if (value is candidate) { ordinal = index; found = true; break; }
            if (!found) throw new Exception("invalid enum array element");
            entries ~= Entry(value, ordinal);
        }
        sort!((a, b) => a.ordinal < b.ordinal, SwapStrategy.stable)(entries);
        foreach (index, entry; entries) array.z8_set(cast(long)index, entry.value);
    }
    ++array.z8_version;
    return array;
}

wstring z8_enum_array_join(T)(Z8Array!T array, T[] values, wstring[] names, wstring separator = ""w) if (is(T == class)) {
    array = z8_nonnull(array);
    wstring result;
    foreach (i; 0 .. array.z8_size()) {
        if (i) result ~= z8_string_required(separator);
        auto value = z8_nonnull(array.z8_get(i));
        bool found;
        foreach (index, candidate; values) if (value is candidate) {
            result ~= names[index]; found = true; break;
        }
        if (!found) throw new Exception("invalid enum array element");
    }
    return result;
}

class Z8Array(T) {
    enum isPrimitive = is(T == long) || is(T == bool) || is(T == Z8Decimal);
    enum sharePrimary = isPrimitive || is(T == wstring) || is(T == Z8Guid) || is(T == Z8File) ||
        is(T == Z8Binary) || is(T == Z8Datespan) || is(T == Z8Date) || is(T == Z8Geometry);
    static if (isPrimitive) alias Element = Z8Optional!T;
    else alias Element = T;
    private Element[] items;
    private size_t arrayVersion;
    static if (is(T == Object)) {
        private Z8Object membersOwner;
        void z8_set_members_owner(Z8Object owner) { membersOwner = owner; }
        private void z8_attach_member(Object value, bool setOrdinal) {
            if (membersOwner is null) return;
            auto member = cast(Z8Object)value;
            if (member is null) throw new Exception("invalid object member");
            member.z8_set_owner(membersOwner);
            if (setOrdinal) member.z8_set_ordinal(cast(long)items.length);
        }
    }
    static if (sharePrimary) private Z8Array!Z8Primary primaryData;
    @property ref size_t z8_version() {
        static if (sharePrimary) if (primaryData !is null) return primaryData.z8_version;
        return arrayVersion;
    }
    this() {}
    static if (isPrimitive) {
        this(T[] values) { foreach (value; values) items ~= z8_optional_wrap(value); }
        this(Element[] values, bool boxed) { items = values; }
    } else this(T[] values) { items = values; }
    // RCollection inherits ArrayList, whose equals compares element values.
    override bool opEquals(Object other) {
        if (this is other) return true;
        auto right = cast(Z8Array!T)other;
        static if (sharePrimary) {
            if (right !is null && (primaryData !is null || right.primaryData !is null))
                return z8_primary_array_view(this).opEquals(z8_primary_array_view(right));
            auto primary = cast(Z8Array!Z8Primary)other;
            if (primary !is null) return z8_primary_array_view(this).opEquals(primary);
        }
        static if (is(T == Z8Primary)) {
            auto strings = cast(Z8Array!wstring)other;
            if (strings !is null) return this.opEquals(z8_primary_array_view(strings));
            auto guids = cast(Z8Array!Z8Guid)other;
            if (guids !is null) return this.opEquals(z8_primary_array_view(guids));
            auto files = cast(Z8Array!Z8File)other;
            if (files !is null) return this.opEquals(z8_primary_array_view(files));
            auto binaries = cast(Z8Array!Z8Binary)other;
            if (binaries !is null) return this.opEquals(z8_primary_array_view(binaries));
            auto spans = cast(Z8Array!Z8Datespan)other;
            if (spans !is null) return this.opEquals(z8_primary_array_view(spans));
            auto dates = cast(Z8Array!Z8Date)other;
            if (dates !is null) return this.opEquals(z8_primary_array_view(dates));
            auto geometries = cast(Z8Array!Z8Geometry)other;
            if (geometries !is null) return this.opEquals(z8_primary_array_view(geometries));
            auto integers = cast(Z8Array!long)other;
            if (integers !is null) return this.opEquals(z8_primary_array_view(integers));
            auto booleans = cast(Z8Array!bool)other;
            if (booleans !is null) return this.opEquals(z8_primary_array_view(booleans));
            auto decimals = cast(Z8Array!Z8Decimal)other;
            if (decimals !is null) return this.opEquals(z8_primary_array_view(decimals));
        }
        if (right is null || items.length != right.items.length) return false;
        foreach (i, item; items) {
            auto value = right.items[i];
            static if (is(T == wstring)) {
                if (!z8_string_equals(item, value)) return false;
            } else static if (is(T == Z8Primary)) {
                if (!z8_primary_collection_equals(item, value)) return false;
            } else static if (is(T == class)) {
                if (item !is value && (item is null || value is null || item != value)) return false;
            } else if (item != value) return false;
        }
        return true;
    }
    long z8_size() { static if (sharePrimary) if (primaryData !is null) return primaryData.z8_size(); return cast(long)items.length; }
    bool z8_isEmpty() { static if (sharePrimary) if (primaryData !is null) return primaryData.z8_isEmpty(); return items.length == 0; }
    static if (is(T == class)) {
        T z8_first() { return z8_isEmpty() ? null : z8_get(0); }
        T z8_last() { return z8_isEmpty() ? null : z8_get(z8_size() - 1); }
    } else static if (is(T == wstring)) {
        T z8_first() { return z8_isEmpty() ? null : z8_get(0); }
        T z8_last() { return z8_isEmpty() ? null : z8_get(z8_size() - 1); }
    } else {
        Element z8_first() { return z8_isEmpty() ? Element.init : z8_get(0); }
        Element z8_last() { return z8_isEmpty() ? Element.init : z8_get(z8_size() - 1); }
    }
    void z8_clear() { static if (sharePrimary) if (primaryData !is null) { primaryData.z8_clear(); return; } ++z8_version; items.length = 0; }
    static if (isPrimitive) {
        Z8Array!T z8_add(T value) { return z8_add(z8_optional_wrap(value)); }
        Z8Array!T z8_add(long index, T value) { return z8_add(index, z8_optional_wrap(value)); }
    }
    Z8Array!T z8_add(Element value) {
        static if (sharePrimary) if (primaryData !is null) { primaryData.z8_add(z8_primary_box(value)); return this; }
        static if (is(T == Object)) z8_attach_member(value, true);
        ++z8_version; items ~= value; return this;
    }
    Z8Array!T z8_add(long index, Element value) {
        static if (sharePrimary) if (primaryData !is null) { primaryData.z8_add(index, z8_primary_box(value)); return this; }
        auto i = cast(int)index;
        if (i < 0 || cast(size_t)i > items.length) throw new Exception("array index out of bounds");
        static if (is(T == Object)) z8_attach_member(value, false);
        items = items[0 .. i] ~ [value] ~ items[i .. $];
        ++z8_version; return this;
    }
    void z8_addAll(Z8Array!T other) {
        z8_addAll(z8_size(), other);
    }
    void z8_addAll(long index, Z8Array!T other) {
        z8_nonnull(other);
        static if (sharePrimary) if (primaryData !is null || other.primaryData !is null) {
            z8_primary_array_view(this).z8_addAll(index, z8_primary_array_view(other)); return;
        }
        auto i = cast(int)index;
        auto expectedVersion = other.z8_version;
        for (long j = 0; j != other.z8_size(); ++j) {
            if (expectedVersion != other.z8_version) throw new Exception("array modified during iteration");
            z8_add(i++, other.z8_get(j));
        }
    }
    static if (isPrimitive) long z8_indexOf(T value) { return z8_indexOf(z8_optional_wrap(value)); }
    long z8_indexOf(Element value) {
        static if (sharePrimary) if (primaryData !is null) return primaryData.z8_indexOf(z8_primary_box(value));
        foreach (i, item; items) {
            static if (isPrimitive) {
                if (!item.present && !value.present) return cast(long)i;
                if (!item.present) throw new Exception("null array element");
                if (value.present && item.value == value.value) return cast(long)i;
            } else static if (is(T == class)) {
                if (item is value) return cast(long)i;
                if (item is null) throw new Exception("null array element");
                static if (is(T == Z8Primary)) { if (z8_primary_collection_equals(item, value)) return cast(long)i; }
                else if (item == value) return cast(long)i;
            } else static if (is(T == wstring)) {
                if (item is null && value is null) return cast(long)i;
                if (item is null) throw new Exception("null array element");
                if (z8_string_equals(item, value)) return cast(long)i;
            } else if (item == value) return cast(long)i;
        }
        return -1;
    }
    static if (isPrimitive) bool z8_contains(T value) { return z8_contains(z8_optional_wrap(value)); }
    bool z8_contains(Element value) { return z8_indexOf(value) >= 0; }
    bool z8_isSubsetOf(Z8Array!T other) {
        z8_nonnull(other);
        static if (sharePrimary) if (primaryData !is null || other.primaryData !is null)
            return z8_primary_array_view(this).z8_isSubsetOf(z8_primary_array_view(other));
        foreach (item; items) if (!other.z8_contains(item)) return false;
        return true;
    }
    Element z8_removeAt(long index) {
        static if (sharePrimary) if (primaryData !is null) return z8_primary_cast!T(primaryData.z8_removeAt(index));
        auto i = cast(int)index;
        if (i < 0 || cast(size_t)i >= items.length) throw new Exception("array index out of bounds");
        auto value = items[i];
        items = items[0 .. i] ~ items[i + 1 .. $];
        ++z8_version; return value;
    }
    static if (isPrimitive) bool z8_remove(T value) { return z8_remove(z8_optional_wrap(value)); }
    bool z8_remove(Element value) {
        static if (sharePrimary) if (primaryData !is null) return primaryData.z8_remove(z8_primary_box(value));
        static if (isPrimitive || is(T == class) || is(T == wstring)) {
            foreach (i, item; items) {
                static if (isPrimitive) bool equal = item.present == value.present && (!value.present || value.value == item.value);
                else static if (is(T == wstring)) bool equal = z8_string_equals(item, value);
                // ArrayList.remove invokes the searched value's equals, even for the same object.
                else bool equal = value is null ? item is null : value.opEquals(item);
                if (equal) {
                    z8_removeAt(cast(long)i); return true;
                }
            }
            return false;
        } else {
            auto index = z8_indexOf(value);
            if (index < 0) return false;
            z8_removeAt(index); return true;
        }
    }
    static if (isPrimitive) Z8Array!T z8_removeAll(T value) { return z8_removeAll(z8_optional_wrap(value)); }
    Z8Array!T z8_removeAll(Element value) { z8_remove(value); return this; }
    Z8Array!T z8_removeAll(Z8Array!T values) {
        z8_nonnull(values);
        static if (sharePrimary) if (primaryData !is null || values.primaryData !is null) {
            z8_primary_array_view(this).z8_removeAll(z8_primary_array_view(values)); return this;
        }
        auto originalLength = items.length;
        size_t read, write;
        try {
            for (; read < originalLength; ++read)
                if (!values.z8_contains(items[read])) items[write++] = items[read];
        } catch (Throwable error) {
            for (; read < originalLength; ++read) items[write++] = items[read];
            if (write != originalLength) { items.length = write; ++z8_version; }
            throw error;
        }
        if (write != originalLength) { items.length = write; ++z8_version; }
        return this;
    }
    Z8Array!T z8_reverse() {
        static if (sharePrimary) if (primaryData !is null) { primaryData.z8_reverse(); return this; }
        for (size_t i = 0; i < items.length / 2; ++i) {
            auto j = items.length - i - 1;
            auto value = items[i]; items[i] = items[j]; items[j] = value;
        }
        return this;
    }
    Z8Array!T z8_sort() {
        import std.algorithm.sorting : sort;
        import std.algorithm.mutation : SwapStrategy;
        static if (is(T == Z8Fts)) {
            if (z8_size() > 1) throw new Exception("object array elements are not comparable");
            ++z8_version;
            return this;
        } else static if (is(T == Z8Guid)) {
            if (z8_size() <= 1) { ++z8_version; return this; }
            Z8Guid[] values;
            foreach (i; 0 .. z8_size()) { auto value = z8_get(i); values ~= z8_nonnull(value); }
            sort!((a, b) => cast(long)a.majorBits == cast(long)b.majorBits ?
                  cast(long)a.minorBits < cast(long)b.minorBits : cast(long)a.majorBits < cast(long)b.majorBits, SwapStrategy.stable)(values);
            if (primaryData !is null) foreach (i, value; values) primaryData.z8_set(cast(long)i, z8_primary_box(value));
            else items = values;
            ++z8_version;
            return this;
        } else {
            static if (isPrimitive) if (primaryData !is null) {
                Element[] values;
                foreach (i; 0 .. z8_size()) values ~= z8_get(i);
                if (values.length > 1) foreach (value; values) if (!value.present) throw new Exception("null array element");
                sort!("a < b", SwapStrategy.stable)(values);
                foreach (i, value; values) primaryData.z8_set(cast(long)i, z8_primary_box(value));
                ++primaryData.z8_version;
                return this;
            }
            static if (is(T == wstring)) if (primaryData !is null) {
                wstring[] values;
                foreach (i; 0 .. z8_size()) values ~= z8_get(i);
                if (values.length > 1) foreach (value; values) if (value is null) throw new Exception("null array element");
                sort!("a < b", SwapStrategy.stable)(values);
                foreach (i, value; values) primaryData.z8_set(cast(long)i, z8_primary_box(value));
                ++primaryData.z8_version;
                return this;
            }
            static if (is(T == wstring)) if (items.length > 1) foreach (item; items) if (item is null) throw new Exception("null array element");
            static if (isPrimitive) if (items.length > 1) foreach (item; items) if (!item.present) throw new Exception("null array element");
            sort!("a < b", SwapStrategy.stable)(items);
            ++z8_version;
            return this;
        }
    }
    Z8Array!T z8_unique() {
        auto result = new Z8Array!T();
        static if (sharePrimary) if (primaryData !is null) { result.primaryData = primaryData.z8_unique(); return result; }
        foreach (item; items) {
            static if (isPrimitive || is(T == class) || is(T == wstring)) {
                bool found;
                foreach (existing; result.items) {
                    static if (isPrimitive) bool equal = item.present == existing.present && (!item.present || item.value == existing.value);
                    else static if (is(T == wstring)) bool equal = z8_string_equals(item, existing);
                    else static if (is(T == Z8Primary)) bool equal = z8_primary_collection_equals(item, existing);
                    else bool equal = item is existing || (item !is null && item == existing);
                    if (equal) { found = true; break; }
                }
                if (!found) result.z8_add(item);
            } else if (!result.z8_contains(item)) result.z8_add(item);
        }
        return result;
    }
    wstring z8_join(wstring separator = ""w) {
        static if (is(T == Z8Fts)) {
            wstring ftsJoined;
            foreach (i; 0 .. z8_size()) {
                if (i) ftsJoined ~= z8_string_required(separator);
                z8_nonnull(z8_get(i));
            }
            return ftsJoined;
        }
        static if (is(T == Z8Encoding) || is(T == Z8FtsQueryType)) {
            wstring enumJoined;
            foreach (i; 0 .. z8_size()) {
                if (i) enumJoined ~= z8_string_required(separator);
                auto item = z8_nonnull(z8_get(i));
                static if (is(T == Z8Encoding)) enumJoined ~= z8_encoding_name(item);
                else enumJoined ~= z8_fts_query_name(item);
            }
            return enumJoined;
        }
        static if (is(T == Z8Guid) || is(T == Z8Binary) || is(T == Z8Datespan) || is(T == Z8Date) ||
                   is(T == Z8Geometry) || is(T == Z8File) || is(T == Z8Primary)) {
            wstring primaryJoined;
            foreach (i; 0 .. z8_size()) {
                if (i) primaryJoined ~= z8_string_required(separator);
                auto item = z8_get(i);
                static if (is(T == Z8Primary)) primaryJoined ~= z8_primary_to_string(item);
                else primaryJoined ~= z8_primary_to_string(z8_primary_box(item));
            }
            return primaryJoined;
        }
        static if (isPrimitive) if (primaryData !is null) {
            wstring result;
            foreach (i; 0 .. z8_size()) {
                if (i) result ~= z8_string_required(separator);
                auto item = z8_get(i);
                static if (is(T == Z8Decimal)) result ~= item.z8_string();
                else static if (is(T == bool)) result ~= item ? "true"w : "false"w;
                else result ~= z8_int_toString(item);
            }
            return result;
        }
        static if (is(T == wstring)) if (primaryData !is null) {
            wstring result;
            foreach (i; 0 .. z8_size()) {
                if (i) result ~= z8_string_required(separator);
                result ~= z8_string_required(z8_get(i));
            }
            return result;
        }
        wstring result;
        foreach (i, item; items) {
            if (i) result ~= z8_string_required(separator);
            static if (is(T == wstring)) result ~= z8_string_required(item);
            else static if (is(T == Z8Decimal)) result ~= item.z8_string();
            else static if (is(T == bool)) result ~= item ? "true"w : "false"w;
            else static if (is(T == long)) result ~= z8_int_toString(item);
            else throw new Exception("join is not supported for object arrays");
        }
        return result;
    }
    ref Element z8_at(long index) {
        static if (sharePrimary) if (primaryData !is null) {
            static if (isPrimitive) throw new Exception("primitive array element requires checked access");
            else {
            auto value = primaryData.z8_get(index);
            if (value is null || !z8_primary_is!T(value)) throw new Exception("array element requires checked access");
            static if (is(T == wstring)) return value.textValue;
            else static if (is(T == Z8Guid)) return value.guidValue;
            else static if (is(T == Z8File)) return value.fileValue;
            else static if (is(T == Z8Binary)) return value.binaryValue;
            else static if (is(T == Z8Datespan)) return value.datespanValue;
            else static if (is(T == Z8Date)) return value.dateValue;
            else static if (is(T == Z8Geometry)) return value.geometryValue;
            }
        }
        auto i = cast(int)index;
        if (i < 0 || cast(size_t)i >= items.length) throw new Exception("array index out of bounds");
        return items[i];
    }
    Element z8_get(long index) {
        static if (sharePrimary) if (primaryData !is null) return z8_primary_cast!T(primaryData.z8_get(index));
        return z8_at(index);
    }
    static if (isPrimitive) void z8_set(long index, T value) { z8_set(index, z8_optional_wrap(value)); }
    void z8_set(long index, Element value) {
        static if (sharePrimary) if (primaryData !is null) { primaryData.z8_set(index, z8_primary_box(value)); return; }
        z8_at(index) = value;
    }
}
// The JVM shares the same erased RCollection when a scalar array widens to primary[].
Z8Array!Z8Primary z8_primary_array_view(T)(Z8Array!T values)
if (is(T == wstring) || is(T == Z8Guid) || is(T == Z8File) || is(T == Z8Binary) || is(T == Z8Datespan) ||
    is(T == Z8Date) || is(T == Z8Geometry) || is(T == long) || is(T == bool) || is(T == Z8Decimal)) {
    if (values is null) return null;
    if (values.primaryData is null) {
        auto data = new Z8Array!Z8Primary();
        foreach (value; values.items) data.z8_add(z8_primary_box(value));
        data.z8_version = values.arrayVersion;
        values.items = null;
        values.primaryData = data;
    }
    return values.primaryData;
}
wstring z8_string_array_get(Z8Array!wstring values, long index) { return z8_nonnull(values).z8_get(index); }
private void z8_string_array_check_index(Z8Array!wstring values, long index) {
    auto position = cast(int)index;
    if (position < 0 || position >= z8_nonnull(values).z8_size()) throw new Exception("array index out of bounds");
}

    z8_string_array_check_index(values, index);
    auto before = values.z8_get(index);
    values.z8_set(index, z8_nullable_string_binary!"+"(before, z8_string_or_empty(value)));
}

void z8_primitive_array_compound(string op, T)(Z8Array!T values, long index, lazy Z8Optional!T value) {
    z8_primitive_array_check_index(values, index);
    auto before = values.z8_get(index);
    auto after = value;
    static if (is(T == long)) values.z8_set(index, z8_optional_wrap(z8_optional_int_binary!op(before, after)));
    else static if (is(T == Z8Decimal)) values.z8_set(index, z8_optional_wrap(z8_optional_decimal_binary!op(before, after)));
    else static assert(0, "unsupported primitive array compound assignment");
}
