// Dashboard.writeLoginInfo for an already authenticated application session.
private void z8_dashboard_write_entry(Z8JsonWriterState writer, Z8Object entry) {
    writer.start('{');
    z8_monitor_property(writer, "request"w, entry.z8_class_id());
    z8_monitor_property(writer, "text"w, entry.z8_get_attribute("displayName"w));
    z8_monitor_property(writer, "description"w, entry.z8_get_attribute("description"w));
    z8_monitor_property(writer, "icon"w, entry.z8_get_attribute("icon"w));
    z8_monitor_property(writer, "url"w, entry.z8_get_attribute("url"w));
    z8_monitor_property(writer, "id"w, entry.z8_id());
    if (z8_native_is_instance(z8_executable_name, entry)) {
        z8_monitor_property(writer, "isJob"w, true);
        writer.start('[', "parameters"w);
        auto state = z8_action_state(entry);
        if (state.parameters !is null && state.parameters.view() !is null) {
            auto parameters = state.parameters.view().z8_erased_data();
            for (long i; i < parameters.z8_size(); ++i) z8_parameter_write_meta(cast(Z8Object)parameters.z8_get(i), writer);
        }
        writer.finish(']');
    }
    writer.finish('}');
}
private void z8_dashboard_write_desktop(Z8JsonWriterState writer, Z8Object desktop, Z8Privileges privileges, wstring title) {
    Z8Object[] visible;
    foreach (member; z8_object_members(desktop)) {
        if (!z8_native_is_instance("org.zenframework.z8.server.base.query.Query"w, member) &&
            !z8_native_is_instance(z8_executable_name, member)) continue;
        if (privileges !is null && privileges.getRequestAccess(z8_guid_create_name(member.z8_class_id())).execute) visible ~= member;
    }
    if (!visible.length) return;
    writer.start('{'); z8_monitor_property(writer, "text"w, title);
    z8_monitor_property(writer, "icon"w, desktop.z8_get_attribute("icon"w));
    writer.start('[', "items"w); foreach (item; visible) z8_dashboard_write_entry(writer, item); writer.finish(']');
    writer.finish('}');
}
private void z8_dashboard_login_info() {
    auto request = z8_application_require_request();
    auto session = z8_nonnull(request.session), user = z8_nonnull(session.user);
    auto params = z8_nonnull(request.parameters);
    if (request.writer.scopes.length != 2) throw new Exception("response object is not open");
    auto newPassword = params.z8_get("newPassword"w);
    if (newPassword !is null) { z8_dashboard_change_password(user, params.z8_get("password"w), newPassword); return; }
    auto email = params.z8_get("email"w);
    if (email !is null) { z8_dashboard_change_email(user, email); return; }
    Z8Object[] entries;
    foreach (entry; user.entries) {
        try { entries ~= z8_object_new_instance(entry.classId); }
        catch (Exception) { } // Loader skips unavailable entry classes in Java.
    }
    auto writer = request.writer;
    z8_monitor_property(writer, "session"w, session.id);
    z8_monitor_property(writer, "maxUploadSize"w, cast(long)z8_application_configuration().uploadMaxMiB);
    writer.start('{', "user"w);
    z8_monitor_property(writer, "id"w, user.id); z8_monitor_property(writer, "login"w, user.login);
    z8_monitor_property(writer, "administrator"w, user.id !is null && user.id == z8_guid_create(2));
    z8_monitor_property(writer, "firstName"w, user.firstName); z8_monitor_property(writer, "middleName"w, user.middleName);
    z8_monitor_property(writer, "lastName"w, user.lastName); z8_monitor_property(writer, "description"w, user.description);
    z8_monitor_property(writer, "email"w, user.email); z8_monitor_property(writer, "phone"w, user.phone);
    z8_monitor_property(writer, "changePassword"w, user.changePassword); z8_monitor_property(writer, "settings"w, user.settings);
    writer.start('[', "entries"w); foreach (entry; entries) z8_dashboard_write_entry(writer, entry); writer.finish(']');
    writer.start('[', "roles"w);
    foreach (index, role; user.roles) {
        writer.start('{'); z8_monitor_property(writer, "id"w, role);
        z8_monitor_property(writer, "name"w, index < user.roleNames.length ? user.roleNames[index] : null); writer.finish('}');
    }
    writer.finish(']');
    if (params.z8_get("experimental"w) !is null) {
        writer.start('[', "data"w);
        foreach (entry; entries) {
            if (z8_native_is_instance("org.zenframework.z8.server.base.form.Desktop"w, entry))
                z8_dashboard_write_desktop(writer, entry, user.privileges, entry.z8_get_attribute("displayName"w));
            else z8_dashboard_write_entry(writer, entry);
        }
        writer.finish(']');
    }
    writer.start('{', "parameters"w);
    auto values = z8_nonnull(user.parameters); auto keys = values.z8_keys();
    for (long i; i < keys.z8_size(); ++i) {
        auto key = z8_string_required(keys.z8_get(i)); writer.property(key); z8_json_writer_primary(writer, values.z8_get(key));
    }
    writer.finish('}'); writer.finish('}');
}
private Z8Object z8_dashboard_users_table() {
    auto registry = z8_runtime_get_registry();
    auto users = registry is null ? null : registry.newTableByName("SystemUsers"w);
    if (users is null) users = z8_native_new(z8_users_name);
    if (!z8_native_is_instance(z8_users_name, users)) throw new Exception("SystemUsers is not a Users table");
    return users;
}
private void z8_dashboard_change_password(Z8UserData user, wstring oldPassword, wstring newPassword) {
    auto connection = z8_connection_current_backend();
    auto verifier = cast(IZ8UserPasswordVerifier)connection;
    if (verifier is null) throw new Exception("database does not support password verification");
    oldPassword = oldPassword is null ? ""w : oldPassword;
    if (!z8_application_configuration().clientHashesPasswords) {
        oldPassword = z8_crypto_md5(oldPassword); newPassword = z8_crypto_md5(newPassword);
    }
    {
        connection.beginTransaction();
        scope(failure) connection.rollback();
        if (!verifier.lockMatchingPassword(z8_nonnull(user.id), oldPassword)) throw new Exception("access denied");
        auto users = z8_dashboard_users_table(); auto fields = z8_users_state(users).fields;
        z8_field_set(fields[2], z8_primary_box(newPassword)); z8_field_set(fields[12], z8_primary_box(false));
        if (!z8_query_update_id_run(users, user.id, true)) throw new Exception("user was not updated");
        connection.commit();
    }
    user.changePassword = false;
}
private void z8_dashboard_change_email(Z8UserData user, wstring email) {
    if (z8_str_compareNoCase(z8_string_or_empty(user.email), email) == 0) return;
    auto users = z8_dashboard_users_table();
    z8_field_set(z8_users_state(users).fields[7], z8_primary_box(email));
    if (!z8_query_update_id_run(users, z8_nonnull(user.id), true)) throw new Exception("user was not updated");
    user.email = email;
}
private void z8_dashboard_save_settings() {
    auto request = z8_application_require_request();
    auto user = z8_nonnull(z8_nonnull(request.session).user);
    if (request.writer.scopes.length != 2) throw new Exception("response object is not open");
    auto settings = z8_nonnull(request.parameters).z8_get("data"w);
    auto users = z8_dashboard_users_table();
    auto field = z8_nonnull(z8_users_state(users).fields[13]);
    z8_field_set(field, z8_primary_box(settings));
    z8_query_update_id_run(users, z8_nonnull(user.id), true);
    user.settings = settings;
}
