private import std.datetime : SysTime, DateTime, Clock, TimeZone, SimpleTimeZone, UTC, dur, AllowDayOverflow;
version (Windows) private import std.datetime : WindowsTimeZone;
else version (Posix) private import std.datetime : PosixTimeZone;

private enum wstring[] z8_month_short = ["янв"w,"фев"w,"мар"w,"апр"w,"мая"w,"июн"w,"июл"w,"авг"w,"сен"w,"окт"w,"ноя"w,"дек"w];
private enum wstring[] z8_month_long = ["января"w,"февраля"w,"марта"w,"апреля"w,"мая"w,"июня"w,"июля"w,"августа"w,"сентября"w,"октября"w,"ноября"w,"декабря"w];
private enum wstring[] z8_month_standalone_short = ["Янв."w,"Февр."w,"Март"w,"Апр."w,"Май"w,"Июнь"w,"Июль"w,"Авг."w,"Сент."w,"Окт."w,"Нояб."w,"Дек."w];
private enum wstring[] z8_month_standalone_long = ["Январь"w,"Февраль"w,"Март"w,"Апрель"w,"Май"w,"Июнь"w,"Июль"w,"Август"w,"Сентябрь"w,"Октябрь"w,"Ноябрь"w,"Декабрь"w];
private enum wstring[] z8_weekday_short = ["Вс"w,"Пн"w,"Вт"w,"Ср"w,"Чт"w,"Пт"w,"Сб"w];
private enum wstring[] z8_weekday_long = ["воскресенье"w,"понедельник"w,"вторник"w,"среда"w,"четверг"w,"пятница"w,"суббота"w];
private int z8_date_parse_month(wstring input, ref size_t index) {
    size_t longest;
    int month;
    foreach (forms; [z8_month_short, z8_month_long, z8_month_standalone_short, z8_month_standalone_long]) {
        foreach (i, candidate; forms) if (candidate.length > longest && index + candidate.length <= input.length &&
            z8_str_toLowerCase(input[index .. index + candidate.length]) == z8_str_toLowerCase(candidate)) {
            longest = candidate.length; month = cast(int)i + 1;
        }
    }
    if (!longest) throw new Exception("invalid date month");
    index += longest;
    return month;
}
private int z8_date_parse_weekday(wstring input, ref size_t index) {
    size_t longest;
    int weekday;
    foreach (forms; [z8_weekday_short, z8_weekday_long]) foreach (i, candidate; forms)
        if (candidate.length > longest && index + candidate.length <= input.length &&
            z8_str_toLowerCase(input[index .. index + candidate.length]) == z8_str_toLowerCase(candidate)) {
            longest = candidate.length; weekday = (cast(int)i + 6) % 7 + 1;
        }
    if (!longest) throw new Exception("invalid date weekday");
    index += longest;
    return weekday;
}

private immutable(TimeZone) z8_date_timezone() {
    version (Windows) return z8_moscow_zone;
    else version (Posix) return PosixTimeZone.getTimeZone("Europe/Moscow");
    else static assert(0, "Moscow timezone is unavailable on this platform");
}
class Z8Date {
    SysTime value;
    bool explicitOffset;
    bool fullFirstWeek;
    this(SysTime value) { this.value = value; }
    override bool opEquals(Object other) {
        auto right = cast(Z8Date)other;
        return right !is null && value == right.value;
    }
}
Z8Date z8_date_now() { return new Z8Date(Clock.currTime(z8_date_timezone())); }
Z8Date z8_date_from_ticks(long ticks) {
    return new Z8Date(SysTime.fromUnixTime(0, z8_date_timezone()) + dur!"msecs"(ticks));
}
__gshared Z8Date z8_date_min, z8_date_max;
shared static this() {
    z8_date_min = z8_date_from_ticks(-62135769600000L);
    z8_date_max = z8_date_from_ticks(95617584000000L);
}
void z8_date_assign(Z8Date value, long ticks) {
    auto target = z8_nonnull(value);
    auto zone = target.explicitOffset ? z8_date_timezone() : target.value.timezone;
    target.value = SysTime.fromUnixTime(0, zone) + dur!"msecs"(ticks);
    target.explicitOffset = false;
}
Z8Date z8_date_literal(string text) {
    import std.string : strip, split;
    auto parts = text.strip().split();
    if (parts.length < 1 || parts.length > 2) throw new Exception("invalid date literal");
    auto dateParts = parts[0].split("/");
    if (dateParts.length != 3) throw new Exception("invalid date literal");
    int hours, minutes, seconds;
    if (parts.length == 2) {
        auto timeParts = parts[1].split(":");
        if (timeParts.length != 3) throw new Exception("invalid date literal");
        hours = timeParts[0].to!int; minutes = timeParts[1].to!int; seconds = timeParts[2].to!int;
    }
    return new Z8Date(SysTime(DateTime(dateParts[2].to!int, dateParts[1].to!int, dateParts[0].to!int,
        hours, minutes, seconds), z8_date_timezone()));
}
long z8_date_ticks(Z8Date x) { return (z8_nonnull(x).value - SysTime.fromUnixTime(0, UTC())).total!"msecs"; }
long z8_date_year(Z8Date x) { return z8_nonnull(x).value.year; }
long z8_date_quarter(Z8Date x) { return (z8_date_month(x) - 1) / 3 + 1; }
long z8_date_month(Z8Date x) { return cast(int)z8_nonnull(x).value.month; }
long z8_date_day(Z8Date x) { return z8_nonnull(x).value.day; }
long z8_date_dayOfWeek(Z8Date x) { return cast(int)z8_nonnull(x).value.dayOfWeek + 1; }
long z8_date_dayOfYear(Z8Date x) { return z8_nonnull(x).value.dayOfYear; }
long z8_date_weekOfYear(Z8Date x) {
    auto target = z8_nonnull(x);
    target.fullFirstWeek = true;
    auto time = target.value;
    int firstMonday(int year) {
        auto jan1 = SysTime(DateTime(year, 1, 1), time.timezone);
        return (1 - cast(int)jan1.dayOfWeek + 7) % 7;
    }
    auto index = time.dayOfYear - 1;
    auto first = firstMonday(time.year);
    if (index >= first) return 1 + (index - first) / 7;
    auto lastYear = SysTime(DateTime(time.year - 1, 12, 31), time.timezone);
    return 1 + (lastYear.dayOfYear - 1 - firstMonday(time.year - 1)) / 7;
}
long z8_date_hashCode(Z8Date x) {
    auto target = z8_nonnull(x);
    auto ticks = cast(ulong)z8_date_ticks(target);
    // OpenJDK ZoneInfo hash for Europe/Moscow and the Russian locale's calendar settings.
    uint settings = 1u | (2u << 1) | ((target.fullFirstWeek ? 7u : 1u) << 4) | (cast(uint)-1071621955 << 7);
    auto hash = cast(uint)ticks ^ cast(uint)(ticks >> 32) ^ settings ^ 577736u;
    return cast(long)cast(int)hash;
}
long z8_date_daysInMonth(Z8Date x) { return z8_nonnull(x).value.daysInMonth; }
long z8_date_hours(Z8Date x) { return z8_nonnull(x).value.hour; }

long z8_date_seconds(Z8Date x) { return z8_nonnull(x).value.second; }
long z8_date_milliseconds(Z8Date x) { return z8_nonnull(x).value.fracSecs.total!"msecs"; }
long z8_date_zoneOffset(Z8Date x) {
    auto time = z8_nonnull(x).value;
    return time.utcOffset.total!"msecs" - (time.dstInEffect ? 3_600_000L : 0);
}

void z8_date_setZoneOffset(Z8Date x, long offset) {
    auto target = z8_nonnull(x);
    auto t = target.value;
    auto zone = new immutable(SimpleTimeZone)(dur!"msecs"(cast(int)offset));
    target.value = SysTime(DateTime(t.year, cast(int)t.month, t.day, t.hour, t.minute, t.second), zone) + t.fracSecs;
    target.explicitOffset = true;
}

void z8_date_set(Z8Date x, long year, long month, long day, long hour, long minute, long second, long millisecond = 0) {
    auto target = z8_nonnull(x);
    auto t = SysTime(DateTime(cast(int)year, 1, 1), target.value.timezone);
    t.add!"months"(cast(int)month - 1, AllowDayOverflow.no);
    t += dur!"days"(cast(int)day - 1) + dur!"hours"(cast(int)hour) + dur!"minutes"(cast(int)minute) +
        dur!"seconds"(cast(int)second) + dur!"msecs"(cast(int)millisecond);
    target.value = t;
}
void z8_date_setDate(Z8Date x, long year, long month, long day) {
    z8_date_set(x, year, month, day, z8_date_hours(x), z8_date_minutes(x), z8_date_seconds(x));
}
void z8_date_setTime(Z8Date x, long hour, long minute, long second, long millisecond = 0) {
    z8_date_set(x, z8_date_year(x), z8_date_month(x), z8_date_day(x), hour, minute, second, millisecond);
}

private Z8Date z8_date_parse_impl(wstring input, bool iso) {
    if (input is null) return null;
    auto text = input;
    if (!text.length) {
        if (iso) return z8_date_min;
        auto result = new Z8Date(z8_date_min.value);
        result.explicitOffset = z8_date_min.explicitOffset;
        result.fullFirstWeek = z8_date_min.fullFirstWeek;
        return result;
    }
    if (!iso) {
        bool digits = true;
        foreach (c; text) if (c < '0' || c > '9') { digits = false; break; }
        if (digits) return z8_date_from_ticks(text.to!long);
    }
    if (text.length < 16 || text[4] != '-' || text[7] != '-' || (text[10] != 'T' && text[10] != ' ') || text[13] != ':')
        throw new Exception("invalid date");
    auto year = z8_java_normalize_digits(text[0..4]).to!int, month = z8_java_normalize_digits(text[5..7]).to!int, day = z8_java_normalize_digits(text[8..10]).to!int;
    auto hour = z8_java_normalize_digits(text[11..13]).to!int, minute = z8_java_normalize_digits(text[14..16]).to!int;
    int second, millisecond;
    size_t index = 16;
    if (index < text.length && text[index] == ':') {
        if (index + 3 > text.length) throw new Exception("invalid date");
        second = z8_java_normalize_digits(text[index+1..index+3]).to!int;
        index += 3;
    }
    if (index < text.length && text[index] == '.') {
        auto begin = ++index;
        while (index < text.length && z8_java_digit(text[index]) >= 0 && z8_java_digit(text[index]) < 10) ++index;
        if (begin == index) throw new Exception("invalid date");
        millisecond = z8_java_normalize_digits(text[begin..index]).to!int;
    }
    bool convertToDefault;
    bool hasOffset;
    int offset;
    if (index < text.length) {
        if (text[index] == 'Z' && index + 1 == text.length) {
            ++index;
        } else {
            if ((text[index] != '+' && text[index] != '-') ||
                (index + 6 != text.length && index + 9 != text.length) || text[index+3] != ':' ||
                (index + 9 == text.length && text[index+6] != ':'))
                throw new Exception("invalid date offset");
            auto sign = text[index] == '-' ? -1 : 1;
            auto offsetHours = z8_java_normalize_digits(text[index+1..index+3]).to!int;
            auto offsetMinutes = z8_java_normalize_digits(text[index+4..index+6]).to!int;
            auto offsetSeconds = index + 9 == text.length ? z8_java_normalize_digits(text[index+7..index+9]).to!int : 0;
            // date.parse uses SimpleDateFormat for this exact ISO spelling;
            // parseIsoString follows the original's separate offset calculation.
            convertToDefault = !iso && index == 19 && text.length == 25;
            if (convertToDefault) foreach (part; [text[0..4], text[5..7], text[8..10], text[11..13], text[14..16], text[17..19], text[20..22], text[23..25]])
                foreach (c; part) if (c < '0' || c > '9') convertToDefault = false;
            offset = sign * offsetHours * 3600 + (convertToDefault ? sign : 1) * offsetMinutes * 60 + offsetSeconds;
            hasOffset = true;
            index = text.length;
        }
    }
    if (index != text.length) throw new Exception("invalid date");
    immutable(TimeZone) zone = hasOffset ? new immutable(SimpleTimeZone)(dur!"seconds"(offset)) : z8_date_timezone();
    auto result = new Z8Date(SysTime(DateTime(year, 1, 1), zone));
    result.explicitOffset = hasOffset && !convertToDefault;
    z8_date_set(result, year, month, day, hour, minute, second, millisecond);
    if (convertToDefault) result.value.timezone = z8_date_timezone();
    return result;
}
Z8Date z8_date_parse(wstring value) { return z8_date_parse_impl(value, false); }
Z8Date z8_date_parseIsoString(wstring value) { return z8_date_parse_impl(value, true); }
Z8Date z8_date_parse(wstring input, wstring pattern) {
    if (input is null) return null;
    pattern = z8_string_required(pattern);
    if (!input.length) return z8_date_parse(input);
    int year = 1970, month = 1, day = 1, hour, minute, second, millisecond, dayOfYear, weekday = 1, week = 1, weekKind;
    bool useDayOfYear, twoDigitYear, twelveHour, afternoon, explicitDay, hasWeekday, hasWeekYear;
    int offset;
    bool hasOffset, quoted;
    size_t pi, ti;
    int number(size_t width, bool fixed) {
        auto begin = ti;
        if (ti < input.length && (input[ti] == '-' || input[ti] == '+')) ++ti;
        auto digits = ti;
        while (ti < input.length && z8_java_digit(input[ti]) >= 0 && z8_java_digit(input[ti]) < 10 && (!fixed || ti - digits < width)) ++ti;
        if (ti == digits || (fixed && ti - digits != width)) throw new Exception("invalid date");
        return z8_java_normalize_digits(input[begin .. ti]).to!int;
    }
    void literal(wchar c) {
        if (ti == input.length || input[ti++] != c) throw new Exception("invalid date");
    }
    while (pi < pattern.length) {
        auto c = pattern[pi++];
        if (c == '\'') {
            if (pi < pattern.length && pattern[pi] == '\'') { ++pi; literal('\''); }
            else quoted = !quoted;
            continue;
        }
        if (quoted || !((c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z'))) { literal(c); continue; }
        size_t width = 1;
        while (pi < pattern.length && pattern[pi] == c) { ++pi; ++width; }
        bool adjacent = pi < pattern.length && ((pattern[pi] >= 'A' && pattern[pi] <= 'Z') || (pattern[pi] >= 'a' && pattern[pi] <= 'z'));
        switch (c) {
            case 'y': case 'Y':
                auto begin = ti;
                year = number(width, adjacent);
                hasWeekYear = c == 'Y';
                twoDigitYear = width <= 2 && ti - begin == 2 && z8_java_digit(input[begin]) >= 0 && z8_java_digit(input[begin]) < 10;
                break;
            case 'M': case 'L': month = width > 2 ? z8_date_parse_month(input, ti) : number(width, adjacent); break;
            case 'd': day = number(width, adjacent); useDayOfYear = false; explicitDay = true; break;
            case 'D': dayOfYear = number(width, adjacent); useDayOfYear = true; explicitDay = true; break;
            case 'E': weekday = z8_date_parse_weekday(input, ti); hasWeekday = true; break;
            case 'u': weekday = number(width, adjacent); hasWeekday = true; break;
            case 'w': week = number(width, adjacent); weekKind = 1; break;
            case 'W': week = number(width, adjacent); weekKind = 2; break;
            case 'F': week = number(width, adjacent); weekKind = 3; break;
            case 'H': hour = number(width, adjacent); twelveHour = false; break;
            case 'k': hour = number(width, adjacent); if (hour == 24) hour = 0; twelveHour = false; break;
            case 'K': hour = number(width, adjacent); twelveHour = true; break;
            case 'h': hour = number(width, adjacent); if (hour == 12) hour = 0; twelveHour = true; break;
            case 'a':
                if (ti + 2 > input.length) throw new Exception("invalid date AM/PM");
                auto ampm = z8_str_toLowerCase(input[ti .. ti + 2]);
                if (ampm != "am"w && ampm != "pm"w) throw new Exception("invalid date AM/PM");
                afternoon = ampm == "pm"w; ti += 2;
                break;
            case 'G':
                if (ti + 4 > input.length || z8_str_toLowerCase(input[ti .. ti + 4]) != "н.э."w)
                    throw new Exception("unsupported date era");
                ti += 4;
                break;
            case 'z':
                hasOffset = true;
                auto remaining = z8_str_toLowerCase(input[ti .. $]);
                import std.string : startsWith;
                if (remaining.startsWith("moscow standard time"w)) { offset = 180; ti += 20; }
                else if (remaining.startsWith("moscow daylight time"w)) { offset = 240; ti += 20; }
                else if (remaining.startsWith("msk"w)) { offset = 180; ti += 3; }
                else if (remaining.startsWith("msd"w)) { offset = 240; ti += 3; }
                else {
                    bool gmt = remaining.startsWith("gmt"w);
                    if (gmt || remaining.startsWith("utc"w)) ti += 3;
                    else if (ti >= input.length || (input[ti] != '+' && input[ti] != '-')) throw new Exception("unsupported date timezone");
                    offset = 0;
                    if (ti < input.length && (input[ti] == '+' || input[ti] == '-')) {
                        auto sign = input[ti++] == '-' ? -1 : 1;
                        auto zoneHours = number(2, true);
                        if (gmt && ti < input.length && input[ti] == ':') ++ti;
                        auto zoneMinutes = number(2, true);
                        if (zoneHours > 23 || zoneMinutes > 59) throw new Exception("invalid date offset");
                        offset = sign * (zoneHours * 60 + zoneMinutes);
                    }
                }
                break;
            case 'm': minute = number(width, adjacent); break;
            case 's': second = number(width, adjacent); break;
            case 'S': millisecond = number(width, adjacent); break;
            case 'X': case 'Z':
                if (c == 'X' && width > 3) throw new Exception("invalid date format");
                hasOffset = true;
                if (c == 'X' && ti < input.length && input[ti] == 'Z') { ++ti; offset = 0; break; }
                if (ti >= input.length || (input[ti] != '+' && input[ti] != '-')) throw new Exception("invalid date offset");
                auto sign = input[ti++] == '-' ? -1 : 1;
                auto hours = number(2, true);
                int minutes;
                if (c == 'Z' || width >= 2) {
                    if (c == 'X' && width == 3) literal(':');
                    minutes = number(2, true);
                }
                if (hours > 23 || minutes > 59) throw new Exception("invalid date offset");
                offset = sign * (hours * 60 + minutes);
                break;
            default: throw new Exception("unsupported date format");
        }
    }
    if (quoted) throw new Exception("unterminated date format quote");
    if (explicitDay && (weekKind || hasWeekYear)) throw new Exception("mixed week and day date fields are unsupported");
    if (twelveHour && afternoon) hour += 12;
    SysTime centuryStart;
    if (twoDigitYear) {
        centuryStart = Clock.currTime(z8_date_timezone());
        centuryStart.add!"years"(-80, AllowDayOverflow.no);
        auto startYear = centuryStart.year;
        year += (startYear / 100) * 100;
        if (year < startYear) year += 100;
    }
    immutable(TimeZone) zone = hasOffset ? new immutable(SimpleTimeZone)(dur!"minutes"(offset)) : z8_date_timezone();
    auto result = new Z8Date(SysTime(DateTime(year, 1, 1), zone));
    void setFromFields(int targetYear) {
        auto kind = hasWeekYear ? 1 : weekKind ? weekKind : hasWeekday && !explicitDay ? 3 : 0;
        if (!kind) {
            z8_date_set(result, targetYear, useDayOfYear ? 1 : month, useDayOfYear ? dayOfYear : day, hour, minute, second, millisecond);
            return;
        }
        auto targetMonth = kind == 1 ? 1 : month;
        z8_date_set(result, targetYear, targetMonth, 1, 0, 0, 0);
        auto first = result.value;
        auto firstWeekday = (cast(int)first.dayOfWeek + 6) % 7;
        int resolvedDay;
        if (kind == 1 || kind == 2) resolvedDay = 1 - firstWeekday + (week - 1) * 7 + weekday - 1;
        else if (week >= 0) resolvedDay = 1 + (weekday - 1 - firstWeekday + 7) % 7 + (week - 1) * 7;
        else {
            auto last = first.daysInMonth;
            auto lastWeekday = (firstWeekday + last - 1) % 7;
            resolvedDay = last - (lastWeekday - weekday + 1 + 7) % 7 + (week + 1) * 7;
        }
        z8_date_set(result, targetYear, targetMonth, resolvedDay, hour, minute, second, millisecond);
    }
    setFromFields(year);
    if (twoDigitYear && year == centuryStart.year && result.value < centuryStart) {
        year += 100;
        setFromFields(year);
    }
    result.value.timezone = z8_date_timezone();
    return result;
}

private Z8Date z8_date_added(Z8Date original, SysTime time, long amount, bool calendarField = false) {
    if (original.explicitOffset && amount != 0) {
        if (calendarField)
            time = SysTime(DateTime(time.year, cast(int)time.month, time.day, time.hour, time.minute, time.second), z8_date_timezone()) + time.fracSecs;
        else time.timezone = z8_date_timezone();
    }
    auto result = new Z8Date(time);
    result.explicitOffset = original.explicitOffset && amount == 0;
    result.fullFirstWeek = original.fullFirstWeek;
    return result;
}
Z8Date z8_date_addYear(Z8Date x, long n) { auto t = z8_nonnull(x).value; t.add!"years"(cast(int)n, AllowDayOverflow.no); return z8_date_added(x, t, n, true); }
Z8Date z8_date_addQuarter(Z8Date x, long n) { return z8_date_addMonth(x, cast(int)(cast(int)n * 3)); }
Z8Date z8_date_addMonth(Z8Date x, long n) { auto t = z8_nonnull(x).value; t.add!"months"(cast(int)n, AllowDayOverflow.no); return z8_date_added(x, t, n, true); }
Z8Date z8_date_addDay(Z8Date x, long n) {
    auto time = z8_nonnull(x).value;
    auto days = cast(int)n;
    auto advanced = time + dur!"days"(days);
    auto adjustment = time.utcOffset - advanced.utcOffset;
    if (adjustment != dur!"seconds"(0)) {
        auto corrected = advanced + adjustment;
        auto expected = DateTime(time.year, cast(int)time.month, time.day, time.hour, time.minute, time.second) + dur!"days"(days);
        if (corrected.year == expected.year && corrected.month == expected.month && corrected.day == expected.day)
            advanced = corrected;
    }
    return z8_date_added(x, advanced, n, true);
}
Z8Date z8_date_addHour(Z8Date x, long n) { return z8_date_added(x, z8_nonnull(x).value + dur!"hours"(cast(int)n), n); }
Z8Date z8_date_addMinute(Z8Date x, long n) { return z8_date_added(x, z8_nonnull(x).value + dur!"minutes"(cast(int)n), n); }
Z8Date z8_date_addSecond(Z8Date x, long n) { return z8_date_added(x, z8_nonnull(x).value + dur!"seconds"(cast(int)n), n); }
Z8Date z8_date_addMillisecond(Z8Date x, long n) { return z8_date_added(x, z8_nonnull(x).value + dur!"msecs"(cast(int)n), n); }

private Z8Date z8_date_truncate(Z8Date x, int unit) {
    auto t = z8_nonnull(x).value;
    int month = unit <= 0 ? 1 : unit == 1 ? ((cast(int)t.month - 1) / 3 * 3 + 1) : cast(int)t.month;
    int day = unit <= 2 ? 1 : t.day;
    int hour = unit <= 3 ? 0 : t.hour;
    int minute = unit <= 4 ? 0 : t.minute;
    return new Z8Date(SysTime(DateTime(t.year, month, day, hour, minute, 0), z8_date_timezone()));
}

Z8Date z8_date_truncMinute(Z8Date x) { return z8_date_truncate(x, 5); }

Z8Date z8_date_tomorrow() { return z8_date_addDay(z8_date_today(), 1); }
Z8Date z8_date_yesterday() { return z8_date_addDay(z8_date_today(), -1); }
bool z8_date_isEqualDate(Z8Date a, Z8Date b) {
    return z8_date_year(a) == z8_date_year(b) && z8_date_month(a) == z8_date_month(b) && z8_date_day(a) == z8_date_day(b);
}
private wstring z8_date_format(Z8Date x, bool milliseconds) {
    auto t = z8_nonnull(x).value;
    if (!milliseconds) t.timezone = z8_date_timezone();
    auto offset = milliseconds ? z8_date_zoneOffset(x) / 60_000 : t.utcOffset.total!"minutes";
    auto absoluteOffset = offset < 0 ? -offset : offset;
    auto result = format("%04d-%02d-%02dT%02d:%02d:%02d", t.year, cast(int)t.month, t.day, t.hour, t.minute, t.second);
    auto fraction = t.fracSecs.total!"msecs";
    if (milliseconds && fraction != 0) result ~= format(".%03d", fraction);
    result ~= format("%s%02d:%02d", offset < 0 ? "-" : "+", absoluteOffset / 60, absoluteOffset % 60);
    return result.to!wstring;
}
wstring z8_date_toString(Z8Date x) { return z8_date_format(x, false); }
wstring z8_date_toIsoString(Z8Date x) { return z8_date_format(x, true); }
private int z8_date_format_week(SysTime t) {
    auto jan = SysTime(DateTime(t.year, 1, 1), t.timezone);
    auto weekday = (cast(int)jan.dayOfWeek + 6) % 7;
    auto next = SysTime(DateTime(t.year + 1, 1, 1), t.timezone);
    auto nextWeekday = (cast(int)next.dayOfWeek + 6) % 7;
    if (nextWeekday && t.dayOfYear >= 1 + (next - jan).total!"days" - nextWeekday) return 1;
    return 1 + (t.dayOfYear - 1 + weekday) / 7;
}
wstring z8_date_toString(Z8Date value, wstring pattern) {
    auto t = z8_nonnull(value).value;
    t.timezone = z8_date_timezone();
    pattern = z8_string_required(pattern);
    wstring result;
    bool quoted;
    size_t i;
    while (i < pattern.length) {
        auto c = pattern[i++];
        if (c == '\'') {
            if (i < pattern.length && pattern[i] == '\'') { result ~= '\''; ++i; }
            else quoted = !quoted;
            continue;
        }
        if (quoted || !((c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z'))) { result ~= c; continue; }
        size_t count = 1;
        while (i < pattern.length && pattern[i] == c) { ++count; ++i; }
        long number;
        bool numeric = true;
        switch (c) {
            case 'y': number = count == 2 ? t.year % 100 : t.year; break;
            case 'Y':
                number = t.year + (cast(int)t.month == 12 && z8_date_format_week(t) == 1 ? 1 : 0);
                if (count == 2) number %= 100;
                break;
            case 'M': case 'L':
                if (count <= 2) number = cast(int)t.month;
                else {
                    numeric = false;
                    auto index = cast(size_t)t.month - 1;
                    bool standalone = c == 'L' || pattern.length == count;
                    result ~= standalone ? (count == 3 ? z8_month_standalone_short[index] : z8_month_standalone_long[index]) :
                        (count == 3 ? z8_month_short[index] : z8_month_long[index]);
                }
                break;
            case 'd': number = t.day; break;
            case 'D': number = t.dayOfYear; break;
            case 'u': number = (cast(int)t.dayOfWeek + 6) % 7 + 1; break;
            case 'w': number = z8_date_format_week(t); break;
            case 'W':
                auto first = SysTime(DateTime(t.year, cast(int)t.month, 1), t.timezone);
                number = 1 + (t.day - 1 + (cast(int)first.dayOfWeek + 6) % 7) / 7;
                break;
            case 'F': number = 1 + (t.day - 1) / 7; break;
            case 'H': number = t.hour; break;
            case 'k': number = t.hour == 0 ? 24 : t.hour; break;
            case 'K': number = t.hour % 12; break;
            case 'h': number = t.hour % 12 == 0 ? 12 : t.hour % 12; break;
            case 'E': numeric = false; result ~= count < 4 ? z8_weekday_short[cast(int)t.dayOfWeek] : z8_weekday_long[cast(int)t.dayOfWeek]; break;
            case 'a': numeric = false; result ~= t.hour < 12 ? "AM"w : "PM"w; break;
            case 'G': numeric = false; result ~= "н.э."w; break;
            case 'z':
                numeric = false;
                result ~= t.dstInEffect ? (count < 4 ? "MSD"w : "Moscow Daylight Time"w) :
                    (count < 4 ? "MSK"w : "Moscow Standard Time"w);
                break;
            case 'm': number = t.minute; break;
            case 's': number = t.second; break;
            case 'S': number = t.fracSecs.total!"msecs"; break;
            case 'Z': case 'X':
                numeric = false;
                if (c == 'X' && count > 3) throw new Exception("invalid date format");
                auto offset = t.utcOffset.total!"minutes";
                if (c == 'X' && offset == 0) { result ~= 'Z'; break; }
                auto absolute = offset < 0 ? -offset : offset;
                auto zone = format("%s%02d", offset < 0 ? "-" : "+", absolute / 60);
                if (c == 'Z' || count >= 2) zone ~= (c == 'X' && count == 3 ? ":" : "") ~ format("%02d", absolute % 60);
                result ~= zone.to!wstring;
                break;
            default: throw new Exception("unsupported date format");
        }
        if (numeric) result ~= format("%0*d", cast(int)count, number).to!wstring;
    }
    if (quoted) throw new Exception("unterminated date format quote");
    return result.length ? result : ""w;
}

auto z8_date_binary(string op)(Z8Date left, Z8Date right) {
    auto a = z8_date_ticks(left);
    static if (op == "-") return new Z8Datespan(a - z8_date_ticks(right));
    else {
        if (right is null) return false;
        auto b = z8_date_ticks(right);
        static if (op == "==") return a == b;
        else static if (op == "!=") return a != b;
        else static if (op == "<") return a < b;
        else static if (op == ">") return a > b;
        else static if (op == "<=") return a <= b;
        else static if (op == ">=") return a >= b;
        else static assert(0, "unsupported date operation");
    }
}
Z8Date z8_date_binary(string op)(Z8Date left, Z8Datespan right) if (op == "+" || op == "-") {
    auto ticks = z8_date_ticks(left), span = z8_datespan_milliseconds(right);
    static if (op == "+") return z8_date_from_ticks(ticks + span);
    else return z8_date_from_ticks(ticks - span);
}
Z8Date z8_date_binary(string op)(Z8Datespan left, Z8Date right) if (op == "+") {
    return z8_date_from_ticks(z8_datespan_milliseconds(left) + z8_date_ticks(right));
}
