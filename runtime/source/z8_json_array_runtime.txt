// Native adapter for org.zenframework.z8.server.base.json.parser.JsonArray.
private enum z8_json_array_name = "org.zenframework.z8.server.base.json.parser.JsonArray"w;
private class Z8JsonArrayState {
    JSONValue value;
    Z8Decimal[size_t] decimals;
    Object[size_t] children;
    this(JSONValue value) { this.value = value; }
}
private JSONValue z8_json_empty_array() { return JSONValue(cast(JSONValue[])null); }
private JSONValue z8_json_array_read(wstring source) {
    auto text = z8_string_required(source).to!string;
    if (!text.length) return z8_json_empty_array();
    if (text[0] != '[') text = "[" ~ text ~ "]";
    auto result = z8_json_parse(text);
    if (result.type != JSONType.array) throw new Exception("JSON array expected");
    return result;
}
private Z8JsonArrayState z8_json_array_state(Z8Object self) {
    self = z8_nonnull(self);
    if (!z8_native_is_instance(z8_json_array_name, self)) throw new Exception("invalid native JSON array");
    auto state = cast(Z8JsonArrayState)self.z8_get_native_state(z8_json_array_name);
    if (state is null) {
        state = new Z8JsonArrayState(z8_json_empty_array());
        self.z8_set_native_state(z8_json_array_name, state);
    }
    return state;
}
private Z8Object z8_json_array_parse(wstring source) {
    if (source is null) return null;
    auto value = z8_json_array_read(source);
    auto result = z8_native_new(z8_json_array_name);
    z8_nonnull(result).z8_set_native_state(z8_json_array_name, new Z8JsonArrayState(value));
    return result;
}
private void z8_json_array_assign(Z8Object self, wstring source) {
    z8_json_array_state(self);
    auto value = z8_json_array_read(source);
    self.z8_set_native_state(z8_json_array_name, new Z8JsonArrayState(value));
}
private size_t z8_json_array_position(Z8JsonArrayState state, Z8Optional!long index, bool insertion = false) {
    auto position = cast(int)z8_optional_require(index);
    if (position < 0 || cast(size_t)position > state.value.array.length ||
        (!insertion && cast(size_t)position == state.value.array.length))
        throw new Exception("JSON array index out of bounds");
    return cast(size_t)position;
}
private JSONValue z8_json_array_get(Z8Object self, Z8Optional!long index) {
    auto state = z8_json_array_state(self);
    return state.value.array[z8_json_array_position(state, index)];
}
private wstring z8_json_array_text_at(Z8JsonArrayState state, size_t position) {
    auto child = position in state.children;
    if (child !is null) return z8_json_child_text(*child);
    auto decimalValue = position in state.decimals;
    return decimalValue is null ? z8_json_value_text(state.value.array[position]) : z8_json_bigdecimal_text(*decimalValue).to!wstring;
}
private Z8Optional!long z8_json_array_size(Z8Object self) {
    return z8_optional_wrap(cast(long)z8_json_array_state(self).value.array.length);
}
private Z8Optional!bool z8_json_array_empty(Z8Object self) {
    return z8_optional_wrap(z8_json_array_state(self).value.array.length == 0);
}
private wstring z8_json_array_render_state(Z8JsonArrayState state) {
    wstring result = "["w;
    foreach (position, value; state.value.array) {
        if (position) result ~= ","w;
        auto decimalValue = position in state.decimals;
        auto child = position in state.children;
        result ~= child !is null ? z8_json_render_state(*child) : decimalValue is null ? z8_json_render(value) : z8_json_number_text(z8_json_bigdecimal_text(*decimalValue)).to!wstring;
    }
    return result ~ "]"w;
}
private wstring z8_json_array_string(Z8Object self) {
    return z8_string_new(z8_json_array_render_state(z8_json_array_state(self)));
}
private wstring z8_json_array_get_string(Z8Object self, Z8Optional!long index) {
    auto state = z8_json_array_state(self);
    return z8_string_new(z8_json_array_text_at(state, z8_json_array_position(state, index)));
}
private Z8Optional!bool z8_json_array_get_bool(Z8Object self, Z8Optional!long index) {
    auto value = z8_json_array_get(self, index);
    if (value.type == JSONType.null_) throw new Exception("null JSON boolean");
    return z8_optional_wrap(value.type == JSONType.true_);
}
private int z8_json_array_integer(JSONValue value) {
    import std.math : isNaN;
    switch (value.type) {
        case JSONType.integer: return cast(int)value.integer;
        case JSONType.uinteger: return cast(int)value.uinteger;
        case JSONType.float_:
            auto number = value.floating;
            if (isNaN(number)) return 0;
            if (number >= cast(double)int.max) return int.max;
            if (number <= cast(double)int.min) return int.min;
            return cast(int)number;
        case JSONType.string:
            if (value.str.length && (value.str[$ - 1] == 'L' || value.str[$ - 1] == 'l'))
                throw new Exception("invalid JSON integer");
            auto number = z8_int_parse(value.str.to!wstring);
            if (number < int.min || number > int.max) throw new Exception("invalid JSON integer");
            return cast(int)number;
        default: throw new Exception("invalid JSON integer");
    }
}
private Z8Optional!long z8_json_array_get_int(Z8Object self, Z8Optional!long index) {
    auto state = z8_json_array_state(self);
    auto position = z8_json_array_position(state, index);
    auto decimalValue = position in state.decimals;
    return z8_optional_wrap(cast(long)(decimalValue is null ? z8_json_array_integer(state.value.array[position]) : cast(int)z8_json_bigdecimal_long(*decimalValue)));
}
private Z8Optional!Z8Decimal z8_json_array_get_decimal(Z8Object self, Z8Optional!long index) {
    auto state = z8_json_array_state(self);
    auto position = z8_json_array_position(state, index);
    auto decimalValue = position in state.decimals;
    return z8_optional_wrap(decimalValue is null ? z8_json_value_decimal(state.value.array[position]) :
        z8_math_result(z8_literal_parse_double(z8_json_bigdecimal_text(*decimalValue))));
}
private Z8Guid z8_json_array_get_guid(Z8Object self, Z8Optional!long index) {
    auto state = z8_json_array_state(self);
    auto position = z8_json_array_position(state, index);
    return state.value.array[position].type == JSONType.null_ ? null : z8_guid_parse(z8_json_array_text_at(state, position));
}
private Z8Array!long z8_json_array_to_ints(Z8Object self) {
    auto result = new Z8Array!long();
    foreach (position; 0 .. z8_json_array_state(self).value.array.length)
        result.z8_add(z8_json_array_get_int(self, z8_optional_wrap(cast(long)position)));
    return result;
}
private Z8Array!wstring z8_json_array_to_strings(Z8Object self) {
    auto result = new Z8Array!wstring();
    auto state = z8_json_array_state(self);
    foreach (position, entry; state.value.array)
        result.z8_add(entry.type == JSONType.null_ ? null : z8_string_new(z8_json_array_text_at(state, position)));
    return result;
}
private Z8Array!bool z8_json_array_to_bools(Z8Object self) {
    auto result = new Z8Array!bool();
    auto state = z8_json_array_state(self);
    foreach (position, entry; state.value.array)
        result.z8_add(entry.type == JSONType.null_ ? false : z8_bool_parse(z8_json_array_text_at(state, position)));
    return result;
}
private Z8Array!Z8Decimal z8_json_array_to_decimals(Z8Object self) {
    auto result = new Z8Array!Z8Decimal();
    foreach (position; 0 .. z8_json_array_state(self).value.array.length)
        result.z8_add(z8_json_array_get_decimal(self, z8_optional_wrap(cast(long)position)));
    return result;
}
private Z8Array!Z8Guid z8_json_array_to_guids(Z8Object self) {
    auto result = new Z8Array!Z8Guid();
    auto state = z8_json_array_state(self);
    foreach (position, entry; state.value.array)
        result.z8_add(entry.type == JSONType.null_ ? null : z8_guid_parse(z8_json_array_text_at(state, position)));
    return result;
}
private Z8Array!Z8Date z8_json_array_to_dates(Z8Object self) {
    auto result = new Z8Array!Z8Date();
    auto state = z8_json_array_state(self);
    foreach (position, entry; state.value.array)
        result.z8_add(entry.type == JSONType.null_ ? null : z8_date_parse(z8_json_array_text_at(state, position)));
    return result;
}
private void z8_json_array_insert_value(Z8JsonArrayState state, size_t position, Z8JsonPrimaryValue value) {
    auto old = state.value.array;
    state.value.array = old[0 .. position] ~ [value.value] ~ old[position .. $];
    Z8Decimal[size_t] shifted;
    foreach (index, number; state.decimals) shifted[index >= position ? index + 1 : index] = number;
    if (value.hasDecimal) shifted[position] = value.decimalValue;
    state.decimals = shifted;
    Object[size_t] children;
    foreach (index, child; state.children) children[index >= position ? index + 1 : index] = child;
    if (value.child !is null) children[position] = value.child;
    state.children = children;
}
private Z8Object z8_json_array_add_primary(Z8Object self, Z8Primary value) {
    auto state = z8_json_array_state(self);
    auto converted = z8_json_from_primary(value);
    z8_json_array_insert_value(state, state.value.array.length, converted);
    return self;
}
private Z8Object z8_json_array_insert_primary(Z8Object self, Z8Optional!long index, Z8Primary value) {
    auto state = z8_json_array_state(self);
    auto converted = z8_json_from_primary(value);
    auto position = z8_json_array_position(state, index, true);
    z8_json_array_insert_value(state, position, converted);
    return self;
}
private Z8Object z8_json_array_remove(Z8Object self, Z8Optional!long index) {
    auto state = z8_json_array_state(self);
    auto position = z8_json_array_position(state, index);
    auto old = state.value.array;
    state.value.array = old[0 .. position] ~ old[position + 1 .. $];
    Z8Decimal[size_t] shifted;
    foreach (key, value; state.decimals) if (key != position) shifted[key > position ? key - 1 : key] = value;
    state.decimals = shifted;
    Object[size_t] children;
    foreach (key, child; state.children) if (key != position) children[key > position ? key - 1 : key] = child;
    state.children = children;
    return self;
}
private Z8JsonArrayState z8_json_array_from_values(Z8Array!Z8Primary values) {
    auto state = new Z8JsonArrayState(z8_json_empty_array());
    if (values !is null) foreach (index; 0 .. values.z8_size()) {
        auto converted = z8_json_from_primary(values.z8_get(index));
        state.value.array ~= converted.value;
        if (converted.hasDecimal) state.decimals[cast(size_t)index] = converted.decimalValue;
        if (converted.child !is null) state.children[cast(size_t)index] = converted.child;
    }
    return state;
}
private void z8_json_array_assign_values(Z8Object self, Z8Array!Z8Primary values) {
    z8_json_array_state(self);
    auto replacement = z8_json_array_from_values(values);
    self.z8_set_native_state(z8_json_array_name, replacement);
}
private Z8Object z8_json_array_from_array(Z8Array!Z8Primary values) {
    if (values is null) return null;
    auto state = z8_json_array_from_values(values);
    auto result = z8_native_new(z8_json_array_name);
    z8_nonnull(result).z8_set_native_state(z8_json_array_name, state);
    return result;
}
private Z8Object z8_json_array_add_array(Z8Object self, Z8Object other) {
    auto left = z8_json_array_state(self), right = z8_json_array_state(other);
    foreach (index, value; left.value.array)
        if ((value.type == JSONType.array || value.type == JSONType.object) && (index in left.children) is null)
            left.children[index] = z8_json_new_state(value);
    foreach (index, value; right.value.array)
        if ((value.type == JSONType.array || value.type == JSONType.object) && (index in right.children) is null)
            right.children[index] = z8_json_new_state(value);
    auto state = new Z8JsonArrayState(JSONValue(left.value.array ~ right.value.array));
    foreach (index, number; left.decimals) state.decimals[index] = number;
    foreach (index, number; right.decimals) state.decimals[left.value.array.length + index] = number;
    foreach (index, child; left.children) state.children[index] = child;
    foreach (index, child; right.children) state.children[left.value.array.length + index] = child;
    auto result = z8_native_new(z8_json_array_name);
    z8_nonnull(result).z8_set_native_state(z8_json_array_name, state);
    return result;
}
private Object z8_json_new_state(JSONValue value) {
    if (value.type == JSONType.array) return new Z8JsonArrayState(value);
    if (value.type == JSONType.object) return new Z8JsonObjectState(value);
    throw new Exception("invalid nested JSON value");
}
private JSONValue z8_json_state_value(Object state) {
    if (auto array = cast(Z8JsonArrayState)state) return array.value;
    if (auto object = cast(Z8JsonObjectState)state) return object.value;
    throw new Exception("invalid JSON state");
}
private wstring z8_json_render_state(Object state) {
    if (auto array = cast(Z8JsonArrayState)state) return z8_json_array_render_state(array);
    if (auto object = cast(Z8JsonObjectState)state) return z8_json_object_render_state(object);
    if (auto geometry = cast(Z8Geometry)state) {
        auto text = z8_geometry_to_string(geometry);
        return text.length ? z8_json_quote(text) : "null"w;
    }
    throw new Exception("invalid JSON state");
}
private wstring z8_json_child_text(Object state) {
    if (auto geometry = cast(Z8Geometry)state) return z8_geometry_to_string(geometry);
    return z8_json_render_state(state);
}
private Object z8_json_state_from_object(Z8Object value) {
    value = z8_nonnull(value);
    if (z8_native_is_instance(z8_json_array_name, value)) return z8_json_array_state(value);
    if (z8_native_is_instance(z8_json_object_name, value)) return z8_json_object_state(value);
    throw new Exception("invalid JSON object");
}
private Z8Object z8_json_wrap_state(Object state) {
    auto name = cast(Z8JsonArrayState)state !is null ? z8_json_array_name : z8_json_object_name;
    auto result = z8_native_new(name);
    z8_nonnull(result).z8_set_native_state(name, state);
    return result;
}
private Z8Object z8_json_array_get_nested(Z8Object self, Z8Optional!long index, JSONType expected) {
    auto state = z8_json_array_state(self);
    auto position = z8_json_array_position(state, index);
    auto value = state.value.array[position];
    if (value.type != expected) throw new Exception("invalid nested JSON value");
    auto child = position in state.children;
    if (child is null) {
        state.children[position] = z8_json_new_state(value);
        child = position in state.children;
    }
    return z8_json_wrap_state(*child);
}
private Z8Object z8_json_array_get_array(Z8Object self, Z8Optional!long index) { return z8_json_array_get_nested(self, index, JSONType.array); }
private Z8Object z8_json_array_get_object(Z8Object self, Z8Optional!long index) { return z8_json_array_get_nested(self, index, JSONType.object); }
private Z8Object z8_json_array_insert_nested(Z8Object self, Z8Optional!long index, Z8Object value) {
    auto state = z8_json_array_state(self);
    auto child = z8_json_state_from_object(value);
    auto position = z8_json_array_position(state, index, true);
    Z8JsonPrimaryValue converted;
    converted.value = z8_json_state_value(child);
    z8_json_array_insert_value(state, position, converted);
    state.children[position] = child;
    return self;
}
private Z8Object z8_json_array_add_nested(Z8Object self, Z8Object value) {
    auto state = z8_json_array_state(self);
    return z8_json_array_insert_nested(self, z8_optional_wrap(cast(long)state.value.array.length), value);
}
