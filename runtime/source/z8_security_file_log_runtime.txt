// A process-local, synchronized rotating sink for the default security log.
class Z8FileSecurityLogSink : IZ8SecurityLogSink {
    private string path;
    private ulong sizeLimit;
    private uint fileCount;
    private wstring pattern;
    this(string path, ulong sizeLimit = 10 * 1024 * 1024, uint fileCount = 10, wstring pattern = null) {
        import std.path : absolutePath, buildPath, isAbsolute;
        if (!path.length || !sizeLimit || !fileCount) throw new Exception("invalid security log configuration");
        this.path = absolutePath(isAbsolute(path) ? path : buildPath(z8_file_working_root, path));
        this.sizeLimit = sizeLimit; this.fileCount = fileCount;
        this.pattern = pattern is null || !pattern.length ?
            "[%1$tF %1$tT] User{%2$s %3$s} Object{%4$s} Action{%5$s} Params%6$s Success: %7$b - %8$s %n"w : pattern.idup;
    }
    void write(Z8Object event, Z8StringMap parameters) {
        import std.datetime : Clock;
        import std.file : exists, getSize, mkdirRecurse, remove, rename;
        import std.format : format;
        import std.path : dirName;
        import std.stdio : File;
        auto request = z8_application_request();
        auto state = z8_security_event_state(event);
        auto object = z8_nonnull(state.object);
        auto type = z8_string_or_empty(object.z8_security_object_virtual_type());
        auto name = z8_string_or_empty(object.z8_security_object_virtual_name());
        auto id = object.z8_security_object_virtual_id();
        wstring objectText;
        if (type.length) objectText ~= type ~ ":"w;
        if (id !is null && id == z8_guid_zero()) objectText ~= z8_guid_to_string(id) ~ ":"w;
        if (name.length) objectText ~= name ~ ":"w;
        if (objectText.length) objectText = objectText[0 .. $ - 1];
        if (request !is null && request.securityLog !is null)
            objectText = z8_nonnull(request.securityLog).z8_security_log_virtual_format_object(object);
        wstring actor, actorId;
        if (request !is null && request.session !is null && request.session.user !is null) {
            auto user = request.session.user;
            if (user.id !is null) actorId = z8_guid_to_string(user.id);
            foreach (part; [user.firstName, user.middleName, user.lastName])
                if (part !is null && part.length) actor ~= part ~ " "w;
            if (actor.length) actor ~= "("w ~ z8_string_or_empty(user.login) ~ ")"w;
            else actor = z8_string_or_empty(user.login);
        } else if (request !is null) actor = z8_string_or_empty(request.securityLogin);
        if (request !is null && request.securityLog !is null) {
            Z8UserData user;
            if (request.session !is null) user = request.session.user;
            if (user is null) { user = new Z8UserData(); user.id = z8_guid_zero(); user.login = request.securityLogin; }
            actor = z8_nonnull(request.securityLog).z8_security_log_virtual_format_user(z8_user_wrap(user));
        }
        wstring params = "{"w;
        auto keys = z8_nonnull(parameters).z8_keys();
        foreach (i; 0 .. keys.z8_size()) {
            auto key = keys.z8_get(i);
            if (i) params ~= ", "w;
            params ~= key ~ "="w ~ z8_string_or_empty(parameters.z8_get(key));
        }
        params ~= "}"w;
        auto now = Clock.currTime();
        wstring[9] args;
        args[2] = actorId; args[3] = actor; args[4] = objectText; args[5] = z8_string_or_empty(state.action);
        args[6] = params; args[7] = state.success.value ? "true"w : "false"w; args[8] = z8_string_or_empty(state.message);
        wstring lineText;
        size_t implicitArgument = 1, lastArgument;
        for (size_t i; i < pattern.length; ++i) {
            if (pattern[i] != '%') { lineText ~= pattern[i]; continue; }
            if (++i >= pattern.length) throw new Exception("incomplete security log format");
            if (pattern[i] == '%') { lineText ~= '%'; continue; }
            if (pattern[i] == 'n') {
                version (Windows) lineText ~= "\r\n"w; else lineText ~= "\n"w;
                continue;
            }
            size_t argument;
            auto start = i;
            size_t number;
            while (i < pattern.length && pattern[i] >= '0' && pattern[i] <= '9') {
                number = number * 10 + pattern[i] - '0'; ++i;
                if (number > 1_000_000) throw new Exception("security log format number is too large");
            }
            bool explicitArgument = i > start && i < pattern.length && pattern[i] == '$';
            if (explicitArgument) { argument = number; ++i; }
            else { i = start; argument = implicitArgument++; }
            bool leftAligned, reuse;
            while (i < pattern.length && (pattern[i] == '-' || pattern[i] == '<')) {
                if (pattern[i] == '-') { if (leftAligned) throw new Exception("duplicate security log flag"); leftAligned = true; }
                else { if (reuse) throw new Exception("duplicate security log flag"); reuse = true; }
                ++i;
            }
            if (reuse) { argument = lastArgument; if (!explicitArgument) --implicitArgument; }
            size_t width;
            while (i < pattern.length && pattern[i] >= '0' && pattern[i] <= '9') {
                width = width * 10 + pattern[i++] - '0';
                if (width > 1_000_000) throw new Exception("security log format width is too large");
            }
            if (leftAligned && !width) throw new Exception("security log format width is required");
            size_t precision; bool hasPrecision;
            if (i < pattern.length && pattern[i] == '.') {
                hasPrecision = true; ++i; auto precisionStart = i;
                while (i < pattern.length && pattern[i] >= '0' && pattern[i] <= '9') {
                    precision = precision * 10 + pattern[i++] - '0';
                    if (precision > 1_000_000) throw new Exception("security log format precision is too large");
                }
                if (i == precisionStart) throw new Exception("security log format precision is missing");
            }
            if (!argument || argument > 8 || i >= pattern.length) throw new Exception("invalid security log argument");
            lastArgument = argument;
            wstring partText;
            if (pattern[i] == 't' || pattern[i] == 'T') {
                if (argument != 1 || hasPrecision || ++i >= pattern.length) throw new Exception("invalid security log date format");
                string part;
                switch (pattern[i]) {
                    case 'F': part = format("%04d-%02d-%02d", now.year, cast(int)now.month, now.day); break;
                    case 'T': part = format("%02d:%02d:%02d", now.hour, now.minute, now.second); break;
                    case 'Y': part = format("%04d", now.year); break;
                    case 'm': part = format("%02d", cast(int)now.month); break;
                    case 'd': part = format("%02d", now.day); break;
                    case 'H': part = format("%02d", now.hour); break;
                    case 'M': part = format("%02d", now.minute); break;
                    case 'S': part = format("%02d", now.second); break;
                    case 'L': part = format("%03d", now.fracSecs.total!"msecs"); break;
                    default: throw new Exception("unsupported security log date format");
                }
                partText = part.to!wstring;
            } else if (pattern[i] == 's' || pattern[i] == 'S' || pattern[i] == 'b' || pattern[i] == 'B') {
                if (pattern[i] == 'b' || pattern[i] == 'B')
                    partText = argument == 7 ? args[7] : argument == 2 && args[2] is null ? "false"w : "true"w;
                else {
                    if (argument == 1) throw new Exception("unsupported security log conversion");
                    partText = z8_string_or_empty(args[argument]);
                }
                if (hasPrecision && precision < partText.length) partText = partText[0 .. precision];
                if (pattern[i] == 'S' || pattern[i] == 'B') partText = z8_str_toUpperCase(partText);
            } else throw new Exception("unsupported security log format");
            if (width > partText.length) {
                auto padding = new wchar[width - partText.length]; padding[] = ' ';
                lineText ~= leftAligned ? partText ~ cast(wstring)padding : cast(wstring)padding ~ partText;
            } else lineText ~= partText;
        }
        auto line = lineText.to!string;
        synchronized (this) {
            auto directory = dirName(path);
            if (directory.length) mkdirRecurse(directory);
            if (exists(path) && getSize(path) && getSize(path) + line.length > sizeLimit) {
                if (fileCount == 1) remove(path);
                else {
                    auto oldest = path ~ "." ~ (fileCount - 1).to!string;
                    if (exists(oldest)) remove(oldest);
                    for (uint i = fileCount - 1; i > 1; --i) {
                        auto from = path ~ "." ~ (i - 1).to!string;
                        if (exists(from)) rename(from, path ~ "." ~ i.to!string);
                    }
                    rename(path, path ~ ".1");
                }
            }
            auto file = File(path, "ab"); file.rawWrite(cast(const(ubyte)[])line); file.close();
        }
    }
}
Z8FileSecurityLogSink z8_security_log_file_from_config() {
    auto path = z8_server_config_value("security.log.file"w);
    return path is null || !path.length ? null : new Z8FileSecurityLogSink(path.to!string, 10 * 1024 * 1024, 10,
        z8_server_config_value("security.log.format"w));
}
