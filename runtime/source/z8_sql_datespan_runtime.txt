enum Z8SqlDatespanKind { constant, group, choose, add, subtract, dateDifference }
class Z8SqlDatespan {
    Z8SqlDatespanKind kind;
    Z8Datespan constant;
    Z8SqlDatespan left, right;
    Z8SqlDate dateLeft, dateRight;
    Z8SqlBool condition;
    this(Z8SqlDatespanKind kind, Z8Datespan constant = null, Z8SqlDatespan left = null,
         Z8SqlDatespan right = null, Z8SqlBool condition = null) {
        this.kind = kind; this.constant = constant; this.left = left; this.right = right; this.condition = condition;
    }
}
Z8SqlDatespan z8_sql_datespan_const(Z8Datespan value) { return new Z8SqlDatespan(Z8SqlDatespanKind.constant, z8_nonnull(value)); }
Z8SqlDatespan z8_sql_datespan_group(Z8SqlDatespan value) { return new Z8SqlDatespan(Z8SqlDatespanKind.group, null, z8_nonnull(value)); }
Z8SqlDatespan z8_sql_datespan_iif(Z8SqlBool condition, Z8SqlDatespan yes, Z8SqlDatespan no) {
    return new Z8SqlDatespan(Z8SqlDatespanKind.choose, null, yes, no, z8_nonnull(condition));
}
Z8SqlDatespan z8_sql_datespan_binary(string op)(Z8SqlDatespan left, Z8SqlDatespan right) {
    static assert(op == "+" || op == "-");
    return new Z8SqlDatespan(op == "+" ? Z8SqlDatespanKind.add : Z8SqlDatespanKind.subtract, null, z8_nonnull(left), right);
}
Z8SqlBool z8_sql_datespan_compare(string op)(Z8SqlDatespan left, Z8SqlDatespan right) {
    auto result = new Z8SqlBool(Z8SqlBoolKind.spanCompare);
    result.spanLeft = z8_nonnull(left); result.spanRight = right; result.spanOperator = op; return result;
}
Z8SqlDate z8_sql_date_arithmetic(string op)(Z8SqlDate left, Z8SqlDatespan right) {
    static assert(op == "+" || op == "-");
    auto result = new Z8SqlDate(op == "+" ? Z8SqlDateKind.addSpan : Z8SqlDateKind.subtractSpan, null, z8_nonnull(left));
    result.spanOperand = right; return result;
}
Z8SqlDate z8_sql_date_arithmetic(string op)(Z8SqlDatespan left, Z8SqlDate right) {
    static assert(op == "+");
    auto result = new Z8SqlDate(Z8SqlDateKind.addSpan, null, right);
    result.spanOperand = z8_nonnull(left); result.spanFirst = true; return result;
}
Z8SqlDatespan z8_sql_date_arithmetic(string op)(Z8SqlDate left, Z8SqlDate right) {
    static assert(op == "-");
    auto result = new Z8SqlDatespan(Z8SqlDatespanKind.dateDifference);
    result.dateLeft = z8_nonnull(left); result.dateRight = right; return result;
}
wstring z8_sql_datespan_format(Z8SqlDatespan value, Z8SqlVendor vendor = Z8SqlVendor.Postgres) {
    z8_nonnull(value);
    final switch (value.kind) {
        case Z8SqlDatespanKind.constant:
            auto ticks = z8_datespan_milliseconds(value.constant).to!wstring;
            return vendor == Z8SqlVendor.SqlServer ? "("w ~ ticks ~ ")"w : ticks;
        case Z8SqlDatespanKind.group: return "("w ~ z8_sql_datespan_format(value.left, vendor) ~ ")"w;
        case Z8SqlDatespanKind.choose:
            return z8_sql_bool_case(z8_sql_bool_format(value.condition, true, vendor), z8_sql_datespan_format(value.left, vendor), z8_sql_datespan_format(value.right, vendor));
        case Z8SqlDatespanKind.add: return z8_sql_datespan_format(value.left, vendor) ~ " + "w ~ z8_sql_datespan_format(value.right, vendor);
        case Z8SqlDatespanKind.subtract: return z8_sql_datespan_format(value.left, vendor) ~ " - ("w ~ z8_sql_datespan_format(value.right, vendor) ~ ")"w;
        case Z8SqlDatespanKind.dateDifference: return z8_sql_date_format(value.dateLeft, vendor) ~ " - ("w ~ z8_sql_date_format(value.dateRight, vendor) ~ ")"w;
    }
}
