// String-keyed scalar maps retain null keys and null scalar values.
class Z8StringValueMap(V) {
    static if (is(V == long) || is(V == bool) || is(V == Z8Decimal)) alias Element = Z8Optional!V;
    else static if (is(V == class)) alias Element = V;
    else static assert(0, "unsupported string-key map value");
    private Element[wstring] entries;
    private bool hasNullKey;
    private Element nullKeyValue;

    private static bool valueEquals(Element left, Element right) {
        static if (is(V == class)) return left is right || (left !is null && left == right);
        else return left.present == right.present && (!left.present || left.value == right.value);
    }

    long z8_size() { return cast(long)entries.length + (hasNullKey ? 1 : 0); }
    bool z8_isEmpty() { return entries.length == 0 && !hasNullKey; }
    void z8_clear() { entries.clear(); hasNullKey = false; nullKeyValue = Element.init; }
    bool z8_containsKey(wstring key) { return key is null ? hasNullKey : (key in entries) !is null; }
    bool z8_containsValue(Element value) {
        if (hasNullKey && valueEquals(value, nullKeyValue)) return true;
        foreach (item; entries) if (valueEquals(value, item)) return true;
        return false;
    }
    Element z8_add(wstring key, Element value) {
        if (key is null) {
            auto previous = hasNullKey ? nullKeyValue : Element.init;
            hasNullKey = true; nullKeyValue = value; return previous;
        }
        auto found = key in entries;
        auto previous = found is null ? Element.init : *found;
        entries[key] = value;
        return previous;
    }
    void z8_add(Z8StringValueMap!V other) {
        z8_nonnull(other);
        if (other is this) return;
        foreach (key, value; other.entries) z8_add(key, value);
        if (other.hasNullKey) z8_add(null, other.nullKeyValue);
    }
    Element z8_get(wstring key) {
        if (key is null) return hasNullKey ? nullKeyValue : Element.init;
        auto found = key in entries;
        return found is null ? Element.init : *found;
    }
    Element z8_remove(wstring key) {
        if (key is null) {
            auto previous = hasNullKey ? nullKeyValue : Element.init;
            hasNullKey = false; nullKeyValue = Element.init; return previous;
        }
        auto found = key in entries;
        if (found is null) return Element.init;
        auto previous = *found;
        entries.remove(key);
        return previous;
    }
    Z8Array!wstring z8_keys() {
        auto result = new Z8Array!wstring();
        foreach (key, value; entries) result.z8_add(key);
        if (hasNullKey) result.z8_add(null);
        return result;
    }
    Z8Array!V z8_values() {
        auto result = new Z8Array!V();
        foreach (value; entries) result.z8_add(value);
        if (hasNullKey) result.z8_add(nullKeyValue);
        return result;
    }
}
alias Z8StringIntMap = Z8StringValueMap!long;
Z8StringValueMap!V z8_map_union(V)(Z8StringValueMap!V left, Z8StringValueMap!V right) {
    auto result = new Z8StringValueMap!V();
    result.z8_add(z8_nonnull(left));
    result.z8_add(z8_nonnull(right));
    return result;
}
void z8_map_add_assign(V)(ref Z8StringValueMap!V left, lazy Z8StringValueMap!V previous, lazy Z8StringValueMap!V right) {
    auto oldValue = previous;
    auto newValue = right;
    left = z8_map_union(oldValue, newValue);
}
