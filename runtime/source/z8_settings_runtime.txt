private Z8Primary z8_settings_parse(wstring value, Z8Primary fallback) {
    if (value is null || !value.length) return fallback;
    if (fallback is null) throw new Exception("setting type is not specified");
    switch (fallback.kind) {
        case Z8PrimaryKind.text: return z8_primary_box(z8_string_new(value));
        case Z8PrimaryKind.guid: return z8_primary_box(z8_guid_parse(value));
        case Z8PrimaryKind.date: return z8_primary_box(z8_date_parse(value));
        case Z8PrimaryKind.integer: return z8_primary_box(z8_int_parse(value));
        case Z8PrimaryKind.decimal: return z8_primary_box(z8_decimal_parse(value));
        case Z8PrimaryKind.boolean: return z8_primary_box(z8_bool_parse(value));
        default: throw new Exception("unsupported setting type");
    }
}

    ids = z8_nonnull(ids); defaults = z8_nonnull(defaults);
    auto result = new Z8GuidMap!Z8Primary();
    for (long i; i < ids.z8_size(); ++i) result.z8_add(ids.z8_get(i), defaults.z8_get(i));
    try {
        for (long i; i < ids.z8_size(); ++i) {
            auto id = ids.z8_get(i);
            auto value = z8_settings_read(id);
            if (value !is null) result.z8_add(id, z8_settings_parse(value, result.z8_get(id)));
        }
    } catch (Throwable) {}
    return result;
}

    if (name !is null) z8_field_set(fields[0], z8_primary_box(name));
    if (description !is null) z8_field_set(fields[1], z8_primary_box(description));
