// JSON filter trees correspond to base.view.filter.Filter and Expression.
private Z8SqlBool z8_filter_combine(Z8SqlBool left, Z8SqlBool right, bool useOr = false) {
    if (left is null) return right;
    if (right is null) return left;
    return useOr ? z8_sql_bool_binary!"||"(left, right) : z8_sql_bool_binary!"&&"(left, right);
}
private wstring z8_filter_property(JSONValue[string] object, string key) {
    auto value = key in object;
    return value is null || value.type == JSONType.null_ ? null : z8_json_value_text(*value);
}
private Z8Primary z8_filter_value(wstring type, wstring value) {
    switch (type) {
        case "int"w: return z8_primary_box(value.length ? z8_int_parse(value) : 0L);
        case "float"w: return z8_primary_box(z8_decimal_parse(value is null ? ""w : value));
        case "boolean"w: return z8_primary_box(z8_bool_parse(value));
        case "guid"w: return z8_primary_box(value is null || !value.length ? z8_guid_zero() : z8_guid_parse(value));
        case "date"w: case "datetime"w: return z8_primary_box(z8_date_parse(value is null ? ""w : value));
        case "string"w: case "text"w: case "file"w: case "attachments"w: return z8_primary_box(value);
        default: throw new Exception("unsupported JSON filter field type");
    }
}
private wstring[] z8_filter_words(wstring text) {
    wstring[] words;
    size_t start;
    bool whitespace(wchar c) { return c == ' ' || (c >= '\t' && c <= '\r'); }
    while (start < text.length) {
        while (start < text.length && whitespace(text[start])) ++start;
        auto end = start;
        while (end < text.length && !whitespace(text[end])) ++end;
        if (end > start) words ~= text[start .. end];
        start = end;
    }
    return words;
}
private Z8SqlBool z8_filter_text_operation(Z8Object field, wstring type, wstring operation, wstring value) {
    import std.string : strip;
    Z8SqlString text = type == "guid"w ? z8_sql_guid_toString(z8_guid_field_sql(field)) :
        type != "string"w ? z8_text_field_sql(field) : z8_string_field_sql(field);
    if (operation == "isempty"w || operation == "isnotempty"w)
        return operation == "isempty"w ? z8_sql_string_compare!"=="(text, z8_sql_string_const(""w)) :
            z8_sql_string_compare!"!="(text, z8_sql_string_const(""w));
    if (value is null || !value.strip().length) return null;
    auto negative = operation == "notbeginswith"w || operation == "notendswith"w ||
        operation == "notcontains"w || operation == "notcontainsword"w;
    bool words = operation == "containsword"w || operation == "notcontainsword"w;
    bool contains = operation == "contains"w || operation == "notcontains"w;
    bool begins = operation == "beginswith"w || operation == "notbeginswith"w;
    bool ends = operation == "endswith"w || operation == "notendswith"w;
    if (!words && !contains && !begins && !ends) throw new Exception("unsupported JSON filter operation");
    wstring[] terms;
    if (words || contains) {
        value = value.strip();
        terms = value.length > 1 && value[0] == '"' && value[$ - 1] == '"' ? [value[1 .. $ - 1]] : z8_filter_words(value);
    } else terms = [value];
    auto lower = z8_sql_string_toLower(text);
    Z8SqlBool result;
    foreach (term; terms) {
        wstring pattern = words ? "\\y"w ~ term ~ "\\y"w : begins ? term ~ "%"w : ends ? "%"w ~ term : "%"w ~ term ~ "%"w;
        auto sqlPattern = z8_sql_string_const(z8_str_toLowerCase(pattern));
        auto part = words ? z8_sql_string_regLike(lower, sqlPattern) : z8_sql_string_like(lower, sqlPattern);
        result = z8_filter_combine(result, part);
    }
    return negative && result !is null ? z8_sql_bool_not(result) : result;
}
private Z8SqlBool z8_filter_relative_date(Z8Object field, wstring operation, wstring value) {
    auto now = z8_date_now();
    Z8Date start, end;
    switch (operation) {
        case "yesterday"w: case "today"w: case "tomorrow"w:
            auto day = operation == "yesterday"w ? -1 : operation == "tomorrow"w ? 1 : 0;
            start = z8_date_truncDay(z8_date_addDay(now, day)); end = z8_date_addDay(start, 1); break;
        case "lastweek"w: case "thisweek"w: case "nextweek"w:
            auto day = z8_date_truncDay(now);
            auto daysSinceMonday = (z8_date_dayOfWeek(day) + 5) % 7;
            auto offset = operation == "lastweek"w ? -7 : operation == "nextweek"w ? 7 : 0;
            start = z8_date_addDay(day, offset - daysSinceMonday); end = z8_date_addDay(start, 7); break;
        case "lastmonth"w: case "thismonth"w: case "nextmonth"w:
            auto offset = operation == "lastmonth"w ? -1 : operation == "nextmonth"w ? 1 : 0;
            start = z8_date_addMonth(z8_date_truncMonth(now), offset); end = z8_date_addMonth(start, 1); break;
        case "lastyear"w: case "thisyear"w: case "nextyear"w:
            auto offset = operation == "lastyear"w ? -1 : operation == "nextyear"w ? 1 : 0;
            start = z8_date_addYear(z8_date_truncYear(now), offset); end = z8_date_addYear(start, 1); break;
        case "lastdays"w: case "nextdays"w: case "lasthours"w: case "nexthours"w:
            if (value is null || !value.length) throw new Exception("relative date filter requires a quantity");
            auto count = cast(int)z8_int_parse(value);
            bool hours = operation == "lasthours"w || operation == "nexthours"w;
            bool previous = operation == "lastdays"w || operation == "lasthours"w;
            auto base = hours ? z8_date_truncHour(now) : z8_date_truncDay(now);
            start = previous ? (hours ? z8_date_addHour(base, -cast(long)count) : z8_date_addDay(base, -cast(long)count)) : base;
            end = hours ? z8_date_addHour(start, count) : z8_date_addDay(start, count); break;
        default: throw new Exception("unsupported relative date filter operation");
    }
    auto column = z8_datetime_field_sql(field);
    return z8_sql_bool_binary!"&&"(z8_sql_date_compare!">="(column, z8_sql_date_const(start)),
        z8_sql_date_compare!"<"(column, z8_sql_date_const(end)));
}
private Z8SqlBool z8_filter_compare(Z8Object field, wstring operation, wstring[] values) {
    auto type = z8_field_type_name(field);
    if (values.length > 1) {
        if ((operation != "eq"w && operation != "noteq"w) || type == "date"w || type == "datetime"w)
            throw new Exception("unsupported JSON filter vector operation");
        auto vector = new Z8SqlBool(Z8SqlBoolKind.inVector);
        vector.vectorValue = new Z8SqlFieldToken(field);
        foreach (value; values) vector.vectorItems ~= new Z8SqlBoxedConstant(z8_filter_value(type, value));
        return operation == "noteq"w ? z8_sql_bool_not(vector) : vector;
    }
    auto value = values.length ? values[0] : null;
    if (type == "boolean"w) {
        bool number = operation == "istrue"w ? true : operation == "isfalse"w ? false : z8_bool_parse(value);
        return z8_sql_bool_binary!"=="(z8_bool_field_sql(field), z8_sql_bool_const(number));
    }
    if (operation == "issimilarto"w || operation == "isnotsimilarto"w) {
        enum prefix = "org.zenframework.z8.server.base.table.value."w;
        if (!z8_native_is_instance(prefix ~ "StringField"w, field) && !z8_native_is_instance(prefix ~ "StringExpression"w, field))
            throw new Exception("full-text JSON filter requires a string field or expression");
        if (value is null) throw new Exception("full-text JSON filter requires a value");
        auto result = z8_sql_string_ftsLike(z8_string_field_sql(field), z8_sql_string_const(value), z8_nonnull(z8_field_state(field).fts));
        return operation == "isnotsimilarto"w ? z8_sql_bool_not(result) : result;
    }
    if ((type == "date"w || type == "datetime"w) && ["yesterday"w, "today"w, "tomorrow"w,
        "lastweek"w, "thisweek"w, "nextweek"w, "lastmonth"w, "thismonth"w, "nextmonth"w,
        "lastyear"w, "thisyear"w, "nextyear"w, "lastdays"w, "nextdays"w, "lasthours"w, "nexthours"w].canFind(operation))
        return z8_filter_relative_date(field, operation, value);
    bool textType = type == "string"w || type == "text"w || type == "file"w || type == "attachments"w;
    if ((textType || type == "guid"w) && ["beginswith"w, "notbeginswith"w, "endswith"w, "notendswith"w,
        "contains"w, "notcontains"w, "containsword"w, "notcontainsword"w, "isempty"w, "isnotempty"w].canFind(operation))
        return z8_filter_text_operation(field, type, operation, value);
    if (!["eq"w, "noteq"w, "lt"w, "gt"w, "le"w, "ge"w].canFind(operation))
        throw new Exception("unsupported JSON filter operation");
    if (textType && value is null) return null;
    Z8SqlBoolKind kind;
    switch (operation) {
        case "eq"w: kind = textType ? Z8SqlBoolKind.textEqual : Z8SqlBoolKind.intEqual; break;
        case "noteq"w: kind = textType ? Z8SqlBoolKind.textNotEqual : Z8SqlBoolKind.intNotEqual; break;
        case "lt"w: kind = textType ? Z8SqlBoolKind.textLess : Z8SqlBoolKind.intLess; break;
        case "gt"w: kind = textType ? Z8SqlBoolKind.textGreater : Z8SqlBoolKind.intGreater; break;
        case "le"w: kind = textType ? Z8SqlBoolKind.textLessEqual : Z8SqlBoolKind.intLessEqual; break;
        case "ge"w: kind = textType ? Z8SqlBoolKind.textGreaterEqual : Z8SqlBoolKind.intGreaterEqual; break;
        default: throw new Exception("unsupported JSON filter operation");
    }
    if (textType) {
        auto result = new Z8SqlBool(kind);
        auto left = type != "string"w ? z8_text_field_sql(field) : z8_string_field_sql(field);
        if (operation == "eq"w || operation == "noteq"w) { left = z8_sql_string_toLower(left); value = z8_str_toLowerCase(value); }
        result.textLeft = left; result.textRight = z8_sql_string_const(value); return result;
    }
    auto boxed = z8_filter_value(type, value);
    if (type == "guid"w) {
        if (operation == "eq"w) return z8_sql_guid_compare!"=="(z8_guid_field_sql(field), z8_sql_guid_const(boxed.guidValue));
        if (operation == "noteq"w) return z8_sql_guid_compare!"!="(z8_guid_field_sql(field), z8_sql_guid_const(boxed.guidValue));
        throw new Exception("unsupported JSON GUID filter operation");
    }
    if (type == "date"w || type == "datetime"w) {
        auto result = new Z8SqlBool(Z8SqlBoolKind.dateCompare);
        result.dateLeft = z8_datetime_field_sql(field); result.dateRight = z8_sql_date_const(boxed.dateValue);
        result.dateOperator = operation == "eq"w ? "==" : operation == "noteq"w ? "!=" : operation == "lt"w ? "<" : operation == "gt"w ? ">" : operation == "le"w ? "<=" : ">=";
        return result;
    }
    if (type != "int"w && type != "float"w) throw new Exception("unsupported JSON filter field type");
    auto result = new Z8SqlBool(kind);
    result.numericLeft = type == "int"w ? cast(Z8SqlNumeric)z8_integer_field_sql(field) : z8_decimal_field_sql(field);
    result.numericRight = type == "int"w ? cast(Z8SqlNumeric)z8_sql_int_const(boxed.integerValue) : z8_sql_decimal_const(boxed.decimalValue);
    return result;
}
private Z8SqlBool z8_filter_node(Z8Object query, JSONValue node) {
    if (node.type != JSONType.object) throw new Exception("JSON filter expression must be an object");
    auto object = node.object;
    if ("logical" in object) {
        auto logical = z8_filter_property(object, "logical");
        logical = logical is null ? null : z8_str_toLowerCase(logical);
        auto children = "expressions" in object;
        if (children is null || children.type != JSONType.array) throw new Exception("JSON filter group requires expressions");
        Z8SqlBool result;
        foreach (child; children.array) result = z8_filter_combine(result, z8_filter_node(query, child), logical == "or"w);
        return result is null ? null : logical == "not"w ? z8_sql_bool_not(result) : z8_sql_bool_group(result);
    }
    auto property = z8_filter_property(object, "property");
    if (property is null) return null;
    auto field = z8_query_find_field_id(query, property);
    auto operation = z8_filter_property(object, "operator");
    operation = operation is null ? "eq"w : z8_str_toLowerCase(operation);
    auto value = z8_filter_property(object, "value");
    wstring[] values;
    if (value !is null && value.length) {
        if (value[0] == '[') {
            JSONValue parsed;
            bool valid;
            try { parsed = z8_json_parse(value.to!string); valid = parsed.type == JSONType.array; } catch (Exception) { }
            if (valid) foreach (item; parsed.array) {
                if (item.type != JSONType.string) throw new Exception("JSON filter values must be strings");
                values ~= z8_json_value_text(item);
            }
            else values ~= value;
        } else values ~= value;
    }
    return z8_filter_compare(field, operation, values);
}
private Z8SqlBool z8_filter_json(Z8Object query, wstring json) {
    if (json is null || !json.length) return null;
    auto root = z8_json_parse(json.to!string);
    if (root.type == JSONType.object) {
        auto result = z8_filter_node(query, root);
        return result is null ? null : z8_sql_bool_group(result);
    }
    if (root.type != JSONType.array) throw new Exception("JSON filter must be an object or array");
    Z8SqlBool result;
    foreach (node; root.array) result = z8_filter_combine(result, z8_filter_node(query, node));
    return result is null ? null : z8_sql_bool_group(result);
}
private Z8SqlBool z8_filter_json_array(Z8Object query, Z8Array!wstring json) {
    Z8SqlBool result;
    for (long i; i < z8_nonnull(json).z8_size(); ++i) result = z8_filter_combine(result, z8_filter_node(query, z8_json_parse(z8_string_required(json.z8_get(i)).to!string)));
    return result is null ? null : z8_sql_bool_group(result);
}
private void z8_query_set_where_json(Z8Object self, wstring json) { if (json !is null) z8_query_set_where(self, z8_filter_json(self, json)); }
private void z8_query_set_where_json_array(Z8Object self, Z8Array!wstring json) { z8_query_set_where(self, z8_filter_json_array(self, json)); }
private void z8_query_add_where_json(Z8Object self, wstring json) { z8_query_add_where(self, z8_filter_json(self, json)); }
private void z8_query_add_where_json_array(Z8Object self, Z8Array!wstring json) { z8_query_add_where(self, z8_filter_json_array(self, json)); }
private void z8_query_add_having_json(Z8Object self, wstring json) { if (json !is null) z8_query_add_having(self, z8_filter_json(self, json)); }
private void z8_query_add_having_json_array(Z8Object self, Z8Array!wstring json) { z8_query_add_having(self, z8_filter_json_array(self, json)); }
