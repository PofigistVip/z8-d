// Native BL methods are bound explicitly by the embedding D application.
private import std.meta : staticMap;
private template Z8NativeErased(T) {
    static if (is(T : Z8Object)) alias Z8NativeErased = Z8Object;
    else static if (is(T == Z8ObjectArray!U, U) && is(U : Z8Object))
        alias Z8NativeErased = Z8ObjectArray!Z8Object;
    else alias Z8NativeErased = T;
}
private template Z8NativeErasedReturn(T) {
    static if (is(T == U*, U) && is(U : Z8Object)) alias Z8NativeErasedReturn = Z8Object*;
    else alias Z8NativeErasedReturn = Z8NativeErased!T;
}
private Z8NativeErased!T z8_native_erased_argument(T)(T value) {
    static if (is(T == Z8ObjectArray!U, U) && is(U : Z8Object))
        return z8_object_array_view!Z8Object(value);
    else return value;
}
private auto z8_native_erased_arguments(Args...)(Args args) {
    import std.typecons : Tuple;
    alias ErasedArgs = staticMap!(Z8NativeErased, Args);
    Tuple!ErasedArgs result;
    static foreach (i; 0 .. Args.length) result[i] = z8_native_erased_argument(args[i]);
    return result;
}
private class Z8NativeBinding(R, Args...) {
    alias Callback = R function(Args);
    Callback callback;
    this(Callback callback) { this.callback = callback; }
}
private __gshared Object[wstring] z8_native_bindings;
private __gshared Object z8_native_monitor;
private struct Z8NativeFactory {
    TypeInfo_Class type;
    Z8Object function() create;
}
private __gshared Z8NativeFactory[][wstring] z8_native_factories;
shared static this() { z8_native_monitor = new Object(); }
private wstring z8_native_binding_key(R, Args...)(wstring name) {
    return name ~ "|"w ~ typeid(Z8NativeBinding!(R, Args)).name.to!wstring;
}
private Z8NativeBinding!(R, Args) z8_native_find(R, Args...)(wstring name) {
    auto key = z8_native_binding_key!(R, Args)(name);
    synchronized (z8_native_monitor) {
        auto found = key in z8_native_bindings;
        return found is null ? null : cast(Z8NativeBinding!(R, Args))*found;
    }
}
void z8_native_bind(R, Args...)(wstring name, R function(Args) callback) {
    if (callback is null) throw new Exception("native binding callback is null");
    auto key = z8_native_binding_key!(R, Args)(name);
    synchronized (z8_native_monitor) z8_native_bindings[key] = new Z8NativeBinding!(R, Args)(callback);
}
// Field callbacks return an address owned by the native implementation.
void z8_native_bind_field(T, Args...)(wstring name, T* function(Args) callback) {
    z8_native_bind(name, callback);
}
// Generated native classes register factories independently of method bindings.
void z8_native_register_factory(wstring name, TypeInfo_Class type, Z8Object function() create) {
    if (type is null || create is null) throw new Exception("invalid native type factory");
    synchronized (z8_native_monitor) {
        auto found = name in z8_native_factories;
        if (found !is null) foreach (ref entry; *found) if (entry.type is type) { entry.create = create; return; }
        z8_native_factories[name] ~= Z8NativeFactory(type, create);
    }
}
Z8Object z8_native_new(wstring name) {
    Z8NativeFactory factory;
    synchronized (z8_native_monitor) {
        auto found = name in z8_native_factories;
        if (found is null || !(*found).length) throw new Exception("native type not registered: " ~ name.to!string);
        if ((*found).length != 1) throw new Exception("native type is ambiguous: " ~ name.to!string);
        factory = (*found)[0];
    }
    auto value = factory.create();
    if (value is null || !factory.type.isBaseOf(value.classinfo))
        throw new Exception("invalid native type factory result: " ~ name.to!string);
    return value;
}
bool z8_native_is_instance(wstring name, Z8Object value) {
    if (value is null) return false;
    synchronized (z8_native_monitor) {
        auto found = name in z8_native_factories;
        if (found !is null) foreach (entry; *found) if (entry.type.isBaseOf(value.classinfo)) return true;
    }
    return false;
}
ref T z8_native_field(T, Args...)(wstring name, Args args) {
    auto address = z8_native_call!(T*)(name, args);
    if (address is null) throw new Exception("native field binding returned null: " ~ name.to!string);
    return *address;
}
// A native value may use Java's default identity equality when no override is supplied.
bool z8_native_try_call(R, Args...)(wstring name, out R result, Args args) if (!is(R == void)) {
    auto binding = z8_native_find!(R, Args)(name);
    if (binding !is null) {
        result = binding.callback(args);
        return true;
    }
    alias ErasedR = Z8NativeErasedReturn!R;
    alias ErasedArgs = staticMap!(Z8NativeErased, Args);
    auto erased = z8_native_find!(ErasedR, ErasedArgs)(name);
    if (erased is null) return false;
    auto erasedArguments = z8_native_erased_arguments(args);
    static if (is(R : Z8Object)) {
        auto value = erased.callback(erasedArguments.expand);
        result = cast(R)value;
        if (value !is null && result is null) throw new Exception("invalid native return type: " ~ name.to!string);
    } else static if (is(R == U*, U) && is(U : Z8Object)) {
        auto address = erased.callback(erasedArguments.expand);
        if (address !is null && *address !is null && cast(U)*address is null)
            throw new Exception("invalid native return type: " ~ name.to!string);
        result = cast(R)address;
    } else static if (is(R == Z8ObjectArray!U, U) && is(U : Z8Object)) {
        result = z8_object_array_view!U(erased.callback(erasedArguments.expand));
    } else result = erased.callback(erasedArguments.expand);
    return true;
}
bool z8_native_try_call_void(Args...)(wstring name, Args args) {
    auto binding = z8_native_find!(void, Args)(name);
    if (binding !is null) {
        binding.callback(args);
        return true;
    }
    alias ErasedArgs = staticMap!(Z8NativeErased, Args);
    auto erased = z8_native_find!(void, ErasedArgs)(name);
    if (erased is null) return false;
    auto erasedArguments = z8_native_erased_arguments(args);
    erased.callback(erasedArguments.expand);
    return true;
}
void z8_native_clear_bindings() { synchronized (z8_native_monitor) z8_native_bindings.clear(); }
R z8_native_call(R, Args...)(wstring name, Args args) {
    static if (is(R == void)) {
        if (!z8_native_try_call_void(name, args))
            throw new Exception("native binding not found: " ~ name.to!string);
    } else {
        R result;
        if (!z8_native_try_call!R(name, result, args))
            throw new Exception("native binding not found: " ~ name.to!string);
        return result;
    }
}
