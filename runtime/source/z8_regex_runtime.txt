// String regex operations. Syntax is the subset shared by Java and std.regex.
private struct Z8MatcherKey { const(wchar)* pointer; size_t length; }
private class Z8MatcherState {
    import std.regex : Captures, RegexMatch;
    wstring subject;
    RegexMatch!wstring matches;
    Captures!wstring current;
    bool hasCurrent;
}
private __gshared Z8MatcherState[Z8MatcherKey] z8_matchers;
private __gshared Object z8_matchers_lock;
shared static this() { z8_matchers_lock = new Object(); }
private Z8MatcherKey z8_matcher_key(wstring value) {
    value = z8_string_required(value);
    return Z8MatcherKey(value.ptr, value.length);
}
private wstring z8_regex_pattern(wstring pattern) {
    pattern = z8_string_required(pattern);
    wstring result;
    size_t classDepth;
    size_t i;
    bool quoted;
    bool unicodeClasses;
    bool[] scopes;
    int captures;
    int[wstring] named;
    while (i < pattern.length) {
        auto c = pattern[i++];
        if (quoted) {
            if (c == '\\' && i < pattern.length && pattern[i] == 'E') { ++i; quoted = false; continue; }
            if ("\\.^$|?*+()[]{}-"w.canFind(c)) result ~= '\\';
            result ~= c;
            continue;
        }
        if (c == '\\') {
            if (i == pattern.length) { result ~= c; continue; }
            auto escape = pattern[i++];
            if (escape == 'Q') { quoted = true; continue; }
            if (!classDepth && escape == 'k' && i < pattern.length && pattern[i] == '<') {
                auto begin = ++i;
                while (i < pattern.length && pattern[i] != '>') ++i;
                if (i == pattern.length || i == begin) throw new Exception("invalid named regex backreference");
                auto group = pattern[begin .. i] in named;
                if (group is null) throw new Exception("unknown named regex group");
                result ~= "(?:\\"w ~ (*group).to!wstring ~ ")"w;
                ++i;
                continue;
            }
            if (unicodeClasses && "dDwWsS"w.canFind(escape)) { result ~= '\\'; result ~= escape; continue; }
            wstring chars;
            if (escape == 'd' || escape == 'D') chars = "0-9"w;
            else if (escape == 'w' || escape == 'W') chars = "A-Za-z0-9_"w;
            else if (escape == 's' || escape == 'S') chars = " \\t\\n\\x0B\\f\\r"w;
            if (chars.length) {
                if (classDepth && escape >= 'a' && escape <= 'z') result ~= chars;
                else result ~= "["w ~ ((escape >= 'A' && escape <= 'Z') ? "^"w : ""w) ~ chars ~ "]"w;
            } else { result ~= c; result ~= escape; }
        } else if (!classDepth && c == '(' && i + 2 < pattern.length && pattern[i] == '?' && pattern[i + 1] == '<' &&
                 pattern[i + 2] != '=' && pattern[i + 2] != '!') {
            scopes ~= unicodeClasses;
            i += 2;
            auto begin = i;
            while (i < pattern.length && pattern[i] != '>') ++i;
            if (i == pattern.length || i == begin) throw new Exception("invalid named regex group");
            auto name = pattern[begin .. i++];
            named[name] = ++captures;
            result ~= "(?P<"w ~ name ~ ">"w;
        } else if (!classDepth && c == '(' && i + 1 < pattern.length && pattern[i] == '?') {
            size_t cursor = i + 1;
            bool negative, sawUnicode, nextUnicode = unicodeClasses;
            wstring enabled, disabled;
            while (cursor < pattern.length) {
                auto flag = pattern[cursor];
                if (flag == '-') { negative = true; ++cursor; continue; }
                if (!"idmsuxU"w.canFind(flag)) break;
                if (flag == 'U') { sawUnicode = true; nextUnicode = !negative; }
                else if (negative) disabled ~= flag;
                else enabled ~= flag;
                ++cursor;
            }
            if (sawUnicode && cursor < pattern.length && (pattern[cursor] == ')' || pattern[cursor] == ':')) {
                auto flags = enabled ~ (disabled.length ? "-"w ~ disabled : ""w);
                if (pattern[cursor] == ':') {
                    scopes ~= unicodeClasses;
                    result ~= flags.length ? "(?"w ~ flags ~ ":"w : "(?:"w;
                } else if (flags.length) result ~= "(?"w ~ flags ~ ")"w;
                unicodeClasses = nextUnicode;
                i = cursor + 1;
                continue;
            }
            scopes ~= unicodeClasses;
            result ~= c;
        } else if (c == '[') { ++classDepth; result ~= c; }
        else if (c == ']' && classDepth) { --classDepth; result ~= c; }
        else {
            if (!classDepth && c == '(') { scopes ~= unicodeClasses; if (i == pattern.length || pattern[i] != '?') ++captures; }
            if (!classDepth && c == ')' && scopes.length) { unicodeClasses = scopes[$ - 1]; scopes.length--; }
            result ~= c;
        }
    }
    return result.length ? result : ""w;
}
void z8_regex_setupMatcher(wstring value, wstring pattern) {
    import std.regex : regex, matchAll;
    auto key = z8_matcher_key(value);
    if (pattern is null) { synchronized (z8_matchers_lock) z8_matchers.remove(key); return; }
    auto state = new Z8MatcherState();
    state.subject = value;
    state.matches = matchAll(value, regex(z8_regex_pattern(pattern)));
    synchronized (z8_matchers_lock) z8_matchers[key] = state;
}
bool z8_regex_next(wstring value) {
    auto key = z8_matcher_key(value);
    synchronized (z8_matchers_lock) {
        auto found = key in z8_matchers;
        if (found is null) return false;
        auto state = *found;
        if (state.matches.empty) { z8_matchers.remove(key); return false; }
        state.current = state.matches.front;
        state.matches.popFront();
        state.hasCurrent = true;
        return true;
    }
}
wstring z8_regex_getGroup(wstring value, long groupNumber) {
    auto key = z8_matcher_key(value);
    synchronized (z8_matchers_lock) {
        auto found = key in z8_matchers;
        if (found is null || !(*found).hasCurrent) throw new Exception("matcher has no current match");
        auto state = *found;
        auto group = cast(int)groupNumber;
        if (group < 0 || cast(size_t)group >= state.current.length) throw new Exception("invalid regex group");
        return state.current[group];
    }
}
bool z8_regex_matches(wstring value, wstring pattern) {
    import std.regex : regex, matchFirst;
    auto input = z8_string_required(value);
    auto match = matchFirst(input, regex("^(?:"w ~ z8_regex_pattern(pattern) ~ ")$"w));
    return cast(bool)match && match.pre.length == 0 && match.post.length == 0;
}
private wstring z8_regex_replace(wstring value, wstring pattern, wstring replacement, bool all) {
    import std.regex : regex, matchAll;
    auto input = z8_string_required(value);
    replacement = z8_string_required(replacement);
    wstring output;
    size_t previous;
    foreach (match; matchAll(input, regex(z8_regex_pattern(pattern)))) {
        auto start = input.length - match.post.length - match.hit.length;
        output ~= input[previous .. start];
        size_t i;
        while (i < replacement.length) {
            auto c = replacement[i++];
            if (c == '\\') {
                if (i == replacement.length) throw new Exception("invalid regex replacement");
                output ~= replacement[i++];
            } else if (c == '$') {
                if (i < replacement.length && replacement[i] == '{') {
                    auto begin = ++i;
                    while (i < replacement.length && replacement[i] != '}') ++i;
                    if (i == begin || i == replacement.length) throw new Exception("invalid regex group reference");
                    if (!((replacement[begin] >= 'A' && replacement[begin] <= 'Z') || (replacement[begin] >= 'a' && replacement[begin] <= 'z')))
                        throw new Exception("invalid regex group reference");
                    foreach (nameChar; replacement[begin + 1 .. i])
                        if (!((nameChar >= 'A' && nameChar <= 'Z') || (nameChar >= 'a' && nameChar <= 'z') || (nameChar >= '0' && nameChar <= '9')))
                            throw new Exception("invalid regex group reference");
                    output ~= match[replacement[begin .. i]];
                    ++i;
                    continue;
                }
                if (i == replacement.length || replacement[i] < '0' || replacement[i] > '9') throw new Exception("invalid regex group reference");
                size_t group = replacement[i++] - '0';
                if (group >= match.length) throw new Exception("invalid regex group reference");
                while (i < replacement.length && replacement[i] >= '0' && replacement[i] <= '9') {
                    auto candidate = group * 10 + replacement[i] - '0';
                    if (candidate >= match.length) break;
                    group = candidate; ++i;
                }
                output ~= match[group];
            } else output ~= c;
        }
        previous = start + match.hit.length;
        if (!all) break;
    }
    output ~= input[previous .. $];
    return output.length ? output : ""w;
}
wstring z8_regex_replaceFirst(wstring value, wstring pattern, wstring replacement) { return z8_regex_replace(value, pattern, replacement, false); }
wstring z8_regex_replaceAll(wstring value, wstring pattern, wstring replacement) { return z8_regex_replace(value, pattern, replacement, true); }
Z8Array!wstring z8_regex_split(wstring value, wstring pattern, long limit = 0) {
    import std.regex : regex, matchAll;
    auto input = z8_string_required(value);
    auto expression = regex(z8_regex_pattern(pattern));
    auto result = new Z8Array!wstring();
    size_t previous;
    bool split;
    if (cast(int)limit != 1) foreach (match; matchAll(input, expression)) {
        auto start = input.length - match.post.length - match.hit.length;
        if (start == 0 && !match.hit.length) continue;
        if (cast(int)limit > 0 && result.z8_size() >= cast(int)limit - 1) break;
        result.z8_add(input[previous .. start]);
        previous = start + match.hit.length; split = true;
    }
    result.z8_add(input[previous .. $]);
    if (cast(int)limit == 0 && split) while (result.z8_size() && !result.z8_last().length) result.z8_removeAt(result.z8_size() - 1);
    foreach (ref item; result.items) item = z8_string_new(item);
    return result;
}
