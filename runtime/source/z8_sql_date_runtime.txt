enum Z8SqlDateKind { constant, group, choose, fromString, truncDay, truncHour, truncMinute, truncSecond, addDay, addHour, addMinute, addSecond, addSpan, subtractSpan }
class Z8SqlDate {
    Z8SqlDateKind kind;
    Z8Date constant;
    Z8SqlDate left, right;
    Z8SqlBool condition;
    Z8SqlString textOperand;
    Z8SqlInt quantity;
    Z8SqlDatespan spanOperand;
    bool spanFirst;
    this(Z8SqlDateKind kind, Z8Date constant = null, Z8SqlDate left = null,
         Z8SqlDate right = null, Z8SqlBool condition = null) {
        this.kind = kind; this.constant = constant; this.left = left; this.right = right; this.condition = condition;
    }
}
Z8SqlDate z8_sql_date_const(Z8Date value) { return new Z8SqlDate(Z8SqlDateKind.constant, z8_nonnull(value)); }
Z8SqlDate z8_sql_date_group(Z8SqlDate value) { return new Z8SqlDate(Z8SqlDateKind.group, null, z8_nonnull(value)); }
Z8SqlDate z8_sql_date_iif(Z8SqlBool condition, Z8SqlDate yes, Z8SqlDate no) {
    return new Z8SqlDate(Z8SqlDateKind.choose, null, yes, no, z8_nonnull(condition));
}
Z8SqlDate z8_sql_string_toDate(Z8SqlString value) {
    auto result = new Z8SqlDate(Z8SqlDateKind.fromString);
    result.textOperand = z8_nonnull(value); return result;
}
Z8SqlInt z8_sql_date_toInt(Z8SqlDate value) {
    auto result = new Z8SqlInt(Z8SqlIntKind.dateToInt);
    result.dateOperand = z8_nonnull(value); return result;
}
Z8SqlInt z8_sql_date_part(string part)(Z8SqlDate value) {
    auto result = new Z8SqlInt(Z8SqlIntKind.datePart);
    result.dateOperand = z8_nonnull(value); result.datePart = part.to!wstring; return result;
}
Z8SqlInt z8_sql_date_minute(Z8SqlDate value) { return z8_sql_date_part!"minute"(value); }
Z8SqlInt z8_sql_date_hour(Z8SqlDate value) { return z8_sql_date_part!"hour"(value); }
Z8SqlInt z8_sql_date_day(Z8SqlDate value) { return z8_sql_date_part!"day"(value); }
Z8SqlInt z8_sql_date_month(Z8SqlDate value) { return z8_sql_date_part!"month"(value); }
Z8SqlInt z8_sql_date_year(Z8SqlDate value) { return z8_sql_date_part!"year"(value); }
Z8SqlDate z8_sql_date_unary(Z8SqlDateKind kind)(Z8SqlDate value) { return new Z8SqlDate(kind, null, z8_nonnull(value)); }
Z8SqlDate z8_sql_date_truncDay(Z8SqlDate value) { return z8_sql_date_unary!(Z8SqlDateKind.truncDay)(value); }
Z8SqlDate z8_sql_date_truncHour(Z8SqlDate value) { return z8_sql_date_unary!(Z8SqlDateKind.truncHour)(value); }
Z8SqlDate z8_sql_date_truncMinute(Z8SqlDate value) { return z8_sql_date_unary!(Z8SqlDateKind.truncMinute)(value); }
Z8SqlDate z8_sql_date_truncSecond(Z8SqlDate value) { return z8_sql_date_unary!(Z8SqlDateKind.truncSecond)(value); }
Z8SqlDate z8_sql_date_add(Z8SqlDateKind kind)(Z8SqlDate value, Z8SqlInt quantity) {
    auto result = new Z8SqlDate(kind, null, z8_nonnull(value)); result.quantity = quantity; return result;
}
Z8SqlDate z8_sql_date_addDay(Z8SqlDate value, Z8SqlInt quantity) { return z8_sql_date_add!(Z8SqlDateKind.addDay)(value, quantity); }
Z8SqlDate z8_sql_date_addHour(Z8SqlDate value, Z8SqlInt quantity) { return z8_sql_date_add!(Z8SqlDateKind.addHour)(value, quantity); }
Z8SqlDate z8_sql_date_addMinute(Z8SqlDate value, Z8SqlInt quantity) { return z8_sql_date_add!(Z8SqlDateKind.addMinute)(value, quantity); }
Z8SqlDate z8_sql_date_addSecond(Z8SqlDate value, Z8SqlInt quantity) { return z8_sql_date_add!(Z8SqlDateKind.addSecond)(value, quantity); }
Z8SqlBool z8_sql_date_compare(string op)(Z8SqlDate left, Z8SqlDate right) {
    static assert(["==", "!=", "<", ">", "<=", ">="].canFind(op));
    auto result = new Z8SqlBool(Z8SqlBoolKind.dateCompare);
    result.dateLeft = z8_nonnull(left); result.dateRight = right; result.dateOperator = op; return result;
}
private wstring z8_sql_date_compare_format(Z8SqlBool value, Z8SqlVendor vendor) {
    auto left = z8_nonnull(value.dateLeft), right = z8_nonnull(value.dateRight);
    auto op = value.dateOperator;
    auto sqlOp = op == "==" ? "="w : op == "!=" ? "<>"w : op.to!wstring;
    bool leftConst = left.kind == Z8SqlDateKind.constant, rightConst = right.kind == Z8SqlDateKind.constant;
    if (leftConst && rightConst) {
        auto a = z8_date_ticks(z8_date_truncDay(left.constant)), b = z8_date_ticks(z8_date_truncDay(right.constant));
        auto result = op == "==" ? a == b : op == "!=" ? a != b : op == "<" ? a < b : op == ">" ? a > b : op == "<=" ? a <= b : a >= b;
        return result ? "1"w : "0"w;
    }
    if (leftConst || rightConst) {
        auto constant = leftConst ? left.constant : right.constant;
        auto field = z8_sql_date_format(leftConst ? right : left, vendor);
        auto ticks = z8_date_ticks(constant);
        if (ticks == z8_date_ticks(z8_date_min) || ticks == z8_date_ticks(z8_date_max))
            return "("w ~ (leftConst ? ticks.to!wstring ~ " "w ~ sqlOp ~ " "w ~ field : field ~ " "w ~ sqlOp ~ " "w ~ ticks.to!wstring) ~ ")"w;
        auto day = z8_date_truncDay(constant);
        auto start = z8_date_ticks(day).to!wstring, end = z8_date_ticks(z8_date_addDay(day, 1)).to!wstring;
        if (op == "==") return "("w ~ start ~ " <= "w ~ field ~ " and "w ~ field ~ " < "w ~ end ~ ")"w;
        if (op == "!=") return "("w ~ field ~ " < "w ~ start ~ " or "w ~ end ~ " <= "w ~ field ~ ")"w;
        if (leftConst) {
            if (op == "<") return "("w ~ end ~ " <= "w ~ field ~ ")"w;
            if (op == "<=") return "("w ~ start ~ " <= "w ~ field ~ ")"w;
            if (op == ">") return "("w ~ start ~ " > "w ~ field ~ ")"w;
            return "("w ~ end ~ " > "w ~ field ~ ")"w;
        }
        if (op == "<") return "("w ~ field ~ " < "w ~ start ~ ")"w;
        if (op == "<=") return "("w ~ field ~ " < "w ~ end ~ ")"w;
        if (op == ">") return "("w ~ field ~ " >= "w ~ end ~ ")"w;
        return "("w ~ field ~ " >= "w ~ start ~ ")"w;
    }
    // Rel rewrites both operands when formatted, so repeated formatting nests TruncDay.
    value.dateLeft = z8_sql_date_truncDay(left); value.dateRight = z8_sql_date_truncDay(right);
    return z8_sql_date_format(value.dateLeft, vendor) ~ sqlOp ~ z8_sql_date_format(value.dateRight, vendor);
}
wstring z8_sql_date_format(Z8SqlDate value, Z8SqlVendor vendor = Z8SqlVendor.Postgres) {
    z8_nonnull(value);
    final switch (value.kind) {
        case Z8SqlDateKind.constant: return z8_date_ticks(value.constant).to!wstring;
        case Z8SqlDateKind.group: return "("w ~ z8_sql_date_format(value.left, vendor) ~ ")"w;
        case Z8SqlDateKind.choose:
            return z8_sql_bool_case(z8_sql_bool_format(value.condition, true, vendor), z8_sql_date_format(value.left, vendor), z8_sql_date_format(value.right, vendor));
        case Z8SqlDateKind.fromString:
            auto text = z8_sql_string_format(value.textOperand, vendor);
            if (vendor == Z8SqlVendor.Oracle) return text;
            if (vendor == Z8SqlVendor.Postgres) return text ~ "::timestamp with time zone"w;
            if (vendor == Z8SqlVendor.SqlServer) return "CONVERT([datetime],"w ~ text ~ ",(103))"w;
            throw new Exception("unsupported SQL database vendor");
        case Z8SqlDateKind.truncDay:
            if (value.left.kind == Z8SqlDateKind.constant) {
                auto date = z8_nonnull(value.left.constant);
                auto truncated = z8_date_truncDay(date);
                date.value = truncated.value;
                return z8_sql_date_format(value.left, vendor);
            }
            auto dayText = z8_sql_date_format(value.left, vendor);
            return "("w ~ dayText ~ " - MOD("w ~ dayText ~ " + (10800000), 86400000))"w;
        case Z8SqlDateKind.truncHour: case Z8SqlDateKind.truncMinute: case Z8SqlDateKind.truncSecond:
            auto text = z8_sql_date_format(value.left, vendor);
            auto unit = value.kind == Z8SqlDateKind.truncHour ? "3600000"w : value.kind == Z8SqlDateKind.truncMinute ? "60000"w : "1000"w;
            return "("w ~ text ~ " - MOD("w ~ text ~ ", "w ~ unit ~ "))"w;
        case Z8SqlDateKind.addDay: case Z8SqlDateKind.addHour: case Z8SqlDateKind.addMinute: case Z8SqlDateKind.addSecond:
            auto text = z8_sql_date_format(value.left, vendor), quantity = z8_sql_int_format(value.quantity, vendor);
            auto unit = value.kind == Z8SqlDateKind.addDay ? " 86400000"w : value.kind == Z8SqlDateKind.addHour ? "3600000"w : value.kind == Z8SqlDateKind.addMinute ? "60000"w : " 1000"w;
            return "("w ~ text ~ " + "w ~ quantity ~ " *"w ~ unit ~ (vendor == Z8SqlVendor.Postgres ? "::bigint"w : ""w) ~ ")"w;
        case Z8SqlDateKind.addSpan: case Z8SqlDateKind.subtractSpan:
            if (value.spanFirst) return z8_sql_datespan_format(value.spanOperand, vendor) ~ " + "w ~ z8_sql_date_format(value.left, vendor);
            if (value.kind == Z8SqlDateKind.addSpan) return z8_sql_date_format(value.left, vendor) ~ " + "w ~ z8_sql_datespan_format(value.spanOperand, vendor);
            return z8_sql_date_format(value.left, vendor) ~ " - ("w ~ z8_sql_datespan_format(value.spanOperand, vendor) ~ ")"w;
    }
}
