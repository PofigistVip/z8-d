// The original primary class is the common base for Z8 scalar values.
enum Z8PrimaryKind { integer, boolean, decimal, text, guid, binary, datespan, geometry, date, file, user }

class Z8Primary {
    Z8PrimaryKind kind;
    long integerValue;
    bool booleanValue;
    Z8Decimal decimalValue;
    wstring textValue;
    Z8Guid guidValue;
    Z8Binary binaryValue;
    Z8Datespan datespanValue;
    Z8Geometry geometryValue;
    Z8Date dateValue;
    Z8File fileValue;
    Z8Object userValue;

    override bool opEquals(Object other) {
        auto right = cast(Z8Primary)other;
        if (right is null || kind != right.kind) return false;
        final switch (kind) {
            case Z8PrimaryKind.integer: return integerValue == right.integerValue;
            case Z8PrimaryKind.boolean: return booleanValue == right.booleanValue;
            case Z8PrimaryKind.decimal: return decimalValue == right.decimalValue;
            case Z8PrimaryKind.text: return z8_string_equals(textValue, right.textValue);
            case Z8PrimaryKind.guid: return z8_guid_compare!"=="(guidValue, right.guidValue);
            case Z8PrimaryKind.binary: return binaryValue is right.binaryValue;
            case Z8PrimaryKind.datespan: return z8_datespan_binary!"=="(datespanValue, right.datespanValue);
            case Z8PrimaryKind.geometry: return z8_geometry_equals(geometryValue, right.geometryValue);
            case Z8PrimaryKind.date: return z8_date_binary!"=="(dateValue, right.dateValue);
            case Z8PrimaryKind.file: return fileValue == right.fileValue;
            case Z8PrimaryKind.user: return userValue is right.userValue || (userValue !is null && userValue.z8_user_primary_equals_impl(right.userValue));
        }
    }
}

// Java collections first compare references, then call equals. A BL primary
// wrapper can be recreated while retaining the same underlying object.
bool z8_primary_collection_equals(Z8Primary left, Z8Primary right) {
    if (left is right) return true;
    if (left is null || right is null || left.kind != right.kind) return false;
    final switch (left.kind) {
        case Z8PrimaryKind.guid: if (left.guidValue is right.guidValue) return true; break;
        case Z8PrimaryKind.binary: if (left.binaryValue is right.binaryValue) return true; break;
        case Z8PrimaryKind.datespan: if (left.datespanValue is right.datespanValue) return true; break;
        case Z8PrimaryKind.geometry: if (left.geometryValue is right.geometryValue) return true; break;
        case Z8PrimaryKind.date: if (left.dateValue is right.dateValue) return true; break;
        case Z8PrimaryKind.file: if (left.fileValue is right.fileValue) return true; break;
        case Z8PrimaryKind.user: if (left.userValue is right.userValue) return true; break;
        case Z8PrimaryKind.integer: case Z8PrimaryKind.boolean: case Z8PrimaryKind.decimal: case Z8PrimaryKind.text: break;
    }
    return left == right;
}

Z8Primary z8_primary_box(T)(T value) {
    static if (is(T == Z8Optional!long) || is(T == Z8Optional!bool) || is(T == Z8Optional!Z8Decimal)) {
        return value.present ? z8_primary_box(value.value) : null;
    } else static if (is(T == wstring)) {
        if (value is null) return null;
        auto boxed = new Z8Primary();
        boxed.kind = Z8PrimaryKind.text;
        boxed.textValue = value;
        return boxed;
    } else static if (is(T == Z8Guid)) {
        if (value is null) return null;
        auto boxed = new Z8Primary();
        boxed.kind = Z8PrimaryKind.guid;
        boxed.guidValue = value;
        return boxed;
    } else static if (is(T == Z8Binary)) {
        if (value is null) return null;
        auto boxed = new Z8Primary();
        boxed.kind = Z8PrimaryKind.binary;
        boxed.binaryValue = value;
        return boxed;
    } else static if (is(T == Z8Datespan)) {
        if (value is null) return null;
        auto boxed = new Z8Primary();
        boxed.kind = Z8PrimaryKind.datespan;
        boxed.datespanValue = value;
        return boxed;
    } else static if (is(T == Z8Geometry)) {
        if (value is null) return null;
        auto boxed = new Z8Primary();
        boxed.kind = Z8PrimaryKind.geometry;
        boxed.geometryValue = value;
        return boxed;
    } else static if (is(T == Z8Date)) {
        if (value is null) return null;
        auto boxed = new Z8Primary();
        boxed.kind = Z8PrimaryKind.date;
        boxed.dateValue = value;
        return boxed;
    } else static if (is(T == Z8File)) {
        if (value is null) return null;
        auto boxed = new Z8Primary();
        boxed.kind = Z8PrimaryKind.file;
        boxed.fileValue = value;
        return boxed;
    } else static if (is(T : Z8Object)) {
        if (value is null) return null;
        auto boxed = new Z8Primary();
        boxed.kind = Z8PrimaryKind.user;
        boxed.userValue = value;
        return boxed;
    } else {
        auto boxed = new Z8Primary();
        static if (is(T == long)) { boxed.kind = Z8PrimaryKind.integer; boxed.integerValue = value; }
        else static if (is(T == bool)) { boxed.kind = Z8PrimaryKind.boolean; boxed.booleanValue = value; }
        else static if (is(T == Z8Decimal)) { boxed.kind = Z8PrimaryKind.decimal; boxed.decimalValue = value; }
        else static assert(0, "unsupported primary value");
        return boxed;
    }
}

wstring z8_primary_to_string(Z8Primary value) {
    if (value is null) throw new Exception("null primary");
    final switch (value.kind) {
        case Z8PrimaryKind.integer: return value.integerValue.to!string.to!wstring;
        case Z8PrimaryKind.boolean: return value.booleanValue ? "true"w : "false"w;
        case Z8PrimaryKind.decimal: return value.decimalValue.z8_string();
        case Z8PrimaryKind.text: return value.textValue;
        case Z8PrimaryKind.guid: return z8_guid_to_string(value.guidValue);
        case Z8PrimaryKind.binary: return z8_binary_to_string(value.binaryValue);
        case Z8PrimaryKind.datespan: return z8_datespan_to_string(value.datespanValue);
        case Z8PrimaryKind.geometry: return z8_geometry_to_string(value.geometryValue);
        case Z8PrimaryKind.date: return z8_date_toString(value.dateValue);
        case Z8PrimaryKind.file: return z8_file_to_string(value.fileValue);
        case Z8PrimaryKind.user: return z8_nonnull(value.userValue).z8_user_primary_string_impl();
    }
}

// Java BigInteger.hashCode hashes unsigned 32-bit magnitude words, most significant first.
private uint z8_bigint_java_hash(BigInt number) {
    auto negative = number < 0;
    auto magnitude = negative ? -number : number;
    auto base = BigInt("4294967296");
    uint[] words;
    while (magnitude != 0) {
        words ~= (magnitude % base).to!string.to!uint;
        magnitude /= base;
    }
    uint hash;
    for (size_t i = words.length; i > 0; --i) hash = 31u * hash + words[i - 1];
    return negative ? 0u - hash : hash;
}

long z8_primary_hash(Z8Primary value) {
    if (value is null) throw new Exception("null primary");
    uint hash;
    final switch (value.kind) {
        case Z8PrimaryKind.integer:
            auto bits = cast(ulong)value.integerValue;
            hash = cast(uint)(bits ^ (bits >> 32));
            break;
        case Z8PrimaryKind.boolean: hash = value.booleanValue ? 1231u : 1237u; break;
        case Z8PrimaryKind.decimal:
            if (value.decimalValue.nan) return 0;
            hash = 31u * z8_bigint_java_hash(value.decimalValue.coefficient) + cast(uint)value.decimalValue.scale;
            break;
        case Z8PrimaryKind.text:
            foreach (wchar c; value.textValue) hash = 31u * hash + c;
            break;
        case Z8PrimaryKind.guid: return z8_guid_hash(value.guidValue);
        case Z8PrimaryKind.binary: return z8_binary_hash(value.binaryValue);
        case Z8PrimaryKind.datespan: return z8_datespan_hashCode(value.datespanValue);
        case Z8PrimaryKind.geometry: return z8_geometry_hashCode(value.geometryValue);
        case Z8PrimaryKind.date: return z8_date_hashCode(value.dateValue);
        case Z8PrimaryKind.file: return z8_file_hashCode(value.fileValue);
        case Z8PrimaryKind.user: return z8_nonnull(value.userValue).z8_user_primary_hash_impl();
    }
    return cast(long)cast(int)hash;
}

long z8_primary_hash_value(T)(T value) { return z8_primary_hash(z8_primary_box(value)); }
wstring z8_user_primary_string(T)(T value) if (is(T : Z8Object)) { return z8_nonnull(value).z8_user_primary_string_impl(); }
long z8_user_primary_hash(T)(T value) if (is(T : Z8Object)) { return z8_nonnull(value).z8_user_primary_hash_impl(); }

bool z8_primary_is(T)(Z8Primary value) {
    if (value is null) return false;
    static if (is(T == Z8Primary)) return true;
    else static if (is(T == long)) return value.kind == Z8PrimaryKind.integer;
    else static if (is(T == bool)) return value.kind == Z8PrimaryKind.boolean;
    else static if (is(T == Z8Decimal)) return value.kind == Z8PrimaryKind.decimal;
    else static if (is(T == wstring)) return value.kind == Z8PrimaryKind.text;
    else static if (is(T == Z8Guid)) return value.kind == Z8PrimaryKind.guid;
    else static if (is(T == Z8Binary)) return value.kind == Z8PrimaryKind.binary;
    else static if (is(T == Z8Datespan)) return value.kind == Z8PrimaryKind.datespan;
    else static if (is(T == Z8Geometry)) return value.kind == Z8PrimaryKind.geometry;
    else static if (is(T == Z8Date)) return value.kind == Z8PrimaryKind.date;
    else static if (is(T == Z8File)) return value.kind == Z8PrimaryKind.file;
    else static if (is(T : Z8Object)) return value.kind == Z8PrimaryKind.user && cast(T)value.userValue !is null;
    else static assert(0, "unsupported primary instanceof target");
}

Z8Optional!T z8_primary_cast(T)(Z8Primary value) if (is(T == long) || is(T == bool) || is(T == Z8Decimal)) {
    if (value is null) return Z8Optional!T.init;
    if (!z8_primary_is!T(value)) throw new Exception("invalid primary cast");
    static if (is(T == long)) return z8_optional_wrap(value.integerValue);
    else static if (is(T == bool)) return z8_optional_wrap(value.booleanValue);
    else return z8_optional_wrap(value.decimalValue);
}
T z8_primary_cast(T)(Z8Primary value) if (is(T == wstring) || is(T == Z8Guid) || is(T == Z8Binary) || is(T == Z8Datespan) || is(T == Z8Geometry) || is(T == Z8Date) || is(T == Z8File)) {
    if (value is null) return null;
    if (!z8_primary_is!T(value)) throw new Exception("invalid primary cast");
    static if (is(T == wstring)) return value.textValue;
    else static if (is(T == Z8Guid)) return value.guidValue;
    else static if (is(T == Z8Binary)) return value.binaryValue;
    else static if (is(T == Z8Datespan)) return value.datespanValue;
    else static if (is(T == Z8Geometry)) return value.geometryValue;
    else static if (is(T == Z8Date)) return value.dateValue;
    else return value.fileValue;
}
T z8_primary_cast(T)(Z8Primary value) if (is(T : Z8Object)) {
    if (value is null) return null;
    if (!z8_primary_is!T(value)) throw new Exception("invalid primary cast");
    return cast(T)value.userValue;
}
