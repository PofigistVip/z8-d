// A TrueType font is supplied by the host, rather than taken from an OS path.
// The template uses the same Roboto font as the original Z8 distribution.
private __gshared ubyte[] z8_pdf_font_data;
private __gshared Object z8_pdf_font_lock;
shared static this() { z8_pdf_font_lock = new Object(); }
void z8_export_configure_pdf_font(string path) {
    import std.file : read;
    auto bytes = cast(ubyte[])read(path, 16 * 1024 * 1024);
    auto font = new Z8PdfFont(bytes); // Validate before publishing it.
    synchronized (z8_pdf_font_lock) z8_pdf_font_data = bytes.dup;
}

private class Z8PdfFont {
    ubyte[] data;
    uint cmap, hmtx, head, hhea, glyphs;
    ushort units, metrics;
    this(ubyte[] bytes) {
        data = bytes;
        if (u32(0) != 0x00010000) throw new Exception("PDF font must be a TrueType font");
        auto count = u16(4);
        uint cmapTable;
        foreach (i; 0 .. count) {
            auto at = 12 + i * 16;
            auto tag = u32(at), offset = u32(at + 8), length = u32(at + 12);
            check(offset, length);
            switch (tag) {
                case 0x636d6170: cmapTable = offset; break; // cmap
                case 0x68656164: head = offset; break;
                case 0x68686561: hhea = offset; break;
                case 0x686d7478: hmtx = offset; break;
                case 0x6d617870: glyphs = offset; break;
                default: break;
            }
        }
        if (!cmapTable || !head || !hhea || !hmtx || !glyphs) throw new Exception("PDF font is missing required tables");
        units = u16(head + 18); metrics = u16(hhea + 34);
        if (!units || !metrics || metrics > u16(glyphs + 4)) throw new Exception("invalid PDF font metrics");
        auto cmapCount = u16(cmapTable + 2);
        foreach (i; 0 .. cmapCount) {
            auto record = cmapTable + 4 + i * 8;
            auto platform = u16(record), encoding = u16(record + 2), subtable = cmapTable + u32(record + 4);
            if (platform == 3 && encoding == 10 && u16(subtable) == 12) { cmap = subtable; break; }
        }
        if (!cmap) throw new Exception("PDF font requires a Unicode format 12 cmap");
        check(cmap, u32(cmap + 4));
    }
    void check(size_t offset, size_t length) {
        if (offset > data.length || length > data.length - offset) throw new Exception("invalid PDF font");
    }
    ushort u16(size_t offset) { check(offset, 2); return cast(ushort)((cast(uint)data[offset] << 8) | data[offset + 1]); }
    uint u32(size_t offset) { return (cast(uint)u16(offset) << 16) | u16(offset + 2); }
    short s16(size_t offset) { return cast(short)u16(offset); }
    ushort glyph(dchar ch) {
        auto n = u32(cmap + 12);
        size_t low, high = n;
        while (low < high) {
            auto mid = low + (high - low) / 2, at = cmap + 16 + mid * 12;
            auto start = u32(at), finish = u32(at + 4);
            if (ch < start) high = mid;
            else if (ch > finish) low = mid + 1;
            else {
                auto result = u32(at + 8) + ch - start;
                if (result >= u16(glyphs + 4)) throw new Exception("invalid PDF font glyph");
                return cast(ushort)result;
            }
        }
        return 0;
    }
    int width(ushort glyph) {
        auto metric = glyph < metrics ? glyph : metrics - 1;
        return cast(int)((cast(long)u16(hmtx + metric * 4) * 1000 + units / 2) / units);
    }
    int scaled(short value) { return cast(int)(cast(long)value * 1000 / units); }
}

private string z8_pdf_hex(uint value, size_t digits) {
    enum hex = "0123456789ABCDEF";
    char[] result = new char[digits];
    foreach_reverse (i; 0 .. digits) { result[i] = hex[value & 15]; value >>= 4; }
    return cast(string)result;
}

private struct Z8PdfPageLayout {
    int width, height, left, right, top, bottom;
    int detailHeight = 18;
    string detailAlign = "middle";
}
// PrintOptions and PageFormat in the Java server use millimeters.
private Z8PdfPageLayout z8_pdf_page_layout(wstring options) {
    import std.json : JSONType;
    import std.math : isFinite, round;
    import std.string : strip, toLower;
    double width = 210, height = 297, left = 10, right = 10, top = 10, bottom = 10;
    bool portrait;
    int detailHeight = 18;
    string detailAlign = "middle";
    if (options !is null && options.length) {
        auto json = z8_json_parse(options.to!string);
        if (json.type != JSONType.object) throw new Exception("invalid print options");
        if (auto format = "pageFormat" in json.object) {
            if (format.type != JSONType.string) throw new Exception("invalid page format");
            switch (format.str.toLower()) {
                case "a0": width = 841; height = 1189; break;
                case "a1": width = 594; height = 841; break;
                case "a2": width = 420; height = 594; break;
                case "a3": width = 297; height = 420; break;
                case "a4": width = 210; height = 297; break;
                case "a5": width = 148; height = 210; break;
                case "a6": width = 105; height = 148; break;
                case "c3": width = 324; height = 458; break;
                case "c4": width = 229; height = 324; break;
                case "c5": width = 162; height = 229; break;
                case "c6": width = 114; height = 162; break;
                case "letter": width = 215.9; height = 355.6; break;
                default: throw new Exception("unknown page format");
            }
        }
        if (auto orientation = "pageOrientation" in json.object) {
            if (orientation.type != JSONType.string) throw new Exception("invalid page orientation");
            auto value = orientation.str.toLower();
            if (value != "portrait" && value != "landscape") throw new Exception("unknown page orientation");
            portrait = value == "portrait";
        }
        double margin(string name, double fallback) {
            auto value = name in json.object;
            if (value is null) return fallback;
            if (value.type != JSONType.string && value.type != JSONType.integer &&
                value.type != JSONType.uinteger && value.type != JSONType.float_)
                throw new Exception("invalid page margin");
            auto number = z8_json_value_text(*value).to!string.strip().to!double;
            if (!isFinite(number) || number < 0 || number > 10000) throw new Exception("invalid page margin");
            return number;
        }
        left = margin("leftMargin", left); right = margin("rightMargin", right);
        top = margin("topMargin", top); bottom = margin("bottomMargin", bottom);
        if (auto alignment = "detailVertAlign" in json.object) {
            if (alignment.type != JSONType.string) throw new Exception("invalid detail alignment");
            detailAlign = alignment.str.toLower();
            if (detailAlign != "top" && detailAlign != "middle" && detailAlign != "bottom")
                throw new Exception("unknown detail alignment");
        }
        if (auto row = "detailRowHeight" in json.object) {
            if (row.type != JSONType.null_) {
                if (row.type != JSONType.string && row.type != JSONType.integer &&
                    row.type != JSONType.uinteger && row.type != JSONType.float_)
                    throw new Exception("invalid detail row height");
                auto value = z8_json_value_text(*row).to!string.strip().toLower();
                size_t split = value.length;
                while (split && value[split - 1] >= 'a' && value[split - 1] <= 'z') --split;
                auto units = value[split .. $];
                auto number = value[0 .. split].strip().to!double;
                double factor;
                switch (units) {
                    case "": case "pt": case "px": factor = 1; break;
                    case "pc": factor = 12; break;
                    case "in": factor = 72; break;
                    case "mm": factor = 72.0 / 25.4; break;
                    case "cm": factor = 72.0 / 2.54; break;
                    default: throw new Exception("invalid detail row height unit");
                }
                number *= factor;
                if (!isFinite(number) || number < 1 || number > 10000) throw new Exception("invalid detail row height");
                detailHeight = cast(int)round(number);
            }
        }
    }
    if (!portrait) { auto temporary = width; width = height; height = temporary; }
    int points(double millimeters) { return cast(int)round(millimeters * 72.0 / 25.4); }
    auto result = Z8PdfPageLayout(points(width), points(height), points(left), points(right), points(top), points(bottom));
    result.detailHeight = detailHeight; result.detailAlign = detailAlign;
    if (result.left + result.right >= result.width || result.top + result.bottom >= result.height)
        throw new Exception("page margins exceed page size");
    return result;
}

private ubyte[] z8_export_pdf(wstring[] ids, wstring[] headers, wstring[] types, JSONValue rows,
                              wstring options, int[] requestedWidths, int[] minimumWidths) {
    import std.json : JSONType;
    import std.math : round;
    ubyte[] bytes;
    synchronized (z8_pdf_font_lock) bytes = z8_pdf_font_data;
    if (bytes is null) throw new Exception("PDF export font is not configured");
    auto font = new Z8PdfFont(bytes);
    ushort[dchar] cids;
    dchar[] characters = [cast(dchar)0];
    string text(wstring value) {
        string result;
        foreach (dchar ch; value) {
            auto found = ch in cids;
            ushort cid;
            if (found is null) {
                if (characters.length >= 65536) throw new Exception("too many PDF characters");
                cid = cast(ushort)characters.length; characters ~= ch; cids[ch] = cid;
            } else cid = *found;
            result ~= z8_pdf_hex(cid, 4);
        }
        return result;
    }
    // Each cell is clipped so a long value cannot overwrite its neighbors.
    auto layout = z8_pdf_page_layout(options);
    int pageWidth = layout.width, pageHeight = layout.height, headerHeight = 18;
    int rowLimit = (pageHeight - layout.top - layout.bottom - headerHeight) / layout.detailHeight;
    if (rowLimit < 1) throw new Exception("page has no room for export rows");
    auto available = pageWidth - layout.left - layout.right;
    assert(requestedWidths.length == ids.length && minimumWidths.length == ids.length);
    double total = 0;
    foreach (width; requestedWidths) total += width;
    double[] scaled = new double[requestedWidths.length];
    double fixedWidth = 0, scaledTotal = 0;
    foreach (i, width; requestedWidths) {
        auto proportional = total > 0 ? available * cast(double)width / total : cast(double)available / requestedWidths.length;
        scaled[i] = proportional < minimumWidths[i] ? minimumWidths[i] : proportional;
        scaledTotal += scaled[i];
        if (scaled[i] == minimumWidths[i]) fixedWidth += scaled[i];
    }
    if (scaledTotal > available && scaledTotal > fixedWidth) {
        auto factor = (available - fixedWidth) / (scaledTotal - fixedWidth);
        scaledTotal = 0;
        foreach (i; 0 .. scaled.length) {
            if (scaled[i] != minimumWidths[i]) {
                scaled[i] *= factor;
                if (scaled[i] < minimumWidths[i]) scaled[i] = minimumWidths[i];
            }
            scaledTotal += scaled[i];
        }
    }
    if (scaledTotal > available) foreach (ref width; scaled) width *= available / scaledTotal;
    int[] cellWidths, positions;
    double cumulative = 0;
    int previous;
    foreach (width; scaled) {
        cumulative += width;
        auto next = cast(int)round(cumulative);
        positions ~= layout.left + previous;
        cellWidths ~= next - previous;
        previous = next;
        if (cellWidths[$ - 1] < 8) throw new Exception("PDF column is too narrow");
    }
    string[] contents;
    size_t position;
    do {
        string content;
        void line(wstring[] values, int row) {
            auto height = row == 0 ? headerHeight : layout.detailHeight;
            foreach (i, value; values) {
                auto x = positions[i], cellWidth = cellWidths[i];
                auto y = pageHeight - layout.top - headerHeight - row * layout.detailHeight;
                auto baseline = row == 0 ? 5 : layout.detailAlign == "top" ? height - 12 :
                    layout.detailAlign == "bottom" ? 3 : (height - 9) / 2 + 1;
                content ~= "q " ~ x.to!string ~ " " ~ y.to!string ~ " " ~ cellWidth.to!string ~ " " ~ height.to!string ~
                    " re W n BT /F1 9 Tf " ~ (x + 2).to!string ~ " " ~ (y + baseline).to!string ~ " Td <" ~ text(value) ~ "> Tj ET Q\n";
            }
        }
        line(headers, 0);
        size_t end = position + rowLimit; if (end > rows.array.length) end = rows.array.length;
        int lineNumber = 1;
        for (; position < end; ++position) {
            auto row = rows.array[position];
            if (row.type != JSONType.object) throw new Exception("invalid export row");
            wstring[] values;
            foreach (i, id; ids) {
                auto value = id.to!string in row.object;
                values ~= value is null ? ""w : z8_export_display_value(types[i], *value);
            }
            line(values, lineNumber++);
        }
        contents ~= content;
    } while (position < rows.array.length);

    string cidMap = "\0\0", widths;
    string unicode = "/CIDInit /ProcSet findresource begin\n12 dict begin\nbegincmap\n/CIDSystemInfo << /Registry (Adobe) /Ordering (UCS) /Supplement 0 >> def\n/CMapName /Z8Unicode def\n/CMapType 2 def\n1 begincodespacerange\n<0000> <FFFF>\nendcodespacerange\n";
    for (size_t start = 1; start < characters.length;) {
        auto end = start + 100; if (end > characters.length) end = characters.length;
        unicode ~= (end - start).to!string ~ " beginbfchar\n";
        foreach (i; start .. end) {
            auto ch = characters[i], glyph = font.glyph(ch);
            cidMap ~= cast(char)(glyph >> 8); cidMap ~= cast(char)(glyph & 255);
            widths ~= font.width(glyph).to!string ~ " ";
            string target = ch <= 0xffff ? z8_pdf_hex(ch, 4) :
                z8_pdf_hex(0xd800 + ((ch - 0x10000) >> 10), 4) ~ z8_pdf_hex(0xdc00 + ((ch - 0x10000) & 1023), 4);
            unicode ~= "<" ~ z8_pdf_hex(cast(uint)i, 4) ~ "> <" ~ target ~ ">\n";
        }
        unicode ~= "endbfchar\n"; start = end;
    }
    unicode ~= "endcmap\nCMapName currentdict /CMap defineresource pop\nend\nend\n";
    string stream(string data, string attributes = "") {
        return "<< /Length " ~ data.length.to!string ~ " " ~ attributes ~ " >>\nstream\n" ~ data ~ "\nendstream";
    }
    string[] objects = new string[8 + contents.length * 2];
    objects[0] = "<< /Type /Catalog /Pages 2 0 R >>";
    string kids;
    foreach (i; 0 .. contents.length) kids ~= (9 + i * 2).to!string ~ " 0 R ";
    objects[1] = "<< /Type /Pages /Count " ~ contents.length.to!string ~ " /Kids [" ~ kids ~ "] >>";
    objects[2] = "<< /Type /Font /Subtype /Type0 /BaseFont /Z8Roboto /Encoding /Identity-H /DescendantFonts [4 0 R] /ToUnicode 8 0 R >>";
    objects[3] = "<< /Type /Font /Subtype /CIDFontType2 /BaseFont /Z8Roboto /CIDSystemInfo << /Registry (Adobe) /Ordering (Identity) /Supplement 0 >> /FontDescriptor 5 0 R /CIDToGIDMap 7 0 R /W [1 [" ~ widths ~ "]] >>";
    objects[4] = "<< /Type /FontDescriptor /FontName /Z8Roboto /Flags 32 /FontBBox [" ~
        font.scaled(font.s16(font.head + 36)).to!string ~ " " ~ font.scaled(font.s16(font.head + 38)).to!string ~ " " ~
        font.scaled(font.s16(font.head + 40)).to!string ~ " " ~ font.scaled(font.s16(font.head + 42)).to!string ~
        "] /ItalicAngle 0 /Ascent " ~ font.scaled(font.s16(font.hhea + 4)).to!string ~ " /Descent " ~
        font.scaled(font.s16(font.hhea + 6)).to!string ~ " /CapHeight 700 /StemV 80 /FontFile2 6 0 R >>";
    objects[5] = stream(cast(string)bytes, "/Length1 " ~ bytes.length.to!string);
    objects[6] = stream(cidMap); objects[7] = stream(unicode);
    foreach (i, content; contents) {
        objects[8 + i * 2] = "<< /Type /Page /Parent 2 0 R /MediaBox [0 0 " ~ pageWidth.to!string ~ " " ~ pageHeight.to!string ~
            "] /Resources << /Font << /F1 3 0 R >> >> /Contents " ~ (10 + i * 2).to!string ~ " 0 R >>";
        objects[9 + i * 2] = stream(content);
    }
    string pdf = "%PDF-1.4\n%\xe2\xe3\xcf\xd3\n", offsets;
    foreach (i, object; objects) {
        auto offset = pdf.length.to!string;
        if (offset.length > 10) throw new Exception("PDF is too large");
        offsets ~= "0000000000"[0 .. 10 - offset.length] ~ offset ~ " 00000 n \n";
        pdf ~= (i + 1).to!string ~ " 0 obj\n" ~ object ~ "\nendobj\n";
    }
    auto xref = pdf.length;
    pdf ~= "xref\n0 " ~ (objects.length + 1).to!string ~ "\n0000000000 65535 f \n" ~ offsets ~
        "trailer\n<< /Size " ~ (objects.length + 1).to!string ~ " /Root 1 0 R >>\nstartxref\n" ~ xref.to!string ~ "\n%%EOF\n";
    if (pdf.length > z8_temporary_archives_bytes_limit) throw new Exception("PDF is too large");
    return cast(ubyte[])pdf;
}
