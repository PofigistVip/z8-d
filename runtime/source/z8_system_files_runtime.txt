// Native members of the framework's SystemFiles table (Files.java).
private enum z8_system_files_name = "org.zenframework.z8.server.base.table.system.Files"w;
private class Z8SystemFilesState { Z8Object[7] fields; }
private void z8_system_files_new(Z8Object self) {
    enum base = "org.zenframework.z8.server.base.table.value."w;
    enum kinds = ["StringField"w, "BinaryField"w, "StringField"w, "IntegerField"w,
        "DatetimeField"w, "DatetimeField"w, "IntegerField"w];
    enum indexes = ["name"w, "data"w, "path"w, "size"w, "time"w, "lastModified"w, "location"w];
    enum names = ["Name"w, "File"w, "Path"w, "Size"w, "Time"w, "Last modified"w, "Location"w];
    auto state = new Z8SystemFilesState();
    foreach (i; 0 .. state.fields.length)
        state.fields[i] = z8_table_system_field(self, base ~ kinds[i], indexes[i], names[i]);
    self.z8_set_native_state(z8_system_files_name, state);
}
private Z8SystemFilesState z8_system_files_state(Z8Object self) {
    auto state = cast(Z8SystemFilesState)z8_nonnull(self).z8_get_native_state(z8_system_files_name);
    if (state is null) throw new Exception("SystemFiles members are not initialized");
    return state;
}
private Z8Object* z8_system_files_field_address(size_t index)(Z8Object self) {
    return &z8_system_files_state(self).fields[index];
}
void z8_system_files_configure(Z8Object self) {
    auto fields = z8_system_files_state(self).fields;
    z8_field_state(fields[0]).length = z8_optional_wrap(512L);
    z8_field_state(fields[2]).length = z8_optional_wrap(512L);
}
private void z8_system_files_on_new(Z8Object self) {
    auto fields = z8_system_files_state(self).fields;
    auto now = z8_date_now();
    z8_field_set(fields[4], z8_primary_box(now));
    z8_field_set(fields[5], z8_primary_box(now));
}
// AttachmentProcessor uses these operations within its caller's transaction.
// Content can be stored in the database or under the configured storage root.
void z8_system_files_write(Z8File file, bool create = true) {
    file = z8_nonnull(file);
    if (file.id is null || file.id == z8_guid_zero()) throw new Exception("file ID is required");
    auto table = z8_native_new(z8_system_files_name); table.z8_ensure();
    auto fields = z8_system_files_state(table).fields;
    z8_field_set(fields[0], z8_primary_box(file.name));
    auto binary = z8_file_to_binary(file);
    if (z8_application_configuration().filesSaveOnDisk) {
        import std.file : mkdirRecurse, write;
        import std.path : dirName;
        ubyte[] bytes;
        { scope(exit) z8_binary_close(binary); bytes = z8_binary_snapshot(binary, size_t.max); }
        auto path = z8_file_system_path(file); auto parent = dirName(path);
        if (parent.length) mkdirRecurse(parent);
        write(path, bytes);
        z8_field_set(fields[6], z8_primary_box(1L));
    } else {
        z8_field_set(fields[1], z8_primary_box(binary));
        z8_field_set(fields[6], z8_primary_box(0L));
    }
    z8_field_set(fields[2], z8_primary_box(file.path));
    z8_field_set(fields[3], z8_primary_box(file.size));
    z8_field_set(fields[4], z8_primary_box(file.time));
    z8_field_set(fields[5], z8_primary_box(file.time));
    if (create) z8_query_create_id(table, file.id);
    else if (!z8_query_update_id_run(table, file.id, true)) throw new Exception("file was not updated");
}
void z8_system_files_remove(Z8Guid id) {
    if (id is null || id == z8_guid_zero()) return;
    auto table = z8_native_new(z8_system_files_name); table.z8_ensure();
    z8_query_destroy_id(table, id);
}
// Resolve metadata separately from content, as the Java Files.get overloads do.
private Z8File z8_system_files_get_id(Z8Guid id) {
    if (id is null || id == z8_guid_zero()) return null;
    auto table = z8_native_new(z8_system_files_name);
    table.z8_ensure();
    auto path = z8_system_files_state(table).fields[2];
    auto fields = new Z8ObjectArray!Z8Object(); fields.z8_add(path);
    scope(exit) z8_query_close(table);
    if (!z8_optional_require(z8_query_read_record(table, id, fields))) return null;
    auto result = z8_file_from_path(z8_primary_cast!wstring(z8_field_primary(path)));
    result.id = new Z8Guid(id.majorBits, id.minorBits);
    return result;
}
private Z8File z8_system_files_get_file(Z8File value) {
    import std.file : exists, isFile, getSize, mkdirRecurse, write;
    import std.path : dirName;
    value = z8_nonnull(value);
    auto path = z8_file_system_path(value);
    if (exists(path) && isFile(path)) {
        value.size = z8_optional_wrap(cast(long)getSize(path));
        value.hasUpload = false; value.upload = null;
        return value;
    }
    if (value.id is null || value.id == z8_guid_zero()) throw new Exception("file was not found");
    auto table = z8_native_new(z8_system_files_name);
    table.z8_ensure();
    auto members = z8_system_files_state(table).fields;
    auto fields = new Z8ObjectArray!Z8Object(); fields.z8_add(members[1]); fields.z8_add(members[3]);
    scope(exit) z8_query_close(table);
    if (!z8_optional_require(z8_query_read_record(table, value.id, fields))) throw new Exception("file was not found");
    auto size = z8_optional_require(z8_primary_cast!long(z8_field_primary(members[3])));
    auto data = z8_primary_cast!Z8Binary(z8_field_primary(members[1]));
    auto bytes = z8_binary_snapshot(z8_nonnull(data), size_t.max);
    if (size != 0 && (size < 0 || cast(ulong)size != bytes.length)) throw new Exception("stored file size does not match its content");
    auto parent = dirName(path); if (parent.length) mkdirRecurse(parent);
    write(path, bytes);
    value.hasUpload = false; value.upload = null;
    value.size = z8_optional_wrap(cast(long)bytes.length);
    return value;
}
// The path must match the database record, never a caller-selected local file.
// As in ConverterAdapter, session authorization is provided by the host.
// ConverterAdapter asks the servlet container for the type of formats that do
// not need conversion. Unknown extensions retain its text/plain fallback.
private wstring z8_application_preview_mime(wstring extension) {
    switch (extension) {
        case "html"w: case "htm"w: return "text/html"w;
        case "css"w: return "text/css"w;
        case "csv"w: return "text/csv"w;
        case "json"w: return "application/json"w;
        case "svg"w: return "image/svg+xml"w;
        case "webp"w: return "image/webp"w;
        case "ico"w: return "image/x-icon"w;
        case "zip"w: return "application/zip"w;
        case "gz"w: return "application/gzip"w;
        case "7z"w: return "application/x-7z-compressed"w;
        case "mp3"w: return "audio/mpeg"w;
        case "ogg"w: return "audio/ogg"w;
        case "wav"w: return "audio/wav"w;
        case "mp4"w: return "video/mp4"w;
        case "webm"w: return "video/webm"w;
        default: return "text/plain"w;
    }
}
// The original uses UniversalDetector. Detect explicit Unicode byte-order
// marks here without consuming the stream; unmarked text defaults to UTF-8.
private wstring z8_application_preview_charset(Z8Binary data) {
    import std.algorithm.comparison : min;
    auto remaining = z8_binary_remaining(z8_nonnull(data));
    ubyte[4] prefix;
    auto count = cast(size_t)min(remaining, 4UL);
    if (data.fileInput is null) prefix[0 .. count] = data.data[data.position .. data.position + count];
    else {
        auto stream = data.fileInput.stream;
        auto position = stream.tell;
        scope(exit) stream.seek(position);
        count = stream.rawRead(prefix[0 .. count]).length;
    }
    if (count >= 4 && prefix[] == [0x00, 0x00, 0xFE, 0xFF]) return "UTF-32BE"w;
    if (count >= 4 && prefix[] == [0xFF, 0xFE, 0x00, 0x00]) return "UTF-32LE"w;
    if (count >= 2 && prefix[0 .. 2] == [0xFE, 0xFF]) return "UTF-16BE"w;
    if (count >= 2 && prefix[0 .. 2] == [0xFF, 0xFE]) return "UTF-16LE"w;
    return "UTF-8"w;
}
private void z8_application_preview_file(Z8ApplicationRequest request, Z8Binary data, wstring name) {
    auto extension = z8_file_converter_extension(name);
    if (z8_file_converter_pdf(extension).value && !request.parameters.z8_containsKey("stamps"w)) {
        z8_application_set_response_inline(request, data, "application/pdf"w);
        return;
    }
    if (z8_file_converter_convertible(extension).value) {
        import std.file : exists, mkdir, rmdirRecurse, tempDir;
        import std.path : absolutePath, buildPath;
        import std.uuid : randomUUID;
        scope(exit) z8_binary_close(data);
        auto temporary = buildPath(absolutePath(tempDir()), "z8-preview-" ~ randomUUID().toString());
        mkdir(temporary);
        scope(exit) if (exists(temporary)) rmdirRecurse(temporary);
        auto target = z8_file_from_path(buildPath(temporary, "preview.pdf").to!wstring);
        auto converted = z8_file_converter_copy(data, extension, target, request.parameters);
        auto stream = z8_file_to_binary(converted);
        ubyte[] content;
        { scope(exit) z8_binary_close(stream); content = z8_binary_snapshot(stream, z8_temporary_archives_bytes_limit); }
        z8_application_set_response_inline(request, new Z8Binary(content), "application/pdf"w);
        return;
    }
    import std.string : startsWith;
    auto mime = z8_application_preview_mime(extension);
    if (mime.startsWith("text/"w)) mime ~= "; charset="w ~ z8_application_preview_charset(data);
    z8_application_set_response_inline(request, data, mime);
}
void z8_application_dispatch_file_download(wstring requestPath) {
    import std.string : startsWith;
    auto request = z8_application_require_request();
    z8_nonnull(request.session);
    auto preview = request.parameters.z8_containsKey("preview"w);
    requestPath = z8_string_required(requestPath);
    if (!requestPath.startsWith("/storage/"w)) throw new Exception("file was not found");
    auto text = request.parameters.z8_get("id"w);
    if (text is null || !text.length) {
        wstring name;
        auto data = z8_temporary_archive_get(requestPath[1 .. $], name);
        if (data is null) throw new Exception("file was not found");
        if (preview) z8_application_preview_file(request, data, name);
        else z8_application_set_response_file(request, data, name);
        return;
    }
    auto file = z8_system_files_get_id(z8_guid_parse(text));
    if (file is null || file.path != requestPath[1 .. $]) throw new Exception("file was not found");
    file = z8_system_files_get_file(file);
    if (preview) z8_application_preview_file(request, z8_file_to_binary(file), file.name);
    else z8_application_set_response_file(request, z8_file_to_binary(file), file.name);
}
shared static this() {
    z8_native_bind(z8_system_files_name ~ ".m_new"w, &z8_system_files_new);
    z8_native_bind(z8_system_files_name ~ ".m_get"w, &z8_system_files_get_id);
    z8_native_bind(z8_system_files_name ~ ".m_get"w, &z8_system_files_get_file);
    static foreach (i, name; ["name"w, "data"w, "path"w, "size"w, "time"w, "lastModified"w, "location"w])
        z8_native_bind_field(z8_system_files_name ~ ".f_"w ~ name, &z8_system_files_field_address!i);
}
