private void z8_integer_field_new(Z8Object self) {
    auto state = new Z8FieldState(z8_primary_box(0L)); state.aggregation = Z8FieldAggregation.sum;
    state.format = "0,000"w;
    state.useSlot!(Z8Optional!long, long)(); z8_field_attach(self, state);
}
private void z8_string_field_new(Z8Object self) {
    auto state = new Z8FieldState(z8_primary_box(""w)); state.aggregation = Z8FieldAggregation.max;
    state.fts = new Z8Fts();
    if (z8_native_is_instance(z8_string_field_name, self)) state.length = z8_optional_wrap(30L);
    state.useSlot!wstring(); z8_field_attach(self, state);
}
private Z8Optional!long* z8_integer_field_default_address(Z8Object self) {
    return &z8_field_state(self).useSlot!(Z8Optional!long, long)().item;
}
private wstring* z8_string_field_default_address(Z8Object self) {
    return &z8_field_state(self).useSlot!wstring().item;
}
private Z8Optional!long z8_integer_field_get(Z8Object self) { return z8_primary_cast!long(z8_field_current(self)); }
private wstring z8_string_field_get(Z8Object self) { return z8_primary_cast!wstring(z8_field_current(self)); }
private Z8Optional!long z8_integer_field_get_default(Z8Object self) { return z8_primary_cast!long(z8_field_state(self).value); }
private wstring z8_string_field_get_default(Z8Object self) { return z8_primary_cast!wstring(z8_field_state(self).value); }
private void z8_integer_field_set_default(Z8Object self, Z8Optional!long value) { z8_field_set_default(self, z8_primary_box(value)); }
private void z8_string_field_set_default(Z8Object self, wstring value) { z8_field_set_default(self, z8_primary_box(value)); }
private Z8Object z8_integer_field_assign(Z8Object self, Z8Optional!long value) { z8_field_set(self, z8_primary_box(value)); return self; }
private void z8_integer_expression_new(Z8Object self) {
    auto state = new Z8FieldState(z8_primary_box(0L), false); state.aggregation = Z8FieldAggregation.sum;
    state.format = "0,000"w; state.useSlot!(Z8Optional!long, long)(); z8_field_attach(self, state);
}

    return z8_sql_int_const(z8_integer_field_get_default(self));

private Z8Object z8_string_field_assign(Z8Object self, wstring value) { z8_field_set(self, z8_primary_box(value)); return self; }
private void z8_bool_field_new(Z8Object self) {
    auto state = new Z8FieldState(z8_primary_box(false)); state.aggregation = Z8FieldAggregation.max;
    state.width = z8_optional_wrap(50L);
    state.useSlot!(Z8Optional!bool, bool)(); z8_field_attach(self, state);

private void z8_decimal_field_new(Z8Object self) {
    auto state = new Z8FieldState(z8_primary_box(z8_decimal_from_int(0))); state.aggregation = Z8FieldAggregation.sum;
    state.format = "0,000.00"w;
    state.useSlot!(Z8Optional!Z8Decimal, Z8Decimal)(); z8_field_attach(self, state);

private void z8_guid_field_new(Z8Object self) {

private void z8_datetime_field_new(Z8Object self) {
    auto state = new Z8FieldState(z8_primary_box(z8_date_min)); state.aggregation = Z8FieldAggregation.max;
    state.format = z8_native_is_instance("org.zenframework.z8.server.base.table.value.DateField"w, self) ? "d.m.Y"w : "d.m.Y H:i"w;
    state.useSlot!Z8Date(); z8_field_attach(self, state);

private T* z8_typed_field_default_address(T, Base = T)(Z8Object self) {

private void z8_register_value_field(T, Base = T)(wstring name, void function(Z8Object) create) {
    z8_native_bind(name ~ ".m_new"w, create);
    z8_native_bind_field(name ~ ".f_defaultValue"w, &z8_typed_field_default_address!(T, Base));
    z8_native_bind(name ~ ".m_get"w, &z8_typed_field_get!(T, Base));
    z8_native_bind(name ~ ".m_getDefault"w, &z8_typed_field_get_default!(T, Base));
    static if (!is(Base == Z8Binary) && !is(Base == Z8Datespan)) z8_native_bind(name ~ ".m_array"w, &z8_field_array!Base);
    z8_native_bind(name ~ ".m_setDefault"w, &z8_typed_field_set_default!T);
    z8_native_bind(name ~ ".z8op_assign"w, &z8_typed_field_assign!T);
}
