// Basic typed-field value operations without a DB cursor.
private enum z8_integer_field_name = "org.zenframework.z8.server.base.table.value.IntegerField"w;
private enum z8_string_field_name = "org.zenframework.z8.server.base.table.value.StringField"w;
private enum z8_bool_field_name = "org.zenframework.z8.server.base.table.value.BoolField"w;
private enum z8_decimal_field_name = "org.zenframework.z8.server.base.table.value.DecimalField"w;
private enum z8_guid_field_name = "org.zenframework.z8.server.base.table.value.GuidField"w;
private enum z8_datetime_field_name = "org.zenframework.z8.server.base.table.value.DatetimeField"w;
private enum z8_datespan_field_name = "org.zenframework.z8.server.base.table.value.DatespanField"w;
private enum z8_binary_field_name = "org.zenframework.z8.server.base.table.value.BinaryField"w;
private enum z8_geometry_field_name = "org.zenframework.z8.server.base.table.value.GeometryField"w;
private enum z8_text_field_name = "org.zenframework.z8.server.base.table.value.TextField"w;

private void z8_integer_field_new(Z8Object self) {
    auto state = new Z8FieldState(z8_primary_box(0L)); state.useSlot!(Z8Optional!long, long)(); z8_field_attach(self, state);
}
private void z8_string_field_new(Z8Object self) {
    auto state = new Z8FieldState(z8_primary_box(""w)); state.useSlot!wstring(); z8_field_attach(self, state);
}
private Z8Optional!long* z8_integer_field_default_address(Z8Object self) {
    return &z8_field_state(self).useSlot!(Z8Optional!long, long)().item;
}
private wstring* z8_string_field_default_address(Z8Object self) {
    return &z8_field_state(self).useSlot!wstring().item;
}
private Z8Optional!long z8_integer_field_get(Z8Object self) { return z8_primary_cast!long(z8_field_state(self).value); }
private wstring z8_string_field_get(Z8Object self) { return z8_primary_cast!wstring(z8_field_state(self).value); }
private void z8_integer_field_set_default(Z8Object self, Z8Optional!long value) { z8_field_set_default(self, z8_primary_box(value)); }
private void z8_string_field_set_default(Z8Object self, wstring value) { z8_field_set_default(self, z8_primary_box(value)); }
private Z8Object z8_integer_field_assign(Z8Object self, Z8Optional!long value) { z8_field_set(self, z8_primary_box(value)); return self; }
private Z8Object z8_string_field_assign(Z8Object self, wstring value) { z8_field_set(self, z8_primary_box(value)); return self; }
private void z8_bool_field_new(Z8Object self) {
    auto state = new Z8FieldState(z8_primary_box(false)); state.useSlot!(Z8Optional!bool, bool)(); z8_field_attach(self, state);
}
private void z8_decimal_field_new(Z8Object self) {
    auto state = new Z8FieldState(z8_primary_box(z8_decimal_from_int(0))); state.useSlot!(Z8Optional!Z8Decimal, Z8Decimal)(); z8_field_attach(self, state);
}
private void z8_guid_field_new(Z8Object self) {
    auto state = new Z8FieldState(z8_primary_box(z8_guid_zero())); state.useSlot!Z8Guid(); z8_field_attach(self, state);
}
private void z8_datetime_field_new(Z8Object self) {
    auto state = new Z8FieldState(z8_primary_box(z8_date_min)); state.useSlot!Z8Date(); z8_field_attach(self, state);
}
private void z8_datespan_field_new(Z8Object self) {
    auto state = new Z8FieldState(z8_primary_box(new Z8Datespan())); state.useSlot!Z8Datespan(); z8_field_attach(self, state);
}
private void z8_binary_field_new(Z8Object self) {
    auto state = new Z8FieldState(z8_primary_box(new Z8Binary())); state.useSlot!Z8Binary(); z8_field_attach(self, state);
}
private void z8_geometry_field_new(Z8Object self) {
    auto state = new Z8FieldState(z8_primary_box(new Z8Geometry())); state.useSlot!Z8Geometry(); z8_field_attach(self, state);
}
private T* z8_typed_field_default_address(T, Base = T)(Z8Object self) {
    return &z8_field_state(self).useSlot!(T, Base)().item;
}
private T z8_typed_field_get(T, Base = T)(Z8Object self) { return z8_primary_cast!Base(z8_field_state(self).value); }
private void z8_typed_field_set_default(T)(Z8Object self, T value) { z8_field_set_default(self, z8_primary_box(value)); }
private Z8Object z8_typed_field_assign(T)(Z8Object self, T value) { z8_field_set(self, z8_primary_box(value)); return self; }
private Z8SqlInt z8_integer_field_sql(Z8Object self) {
    auto value = new Z8SqlInt(Z8SqlIntKind.field); value.field = z8_nonnull(self); return value;
}
private Z8SqlString z8_string_field_sql(Z8Object self) {
    auto value = new Z8SqlString(Z8SqlStringKind.field); value.field = z8_nonnull(self); return value;
}
private Z8SqlBool z8_bool_field_sql(Z8Object self) {
    auto value = new Z8SqlBool(Z8SqlBoolKind.field); value.field = z8_nonnull(self); return value;
}
private Z8SqlDecimal z8_decimal_field_sql(Z8Object self) {
    auto value = new Z8SqlDecimal(Z8SqlDecimalKind.field); value.field = z8_nonnull(self); return value;
}
private Z8SqlGuid z8_guid_field_sql(Z8Object self) {
    auto value = new Z8SqlGuid(Z8SqlGuidKind.field); value.field = z8_nonnull(self); return value;
}
private void z8_register_value_field(T, Base = T)(wstring name, void function(Z8Object) create) {
    z8_native_bind(name ~ ".m_new"w, create);
    z8_native_bind_field(name ~ ".f_defaultValue"w, &z8_typed_field_default_address!(T, Base));
    z8_native_bind(name ~ ".m_get"w, &z8_typed_field_get!(T, Base));
    z8_native_bind(name ~ ".m_getDefault"w, &z8_typed_field_get!(T, Base));
    z8_native_bind(name ~ ".m_setDefault"w, &z8_typed_field_set_default!T);
    z8_native_bind(name ~ ".z8op_assign"w, &z8_typed_field_assign!T);
}

void z8_bind_builtin_typed_fields() {
    z8_native_bind(z8_integer_field_name ~ ".m_new"w, &z8_integer_field_new);
    z8_native_bind_field(z8_integer_field_name ~ ".f_defaultValue"w, &z8_integer_field_default_address);
    z8_native_bind(z8_integer_field_name ~ ".m_get"w, &z8_integer_field_get);
    z8_native_bind(z8_integer_field_name ~ ".m_getDefault"w, &z8_integer_field_get);
    z8_native_bind(z8_integer_field_name ~ ".m_setDefault"w, &z8_integer_field_set_default);
    z8_native_bind(z8_integer_field_name ~ ".z8op_assign"w, &z8_integer_field_assign);
    z8_native_bind(z8_string_field_name ~ ".m_new"w, &z8_string_field_new);
    z8_native_bind_field(z8_string_field_name ~ ".f_defaultValue"w, &z8_string_field_default_address);
    z8_native_bind(z8_string_field_name ~ ".m_get"w, &z8_string_field_get);
    z8_native_bind(z8_string_field_name ~ ".m_getDefault"w, &z8_string_field_get);
    z8_native_bind(z8_string_field_name ~ ".m_setDefault"w, &z8_string_field_set_default);
    z8_native_bind(z8_string_field_name ~ ".z8op_assign"w, &z8_string_field_assign);
    z8_register_value_field!(Z8Optional!bool, bool)(z8_bool_field_name, &z8_bool_field_new);
    z8_register_value_field!(Z8Optional!Z8Decimal, Z8Decimal)(z8_decimal_field_name, &z8_decimal_field_new);
    z8_register_value_field!Z8Guid(z8_guid_field_name, &z8_guid_field_new);
    z8_register_value_field!Z8Date(z8_datetime_field_name, &z8_datetime_field_new);
    z8_register_value_field!Z8Datespan(z8_datespan_field_name, &z8_datespan_field_new);
    z8_register_value_field!Z8Binary(z8_binary_field_name, &z8_binary_field_new);
    z8_register_value_field!Z8Geometry(z8_geometry_field_name, &z8_geometry_field_new);
    z8_register_value_field!wstring(z8_text_field_name, &z8_string_field_new);
    z8_native_bind(z8_integer_field_name ~ ".z8cast_sql_int"w, &z8_integer_field_sql);
    z8_native_bind(z8_string_field_name ~ ".z8cast_sql_string"w, &z8_string_field_sql);
    z8_native_bind(z8_bool_field_name ~ ".z8cast_sql_bool"w, &z8_bool_field_sql);
    z8_native_bind(z8_decimal_field_name ~ ".z8cast_sql_decimal"w, &z8_decimal_field_sql);
    z8_native_bind(z8_guid_field_name ~ ".z8cast_sql_guid"w, &z8_guid_field_sql);
}
shared static this() { z8_bind_builtin_typed_fields(); }
