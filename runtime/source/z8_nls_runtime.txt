// Java XML-properties resources used by Z8 .nls files.
size_t z8_string_load_nls(string path) {
    auto entries = z8_nls_read(path);
    foreach (key, value; entries) z8_string_set_resource(key, value);
    auto trueValue = "bool.true"w in entries, falseValue = "bool.false"w in entries;
    if (trueValue !is null || falseValue !is null)
        z8_bool_set_resource_strings(trueValue is null ? z8_bool_true_resource : *trueValue,
                                     falseValue is null ? z8_bool_false_resource : *falseValue);
    return entries.length;
}

// Match the server's language-specific bundle names, with deterministic ordering.
size_t z8_string_load_nls_directory(string directory, string language) {
    auto files = z8_nls_find_files(directory, language);
    foreach (file; files) z8_string_load_nls(file);
    return files.length;
}

// Reload at host startup after server.properties supplies the application root
// and language. Environment overrides keep the standalone runtime behavior.
void z8_string_configure_nls(string applicationRoot, wstring configuredLanguage) {
    import std.path : buildPath, pathSeparator;
    import std.process : environment;
    import std.string : split;
    auto language = environment.get("Z8_LANGUAGE", "");
    if (!language.length) language = configuredLanguage is null || !configuredLanguage.length ? "ru" : configuredLanguage.to!string;
    auto directories = environment.get("Z8_NLS_DIR", "");
    if (!directories.length) directories = buildPath(applicationRoot, "resources");
    string[] files;
    foreach (directory; directories.split(pathSeparator))
        if (directory.length) files ~= z8_nls_find_files(directory, language);
    foreach (file; environment.get("Z8_NLS_FILES", "").split(pathSeparator)) if (file.length) files ~= file;
    wstring[wstring] entries;
    foreach (file; files) foreach (key, value; z8_nls_read(file)) entries[key] = value;
    z8_string_clear_resources();
    foreach (key, value; entries) z8_string_set_resource(key, value);
    auto trueValue = "bool.true"w in entries, falseValue = "bool.false"w in entries;
    z8_bool_set_resource_strings(trueValue is null ? "bool.true"w : *trueValue,
                                 falseValue is null ? "bool.false"w : *falseValue);
}

shared static this() {
    import std.file : thisExePath;
    import std.path : buildPath, dirName, pathSeparator;
    import std.process : environment;
    import std.string : split;
    auto directories = environment.get("Z8_NLS_DIR", "");
    auto language = environment.get("Z8_LANGUAGE", "ru");
    if (directories.length) {
        foreach (directory; directories.split(pathSeparator))
            if (directory.length) z8_string_load_nls_directory(directory, language);
    } else z8_string_load_nls_directory(buildPath(dirName(thisExePath()), "resources"), language);
    auto files = environment.get("Z8_NLS_FILES", "");
    foreach (path; files.split(pathSeparator)) if (path.length) z8_string_load_nls(path);
}
