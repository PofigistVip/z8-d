// Parse every filter leaf explicitly: silently discarding an invalid leaf can
// widen an AND filter and return records the caller did not ask for.
private Z8SqlBool z8_application_query_filter_node(Z8Object query, JSONValue node) {
    if (node.type != JSONType.object) throw new Exception("invalid query filter node");
    auto logical = "logical" in node.object;
    if (logical is null) {
        auto property = "property" in node.object;
        if (property is null || property.type != JSONType.string || !property.str.length)
            throw new Exception("query filter requires a field");
        auto expression = z8_filter_node(query, node);
        if (expression is null) throw new Exception("empty query filter expression");
        return expression;
    }
    if (logical.type != JSONType.string) throw new Exception("invalid query filter operator");
    auto op = z8_str_toLowerCase(logical.str.to!wstring);
    if (op != "and"w && op != "or"w && op != "not"w) throw new Exception("invalid query filter operator");
    auto children = "expressions" in node.object;
    if (children is null || children.type != JSONType.array || !children.array.length)
        throw new Exception("query filter group requires expressions");
    Z8SqlBool result;
    foreach (child; children.array) result = z8_filter_combine(result, z8_application_query_filter_node(query, child), op == "or"w);
    return op == "not"w ? z8_sql_bool_not(result) : z8_sql_bool_group(result);
}
private Z8SqlBool z8_application_query_filter(Z8Object query, wstring json) {
    auto root = z8_json_parse(json.to!string);
    if (root.type != JSONType.array) return z8_application_query_filter_node(query, root);
    if (!root.array.length) return null;
    Z8SqlBool result;
    foreach (node; root.array) result = z8_filter_combine(result, z8_application_query_filter_node(query, node));
    return z8_sql_bool_group(result);
}
private Z8SqlBool z8_application_query_where(Z8Object context, Z8Object query, wstring json) {
    auto root = z8_json_parse(json.to!string);
    if (context !is query && root.type == JSONType.array && root.array.length == 2) {
        auto first = root.array[0], second = root.array[1];
        if (first.type == JSONType.object && first.object.length == 1 && ("value" in first.object) !is null &&
            second.type == JSONType.object && ("property" in second.object) !is null && ("value" in second.object) !is null) {
            auto property = second.object["property"];
            if (property.type == JSONType.string) {
                auto field = z8_query_find_field_id(query, property.str.to!wstring);
                foreach (control; z8_query_array(context, "controls"w)) {
                    if (!z8_native_is_instance(z8_listbox_name, control)) continue;
                    auto list = z8_listbox_state(control);
                    // The original Listbox emits a value-only dependency when
                    // no dependency field is configured. Java Filter ignores
                    // it; retain the list's actual parent-link filter.
                    if (list.query is query && list.link is field && z8_control_state(control).dependency is null)
                        return z8_application_query_filter_node(query, second);
                }
            }
        }
    }
    return z8_application_query_filter(query, json);
}

// Follow the planner's first Link/Join path, checking each table and relation
// before exposing a field from a related query.
private Z8Object[] z8_application_query_field_path(Z8Object root, Z8Object field, Z8Privileges privileges) {
    enum queryName = "org.zenframework.z8.server.base.query.Query"w;
    enum tableName = "org.zenframework.z8.server.base.table.Table"w;
    enum linkName = "org.zenframework.z8.server.base.table.value.Link"w;
    enum joinName = "org.zenframework.z8.server.base.table.value.Join"w;
    auto owner = z8_object_container(field);
    while (owner !is null && !z8_native_is_instance(queryName, owner)) owner = z8_object_container(owner);
    if (owner is null) throw new Exception("query field has no owner");
    Z8Object[] path;
    auto current = owner;
    while (current !is root) {
        path ~= current;
        current = z8_object_container(current);
        while (current !is null && !z8_native_is_instance(queryName, current)) current = z8_object_container(current);
        if (current is null) throw new Exception("query field is not reachable through a link");
    }
    Z8Object[] links;
    auto parent = root;
    for (size_t remaining = path.length; remaining > 0; --remaining) {
        auto target = path[remaining - 1];
        if (!z8_native_is_instance(tableName, target)) throw new Exception("linked query must be a table");
        Z8Object link;
        foreach (candidate; z8_query_declared_fields(parent))
            if ((z8_native_is_instance(linkName, candidate) || z8_native_is_instance(joinName, candidate)) && z8_link_query(candidate) is target) {
                link = candidate; break;
            }
        if (link is null) throw new Exception("linked query requires a Link or Join");
        auto parentKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, parent);
        auto linkKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, link);
        auto targetKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, target);
        if (!privileges.getFieldAccess(parentKey, linkKey).read || !privileges.getTableAccess(targetKey).read)
            throw new Exception("linked query access denied");
        links ~= link;
        parent = target;
    }
    bool declared;
    foreach (candidate; z8_query_declared_fields(owner)) if (candidate is field) { declared = true; break; }
    if (!declared) throw new Exception("query field is not declared by its owner");
    return links;
}
// Resolve a nested query, checking each containing query and declared relation.
// Table operations keep the more restrictive table-only default.
private Z8Object z8_application_nested_query(Z8Object context, Z8Privileges privileges, wstring queryId, bool tableOnly = true) {
    if (queryId is null || !queryId.length) return context;
    auto query = z8_action_find_member(context, queryId);
    if (!z8_native_is_instance(z8_query_name, query) || (tableOnly && !z8_native_is_instance(z8_table_name, query)))
        throw new Exception(tableOnly ? "nested query requires a Table" : "nested query requires a Query");
    auto current = query;
    Z8Object[] parents;
    while (current !is context) {
        auto parent = z8_object_container(current);
        while (parent !is null && !z8_native_is_instance(z8_query_name, parent)) parent = z8_object_container(parent);
        if (parent is null || (tableOnly && !z8_native_is_instance(z8_table_name, parent)))
            throw new Exception("nested query is outside the request");
        auto parentKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, parent);
        if (!privileges.getTableAccess(parentKey).read) throw new Exception("nested query parent access denied");
        foreach (relation; z8_query_declared_fields(parent)) {
            if ((z8_native_is_instance("org.zenframework.z8.server.base.table.value.Link"w, relation) ||
                 z8_native_is_instance("org.zenframework.z8.server.base.table.value.Join"w, relation)) && z8_link_query(relation) is current) {
                auto relationKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, relation);
                if (!privileges.getFieldAccess(parentKey, relationKey).read) throw new Exception("nested query relation access denied");
                break;
            }
        }
        parents ~= parent;
        current = parent;
    }
    // Recreate the parent controls' dependent-form bindings on this request.
    foreach_reverse (parent; parents) z8_query_form_fields(parent);
    return query;
}
// ActionFactory gives a nonempty link precedence over the query parameter.
private Z8Object z8_application_action_query(Z8Object context, Z8Privileges privileges, wstring queryId, wstring linkId, bool tableOnly = true) {
    if (linkId is null || !linkId.length) return z8_application_nested_query(context, privileges, queryId, tableOnly);
    auto link = z8_application_context_field(context, privileges, linkId, !tableOnly);
    if (!z8_native_is_instance(z8_link_name, link) && !z8_native_is_instance(z8_join_name, link))
        throw new Exception("query link access denied");
    auto target = z8_link_query(link);
    if (target is null) throw new Exception("query link has no target");
    auto query = target is context ? context : z8_application_nested_query(context, privileges, target.z8_id(), tableOnly);
    if (query !is target) throw new Exception("query link target is outside the request");
    return query;
}
private bool z8_application_query_field_readable(Z8Object root, Z8Object field, Z8Privileges privileges) {
    Z8Object[] seen;
    bool check(Z8Object candidate) {
        foreach (existing; seen) if (existing is candidate) return true;
        seen ~= candidate;
        z8_application_query_field_path(root, candidate, privileges);
        auto owner = z8_nonnull(candidate.z8_get_owner());
        auto ownerKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, owner);
        auto fieldKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, candidate);
        if (candidate !is z8_query_primary_key(root) && !privileges.getFieldAccess(ownerKey, fieldKey).read) return false;
        foreach (dependency; z8_field_query_dependencies(candidate))
            if (dependency !is candidate && !check(dependency)) return false;
        return true;
    }
    return check(field);
}
private Z8Object z8_application_context_field(Z8Object context, Z8Privileges privileges, wstring name, bool allowNonTable = false, bool allowMissing = false) {
    auto field = z8_query_try_find_field_id(context, name);
    if (field is null) {
        if (allowMissing) return null;
        throw new Exception("query field not found");
    }
    auto owner = z8_object_container(field);
    while (owner !is null && !z8_native_is_instance(z8_query_name, owner)) owner = z8_object_container(owner);
    if (owner is null) throw new Exception("query context field has no owner");
    if (owner !is context && z8_application_nested_query(context, privileges, owner.z8_id(), !allowNonTable) !is owner)
        throw new Exception("query context field is outside the request");
    auto ownerKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, owner);
    if (!privileges.getTableAccess(ownerKey).read || !z8_application_query_field_readable(owner, field, privileges))
        throw new Exception("query context field access denied");
    return field;
}
private Z8Object[] z8_application_query_context_values(Z8Object query, Z8Privileges privileges, Z8StringMap params, bool apply, Z8Object context = null, bool allowNonTable = false) {
    Z8Object[] applied;
    auto contextValues = params.z8_get("values"w);
    if (contextValues is null || !contextValues.length) return applied;
    auto parsed = z8_json_parse(contextValues.to!string);
    if (parsed.type == JSONType.array) parsed = parsed.array.length ? parsed.array[0] : JSONValue.init;
    if (parsed.type == JSONType.null_) return applied;
    if (parsed.type != JSONType.object) throw new Exception("query values must be an object");
    foreach (name, value; parsed.object) {
        auto field = context is null ? z8_query_try_find_field_id(query, name.to!wstring) :
            z8_application_context_field(context, privileges, name.to!wstring, allowNonTable, true);
        if (field is null) continue;
        if (context is null && !z8_application_query_field_readable(query, field, privileges)) throw new Exception("query context field access denied");
        if (z8_field_type_id(field) == "binary"w) continue;
        auto boxed = z8_application_query_parse_value(field, value);
        if (apply) { z8_field_set(field, boxed); applied ~= field; }
    }
    return applied;
}

// Shared by read and mutation responses; field selection is checked before a
// mutation starts, so malformed response options cannot fail after commit.
private Z8Object[] z8_application_query_fields(Z8Object query, Z8Privileges privileges, Z8StringMap params, bool includeKeys = true) {
    auto key = z8_query_primary_key(query);
    Z8Object[] requested;
    // Resolve dependent forms even when the client supplies an explicit field list.
    auto formFields = z8_query_form_fields(query);
    auto fields = params.z8_get("fields"w);
    bool explicitFields;
    if (fields !is null && fields.length) {
        auto parsed = z8_json_parse(fields.to!string);
        if (parsed.type != JSONType.array) throw new Exception("query fields must be a JSON array");
        explicitFields = parsed.array.length != 0;
        foreach (item; parsed.array) {
            wstring id;
            if (item.type == JSONType.string) id = item.str.to!wstring;
            else if (item.type == JSONType.object) {
                auto member = "id" in item.object;
                if (member is null || member.type != JSONType.string) throw new Exception("query field has no id");
                id = member.str.to!wstring;
            } else throw new Exception("invalid query field");
            auto field = z8_query_try_find_field_id(query, id);
            if (field !is null) requested ~= field;
        }
    }
    if (!requested.length && !explicitFields) {
        requested = formFields;
    }
    if (includeKeys) {
        if (key !is null) requested ~= key;
        foreach (field; z8_query_declared_fields(query)) {
            auto parent = field.z8_get_attribute("parentKey"w);
            if (field.z8_get_attribute("lockKey"w) !is null ||
                (parent !is null && z8_str_compareNoCase(parent, "false"w) != 0)) requested ~= field;
        }
    }
    Z8Object[] expanded;
    foreach (field; requested) {
        expanded ~= field;
        if (!z8_native_is_instance("org.zenframework.z8.server.base.table.value.Join"w, field))
            expanded ~= z8_field_used_fields(field);
    }
    Z8Object[] selected;
    foreach (field; expanded) {
        // Join is a relation expression, not a data field (ReadAction.addSelectField).
        if (z8_native_is_instance("org.zenframework.z8.server.base.table.value.Join"w, field)) continue;
        bool duplicate;
        foreach (existing; selected) if (existing is field) { duplicate = true; break; }
        if (duplicate) continue;
        auto path = z8_application_query_field_path(query, field, privileges);
        if (!z8_application_query_field_readable(query, field, privileges)) continue;
        foreach (link; path) {
            bool present;
            foreach (existing; selected) if (existing is link) { present = true; break; }
            if (!present && !z8_native_is_instance("org.zenframework.z8.server.base.table.value.Join"w, link)) selected ~= link;
            if (z8_link_join_type(link) == Z8JoinType.right) {
                auto target = z8_nonnull(z8_link_query(link));
                auto targetKey = z8_nonnull(z8_query_primary_key(target));
                present = false;
                foreach (existing; selected) if (existing is targetKey) { present = true; break; }
                if (!present) selected ~= targetKey;
            }
        }
        selected ~= field;
    }
    return selected;
}
private void z8_application_query_write_fields(Z8Object query, Z8JsonWriterState writer, Z8Object[] selected) {
    auto key = z8_query_primary_key(query);
    foreach (field; selected) {
        bool present;
        foreach (column; z8_query_state(query).cursorFields) if (column is field) { present = true; break; }
        if (!present) continue;
        auto value = z8_field_primary(field);
        auto wasNull = z8_optional_require(z8_field_was_null(field));
        if (field is key && wasNull) continue;
        if (wasNull) {
            enum fieldPrefix = "org.zenframework.z8.server.base.table.value."w;
            if (z8_native_is_instance(fieldPrefix ~ "IntegerField"w, field)) value = z8_primary_box(0L);
            else if (z8_native_is_instance(fieldPrefix ~ "DecimalField"w, field)) value = z8_primary_box(z8_decimal_from_int(0));
            else if (z8_native_is_instance(fieldPrefix ~ "StringField"w, field)) value = z8_primary_box(""w);
            else if (z8_native_is_instance(fieldPrefix ~ "Link"w, field)) {
                auto target = z8_link_query(field);
                auto parent = field.z8_get_attribute("parentKey"w);
                if (target is null && parent !is null && z8_str_compareNoCase(parent, "false"w) != 0) target = field.z8_get_owner();
                auto referer = target is null ? null : z8_query_primary_key(target);
                value = referer is null ? z8_primary_box(z8_guid_zero()) : z8_field_primary(referer);
            } else value = field.z8_field_virtual_default();
        }
        writer.property(field.z8_id());
        if (value !is null && value.kind == Z8PrimaryKind.geometry &&
            (wasNull || (value.geometryValue !is null && value.geometryValue.rawHex is null && value.geometryValue.shape == -1)))
            writer.raw("null"w);
        else z8_json_writer_primary(writer, value);
    }
}
private void z8_application_query_write_rows(Z8Object query, Z8JsonWriterState writer, Z8Object[] selected) {
    writer.start('[', "data"w);
    while (z8_optional_require(z8_query_next(query))) {
        writer.start('{');
        z8_application_query_write_fields(query, writer, selected);
        writer.finish('}');
    }
    writer.finish(']');
}

// Initial ReadAction adapter for table queries. Unsupported request options
// are rejected rather than silently changing the requested result.
void z8_application_dispatch_query_read(wstring className) {
    z8_application_dispatch_query_read_target(className, null);
}
private void z8_application_dispatch_query_read_target(wstring className, Z8Object query, bool defaultAction = false) {
    import std.json : JSONType;
    auto request = z8_application_require_request();
    auto session = z8_nonnull(request.session);
    auto params = z8_nonnull(request.parameters);
    auto writer = request.writer;
    if (writer.scopes.length != 2) throw new Exception("response object is not open");
    auto action = params.z8_get("action"w);
    if (action != "read"w && !(defaultAction && action is null)) throw new Exception("unsupported query action");
    if (!z8_application_may_execute(session, className, action)) throw new Exception("request access denied");
    auto parameterKeys = params.z8_keys();
    for (long i; i < parameterKeys.z8_size(); ++i) {
        auto name = parameterKeys.z8_get(i);
        if (name is null) throw new Exception("invalid query read parameter");
        // Additional parameters are available to BL request hooks, just as in
        // the original RequestAction; only recognized options affect this reader.
    }
    if (query is null) query = z8_object_new_instance_native(z8_string_required(className));
    if (!z8_native_is_instance(z8_query_name, query)) throw new Exception("query read requires a Query");
    auto privileges = z8_nonnull(session.user).privileges;
    if (privileges is null) throw new Exception("query read access denied");
    auto context = query;
    query = z8_application_action_query(context, privileges, params.z8_get("query"w), params.z8_get("link"w), false);
    auto queryKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, query);
    if (!privileges.getTableAccess(queryKey).read) throw new Exception("query read access denied");
    if (!z8_native_is_instance(z8_table_name, query)) {
        z8_application_query_context_values(query, privileges, params, true, context, true);
        if (z8_query_user_events_enabled()) query.z8_query_virtual_onReadAction();
        z8_query_before_read(query);
        auto selected = z8_application_query_fields(query, privileges, params);
        Z8QueryReadRequest read;
        z8_query_capture_read(query, read);
        scope(exit) if (z8_query_user_events_enabled()) query.z8_query_virtual_afterRead();
        if (defaultAction) z8_application_query_write_meta(query, selected, null, privileges, context);
        auto data = query.z8_object_get_data(params);
        if (data is null || !z8_native_is_instance(z8_json_array_name, data))
            throw new Exception("non-table query requires JSON data");
        writer.property("data"w);
        writer.raw(z8_json_array_render_state(z8_json_array_state(data)));
        return;
    }
    auto key = z8_query_primary_key(query);
    if (key is null) throw new Exception("query has no primary key");
    auto contextFields = z8_application_query_context_values(query, privileges, params, true, context, true);
    if (z8_query_user_events_enabled()) query.z8_query_virtual_onReadAction();
    z8_query_before_read(query);
    auto groups = z8_query_array(query, "groupBy"w);
    auto aggregates = z8_query_array(query, "aggregateBy"w);
    auto selected = z8_application_query_fields(query, privileges, params, !groups.length && !aggregates.length);
    foreach (field; groups ~ aggregates) {
        if (!z8_application_query_field_readable(query, field, privileges)) throw new Exception("query group field access denied");
        bool exists;
        foreach (existing; selected) if (existing is field) { exists = true; break; }
        if (!exists) selected ~= field;
    }
    void verifyGroups() {
        auto actual = z8_query_array(query, "groupBy"w);
        if (actual.length != groups.length) throw new Exception("changing query groups after beforeRead is not supported");
        foreach (i, field; actual) if (field !is groups[i]) throw new Exception("changing query groups after beforeRead is not supported");
        actual = z8_query_array(query, "aggregateBy"w);
        if (actual.length != aggregates.length) throw new Exception("changing query aggregateBy after beforeRead is not supported");
        foreach (i, field; actual) if (field !is aggregates[i]) throw new Exception("changing query aggregateBy after beforeRead is not supported");
    }
    long number(wstring key, long fallback) {
        auto value = params.z8_get(key);
        return value is null || !value.length ? fallback : z8_int_parse(value);
    }
    auto start = number("start"w, 0), limit = number("limit"w, z8_query_limit(query));
    if (start < int.min || start > int.max || limit < int.min || limit > int.max) throw new Exception("invalid query pagination");
    if (start < 0) start = 0;
    if (limit <= 0) limit = z8_query_limit(query);
    if (!groups.length && !aggregates.length) foreach (field; z8_query_declared_fields(query)) {
        auto parent = field.z8_get_attribute("parentKey"w);
        if (parent !is null && z8_str_compareNoCase(parent, "false"w) != 0) { limit = -1; break; }
    }
    Z8QueryReadRequest read;
    read.fields = selected; read.start = cast(int)start; read.limit = cast(int)limit; read.excludeNullRecord = true;
    auto recordIds = params.z8_get("recordId"w);
    if (recordIds !is null && recordIds.length) {
        auto json = recordIds[0] == '[' ? recordIds : recordIds[0] == '"' ? "["w ~ recordIds ~ "]"w : "[\""w ~ recordIds ~ "\"]"w;
        auto parsed = z8_json_parse(json.to!string);
        if (parsed.type != JSONType.array) throw new Exception("recordId must be a GUID or a JSON array of GUIDs");
        if (parsed.array.length) {
            auto filter = new Z8SqlBool(Z8SqlBoolKind.inVector);
            filter.vectorValue = new Z8SqlFieldToken(key);
            foreach (item; parsed.array) {
                if (item.type != JSONType.string) throw new Exception("recordId must contain GUIDs");
                auto id = z8_guid_parse(item.str.to!wstring);
                if (id is null) throw new Exception("invalid recordId");
                filter.vectorItems ~= new Z8SqlBoxedConstant(z8_primary_box(id));
            }
            read.where = filter;
            read.byRecordIds = true;
        }
    }
    Z8Object[] groupOrder, defaultGroups;
    auto group = params.z8_get("group"w);
    if (defaultAction) foreach (field; z8_query_array(query, "groupFields"w)) {
        foreach (item; selected) if (item is field) { defaultGroups ~= field; break; }
    }
    if (!defaultAction && group !is null && group.length) {
        auto parsed = z8_json_parse(group.to!string);
        JSONValue[] entries;
        if (parsed.type == JSONType.array) entries = parsed.array;
        else if (parsed.type == JSONType.object) entries = [parsed];
        else throw new Exception("query group must contain objects");
        foreach (item; entries) {
            if (item.type != JSONType.object) throw new Exception("invalid query group entry");
            auto property = "property" in item.object;
            if (property is null || property.type != JSONType.string || !property.str.length) throw new Exception("invalid query group entry");
            auto field = z8_query_try_find_field_id(query, property.str.to!wstring);
            if (field is null) continue;
            bool exists;
            foreach (existing; groupOrder) if (existing is field) { exists = true; break; }
            if (!exists) groupOrder ~= field;
        }
    }
    auto sorting = params.z8_get("sort"w);
    if (!defaultAction && sorting !is null && sorting.length) {
        auto parsed = z8_json_parse(sorting.to!string);
        if (parsed.type != JSONType.array) throw new Exception("query sort must be a JSON array");
        foreach (item; parsed.array) {
            if (item.type != JSONType.object) throw new Exception("invalid query sort entry");
            auto property = "property" in item.object, direction = "direction" in item.object;
            if (property is null || property.type != JSONType.string || !property.str.length ||
                (direction !is null && direction.type != JSONType.string)) throw new Exception("invalid query sort entry");
            auto field = z8_query_try_find_field_id(query, property.str.to!wstring);
            if (field is null) continue;
            auto order = direction is null ? "asc"w : z8_str_toLowerCase(direction.str.to!wstring);
            z8_field_set_sort_direction(field, order);
            bool exists;
            foreach (existing; read.sortFields) if (existing is field) { exists = true; break; }
            if (!exists) read.sortFields ~= field;
        }
    }
    if (defaultAction) {
        groupOrder = null;
        foreach (field; defaultGroups) {
            bool exists;
            foreach (existing; groupOrder) if (existing is field) { exists = true; break; }
            if (!exists) groupOrder ~= field;
        }
        foreach (field; z8_query_array(query, "sortFields"w)) {
            bool selectedField;
            foreach (item; selected) if (item is field) { selectedField = true; break; }
            if (selectedField) read.sortFields ~= field;
        }
    }
    foreach (field; read.sortFields) {
        bool exists;
        foreach (existing; groupOrder) if (existing is field) { exists = true; break; }
        if (!exists) groupOrder ~= field;
    }
    read.sortFields = groupOrder;
    foreach (field; read.sortFields) {
        if (!z8_application_query_field_readable(query, field, privileges)) throw new Exception("query sort field access denied");
    }
    auto defaultSort = read.sortFields.dup;
    void writeDefaultMetadata() {
        if (!defaultAction) return;
        z8_application_query_write_meta(query, selected, groups, privileges, context);
        z8_application_query_write_order(defaultSort, defaultGroups);
    }
    auto lookupFields = params.z8_get("lookupFields"w), lookup = params.z8_get("lookup"w);
    if (!read.byRecordIds && lookupFields !is null && lookupFields.length) {
        auto parsed = z8_json_parse(lookupFields.to!string);
        if (parsed.type != JSONType.array) throw new Exception("query lookup fields must be a JSON array");
        Z8SqlBool match;
        foreach (item; parsed.array) {
            if (item.type != JSONType.string || !item.str.length) throw new Exception("invalid query lookup field");
            if (lookup is null || !lookup.length) continue;
            auto field = z8_query_find_field_id(query, item.str.to!wstring);
            if (!z8_application_query_field_readable(query, field, privileges)) throw new Exception("query lookup field access denied");
            auto type = z8_field_type_id(field);
            if (type != "string"w && type != "text"w && type != "file"w && type != "attachments"w) continue;
            auto column = type == "string"w ? z8_string_field_sql(field) : z8_text_field_sql(field);
            auto pattern = z8_sql_string_const("%"w ~ z8_str_toLowerCase(lookup) ~ "%"w);
            match = z8_filter_combine(match, z8_sql_string_like(z8_sql_string_toLower(column), pattern), true);
        }
        if (match !is null) read.where = z8_filter_combine(read.where, z8_sql_bool_group(match));
    }
    auto period = params.z8_get("period"w);
    if (!read.byRecordIds && period !is null && period.length) {
        auto parsed = z8_json_parse(period.to!string);
        if (parsed.type != JSONType.object) throw new Exception("query period must be an object");
        auto active = "active" in parsed.object;
        if (active !is null && z8_bool_parse(z8_json_value_text(*active))) {
            auto field = z8_query_state(query).period;
            if (field is null || field.z8_get_owner() !is query || z8_native_is_instance("org.zenframework.z8.server.base.table.value.Expression"w, field))
                throw new Exception("query period requires a direct date field");
            auto type = z8_field_type_id(field);
            if (type != "date"w && type != "datetime"w) throw new Exception("query period requires a date field");
            auto fieldKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, field);
            if (!privileges.getFieldAccess(queryKey, fieldKey).read) throw new Exception("query period field access denied");
            auto column = z8_datetime_field_sql(field);
            auto startDate = "start" in parsed.object, finishDate = "finish" in parsed.object;
            if (startDate !is null && startDate.type != JSONType.null_) {
                auto boundary = z8_date_parse(z8_json_value_text(*startDate));
                if (z8_date_ticks(boundary) != z8_date_ticks(z8_date_min))
                    read.where = z8_filter_combine(read.where, z8_sql_date_compare!">="(column, z8_sql_date_const(boundary)));
            }
            if (finishDate !is null && finishDate.type != JSONType.null_) {
                auto finish = z8_date_parse(z8_json_value_text(*finishDate));
                if (z8_date_ticks(finish) != z8_date_ticks(z8_date_max))
                    read.where = z8_filter_combine(read.where, z8_sql_date_compare!"<="(column, z8_sql_date_const(finish)));
            }
        }
    }
    foreach (name; ["filter"w, "where"w, "quickFilter"w]) {
        // ReadAction in the original Z8 uses the record IDs instead of the
        // request and query filters when a nonempty ID list is provided.
        if (read.byRecordIds) break;
        auto json = params.z8_get(name);
        if (json is null || !json.length) continue;
        auto filter = name == "where"w ? z8_application_query_where(context, query, json) : z8_application_query_filter(query, json);
        if (filter is null) continue; // An explicit empty array means no filter.
        foreach (field; z8_sql_collect_fields(filter)) {
            if (!z8_application_query_field_readable(query, field, privileges)) throw new Exception("query filter field access denied");
        }
        read.where = z8_filter_combine(read.where, filter);
    }
    auto having = params.z8_get("having"w);
    if (!read.byRecordIds && having !is null && having.length) {
        if (!groups.length && !aggregates.length) throw new Exception("query having requires grouped results");
        read.having = z8_application_query_filter(query, having);
        foreach (field; z8_sql_collect_fields(read.having)) {
            if (!z8_application_query_field_readable(query, field, privileges)) throw new Exception("query having field access denied");
        }
    }
    auto totals = params.z8_get("totals"w);
    bool isTotals = totals !is null && z8_bool_parse(totals);
    Z8Object[] totalsFields;
    Z8FieldAggregation[] aggregations;
    if (isTotals) {
        foreach (field; selected) {
            auto fieldState = z8_field_state(field);
            auto aggregation = z8_field_aggregation(field);
            if ((fieldState.totals.present && !fieldState.totals.value) || aggregation == Z8FieldAggregation.none ||
                aggregation == Z8FieldAggregation.min || aggregation == Z8FieldAggregation.max) continue;
            if (aggregation == Z8FieldAggregation.count) { aggregation = Z8FieldAggregation.sum; z8_field_set_aggregation(field, aggregation); }
            if (aggregation != Z8FieldAggregation.sum && aggregation != Z8FieldAggregation.average)
                throw new Exception("unsupported query totals aggregation");
            totalsFields ~= field; aggregations ~= aggregation;
        }
        read.aggregateSourceFields = selected.dup;
        read.fields = totalsFields; read.sortFields = null; read.start = 0; read.limit = -1;
        read.aggregateAll = true;
    }
    z8_query_capture_read(query, read);
    Z8Object customData;
    try {
        customData = query.z8_object_get_data(params);
        if (customData !is null) {
            if (!z8_native_is_instance(z8_json_array_name, customData)) throw new Exception("invalid query response data");
            // ReadAction already captured the request metadata. A custom
            // getData result does not execute the grouped SQL query.
            writeDefaultMetadata();
            writer.property("data"w); writer.raw(z8_json_array_render_state(z8_json_array_state(customData)));
        }
    } catch (Throwable error) {
        if (z8_query_user_events_enabled()) query.z8_query_virtual_afterRead();
        throw error;
    }
    if (customData !is null) {
        if (z8_query_user_events_enabled()) query.z8_query_virtual_afterRead();
        return;
    }
    if (isTotals) {
        try {
            verifyGroups();
            if (!groups.length && !aggregates.length && read.queryHaving !is null && (read.queryHaving.kind != Z8SqlBoolKind.constant || !read.queryHaving.constant))
                throw new Exception("query having is not supported for totals");
            foreach (i, field; totalsFields) if (z8_field_aggregation(field) != aggregations[i])
                throw new Exception("changing query aggregations after beforeRead is not supported");
            writeDefaultMetadata();
        } catch (Throwable error) {
            if (z8_query_user_events_enabled()) query.z8_query_virtual_afterRead();
            throw error;
        }
        auto saved = new Z8QuerySavedState(query, contextFields ~ selected);
        scope(exit) saved.restore();
        z8_query_open(query, read, false, true);
        writer.start('{', "data"w);
        if (z8_optional_require(z8_query_next(query))) z8_application_query_write_fields(query, writer, totalsFields);
        writer.finish('}');
        return;
    }
    bool prepared;
    auto count = params.z8_get("count"w);
    if (count !is null && z8_bool_parse(count)) {
        auto counter = cast(IZ8QueryCounter)z8_query_executor(query);
        if (counter is null) {
            if (z8_query_user_events_enabled()) query.z8_query_virtual_afterRead();
            throw new Exception("query executor does not support count");
        }
        long total;
        try { verifyGroups(); writeDefaultMetadata(); total = counter.count(query, read); }
        catch (Throwable error) {
            if (z8_query_user_events_enabled()) query.z8_query_virtual_afterRead();
            throw error;
        }
        z8_monitor_property(writer, "total"w, total);
        if (!total || read.start <= total) {
            if (z8_query_user_events_enabled()) query.z8_query_virtual_afterRead();
            return;
        }
        if (read.limit <= 0) {
            if (z8_query_user_events_enabled()) query.z8_query_virtual_afterRead();
            throw new Exception("query count pagination requires a positive limit");
        }
        read.start = cast(int)(((total - 1) / read.limit) * read.limit);
        prepared = true;
    }
    if (!prepared) {
        try { verifyGroups(); writeDefaultMetadata(); }
        catch (Throwable error) {
            if (z8_query_user_events_enabled()) query.z8_query_virtual_afterRead();
            throw error;
        }
    }
    // ReadAction's FramedSelect saves changed field values before attaching
    // the cursor, then restores them after writing the rows.
    auto saved = new Z8QuerySavedState(query, contextFields ~ selected);
    scope(exit) saved.restore();
    // Pagination was already resolved above; a TreeTable deliberately uses
    // an unlimited read even when the query has a configured default limit.
    z8_query_open(query, read, false, true);
    z8_application_query_write_rows(query, writer, selected);
}
