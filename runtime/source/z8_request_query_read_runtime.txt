// Parse every filter leaf explicitly: silently discarding an invalid leaf can
// widen an AND filter and return records the caller did not ask for.
private Z8SqlBool z8_application_query_filter_node(Z8Object query, JSONValue node) {
    if (node.type != JSONType.object) throw new Exception("invalid query filter node");
    auto logical = "logical" in node.object;
    if (logical is null) {
        auto property = "property" in node.object;
        if (property is null || property.type != JSONType.string || !property.str.length)
            throw new Exception("query filter requires a field");
        auto expression = z8_filter_node(query, node);
        if (expression is null) throw new Exception("empty query filter expression");
        return expression;
    }
    if (logical.type != JSONType.string) throw new Exception("invalid query filter operator");
    auto op = z8_str_toLowerCase(logical.str.to!wstring);
    if (op != "and"w && op != "or"w && op != "not"w) throw new Exception("invalid query filter operator");
    auto children = "expressions" in node.object;
    if (children is null || children.type != JSONType.array || !children.array.length)
        throw new Exception("query filter group requires expressions");
    Z8SqlBool result;
    foreach (child; children.array) result = z8_filter_combine(result, z8_application_query_filter_node(query, child), op == "or"w);
    return op == "not"w ? z8_sql_bool_not(result) : z8_sql_bool_group(result);
}
private Z8SqlBool z8_application_query_filter(Z8Object query, wstring json) {
    auto root = z8_json_parse(json.to!string);
    if (root.type != JSONType.array) return z8_application_query_filter_node(query, root);
    if (!root.array.length) return null;
    Z8SqlBool result;
    foreach (node; root.array) result = z8_filter_combine(result, z8_application_query_filter_node(query, node));
    return z8_sql_bool_group(result);
}

// Initial ReadAction adapter for table queries. Unsupported request options
// are rejected rather than silently changing the requested result.
void z8_application_dispatch_query_read(wstring className) {
    import std.json : JSONType;
    auto request = z8_application_require_request();
    auto session = z8_nonnull(request.session);
    auto params = z8_nonnull(request.parameters);
    auto writer = request.writer;
    if (writer.scopes.length != 2) throw new Exception("response object is not open");
    auto action = params.z8_get("action"w);
    if (action != "read"w) throw new Exception("unsupported query action");
    if (!z8_application_may_execute(session, className, action)) throw new Exception("request access denied");
    auto parameterKeys = params.z8_keys();
    for (long i; i < parameterKeys.z8_size(); ++i) {
        auto name = parameterKeys.z8_get(i);
        if (name is null) throw new Exception("invalid query read parameter");
        if (name == "request"w || name == "session"w || name == "action"w || name == "fields"w || name == "start"w || name == "limit"w ||
            name == "filter"w || name == "where"w || name == "quickFilter"w || name == "sort"w || name == "recordId"w) continue;
        if (params.z8_get(name).length) throw new Exception("unsupported query read parameter: " ~ name.to!string);
    }
    auto query = z8_object_new_instance_native(z8_string_required(className));
    if (!z8_native_is_instance("org.zenframework.z8.server.base.table.Table"w, query))
        throw new Exception("query read requires a Table");
    auto privileges = z8_nonnull(session.user).privileges;
    auto queryKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, query);
    if (privileges is null || !privileges.getTableAccess(queryKey).read) throw new Exception("query read access denied");
    auto key = z8_query_primary_key(query);
    if (key is null) throw new Exception("query has no primary key");
    Z8Object[] requested;
    auto fields = params.z8_get("fields"w);
    if (fields !is null && fields.length) {
        auto parsed = z8_json_parse(fields.to!string);
        if (parsed.type != JSONType.array) throw new Exception("query fields must be a JSON array");
        foreach (item; parsed.array) {
            wstring id;
            if (item.type == JSONType.string) id = item.str.to!wstring;
            else if (item.type == JSONType.object) {
                auto member = "id" in item.object;
                if (member is null || member.type != JSONType.string) throw new Exception("query field has no id");
                id = member.str.to!wstring;
            } else throw new Exception("invalid query field");
            requested ~= z8_query_find_field_id(query, id);
        }
    }
    if (!requested.length) requested = z8_query_declared_fields(query);
    requested ~= key;
    foreach (field; z8_query_declared_fields(query))
        if (field.z8_get_attribute("lockKey"w) !is null) requested ~= field;
    Z8Object[] selected;
    foreach (field; requested) {
        bool duplicate;
        foreach (existing; selected) if (existing is field) { duplicate = true; break; }
        if (duplicate) continue;
        auto owner = field.z8_get_owner();
        auto tableKey = owner is null ? queryKey : z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, owner);
        auto fieldKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, field);
        if (field !is key && !privileges.getFieldAccess(tableKey, fieldKey).read) continue;
        // Expressions and linked fields require authorization of every SQL
        // dependency; do not expose them until that check is implemented.
        if (owner !is query || z8_native_is_instance("org.zenframework.z8.server.base.table.value.Expression"w, field))
            throw new Exception("linked and computed fields are not supported by query read");
        selected ~= field;
    }
    long number(wstring key, long fallback) {
        auto value = params.z8_get(key);
        return value is null || !value.length ? fallback : z8_int_parse(value);
    }
    auto start = number("start"w, 0), limit = number("limit"w, z8_query_limit(query));
    if (start < 0 || start > int.max || limit < 0 || limit > int.max) throw new Exception("invalid query pagination");
    if (!limit) limit = z8_query_limit(query);
    Z8QueryReadRequest read;
    read.fields = selected; read.start = cast(int)start; read.limit = cast(int)limit;
    auto recordIds = params.z8_get("recordId"w);
    if (recordIds !is null && recordIds.length) {
        auto json = recordIds[0] == '[' ? recordIds : recordIds[0] == '"' ? "["w ~ recordIds ~ "]"w : "[\""w ~ recordIds ~ "\"]"w;
        auto parsed = z8_json_parse(json.to!string);
        if (parsed.type != JSONType.array) throw new Exception("recordId must be a GUID or a JSON array of GUIDs");
        if (parsed.array.length) {
            auto filter = new Z8SqlBool(Z8SqlBoolKind.inVector);
            filter.vectorValue = new Z8SqlFieldToken(key);
            foreach (item; parsed.array) {
                if (item.type != JSONType.string) throw new Exception("recordId must contain GUIDs");
                auto id = z8_guid_parse(item.str.to!wstring);
                if (id is null) throw new Exception("invalid recordId");
                filter.vectorItems ~= new Z8SqlBoxedConstant(z8_primary_box(id));
            }
            read.where = filter;
            read.byRecordIds = true;
        }
    }
    auto sorting = params.z8_get("sort"w);
    if (sorting !is null && sorting.length) {
        auto parsed = z8_json_parse(sorting.to!string);
        if (parsed.type != JSONType.array) throw new Exception("query sort must be a JSON array");
        foreach (item; parsed.array) {
            if (item.type != JSONType.object) throw new Exception("invalid query sort entry");
            auto property = "property" in item.object, direction = "direction" in item.object;
            if (property is null || property.type != JSONType.string || !property.str.length ||
                (direction !is null && direction.type != JSONType.string)) throw new Exception("invalid query sort entry");
            auto field = z8_query_find_field_id(query, property.str.to!wstring);
            auto order = direction is null ? "asc"w : z8_str_toLowerCase(direction.str.to!wstring);
            z8_field_set_sort_direction(field, order);
            bool exists;
            foreach (existing; read.sortFields) if (existing is field) { exists = true; break; }
            if (!exists) read.sortFields ~= field;
        }
    }
    if (!read.sortFields.length) read.sortFields = z8_query_array(query, "sortFields"w);
    foreach (field; read.sortFields) {
        if (field.z8_get_owner() !is query || z8_native_is_instance("org.zenframework.z8.server.base.table.value.Expression"w, field))
            throw new Exception("linked and computed sort fields are not supported");
        auto fieldKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, field);
        if (field !is key && !privileges.getFieldAccess(queryKey, fieldKey).read) throw new Exception("query sort field access denied");
    }
    foreach (name; ["filter"w, "where"w, "quickFilter"w]) {
        // ReadAction in the original Z8 uses the record IDs instead of the
        // request and query filters when a nonempty ID list is provided.
        if (read.byRecordIds) break;
        auto json = params.z8_get(name);
        if (json is null || !json.length) continue;
        auto filter = z8_application_query_filter(query, json);
        if (filter is null) continue; // An explicit empty array means no filter.
        foreach (field; z8_sql_collect_fields(filter)) {
            if (field.z8_get_owner() !is query || z8_native_is_instance("org.zenframework.z8.server.base.table.value.Expression"w, field))
                throw new Exception("linked and computed filter fields are not supported");
            auto fieldKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, field);
            if (field !is key && !privileges.getFieldAccess(queryKey, fieldKey).read) throw new Exception("query filter field access denied");
        }
        read.where = z8_filter_combine(read.where, filter);
    }
    z8_query_open(query, read);
    scope(exit) z8_query_close(query);
    writer.start('[', "data"w);
    while (z8_optional_require(z8_query_next(query))) {
        writer.start('{');
        foreach (field; selected) {
            auto value = z8_field_primary(field);
            auto wasNull = z8_optional_require(z8_field_was_null(field));
            if (field is key && wasNull) continue;
            writer.property(field.z8_id());
            if (value !is null && value.kind == Z8PrimaryKind.geometry &&
                (wasNull || (value.geometryValue !is null && value.geometryValue.rawHex is null && value.geometryValue.shape == -1)))
                writer.raw("null"w);
            else z8_json_writer_primary(writer, value);
        }
        writer.finish('}');
    }
    writer.finish(']');
}
