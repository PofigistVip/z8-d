// Map.equals in the original runtime compares entries, not map identity.
private bool z8_map_content_equals(M)(M left, M right) {
    if (left is right) return true;
    if (left.z8_size() != right.z8_size()) return false;
    auto keys = left.z8_keys();
    auto values = left.z8_values();
    for (long i; i < keys.z8_size(); ++i) {
        auto key = keys.z8_get(i);
        if (!right.z8_containsKey(key)) return false;
        auto a = values.z8_get(i), b = right.z8_get(key);
        static if (is(typeof(a) == wstring)) {
            if (!z8_string_equals(a, b)) return false;
        } else static if (is(typeof(a) == class)) {
            // AbstractMap.equals calls the stored value's equals directly.
            if (a is null ? b !is null : (b is null || !a.opEquals(b))) return false;
        } else if (a != b) return false;
    }
    return true;
}
class Z8StringMap {
    private wstring[wstring] entries;
    // Retain insertion order so serializers can reproduce Java HashMap bucket ties.
    private wstring[] insertionKeys;
    private bool hasNullKey;
    private wstring nullKeyValue;
    override bool opEquals(Object other) {
        auto right = cast(Z8StringMap)other;
        return right !is null && z8_map_content_equals(this, right);
    }
    long z8_size() { return cast(long)entries.length + (hasNullKey ? 1 : 0); }
    bool z8_isEmpty() { return entries.length == 0 && !hasNullKey; }
    void z8_clear() { entries.clear(); insertionKeys = null; hasNullKey = false; nullKeyValue = null; }
    bool z8_containsKey(wstring key) { return key is null ? hasNullKey : (key in entries) !is null; }
    bool z8_containsValue(wstring value) {
        if (hasNullKey && z8_string_equals(nullKeyValue, value)) return true;
        foreach (item; entries) if (z8_string_equals(item, value)) return true;
        return false;
    }
    wstring z8_add(wstring key, wstring value) {
        if (key is null) {
            auto previous = hasNullKey ? nullKeyValue : null;
            hasNullKey = true; nullKeyValue = value; return previous;
        }
        auto found = key in entries;
        wstring previous = found is null ? null : *found;
        if (found is null) insertionKeys ~= key;
        entries[key] = value;
        return previous;
    }
    void z8_add(Z8StringMap other) {
        z8_nonnull(other);
        if (other is this) return;
        foreach (key, value; other.entries) z8_add(key, value);
        if (other.hasNullKey) z8_add(null, other.nullKeyValue);
    }
    Z8Array!wstring z8_keys() {
        auto result = new Z8Array!wstring();
        foreach (key, value; entries) result.z8_add(key);
        if (hasNullKey) result.z8_add(null);
        return result;
    }
    Z8Array!wstring z8_values() {
        auto result = new Z8Array!wstring();
        foreach (value; entries) result.z8_add(value);
        if (hasNullKey) result.z8_add(nullKeyValue);
        return result;
    }
    wstring z8_get(wstring key) {
        if (key is null) return hasNullKey ? nullKeyValue : null;
        auto found = key in entries;
        if (found is null) return null;
        return *found;
    }
    wstring z8_remove(wstring key) {
        if (key is null) {
            auto previous = hasNullKey ? nullKeyValue : null;
            hasNullKey = false; nullKeyValue = null; return previous;
        }
        auto found = key in entries;
        if (found is null) return null;
        auto previous = *found;
        entries.remove(key);
        foreach (i, existing; insertionKeys) if (existing == key) {
            insertionKeys = insertionKeys[0 .. i] ~ insertionKeys[i + 1 .. $];
            break;
        }
        return previous;
    }
}
Z8StringMap z8_map_union(Z8StringMap left, Z8StringMap right) {
    auto result = new Z8StringMap();
    result.z8_add(z8_nonnull(left));
    result.z8_add(z8_nonnull(right));
    return result;
}
void z8_map_add_assign(ref Z8StringMap left, Z8StringMap right) { left = z8_map_union(left, right); }
void z8_map_add_assign(ref Z8StringMap left, lazy Z8StringMap previous, lazy Z8StringMap right) {
    auto oldValue = previous;
    auto newValue = right;
    left = z8_map_union(oldValue, newValue);
}
