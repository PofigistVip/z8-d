// Discarded map results must not trigger a generic value cast on the JVM.
void z8_map_add_discard(M, K, V)(M map, K key, V value) {
    z8_nonnull(map);
    static if (__traits(compiles, map.z8_add_discard(key, value))) map.z8_add_discard(key, value);
    else map.z8_add(key, value);
}
void z8_map_remove_discard(M, K)(M map, K key) {
    z8_nonnull(map);
    static if (__traits(compiles, map.z8_remove_discard(key))) map.z8_remove_discard(key);
    else map.z8_remove(key);
}

// Indexed assignment is a z8-d extension. Resolve its receiver and key once,
// then read the old entry before evaluating the right-hand side.
void z8_map_entry_compound(string op, M, K, V)(M map, K key, lazy V value) {
    z8_nonnull(map);
    auto before = map.z8_get(key);
    auto after = value;
    static if (is(typeof(before) == Z8Optional!long))
        map.z8_add(key, z8_optional_wrap(z8_optional_int_binary!op(before, after)));
    else static if (is(typeof(before) == Z8Optional!Z8Decimal))
        map.z8_add(key, z8_optional_wrap(z8_optional_decimal_binary!op(before, after)));
    else static if (is(typeof(before) == wstring) && op == "+")
        map.z8_add(key, z8_nullable_string_binary!"+"(before, z8_string_or_empty(after)));
    else static if (op == "+" && __traits(compiles, z8_array_union(before, after)))
        map.z8_add(key, z8_array_union(before, after));
    else static if (op == "+" && __traits(compiles, z8_map_union(before, after)))
        map.z8_add(key, z8_map_union(before, after));
    else static assert(0, "unsupported map entry compound assignment");
}

// RLinkedHashMap.java extends HashMap; ordinary buckets iterate in bucket order,
// retaining insertion order within a bucket. Keep its capacity across clear.
// Treeified collision buckets have additional Java-specific ordering.
private class Z8JavaMapOrder(K) {
    private struct Entry { K key; uint hash; }
    private Entry[] keys;
    private size_t capacity;
    private size_t[size_t] bucketCounts;
    private static uint hash(K key) {
        uint value;
        static if (is(K == wstring)) {
            if (key !is null) foreach (wchar c; key) value = value * 31 + c;
        } else static if (is(K == ulong)) {
            value = cast(uint)(key ^ (key >> 32));
        } else {
            if (key.present) value = cast(uint)z8_primary_hash(z8_primary_box(key));
        }
        return value ^ (value >> 16);
    }
    private static bool same(K left, K right) {
        static if (is(K == wstring)) return left is null ? right is null : (right !is null && left == right);
        else static if (is(K == ulong)) return left == right;
        else {
            if (left.present != right.present) return false;
            if (!left.present) return true;
            static if (is(K == Z8Optional!Z8Decimal))
                return left.value.nan ? right.value.nan : (!right.value.nan && left.value.scale == right.value.scale && left.value.coefficient == right.value.coefficient);
            else return left.value == right.value;
        }
    }
    private void grow() {
        capacity *= 2; bucketCounts.clear();
        foreach (key; keys) {
            auto bucket = key.hash & (capacity - 1);
            auto count = bucket in bucketCounts;
            if (count is null) bucketCounts[bucket] = 1; else ++*count;
        }
    }
    void preparePutAll(size_t incoming) {
        if (!incoming) return;
        if (!capacity) {
            size_t needed = incoming + incoming / 3 + 1;
            capacity = 1;
            while (capacity < needed && capacity < (cast(size_t)1 << 30)) capacity *= 2;
        } else if (incoming > capacity * 3 / 4 && capacity < (cast(size_t)1 << 30)) grow();
    }
    void add(K key) { addWithHash(key, hash(key)); }
    void addWithHash(K key, uint keyHash) {
        if (!capacity) capacity = 16;
        auto bucket = keyHash & (capacity - 1);
        auto count = bucket in bucketCounts;
        auto collisions = count is null ? 0 : *count;
        keys ~= Entry(key, keyHash);
        bucketCounts[bucket] = collisions + 1;
        if (collisions >= 8 && capacity < 64) grow();
        if (keys.length > capacity * 3 / 4 && capacity < (cast(size_t)1 << 30)) grow();
    }
    void remove(K key) {
        foreach (i, existing; keys) if (same(existing.key, key)) {
            auto bucket = existing.hash & (capacity - 1);
            if (--bucketCounts[bucket] == 0) bucketCounts.remove(bucket);
            keys = keys[0 .. i] ~ keys[i + 1 .. $]; return;
        }
    }
    void clear() { keys = null; bucketCounts.clear(); }
    K[] ordered() {
        import std.algorithm.sorting : sort;
        struct OrderedKey { K value; size_t bucket, index; }
        OrderedKey[] entries; entries.reserve(keys.length);
        foreach (i, key; keys) entries ~= OrderedKey(key.key, key.hash & (capacity - 1), i);
        sort!((a, b) => a.bucket < b.bucket || (a.bucket == b.bucket && a.index < b.index))(entries);
        K[] result; result.reserve(entries.length);
        foreach (entry; entries) result ~= entry.value;
        return result;
    }
}

// Map.equals in the original runtime compares entries, not map identity.
private bool z8_map_content_equals(M)(M left, M right) {
    if (left is right) return true;
    if (left.z8_size() != right.z8_size()) return false;
    auto keys = left.z8_keys();
    auto values = left.z8_values();
    for (long i; i < keys.z8_size(); ++i) {
        auto key = keys.z8_get(i);
        if (!right.z8_containsKey(key)) return false;
        auto a = values.z8_get(i), b = right.z8_get(key);
        static if (is(typeof(a) == wstring)) {
            if (!z8_string_equals(a, b)) return false;
        } else static if (is(typeof(a) == class)) {
            // AbstractMap.equals calls the stored value's equals directly.
            if (a is null ? b !is null : (b is null || !a.opEquals(b))) return false;
        } else if (a != b) return false;
    }
    return true;
}
class Z8StringMap {
    private Z8PrimaryMap primaryData;
    private Z8PrimaryKeyMap!wstring primaryKeyData;
    private wstring[wstring] entries;
    private Z8JavaMapOrder!wstring order;
    // Retain insertion order so serializers can reproduce Java HashMap bucket ties.
    private wstring[] insertionKeys;
    private bool hasNullKey;
    private wstring nullKeyValue;
    override bool opEquals(Object other) {
        auto right = cast(Z8StringMap)other;
        if (right !is null && (primaryKeyData !is null || right.primaryKeyData !is null))
            return z8_primary_key_map_view(this).opEquals(z8_primary_key_map_view(right));
        auto primaryKeys = cast(Z8PrimaryKeyMap!wstring)other;
        if (primaryKeys !is null) return z8_primary_key_map_view(this).opEquals(primaryKeys);
        if (right !is null && (primaryData !is null || right.primaryData !is null))
            return z8_primary_map_view(this).opEquals(z8_primary_map_view(right));
        auto primary = cast(Z8PrimaryMap)other;
        if (primary !is null) return z8_primary_map_view(this).opEquals(primary);
        return right !is null && z8_map_content_equals(this, right);
    }
    long z8_size() { if (primaryKeyData !is null) return primaryKeyData.z8_size(); if (primaryData !is null) return primaryData.z8_size(); return cast(long)entries.length + (hasNullKey ? 1 : 0); }
    bool z8_isEmpty() { if (primaryKeyData !is null) return primaryKeyData.z8_isEmpty(); if (primaryData !is null) return primaryData.z8_isEmpty(); return entries.length == 0 && !hasNullKey; }
    private Z8JavaMapOrder!wstring ordering() { if (order is null) order = new Z8JavaMapOrder!wstring(); return order; }
    void z8_clear() { if (primaryKeyData !is null) { primaryKeyData.z8_clear(); return; } if (primaryData !is null) { primaryData.z8_clear(); return; } entries.clear(); insertionKeys = null; hasNullKey = false; nullKeyValue = null; if (order !is null) order.clear(); }
    bool z8_containsKey(wstring key) { if (primaryKeyData !is null) return primaryKeyData.z8_containsKey(z8_primary_box(key)); if (primaryData !is null) return primaryData.z8_containsKey(key); return key is null ? hasNullKey : (key in entries) !is null; }
    bool z8_containsValue(wstring value) {
        if (primaryKeyData !is null) return primaryKeyData.z8_containsValue(value);
        if (primaryData !is null) return primaryData.z8_containsValue(z8_primary_box(value));
        if (hasNullKey && z8_string_equals(nullKeyValue, value)) return true;
        foreach (item; entries) if (z8_string_equals(item, value)) return true;
        return false;
    }
    wstring z8_add(wstring key, wstring value) {
        if (primaryKeyData !is null) return primaryKeyData.z8_add(z8_primary_box(key), value);
        if (primaryData !is null) return z8_primary_cast!wstring(primaryData.z8_add(key, z8_primary_box(value)));
        if (key is null) {
            auto previous = hasNullKey ? nullKeyValue : null;
            if (!hasNullKey) ordering().add(null);
            hasNullKey = true; nullKeyValue = value; return previous;
        }
        auto found = key in entries;
        wstring previous = found is null ? null : *found;
        if (found is null) { insertionKeys ~= key; ordering().add(key); }
        entries[key] = value;
        return previous;
    }
    void z8_add_discard(wstring key, wstring value) {
        if (primaryKeyData !is null) { primaryKeyData.z8_add_discard(z8_primary_box(key), value); return; }
        if (primaryData !is null) { primaryData.z8_add(key, z8_primary_box(value)); return; }
        z8_add(key, value);
    }
    void z8_remove_discard(wstring key) {
        if (primaryKeyData !is null) { primaryKeyData.z8_remove_discard(z8_primary_box(key)); return; }
        if (primaryData !is null) { primaryData.z8_remove(key); return; }
        z8_remove(key);
    }
    void z8_add(Z8StringMap other) {
        z8_nonnull(other);
        if (primaryKeyData !is null || other.primaryKeyData !is null) {
            z8_primary_key_map_view(this).z8_add(z8_primary_key_map_view(other)); return;
        }
        if (primaryData !is null || other.primaryData !is null) {
            z8_primary_map_view(this).z8_add(z8_primary_map_view(other)); return;
        }
        if (other is this) return;
        ordering().preparePutAll(cast(size_t)other.z8_size());
        auto keys = other.z8_keys();
        for (long i; i < keys.z8_size(); ++i) { auto key = keys.z8_get(i); z8_add(key, other.z8_get(key)); }
    }
    Z8Array!wstring z8_keys() {
        if (primaryKeyData !is null) { auto result = new Z8Array!wstring(); result.primaryData = primaryKeyData.z8_keys(); return result; }
        if (primaryData !is null) return primaryData.z8_keys();
        auto result = new Z8Array!wstring();
        if (order !is null) foreach (key; order.ordered()) result.z8_add(key);
        return result;
    }
    // Stable order is needed when copying file details with equal hash buckets.
    Z8Array!wstring z8_insertion_keys() {
        if (primaryKeyData !is null) { auto result = new Z8Array!wstring(); result.primaryData = primaryKeyData.z8_insertion_keys(); return result; }
        if (primaryData !is null) return primaryData.z8_insertion_keys();
        auto result = new Z8Array!wstring();
        foreach (key; insertionKeys) result.z8_add(key);
        if (hasNullKey) result.z8_add(null);
        return result;
    }
    Z8Array!wstring z8_values() {
        if (primaryKeyData !is null) return primaryKeyData.z8_values();
        if (primaryData !is null) {
            auto result = new Z8Array!wstring(); result.primaryData = primaryData.z8_values(); return result;
        }
        auto result = new Z8Array!wstring();
        if (order !is null) foreach (key; order.ordered()) result.z8_add(z8_get(key));
        return result;
    }
    wstring z8_get(wstring key) {
        if (primaryKeyData !is null) return primaryKeyData.z8_get(z8_primary_box(key));
        if (primaryData !is null) return z8_primary_cast!wstring(primaryData.z8_get(key));
        if (key is null) return hasNullKey ? nullKeyValue : null;
        auto found = key in entries;
        if (found is null) return null;
        return *found;
    }
    wstring z8_remove(wstring key) {
        if (primaryKeyData !is null) return primaryKeyData.z8_remove(z8_primary_box(key));
        if (primaryData !is null) return z8_primary_cast!wstring(primaryData.z8_remove(key));
        if (key is null) {
            auto previous = hasNullKey ? nullKeyValue : null;
            if (hasNullKey) ordering().remove(null);
            hasNullKey = false; nullKeyValue = null; return previous;
        }
        auto found = key in entries;
        if (found is null) return null;
        auto previous = *found;
        entries.remove(key);
        ordering().remove(key);
        foreach (i, existing; insertionKeys) if (existing == key) {
            insertionKeys = insertionKeys[0 .. i] ~ insertionKeys[i + 1 .. $];
            break;
        }
        return previous;
    }
}
Z8StringMap z8_map_union(Z8StringMap left, Z8StringMap right) {
    auto result = new Z8StringMap();
    result.z8_add(z8_nonnull(left));
    result.z8_add(z8_nonnull(right));
    return result;
}
void z8_map_add_assign(ref Z8StringMap left, Z8StringMap right) { left = z8_map_union(left, right); }
void z8_map_add_assign(ref Z8StringMap left, lazy Z8StringMap previous, lazy Z8StringMap right) {
    auto oldValue = previous;
    auto newValue = right;
    left = z8_map_union(oldValue, newValue);
}

Z8PrimaryMap z8_primary_map_view(Z8StringMap values) {
    if (values is null) return null;
    if (values.primaryData is null) {
        if (values.primaryKeyData !is null) {
            auto typed = new Z8PrimaryMap(); typed.primaryKeyData = z8_primary_map_view(values.primaryKeyData);
            values.primaryData = typed; return typed;
        }
        auto result = new Z8PrimaryMap();
        foreach (key, value; values.entries) result.entries[key] = z8_primary_box(value);
        result.order = values.order; result.insertionKeys = values.insertionKeys;
        result.hasNullKey = values.hasNullKey; result.nullKeyValue = z8_primary_box(values.nullKeyValue);
        values.entries = null; values.order = null; values.insertionKeys = null;
        values.primaryData = result;
    }
    return values.primaryData;
}

Z8PrimaryKeyMap!wstring z8_primary_key_map_view(Z8StringMap values) {
    if (values is null) return null;
    if (values.primaryKeyData is null) {
        if (values.primaryData !is null) {
            auto primaryValues = z8_primary_key_map_view(values.primaryData);
            auto typed = new Z8PrimaryKeyMap!wstring(); typed.primaryData = primaryValues;
            values.primaryKeyData = typed; return typed;
        }
        auto result = new Z8PrimaryKeyMap!wstring();
        if (values.order !is null) {
            result.order = new Z8JavaMapOrder!ulong(); result.order.capacity = values.order.capacity;
            foreach (entry; values.order.keys) {
                auto key = entry.key;
                auto value = key is null ? values.nullKeyValue : values.entries[key];
                result.z8_add(z8_primary_box(key), value);
            }
        }
        values.entries = null; values.order = null; values.insertionKeys = null; values.primaryKeyData = result;
    }
    return values.primaryKeyData;
}
