// Native implementation of base.form.action.Parameter.
private enum z8_parameter_name = "org.zenframework.z8.server.base.form.action.Parameter"w;
private class Z8ParameterState {
    wstring text = ""w, type = "none"w;
    Z8Object source;
    Z8Primary value;
    Z8Optional!bool visible = z8_optional_wrap(true);
    this() { value = z8_primary_box(""w); }
}
private Z8ParameterState z8_parameter_state(Z8Object self) {
    self = z8_nonnull(self);
    if (!z8_native_is_instance(z8_parameter_name, self)) throw new Exception("invalid Parameter object");
    auto state = cast(Z8ParameterState)self.z8_get_native_state(z8_parameter_name);
    if (state is null) { state = new Z8ParameterState(); self.z8_set_native_state(z8_parameter_name, state); }
    return state;
}
private auto z8_parameter_address(string member)(Z8Object self) { return &__traits(getMember, z8_parameter_state(self), member); }
private wstring z8_parameter_type(Z8ParameterState state) {
    if (state.type != "none"w) return state.type;
    if (state.value is null) return "string"w;
    final switch (state.value.kind) {
        case Z8PrimaryKind.boolean: return "boolean"w;
        case Z8PrimaryKind.integer: return "int"w;
        case Z8PrimaryKind.decimal: return "float"w;
        case Z8PrimaryKind.date: return "date"w;
        case Z8PrimaryKind.guid: return "guid"w;
        case Z8PrimaryKind.text: case Z8PrimaryKind.binary: case Z8PrimaryKind.datespan:
        case Z8PrimaryKind.geometry: case Z8PrimaryKind.file: case Z8PrimaryKind.user: return "string"w;
    }
}
private Z8Object z8_parameter_create(wstring text, Z8Primary value, Z8Optional!bool visible) {
    auto item = z8_native_new(z8_parameter_name);
    auto state = z8_parameter_state(item);
    state.text = text; state.value = value; state.visible = visible;
    return item;
}
private Z8Object z8_parameter_create_scalar(T)(wstring text, T value) {
    return z8_parameter_create(text, z8_primary_box(value), z8_optional_wrap(true));
}
private Z8Object z8_parameter_create_source(wstring text, Z8Object source) {
    auto sourceState = z8_parameter_source_state(source);
    auto item = z8_parameter_create(text, z8_primary_box(""w), z8_optional_wrap(true));
    auto state = z8_parameter_state(item);
    state.type = sourceState.valueField is null ? "guid"w : z8_field_type_name(sourceState.valueField);
    state.source = source;
    return item;
}
private Z8Object z8_parameter_create_value_source(wstring text, Z8Primary value, Z8Object source) {
    auto item = z8_parameter_create(text, value, z8_optional_wrap(true));
    z8_parameter_state(item).source = source;
    return item;
}
T z8_parameter_create_type(T, E)(wstring text, E type, Z8Optional!bool visible) if (is(T : Z8Object)) {
    if (type is null) throw new Exception("null parameter type");
    auto item = cast(T)z8_native_new(z8_parameter_name);
    if (item is null) throw new Exception("invalid native Parameter type");
    auto state = z8_parameter_state(item);
    state.text = text; state.visible = visible;
    if (type is E.e_None) state.type = "none"w;
    else if (type is E.e_Guid) state.type = "guid"w;
    else if (type is E.e_Geometry) state.type = "geometry"w;
    else if (type is E.e_Boolean) state.type = "boolean"w;
    else if (type is E.e_Integer) state.type = "int"w;
    else if (type is E.e_String) state.type = "string"w;
    else if (type is E.e_Date) state.type = "date"w;
    else if (type is E.e_Datetime) state.type = "datetime"w;
    else if (type is E.e_Datespan) state.type = "datespan"w;
    else if (type is E.e_Decimal) state.type = "float"w;
    else if (type is E.e_Binary) state.type = "binary"w;
    else if (type is E.e_Text) state.type = "text"w;
    else if (type is E.e_Attachments) state.type = "attachments"w;
    else if (type is E.e_File) state.type = "file"w;
    else if (type is E.e_Null) state.type = "null"w;
    else throw new Exception("unknown parameter type");
    return item;
}
private auto z8_parameter_cast(T)(Z8Object self) { return z8_primary_cast!T(z8_parameter_state(self).value); }
private void z8_parameter_parse(Z8Object item, wstring text) {
    auto state = z8_parameter_state(item);
    auto type = z8_parameter_type(state);
    // Parameter.getType caches an inferred type before replacing its value.
    state.type = type;
    switch (type) {
        case "string"w: case "text"w: state.value = z8_primary_box(text); break;
        case "int"w: state.value = z8_primary_box(z8_int_parse(text)); break;
        case "float"w: state.value = z8_primary_box(z8_decimal_parse(text)); break;
        case "boolean"w: state.value = z8_primary_box(z8_bool_parse(text)); break;
        case "guid"w: state.value = z8_primary_box(z8_guid_parse(text)); break;
        case "date"w: case "datetime"w: state.value = z8_primary_box(text.length ? z8_date_parse(text) : z8_date_now()); break;
        case "datespan"w: state.value = z8_primary_box(text.length ? z8_datespan_parse(text) : new Z8Datespan()); break;
        default: throw new Exception("unsupported parameter type: " ~ type.to!string);
    }
}
private Z8Object z8_action_parameter_by_id(Z8ActionState state, wstring id) {
    if (state.parameters !is null && state.parameters.view() !is null) {
        auto candidates = state.parameters.view().z8_erased_data();
        for (long i; i < candidates.z8_size(); ++i) {
            auto candidate = cast(Z8Object)candidates.z8_get(i);
            if (z8_parameter_state(candidate).text == id) return candidate;
        }
    }
    throw new Exception("unknown action parameter: " ~ z8_string_required(id).to!string);
}
private void z8_parameter_write_meta(Z8Object item, Z8JsonWriterState writer) {
    auto state = z8_parameter_state(item);
    auto type = z8_parameter_type(state);
    writer.start('{');
    z8_monitor_property(writer, "id"w, state.text); z8_monitor_property(writer, "text"w, state.text);
    z8_monitor_property(writer, "type"w, type); z8_monitor_property(writer, "visible"w, z8_optional_require(state.visible));
    writer.property("value"w); z8_json_writer_primary(writer, state.value);
    writer.start('{', "field"w); z8_monitor_property(writer, "header"w, state.text);
    if (state.source is null) z8_monitor_property(writer, "type"w, type);
    else z8_parameter_source_write_meta(state.source, writer);
    writer.finish('}');
    writer.finish('}');
}
