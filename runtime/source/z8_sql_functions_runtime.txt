private class Z8SqlFieldToken : Z8SqlPrimary {
    Z8Object field;
    this(Z8Object field) { this.field = field; }
}
private class Z8SqlBoxedConstant : Z8SqlPrimary {
    Z8Primary value;
    this(Z8Primary value) { this.value = value; }
}
private wstring z8_sql_functions_format(Z8SqlPrimary value, Z8SqlVendor vendor) {
    z8_nonnull(value);
    if (auto item = cast(Z8SqlFieldToken)value) return z8_field_sql_format(item.field, vendor);
    if (auto item = cast(Z8SqlBoxedConstant)value) {
        auto primary = z8_nonnull(item.value);
        final switch (primary.kind) {
            case Z8PrimaryKind.integer: return z8_sql_int_format(z8_sql_int_const(primary.integerValue), vendor);
            case Z8PrimaryKind.boolean: return z8_sql_bool_format(z8_sql_bool_const(primary.booleanValue), false, vendor);
            case Z8PrimaryKind.decimal: return z8_sql_decimal_format(z8_sql_decimal_const(primary.decimalValue), vendor);
            case Z8PrimaryKind.text: return z8_sql_string_format(z8_sql_string_const(primary.textValue), vendor);
            case Z8PrimaryKind.guid: return z8_sql_guid_format(z8_sql_guid_const(primary.guidValue), vendor);
            case Z8PrimaryKind.date: return z8_sql_date_format(z8_sql_date_const(primary.dateValue), vendor);
            case Z8PrimaryKind.datespan: return z8_sql_datespan_format(z8_sql_datespan_const(primary.datespanValue), vendor);
            case Z8PrimaryKind.binary: return z8_sql_binary_format(new Z8SqlBinary(), vendor);
            case Z8PrimaryKind.geometry:
                auto text = z8_sql_geometry_format(z8_sql_geometry_const(primary.geometryValue), vendor);
                return text is null ? "null"w : text;
            case Z8PrimaryKind.file: case Z8PrimaryKind.user: throw new Exception("unsupported SQL primary constant");
        }
    }
    if (auto item = cast(Z8SqlDatespan)value) return z8_sql_datespan_format(item, vendor);
    if (auto item = cast(Z8SqlDate)value) return z8_sql_date_format(item, vendor);
    if (auto item = cast(Z8SqlDecimal)value) return z8_sql_decimal_format(item, vendor);
    if (auto item = cast(Z8SqlGuid)value) return z8_sql_guid_format(item, vendor);
    if (auto item = cast(Z8SqlInt)value) return z8_sql_int_format(item, vendor);
    if (auto item = cast(Z8SqlString)value) return z8_sql_string_format(item, vendor);
    throw new Exception("unsupported SQL inVector value");
}
private Z8SqlBool z8_sql_in_vector(S, T)(S value, Z8Array!T values) {
    auto result = new Z8SqlBool(Z8SqlBoolKind.inVector);
    result.vectorValue = value;
    z8_nonnull(values);
    for (long i = 0; i < values.z8_size(); ++i) {
        auto item = values.z8_get(i);
        static if (is(T == long)) result.vectorItems ~= item.present ? z8_sql_int_const(item.value) : null;
        else static if (is(T == Z8Decimal)) result.vectorItems ~= item.present ? z8_sql_decimal_const(item.value) : null;
        else static if (is(T == wstring)) result.vectorItems ~= item is null ? null : z8_sql_string_const(item);
        else static if (is(T == Z8Date)) result.vectorItems ~= item is null ? null : z8_sql_date_const(item);
        else static if (is(T == Z8Datespan)) result.vectorItems ~= item is null ? null : z8_sql_datespan_const(item);
        else static if (is(T == Z8Guid)) result.vectorItems ~= item is null ? null : z8_sql_guid_const(item);
        else static assert(0, "unsupported SQL inVector array");
    }
    return result;
}
