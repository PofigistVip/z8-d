// OOXML column visibility for the POI report engine. Other workbook parts are
// retained as archive members; worksheet XML is edited without rewriting cells.
private struct Z8PoiColumn {
    int first, last;
    wstring name;
    wstring[wstring] attributes;
}

private struct Z8PoiReportBlock { long firstRow, lastRow, firstColumn, lastColumn; }
private Z8PoiReportBlock z8_poi_report_block(wstring address) {
    import std.string : split;
    auto parts = z8_string_required(address).split(":"w);
    if (parts.length != 1 && parts.length != 2) throw new Exception("invalid POI report range address");
    void parseCell(wstring text, out long row, out long column) {
        size_t position;
        if (position < text.length && text[position] == '$') ++position;
        auto start = position;
        while (position < text.length && text[position] >= 'A' && text[position] <= 'Z') {
            column = column * 26 + text[position++] - 'A' + 1;
            if (column > 16384) throw new Exception("invalid POI report column");
        }
        if (position == start) throw new Exception("invalid POI report range address");
        if (position < text.length && text[position] == '$') ++position;
        start = position;
        while (position < text.length && text[position] >= '0' && text[position] <= '9') {
            row = row * 10 + text[position++] - '0';
            if (row > 1048576) throw new Exception("invalid POI report row");
        }
        if (position == start || position != text.length || !row) throw new Exception("invalid POI report range address");
        --row; --column;
    }
    Z8PoiReportBlock result;
    parseCell(parts[0], result.firstRow, result.firstColumn);
    if (parts.length == 2) parseCell(parts[1], result.lastRow, result.lastColumn);
    else { result.lastRow = result.firstRow; result.lastColumn = result.firstColumn; }
    if (result.lastRow < result.firstRow || result.lastColumn < result.firstColumn)
        throw new Exception("invalid POI report range order");
    return result;
}

// A single, non-expanding simple source can be applied without editing cells
// if the sheet contains no report expressions or reserved text delimiters.
private void z8_poi_report_apply_simple_range(Z8Object report, Z8Object range, Z8PoiWorkbook workbook,
                                              out int renameSheet, out wstring renameName) {
    import std.algorithm.searching : canFind;
    range = z8_nonnull(range); range.z8_ensure();
    auto rangeState = z8_report_range_state(range);
    if (rangeState.ranges !is null && rangeState.ranges.view().z8_erased_data().z8_size())
        throw new Exception("nested POI report ranges are not implemented");
    foreach (attribute; ["totals"w, "subtotal"w, "subtotalsBy"w, "subtotalsMerge"w, "merge"w])
        if (range.z8_get_attribute(attribute) !is null) throw new Exception("POI report range option is not implemented");
    auto axisText = range.z8_get_attribute("axis"w);
    if (axisText !is null && axisText.to!int != 0 && axisText.to!int != 1) throw new Exception("invalid POI report range axis");
    auto sheetText = range.z8_get_attribute("sheet"w);
    auto sheet = sheetText is null ? 0 : sheetText.to!int;
    if (sheet < 0 || sheet >= workbook.sheets.length) throw new Exception("POI report sheet expansion is not implemented");
    renameSheet = sheet; renameName = range.z8_get_attribute("sheetName"w);
    if (workbook.sheets[sheet].hasMergedRegions) throw new Exception("merged POI report range cells are not implemented");
    auto block = z8_poi_report_block(range.z8_get_attribute("address"w));
    auto boundariesText = range.z8_get_attribute("boundaries"w);
    if (boundariesText !is null) {
        auto boundaries = z8_poi_report_block(boundariesText);
        if (block.firstRow < boundaries.firstRow || block.lastRow > boundaries.lastRow ||
            block.firstColumn < boundaries.firstColumn || block.lastColumn > boundaries.lastColumn)
            throw new Exception("POI report block is outside its boundaries");
    } else {
        long rows = 1, columns = 0;
        foreach (rowNumber, row; workbook.sheets[sheet].rows) {
            if (rowNumber + 1 > rows) rows = rowNumber + 1;
            if (rowNumber < 1000) foreach (columnNumber, cell; row.cells) if (columnNumber + 1 > columns) columns = columnNumber + 1;
        }
        if (rows > 1000) rows = 1000;
        if (++columns > 1000) columns = 1000;
        if (block.lastRow >= rows || block.lastColumn >= columns)
            throw new Exception("POI report block is outside its sheet boundaries");
    }
    auto source = *z8_report_range_source(range);
    if (z8_report_source_state(source).kind != Z8ReportSourceKind.simple)
        throw new Exception("expanding POI report data sources are not implemented");
    z8_report_register_data_source(report, source);
    foreach (row; workbook.sheets[sheet].rows) foreach (cell; row.cells)
        if (!cell.hasFormula && (cell.kind == "str"w || cell.kind == "inlineStr"w) &&
            (cell.value.canFind('$') || cell.value.canFind('{') || cell.value.canFind('}')))
            throw new Exception("POI report text expressions are not implemented");
    z8_report_source_open(source, sheet);
    scope(exit) z8_report_source_close(source);
    if (!z8_report_source_next(source) || z8_report_source_next(source)) throw new Exception("invalid simple POI report source");
}

private ubyte[] z8_poi_report_rename_sheet(const(ubyte)[] input, Z8PoiWorkbook workbook, int index, wstring name) {
    import std.algorithm.searching : canFind, endsWith;
    import std.zip : ZipArchive;
    if (name.length > 31) name = name[0 .. 31];
    if (!name.length || name[0] == '\'' || name[$ - 1] == '\'') throw new Exception("invalid POI report sheet name");
    foreach (wchar c; name)
        if (c == ':' || c == '\\' || c == '/' || c == '?' || c == '*' || c == '[' || c == ']' || c < 0x20)
            throw new Exception("invalid POI report sheet name");
    foreach (i, sheet; workbook.sheets)
        if (i != index && z8_str_compareNoCase(z8_string_or_empty(sheet.name), name) == 0)
            throw new Exception("duplicate POI report sheet name");
    if (workbook.sheets[index].name == name) return input.dup;
    foreach (sheet; workbook.sheets) foreach (row; sheet.rows) foreach (cell; row.cells)
        if (cell.hasFormula) throw new Exception("renaming a POI report sheet with formulas is not implemented");
    auto archive = new ZipArchive(cast(ubyte[])input.dup);
    auto packageRels = z8_poi_read_xml(archive, "_rels/.rels");
    string workbookPath;
    foreach (rel; packageRels.children) if (z8_poi_xml_local(rel.name) == "Relationship"w &&
        rel.attr("TargetMode"w) != "External"w && rel.attr("Type"w).endsWith("/officeDocument"w))
        workbookPath = z8_poi_resolve_path("", rel.attr("Target"w));
    if (!workbookPath.length) throw new Exception("missing XLSX workbook relationship");
    auto member = workbookPath in archive.directory;
    if (member is null) throw new Exception("missing XLSX workbook");
    archive.expand(*member);
    auto original = (*member).expandedData.dup;
    auto xml = z8_poi_xml_decode(original);
    auto root = (new Z8PoiXmlReader(xml)).parseDocument();
    auto definedNames = root.child("definedNames"w);
    if (definedNames !is null && definedNames.children.length)
        throw new Exception("renaming a POI report sheet with defined names is not implemented");
    auto sheets = root.child("sheets"w);
    if (sheets is null) throw new Exception("missing XLSX sheets");
    Z8PoiXmlNode selected; int current;
    foreach (item; sheets.children) if (z8_poi_xml_local(item.name) == "sheet"w) {
        if (current++ == index) { selected = item; break; }
    }
    if (selected is null) throw new Exception("invalid POI report sheet index");
    auto opening = "<"w ~ selected.name;
    foreach (key, value; selected.attributes) {
        if (z8_poi_xml_local(key) == "name"w) value = name;
        opening ~= " "w ~ key ~ "=\""w ~ z8_poi_xml_attribute_escape(value) ~ "\""w;
    }
    opening ~= selected.empty ? "/>"w : ">"w;
    xml = xml[0 .. selected.start] ~ opening ~ xml[selected.startEnd .. $];
    (*member).expandedData = z8_poi_report_encode_xml(xml, original);
    auto output = new ZipArchive(); output.comment = archive.comment;
    foreach (entry; archive.directory.byValue) output.addMember(entry);
    return cast(ubyte[])output.build();
}

private wstring z8_poi_xml_attribute_escape(wstring value) {
    import std.string : replace;
    return value.replace("&"w, "&amp;"w).replace("<"w, "&lt;"w).replace("\""w, "&quot;"w)
        .replace("\r"w, "&#13;"w).replace("\n"w, "&#10;"w).replace("\t"w, "&#9;"w);
}

private int z8_poi_report_column(wstring name) {
    name = z8_string_required(name);
    uint index;
    foreach (c; name) index = index * 26 + cast(uint)(cast(int)c - 'A' + 1);
    auto result = cast(int)(index - 1);
    if (result < 0 || result >= 16384) throw new Exception("invalid POI report column");
    return result + 1; // OOXML uses one-based column indexes.
}

private ubyte[] z8_poi_report_encode_xml(wstring xml, const(ubyte)[] original) {
    bool little, utf16, bom;
    if (original.length >= 2 && original[0] == 0xFF && original[1] == 0xFE) { utf16 = little = bom = true; }
    else if (original.length >= 2 && original[0] == 0xFE && original[1] == 0xFF) { utf16 = bom = true; }
    else if (original.length >= 4 && original[0] == '<' && original[1] == 0 && original[2] != 0 && original[3] == 0) utf16 = little = true;
    else if (original.length >= 4 && original[0] == 0 && original[1] == '<' && original[2] == 0 && original[3] != 0) utf16 = true;
    if (!utf16) return cast(ubyte[])xml.to!string.dup;
    ubyte[] result;
    if (bom) result ~= little ? [cast(ubyte)0xFF, 0xFE] : [cast(ubyte)0xFE, 0xFF];
    foreach (c; xml) {
        if (little) { result ~= cast(ubyte)c; result ~= cast(ubyte)(c >> 8); }
        else { result ~= cast(ubyte)(c >> 8); result ~= cast(ubyte)c; }
    }
    return result;
}

private ubyte[] z8_poi_report_hide_columns(const(ubyte)[] input, Z8IntMap!(Z8Array!wstring) hidden) {
    import std.algorithm.sorting : sort;
    import std.algorithm.searching : endsWith;
    import std.string : lastIndexOf;
    import std.zip : ZipArchive;
    auto archive = new ZipArchive(cast(ubyte[])input.dup);
    auto packageRels = z8_poi_read_xml(archive, "_rels/.rels");
    string workbookPath;
    foreach (rel; packageRels.children) if (z8_poi_xml_local(rel.name) == "Relationship"w &&
        rel.attr("TargetMode"w) != "External"w && rel.attr("Type"w).endsWith("/officeDocument"w))
        workbookPath = z8_poi_resolve_path("", rel.attr("Target"w));
    if (!workbookPath.length) throw new Exception("missing XLSX workbook relationship");
    auto slash = workbookPath.lastIndexOf('/');
    auto base = slash < 0 ? "" : workbookPath[0 .. slash + 1];
    auto filename = slash < 0 ? workbookPath : workbookPath[slash + 1 .. $];
    auto root = z8_poi_read_xml(archive, workbookPath);
    auto relationships = z8_poi_read_xml(archive, base ~ "_rels/" ~ filename ~ ".rels");
    wstring[wstring] targets;
    foreach (rel; relationships.children) if (z8_poi_xml_local(rel.name) == "Relationship"w &&
        rel.attr("TargetMode"w) != "External"w && rel.attr("Type"w).endsWith("/worksheet"w))
        targets[rel.attr("Id"w)] = rel.attr("Target"w);
    string[] sheets;
    auto sheetNodes = root.child("sheets"w);
    if (sheetNodes is null) throw new Exception("missing XLSX sheets");
    foreach (item; sheetNodes.children) if (z8_poi_xml_local(item.name) == "sheet"w) {
        auto target = item.attr("id"w) in targets;
        if (target is null) throw new Exception("missing XLSX sheet relationship");
        sheets ~= z8_poi_resolve_path(base, *target);
    }
    auto keys = hidden.z8_keys();
    foreach (keyIndex; 0 .. keys.z8_size()) {
        auto key = keys.z8_get(keyIndex);
        auto index = cast(int)z8_optional_require(key);
        if (index < 0 || index >= sheets.length) throw new Exception("invalid POI report sheet index");
        bool[int] wanted;
        auto wantedOrder = new Z8JavaMapOrder!(Z8Optional!long)();
        auto columns = z8_nonnull(hidden.z8_get(key));
        foreach (i; 0 .. columns.z8_size()) {
            auto number = z8_poi_report_column(columns.z8_get(i));
            if (number !in wanted) { wanted[number] = true; wantedOrder.add(z8_optional_wrap(cast(long)number - 1)); }
        }
        if (!wanted.length) continue;
        auto member = sheets[index] in archive.directory;
        if (member is null) throw new Exception("missing XLSX worksheet");
        archive.expand(*member);
        auto original = (*member).expandedData.dup;
        auto xml = z8_poi_xml_decode(original);
        auto sheet = (new Z8PoiXmlReader(xml)).parseDocument();
        if (z8_poi_xml_local(sheet.name) != "worksheet"w || sheet.empty) throw new Exception("invalid XLSX worksheet");
        Z8PoiXmlNode[] columnGroups;
        foreach (child; sheet.children) if (z8_poi_xml_local(child.name) == "cols"w) columnGroups ~= child;
        auto cols = columnGroups.length ? columnGroups[0] : null;
        auto prefixEnd = sheet.name.lastIndexOf(':');
        wstring prefix = prefixEnd < 0 ? ""w : sheet.name[0 .. prefixEnd + 1];
        Z8PoiColumn[] segments, sourceColumns;
        foreach (group; columnGroups) foreach (column; group.children) {
            if (z8_poi_xml_local(column.name) != "col"w || column.children.length)
                throw new Exception("unsupported XLSX column element");
            auto minimum = column.attr("min"w).to!int, maximum = column.attr("max"w).to!int;
            if (minimum < 1 || maximum > 16384 || minimum > maximum) throw new Exception("invalid XLSX column range");
            auto attributes = column.attributes.dup;
            // Moving a column out of a separate <cols> must not lose namespace
            // declarations scoped to that group. Child declarations win.
            if (group !is cols) foreach (name, value; group.attributes)
                if ((name == "xmlns"w || (name.length > 6 && name[0 .. 6] == "xmlns:"w)) && name !in attributes)
                    attributes[name] = value;
            sourceColumns ~= Z8PoiColumn(minimum, maximum, column.name, attributes);
        }
        {
            // POI 3.16's ColumnHelper can retain overlapping ranges after
            // cleaning. Model its insertion/splitting order before flattening
            // them, otherwise a broad range can acquire the wrong width.
            void mergeAttributes(ref Z8PoiColumn target, Z8PoiColumn source) {
                bool collapsed;
                foreach (name, value; source.attributes) {
                    auto local = z8_poi_xml_local(name);
                    if (local == "min"w || local == "max"w) continue;
                    if (local == "collapsed"w) collapsed = true;
                    target.attributes[name] = value;
                }
                foreach (name; target.attributes.keys) if (z8_poi_xml_local(name) == "collapsed"w) target.attributes.remove(name);
                target.attributes["collapsed"w] = collapsed ? "1"w : "0"w;
            }
            Z8PoiColumn clone(Z8PoiColumn source, int first, int last) {
                Z8PoiColumn result = Z8PoiColumn(first, last, source.name, null);
                mergeAttributes(result, source); return result;
            }
            Z8PoiColumn[] clean;
            void addUnique(Z8PoiColumn item) {
                foreach (previous; clean) if (previous.first == item.first && previous.last == item.last) return;
                clean ~= item;
            }
            void insertUnique(Z8PoiColumn item) {
                foreach (previous; clean) if (previous.first == item.first && previous.last == item.last) return;
                clean = [item] ~ clean;
            }
            foreach (column; sourceColumns) {
                Z8PoiColumn[] intersections, remainder;
                foreach (item; clean) {
                    if (column.first <= item.last && column.last >= item.first) intersections ~= item;
                    else remainder ~= item;
                }
                clean = remainder;
                if (!intersections.length) addUnique(clone(column, column.first, column.last));
                else foreach (old; intersections) {
                    auto start = old.first > column.first ? old.first : column.first;
                    auto end = old.last < column.last ? old.last : column.last;
                    auto common = clone(old, start, end); mergeAttributes(common, column); addUnique(common);
                    auto left = old.first < column.first ? old : column;
                    auto leftStart = old.first < column.first ? old.first : column.first;
                    if (leftStart < start) addUnique(clone(left, leftStart, start - 1));
                    auto right = old.last > column.last ? old : column;
                    auto rightEnd = old.last > column.last ? old.last : column.last;
                    if (end < rightEnd) addUnique(clone(right, end + 1, rightEnd));
                }
                sort!((a, b) => a.first == b.first ? a.last < b.last : a.first < b.first)(clean);
            }
            foreach (wantedKey; wantedOrder.ordered()) {
                auto n = cast(int)z8_optional_require(wantedKey) + 1;
                size_t columnIndex;
                while (columnIndex < clean.length && !(clean[columnIndex].first <= n && clean[columnIndex].last >= n)) ++columnIndex;
                if (columnIndex == clean.length) clean ~= Z8PoiColumn(n, n, prefix ~ "col"w, ["hidden"w: "1"w]);
                else {
                    auto current = clean[columnIndex];
                    current.attributes = current.attributes.dup;
                    clean[columnIndex].first = n; clean[columnIndex].last = n;
                    clean[columnIndex].attributes["hidden"w] = "1"w;
                    if (current.first < n) insertUnique(clone(current, current.first, n - 1));
                    if (current.last > n) insertUnique(clone(current, n + 1, current.last));
                }
            }
            auto slots = new Z8PoiColumn[16385];
            bool[16385] occupied;
            foreach (column; clean) foreach (n; column.first .. column.last + 1) if (!occupied[n]) {
                slots[n] = column; occupied[n] = true;
            }
            foreach (n; 1 .. 16385) if (occupied[n]) {
                auto item = slots[n];
                item.first = item.last = n;
                if (segments.length && segments[$ - 1].last + 1 == n && segments[$ - 1].name == item.name &&
                    segments[$ - 1].attributes == item.attributes) segments[$ - 1].last = n;
                else segments ~= item;
            }
        }
        sort!((a, b) => a.first < b.first)(segments);
        wstring content;
        foreach (column; segments) {
            content ~= "<"w ~ column.name ~ " min=\""w ~ column.first.to!wstring ~ "\" max=\""w ~ column.last.to!wstring ~ "\""w;
            foreach (name, value; column.attributes) {
                auto local = z8_poi_xml_local(name);
                // On load POI 3.16 sets collapsed from the attribute's
                // presence, so an absent attribute reproduces false.
                if (local == "min"w || local == "max"w || (local == "collapsed"w && value == "0"w)) continue;
                content ~= " "w ~ name ~ "=\""w ~ z8_poi_xml_attribute_escape(value) ~ "\""w;
            }
            content ~= "/>"w;
        }
        if (cols !is null) {
            auto opening = xml[cols.start .. cols.startEnd];
            if (cols.empty) opening = opening[0 .. $ - 2] ~ ">"w;
            auto updated = xml[0 .. cols.start] ~ opening ~ content ~ "</"w ~ cols.name ~ ">"w;
            auto previous = cols.end;
            foreach (group; columnGroups[1 .. $]) {
                updated ~= xml[previous .. group.start];
                previous = group.end;
            }
            xml = updated ~ xml[previous .. $];
        } else {
            size_t position = sheet.endStart;
            auto data = sheet.child("sheetData"w);
            if (data !is null) position = data.start;
            xml = xml[0 .. position] ~ "<"w ~ prefix ~ "cols>"w ~ content ~ "</"w ~ prefix ~ "cols>"w ~ xml[position .. $];
        }
        (*member).expandedData = z8_poi_report_encode_xml(xml, original);
    }
    auto output = new ZipArchive(); output.comment = archive.comment;
    foreach (member; archive.directory.byValue) output.addMember(member);
    return cast(ubyte[])output.build();
}
