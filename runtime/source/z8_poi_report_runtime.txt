// OOXML column visibility for the POI report engine. Other workbook parts are
// retained as archive members; worksheet XML is edited without rewriting cells.
private struct Z8PoiColumn {
    int first, last;
    wstring name;
    wstring[wstring] attributes;
}

private struct Z8PoiReportBlock { long firstRow, lastRow, firstColumn, lastColumn; }
// OpenXLSX may rewrite XML parts on save. Keep the compatible editor for
// archives whose encoding or ZIP comment it would otherwise change.
private bool z8_poi_report_openxlsx_safe(ZipArchive archive, size_t sheets) {
    import std.algorithm.searching : endsWith;
    import std.string : toLower;
    if (sheets != 1 || archive.comment.length) return false;
    foreach (name, member; archive.directory) {
        if (!name.toLower().endsWith(".xml")) continue;
        try archive.expand(member);
        catch (Exception) return false;
        auto bytes = member.expandedData;
        if (bytes.length >= 2 && ((bytes[0] == 0xFF && bytes[1] == 0xFE) ||
            (bytes[0] == 0xFE && bytes[1] == 0xFF) || bytes[0] == 0 || bytes[1] == 0)) return false;
    }
    return true;
}
private Z8PoiReportBlock z8_poi_report_block(wstring address) {
    import std.string : split;
    auto parts = z8_string_required(address).split(":"w);
    if (parts.length != 1 && parts.length != 2) throw new Exception("invalid POI report range address");
    void parseCell(wstring text, out long row, out long column) {
        size_t position;
        if (position < text.length && text[position] == '$') ++position;
        auto start = position;
        while (position < text.length && text[position] >= 'A' && text[position] <= 'Z') {
            column = column * 26 + text[position++] - 'A' + 1;
            if (column > 16384) throw new Exception("invalid POI report column");
        }
        if (position == start) throw new Exception("invalid POI report range address");
        if (position < text.length && text[position] == '$') ++position;
        start = position;
        while (position < text.length && text[position] >= '0' && text[position] <= '9') {
            row = row * 10 + text[position++] - '0';
            if (row > 1048576) throw new Exception("invalid POI report row");
        }
        if (position == start || position != text.length || !row) throw new Exception("invalid POI report range address");
        --row; --column;
    }
    Z8PoiReportBlock result;
    parseCell(parts[0], result.firstRow, result.firstColumn);
    if (parts.length == 2) parseCell(parts[1], result.lastRow, result.lastColumn);
    else { result.lastRow = result.firstRow; result.lastColumn = result.firstColumn; }
    if (result.lastRow < result.firstRow || result.lastColumn < result.firstColumn)
        throw new Exception("invalid POI report range order");
    return result;
}

// POI DateUtil uses the date's Calendar zone and includes Excel's fictional
// 1900-02-29 in the 1900 date system.
private double z8_poi_report_excel_date(Z8Date input, bool date1904) {
    auto value = z8_nonnull(input);
    auto fields = z8_calendar_fields(value.value);
    if (fields.year <= 0) throw new Exception("POI report BCE dates are not implemented");
    if (fields.year < (date1904 ? 1904 : 1900)) return -1.0;
    auto epoch = date1904 ? z8_calendar_day_number(1904, 1, 1, false) :
        z8_calendar_day_number(1899, 12, 31, false);
    auto days = fields.dayNumber - epoch;
    if (!date1904 && days >= 60) ++days;
    auto milliseconds = ((z8_date_hours(value) * 60 + z8_date_minutes(value)) * 60 +
        z8_date_seconds(value)) * 1000 + z8_date_milliseconds(value);
    return cast(double)days + cast(double)milliseconds / 86400000.0;
}

// POI validates copied formulas. Accept a deliberately bounded Excel subset
// until a full formula parser is available.
private struct Z8PoiFormulaChecker {
    wstring text;
    size_t position;
    uint depth;
    void whitespace() { while (position < text.length && (text[position] == ' ' || text[position] == '\t')) ++position; }
    bool digit(wchar c) { return c >= '0' && c <= '9'; }
    bool letter(wchar c) { return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z'); }
    bool take(wchar c) { whitespace(); if (position < text.length && text[position] == c) { ++position; return true; } return false; }
    bool reference() {
        if (text[position] == '$') ++position;
        auto first = position; uint column;
        while (position < text.length && letter(text[position])) {
            auto c = text[position++];
            column = column * 26 + cast(uint)((c >= 'a' ? c - 'a' : c - 'A') + 1);
            if (column > 16384) return false;
        }
        if (position == first || !column) return false;
        if (position < text.length && text[position] == '$') ++position;
        first = position; uint row;
        while (position < text.length && digit(text[position])) {
            row = row * 10 + cast(uint)(text[position++] - '0');
            if (row > 1048576) return false;
        }
        return position != first && row != 0;
    }
    bool expression() {
        if (++depth > 128) { --depth; return false; }
        scope(exit) --depth;
        if (!atom()) return false;
        for (;;) {
            whitespace();
            if (position == text.length || text[position] == ')' || text[position] == ',') return true;
            if (text[position] != '+' && text[position] != '-' && text[position] != '*' && text[position] != '/' && text[position] != '^') return false;
            ++position;
            if (!atom()) return false;
        }
    }
    bool atom() {
        whitespace();
        if (position < text.length && (text[position] == '+' || text[position] == '-')) { ++position; whitespace(); }
        if (position == text.length) return false;
        if (take('(')) return expression() && take(')');
        if (digit(text[position])) {
            while (position < text.length && digit(text[position])) ++position;
            if (position < text.length && text[position] == '.') {
                auto first = ++position;
                while (position < text.length && digit(text[position])) ++position;
                if (position == first) return false;
            }
            return true;
        }
        auto start = position;
        if (letter(text[position])) {
            wstring name;
            while (position < text.length && letter(text[position])) {
                auto c = text[position++]; name ~= c >= 'a' ? cast(wchar)(c - 'a' + 'A') : c;
            }
            whitespace();
            if (position < text.length && text[position] == '(') {
                if (name != "SUM"w && name != "MIN"w && name != "MAX"w && name != "AVERAGE"w && name != "COUNT"w) return false;
                ++position;
                uint count;
                do { if (!expression() || ++count > 30) return false; } while (take(','));
                return take(')');
            }
        }
        position = start;
        if (text[position] != '$' && !letter(text[position])) return false;
        if (!reference()) return false;
        if (take(':')) {
            whitespace();
            if (position == text.length || (text[position] != '$' && !letter(text[position]))) return false;
            return reference();
        }
        return true;
    }
}
private bool z8_poi_report_simple_formula(wstring text) {
    auto checker = Z8PoiFormulaChecker(text);
    if (!checker.expression()) return false;
    checker.whitespace();
    return checker.position == text.length;
}
private bool z8_poi_report_copyable_cell(Z8PoiCell cell) {
    return cell.safeCopy && !(cell.hasFormula && ((cell.rawKind !is null && cell.rawKind != "n"w) ||
        !z8_poi_report_simple_formula(cell.formula))) && (cell.rawKind is null || cell.rawKind == "s"w ||
        cell.rawKind == "inlineStr"w || cell.rawKind == "n"w || cell.rawKind == "b"w);
}

// Collect directly named Field members before opening a query source, as
// QuerySource.collectFields does in Java. Quoted text is never a reference.
private Z8Object[] z8_poi_report_query_fields(Z8Object report, Z8Object query, Z8PoiSheet sheet,
                                               Z8PoiReportBlock block) {
    import std.algorithm.searching : startsWith;
    import std.string : indexOf;
    import std.uni : isAlpha, isNumber;
    Z8Object[] result;
    void add(Z8Object field) {
        foreach (existing; result) if (existing is field) return;
        result ~= field;
    }
    foreach (field; z8_query_array(query, "extraFields"w)) add(z8_nonnull(field));
    foreach (rowNumber, row; sheet.rows) foreach (columnNumber, cell; row.cells) {
        if (rowNumber < block.firstRow || rowNumber > block.lastRow || columnNumber < block.firstColumn ||
            columnNumber > block.lastColumn || cell.hasFormula || (cell.kind != "str"w && cell.kind != "inlineStr"w)) continue;
        auto text = cell.value;
        size_t offset;
        while (offset < text.length) {
            auto start = text[offset .. $].indexOf("${"w);
            if (start < 0) break;
            auto begin = offset + cast(size_t)start + 2;
            auto end = text[begin .. $].indexOf('}');
            if (end < 0) break;
            auto expression = text[begin .. begin + cast(size_t)end];
            offset = begin + cast(size_t)end + 1;
            size_t position;
            while (position < expression.length) {
                auto c = expression[position];
                if (c == '\'' || c == '"') {
                    ++position;
                    while (position < expression.length && expression[position] != c) {
                        if (expression[position] == '\\' && position + 1 < expression.length) ++position;
                        ++position;
                    }
                    if (position < expression.length) ++position;
                    continue;
                }
                if (!(isAlpha(c) || c == '_')) { ++position; continue; }
                auto object = report;
                for (;;) {
                    auto nameStart = position++;
                    while (position < expression.length && (isAlpha(expression[position]) ||
                           isNumber(expression[position]) || expression[position] == '_')) ++position;
                    auto name = expression[nameStart .. position];
                    Z8Object member;
                    if (object !is null) foreach (candidate; z8_object_members(object))
                        if (candidate.z8_index() == name) { member = candidate; break; }
                    if (member !is null) {
                        member.z8_ensure();
                        if (z8_native_is_instance(z8_field_name, member) &&
                            z8_string_or_empty(member.z8_id()).startsWith(z8_string_or_empty(query.z8_id()))) add(member);
                    }
                    object = member;
                    while (position < expression.length && (expression[position] == ' ' || expression[position] == '\t' ||
                           expression[position] == '\r' || expression[position] == '\n')) ++position;
                    if (position >= expression.length || expression[position] != '.') break;
                    auto next = position + 1;
                    while (next < expression.length && (expression[next] == ' ' || expression[next] == '\t' ||
                           expression[next] == '\r' || expression[next] == '\n')) ++next;
                    if (next >= expression.length || !(isAlpha(expression[next]) || expression[next] == '_')) break;
                    position = next;
                }
            }
        }
    }
    return result;
}

private void z8_poi_report_evaluate_cell(Z8EvalState state, Z8PoiWorkbook workbook, Z8PoiCell cell,
                                         long sheet, long row, long column, ref Z8PoiCellEdit[] edits, bool copied = false) {
    import std.algorithm.searching : canFind;
    import std.string : indexOf;
    auto firstEnd = cell.value.indexOf('}');
    auto single = cell.value.length >= 3 && cell.value[0 .. 2] == "${"w &&
        firstEnd >= 0 && cast(size_t)firstEnd + 1 == cell.value.length;
    Z8PoiCellEdit edit;
    edit.sheet = cast(uint)sheet; edit.row = cast(uint)row; edit.column = cast(uint)column;
    if (!single) {
        auto text = z8_eval_text_from_state(state, cell.value, true, true);
        if (cell.kind == "inlineStr"w && !copied) return;
        edit.kind = Z8PoiCellEditKind.text; edit.text = text; edits ~= edit;
        return;
    }
    auto expression = cell.value[2 .. $ - 1];
    // The original report grammar accepts single-quoted strings. The Eval
    // parser also accepts double quotes, which are not equivalent here.
    if (expression.canFind('"')) throw new Exception("POI report expression syntax is not implemented");
    auto evaluated = z8_eval_expression(state, expression);
    if (!evaluated.evaluated || evaluated.context != Z8EvalContext.none || evaluated.json !is null ||
        evaluated.requestParameters !is null || evaluated.reportObject !is null)
        throw new Exception("POI report expression context is not implemented");
    auto value = evaluated.primary;
    // POI 3.16 leaves an inline-string cell unchanged when assigning a
    // string result; numeric and boolean results replace its type.
    if (cell.kind == "inlineStr"w && !copied && (value is null || value.kind == Z8PrimaryKind.text ||
        value.kind == Z8PrimaryKind.guid || value.kind == Z8PrimaryKind.datespan)) return;
    if (value is null) { edit.kind = Z8PoiCellEditKind.text; edit.text = ""w; }
    else switch (value.kind) {
        case Z8PrimaryKind.boolean: edit.kind = Z8PoiCellEditKind.boolean; edit.booleanValue = value.booleanValue; break;
        // Range.evaluateAndSet uses integer.getInt(), including narrowing.
        case Z8PrimaryKind.integer: edit.kind = Z8PoiCellEditKind.number; edit.number = cast(double)cast(int)value.integerValue; break;
        case Z8PrimaryKind.decimal: edit.kind = Z8PoiCellEditKind.number; edit.number = value.decimalValue.z8_as_double(); break;
        case Z8PrimaryKind.date: edit.kind = Z8PoiCellEditKind.number; edit.number = z8_poi_report_excel_date(value.dateValue, workbook.date1904); break;
        case Z8PrimaryKind.text: edit.kind = Z8PoiCellEditKind.text; edit.text = value.textValue; break;
        case Z8PrimaryKind.guid: case Z8PrimaryKind.datespan:
            edit.kind = Z8PoiCellEditKind.text; edit.text = z8_primary_to_string(value); break;
        default: throw new Exception("POI report expression result type is not implemented");
    }
    edits ~= edit;
}

// Apply a single, non-expanding source to the selected report block.
private void z8_poi_report_apply_simple_range(Z8Object report, Z8Object range, Z8PoiWorkbook workbook,
                                              out int renameSheet, out wstring renameName, out Z8PoiCellEdit[] edits,
                                              out Z8PoiCellCopy[] copies, out Z8PoiMergedRegion[] addedMerges,
                                              bool allowExpansion = false, bool allowMerges = false) {
    import std.string : indexOf;
    range = z8_nonnull(range); range.z8_ensure();
    auto rangeState = z8_report_range_state(range);
    if (rangeState.ranges !is null && rangeState.ranges.view().z8_erased_data().z8_size())
        throw new Exception("nested POI report ranges are not implemented");
    foreach (attribute; ["totals"w, "subtotal"w, "subtotalsBy"w, "subtotalsMerge"w, "merge"w])
        if (range.z8_get_attribute(attribute) !is null) throw new Exception("POI report range option is not implemented");
    auto axisText = range.z8_get_attribute("axis"w);
    if (axisText !is null && axisText.to!int != 0 && axisText.to!int != 1) throw new Exception("invalid POI report range axis");
    auto sheetText = range.z8_get_attribute("sheet"w);
    auto sheet = sheetText is null ? 0 : sheetText.to!int;
    if (sheet < 0 || sheet >= workbook.sheets.length) throw new Exception("POI report sheet expansion is not implemented");
    renameSheet = sheet; renameName = range.z8_get_attribute("sheetName"w);
    auto block = z8_poi_report_block(range.z8_get_attribute("address"w));
    Z8PoiReportBlock[] templateMerges;
    if (workbook.sheets[sheet].hasMergedRegions) {
        if (!allowMerges || workbook.sheets[sheet].mergedReferences.length == 0)
            throw new Exception("merged POI report range cells are not implemented");
        foreach (reference; workbook.sheets[sheet].mergedReferences) {
            auto merge = z8_poi_report_block(reference);
            if (merge.firstRow < block.firstRow || merge.lastRow > block.lastRow ||
                merge.firstColumn < block.firstColumn || merge.lastColumn > block.lastColumn ||
                (merge.firstRow == merge.lastRow && merge.firstColumn == merge.lastColumn))
                throw new Exception("merged POI report range cells are not implemented");
            foreach (previous; templateMerges) if (merge.firstRow <= previous.lastRow && previous.firstRow <= merge.lastRow &&
                merge.firstColumn <= previous.lastColumn && previous.firstColumn <= merge.lastColumn)
                throw new Exception("overlapping POI report merged regions");
            templateMerges ~= merge;
        }
    }
    auto boundariesText = range.z8_get_attribute("boundaries"w);
    if (boundariesText !is null) {
        auto boundaries = z8_poi_report_block(boundariesText);
        if (block.firstRow < boundaries.firstRow || block.lastRow > boundaries.lastRow ||
            block.firstColumn < boundaries.firstColumn || block.lastColumn > boundaries.lastColumn)
            throw new Exception("POI report block is outside its boundaries");
    } else {
        long rows = 1, columns = 0;
        foreach (rowNumber, row; workbook.sheets[sheet].rows) {
            if (rowNumber + 1 > rows) rows = rowNumber + 1;
            if (rowNumber < 1000) foreach (columnNumber, cell; row.cells) if (columnNumber + 1 > columns) columns = columnNumber + 1;
        }
        if (rows > 1000) rows = 1000;
        if (++columns > 1000) columns = 1000;
        if (block.lastRow >= rows || block.lastColumn >= columns)
            throw new Exception("POI report block is outside its sheet boundaries");
    }
    auto source = *z8_report_range_source(range);
    auto sourceKind = z8_report_source_state(source).kind;
    z8_report_register_data_source(report, source);
    // Java opens and advances the source before evaluating cells. In
    // particular, CustomData.open may populate report context members.
    scope(exit) z8_report_source_close(source);
    auto fields = sourceKind == Z8ReportSourceKind.query ?
        z8_poi_report_query_fields(report, z8_report_source_object(source), workbook.sheets[sheet], block) : null;
    z8_report_source_open(source, sheet, fields);
    if (!z8_report_source_next(source)) throw new Exception("empty POI report data sources are not implemented");
    auto evalState = new Z8EvalState();
    evalState.reportContext = report;
    evalState.reportSource = source;
    auto request = z8_application_request();
    if (request !is null && request.parameters !is null)
        evalState.variables["parameters"w] = Z8EvalVariable(null, null, request.parameters);
    bool horizontal = axisText !is null && axisText.to!int == 1;
    long iteration, height = block.lastRow - block.firstRow + 1, width = block.lastColumn - block.firstColumn + 1;
    bool hasFooter;
    do {
        if (iteration > (horizontal ? (16383 - block.lastColumn) / width : (1048575 - block.lastRow) / height))
            throw new Exception("POI report range expansion exceeds worksheet bounds");
        auto offset = iteration * (horizontal ? width : height);
        if (iteration) {
            if (!allowExpansion || workbook.sheets.length != 1 || boundariesText is null)
                throw new Exception("POI report range expansion layout is not implemented");
            auto boundaries = z8_poi_report_block(boundariesText);
            if (horizontal ? (boundaries.firstRow != block.firstRow || boundaries.lastRow != block.lastRow) :
                             (boundaries.firstColumn != block.firstColumn || boundaries.lastColumn != block.lastColumn))
                throw new Exception("POI report range expansion layout is not implemented");
            hasFooter = horizontal ? boundaries.lastColumn > block.lastColumn : boundaries.lastRow > block.lastRow;
            foreach (sourceRow; block.firstRow .. block.lastRow + 1) {
                auto targetRow = sourceRow + (horizontal ? 0 : offset);
                auto firstColumn = block.firstColumn + (horizontal ? offset : 0);
                auto lastColumn = block.lastColumn + (horizontal ? offset : 0);
                auto existingRow = targetRow in workbook.sheets[sheet].rows;
                auto templateRow = sourceRow in workbook.sheets[sheet].rows;
                if (templateRow is null) {
                    if (hasFooter) throw new Exception("POI report sparse block with footer is not implemented");
                    // POI clears existing cells in this part of an absent
                    // source row. Cell deletion is not supported yet.
                    if (existingRow !is null) foreach (column, cell; (*existingRow).cells)
                        if (column >= firstColumn && column <= lastColumn)
                            throw new Exception("POI report range expansion target cell is not implemented");
                    continue;
                }
                if (existingRow !is null) {
                    if (!(*existingRow).safeExpansionTarget) throw new Exception("POI report range expansion row is not implemented");
                    foreach (column, cell; (*existingRow).cells) {
                        if (column < firstColumn || column > lastColumn) continue;
                        auto sourceColumn = column - (horizontal ? offset : 0);
                        if ((sourceColumn in (*templateRow).cells) is null || !cell.safeCopy || cell.hasFormula || cell.rawKind == "inlineStr"w)
                            throw new Exception("POI report range expansion target cell is not implemented");
                    }
                }
                long templateCellCount;
                foreach (column, cell; (*templateRow).cells) {
                    if (column < block.firstColumn || column > block.lastColumn) continue;
                    ++templateCellCount;
                    if (!z8_poi_report_copyable_cell(cell))
                        throw new Exception("POI report range expansion cell is not implemented");
                    copies ~= Z8PoiCellCopy(cast(uint)sheet, cast(uint)sourceRow, cast(uint)column,
                                            cast(uint)targetRow, cast(uint)(column + (horizontal ? offset : 0)), cell.hasFormula);
                    // POI copies a string's plain value into a new shared-string
                    // cell, even when the template used inline or rich text.
                    if ((cell.rawKind == "s"w || cell.rawKind == "inlineStr"w) && cell.value.indexOf("${"w) < 0) {
                        Z8PoiCellEdit edit;
                        edit.sheet = cast(uint)sheet; edit.row = cast(uint)targetRow; edit.column = cast(uint)(column + (horizontal ? offset : 0));
                        edit.kind = Z8PoiCellEditKind.text; edit.text = cell.value; edits ~= edit;
                    }
                }
                // POI creates a physical target row even when the source row
                // has no cells in this block. Avoid silently omitting it.
                if ((!templateCellCount && existingRow is null) || (hasFooter && templateCellCount != width))
                    throw new Exception("POI report empty target row is not implemented");
            }
            foreach (merge; templateMerges) {
                auto rowOffset = horizontal ? 0 : offset, columnOffset = horizontal ? offset : 0;
                addedMerges ~= Z8PoiMergedRegion(cast(uint)sheet, cast(uint)(merge.firstRow + rowOffset),
                    cast(uint)(merge.firstColumn + columnOffset), cast(uint)(merge.lastRow + rowOffset),
                    cast(uint)(merge.lastColumn + columnOffset));
            }
        }
        foreach (rowNumber, row; workbook.sheets[sheet].rows) foreach (columnNumber, cell; row.cells) {
            if (cell.hasFormula || (cell.kind != "str"w && cell.kind != "inlineStr"w) ||
                cell.value.indexOf("${"w) < 0) continue;
            if (rowNumber < block.firstRow || rowNumber > block.lastRow ||
                columnNumber < block.firstColumn || columnNumber > block.lastColumn) continue;
            z8_poi_report_evaluate_cell(evalState, workbook, cell, sheet, rowNumber + (horizontal ? 0 : offset),
                                        columnNumber + (horizontal ? offset : 0), edits, iteration != 0);
        }
        ++iteration;
    } while (z8_report_source_next(source));
    if (hasFooter && iteration > 1) {
        auto offset = (iteration - 1) * (horizontal ? width : height);
        Z8PoiCellCopy[] footerCopies;
        Z8PoiCellEdit[] footerEdits;
        auto boundaries = z8_poi_report_block(boundariesText);
        auto firstRow = horizontal ? block.firstRow : block.lastRow + 1;
        auto lastRow = horizontal ? block.lastRow : boundaries.lastRow;
        auto firstColumn = horizontal ? block.lastColumn + 1 : block.firstColumn;
        auto lastColumn = horizontal ? boundaries.lastColumn : block.lastColumn;
        // The footer may overlap its destination. Copy backwards along the
        // expansion axis, preserving source values until they are copied.
        foreach_reverse (rowNumber; firstRow .. lastRow + 1) {
            auto targetRow = rowNumber + (horizontal ? 0 : offset);
            if (targetRow >= 1048576 || lastColumn + (horizontal ? offset : 0) >= 16384)
                throw new Exception("POI report footer exceeds worksheet bounds");
            auto sourceRow = rowNumber in workbook.sheets[sheet].rows;
            if (sourceRow is null) throw new Exception("POI report sparse footer is not implemented");
            auto target = targetRow in workbook.sheets[sheet].rows;
            if (target !is null && !(*target).safeExpansionTarget)
                throw new Exception("POI report footer target row is not implemented");
            foreach_reverse (column; firstColumn .. lastColumn + 1) {
                auto original = column in (*sourceRow).cells;
                if (original is null || !z8_poi_report_copyable_cell(*original))
                    throw new Exception("POI report sparse footer is not implemented");
                auto destinationColumn = column + (horizontal ? offset : 0);
                if (target !is null) {
                    auto destination = destinationColumn in (*target).cells;
                    if (destination !is null && (!(*destination).safeCopy || (*destination).hasFormula || (*destination).rawKind == "inlineStr"w))
                        throw new Exception("POI report footer target cell is not implemented");
                }
                auto cell = *original;
                footerCopies ~= Z8PoiCellCopy(cast(uint)sheet, cast(uint)rowNumber, cast(uint)column,
                    cast(uint)targetRow, cast(uint)destinationColumn, cell.hasFormula);
                if (cell.rawKind == "s"w || cell.rawKind == "inlineStr"w) {
                    Z8PoiCellEdit edit;
                    edit.sheet = cast(uint)sheet; edit.row = cast(uint)targetRow; edit.column = cast(uint)destinationColumn;
                    edit.kind = Z8PoiCellEditKind.text; edit.text = cell.value; footerEdits ~= edit;
                }
            }
        }
        copies = footerCopies ~ copies;
        edits ~= footerEdits;
    }
}

private struct Z8PoiReportEntry { Z8Object range; Z8PoiReportBlock block, boundaries; }
// Replicate Block.Comparator's stable ordering for sibling ranges.
private int z8_poi_report_compare_blocks(Z8PoiReportBlock a, Z8PoiReportBlock b) {
    int compare(long left, long right) { return left < right ? -1 : left > right ? 1 : 0; }
    if (a.firstRow <= b.lastRow && b.firstRow <= a.lastRow) return compare(a.firstColumn, b.firstColumn);
    if (a.firstColumn <= b.lastColumn && b.firstColumn <= a.lastColumn) return compare(a.firstRow, b.firstRow);
    return compare(a.firstColumn, b.firstColumn) + compare(a.firstRow, b.firstRow);
}
// Java's synthetic sheet range visits the resulting cells once more after all
// child sources close. A text result can itself contain another expression.
private void z8_poi_report_apply_root(Z8Object report, Z8PoiWorkbook workbook, int sheet, Z8PoiReportBlock root,
                                      const(Z8PoiCellCopy)[] copies, ref Z8PoiCellEdit[] edits) {
    import std.algorithm.sorting : sort;
    import std.string : indexOf;
    Z8PoiCell[ulong] values;
    ulong key(long row, long column) { return (cast(ulong)row << 14) | cast(ulong)column; }
    bool inside(long row, long column) {
        return row >= root.firstRow && row <= root.lastRow && column >= root.firstColumn && column <= root.lastColumn;
    }
    foreach (rowNumber, row; workbook.sheets[sheet].rows) foreach (columnNumber, cell; row.cells)
        if (inside(rowNumber, columnNumber) && !cell.hasFormula && (cell.kind == "str"w || cell.kind == "inlineStr"w))
            values[key(rowNumber, columnNumber)] = cell;
    foreach (copy; copies) {
        auto position = key(copy.targetRow, copy.targetColumn);
        values.remove(position);
        auto sourceRow = cast(long)copy.sourceRow in workbook.sheets[sheet].rows;
        if (sourceRow is null) continue;
        auto sourceCell = cast(long)copy.sourceColumn in (*sourceRow).cells;
        if (sourceCell is null || ((*sourceCell).kind != "str"w && (*sourceCell).kind != "inlineStr"w)) continue;
        auto cell = new Z8PoiCell(); cell.kind = "str"w; cell.value = (*sourceCell).value;
        values[position] = cell;
    }
    foreach (edit; edits) {
        auto position = key(edit.row, edit.column);
        if (edit.kind == Z8PoiCellEditKind.text) {
            auto cell = new Z8PoiCell(); cell.kind = "str"w; cell.value = z8_poi_decode_text(edit.text);
            values[position] = cell;
        } else values.remove(position);
    }
    auto state = new Z8EvalState(); state.reportContext = report;
    auto request = z8_application_request();
    if (request !is null && request.parameters !is null)
        state.variables["parameters"w] = Z8EvalVariable(null, null, request.parameters);
    auto positions = values.keys; sort(positions);
    foreach (position; positions) {
        auto cell = values[position];
        if (cell.value.indexOf("${"w) < 0) continue;
        z8_poi_report_evaluate_cell(state, workbook, cell, sheet, cast(long)(position >> 14),
                                    cast(long)(position & 16383), edits);
    }
}
private void z8_poi_report_apply_ranges(Z8Object report, Z8Object[] ranges, Z8PoiWorkbook workbook,
                                        out int renameSheet, out wstring renameName, out Z8PoiCellEdit[] edits,
                                        out Z8PoiCellCopy[] copies, out Z8PoiMergedRegion[] addedMerges) {
    import std.string : indexOf;
    if (!ranges.length) return;
    if (ranges.length == 1) {
        z8_poi_report_apply_simple_range(report, ranges[0], workbook, renameSheet, renameName, edits, copies, addedMerges, true, true);
        auto boundaries = ranges[0].z8_get_attribute("boundaries"w);
        Z8PoiReportBlock root;
        if (boundaries !is null) root = z8_poi_report_block(boundaries);
        else {
            foreach (rowNumber, row; workbook.sheets[renameSheet].rows) {
                if (rowNumber > root.lastRow) root.lastRow = rowNumber;
                foreach (columnNumber, cell; row.cells) if (columnNumber > root.lastColumn) root.lastColumn = columnNumber;
            }
            if (root.lastRow >= 1000) root.lastRow = 999;
            if (root.lastColumn >= 1000) root.lastColumn = 999;
        }
        foreach (copy; copies) {
            if (copy.targetRow > root.lastRow) root.lastRow = copy.targetRow;
            if (copy.targetColumn > root.lastColumn) root.lastColumn = copy.targetColumn;
        }
        z8_poi_report_apply_root(report, workbook, renameSheet, root, copies, edits);
        return;
    }
    Z8PoiReportEntry[] entries;
    int sheet = -1;
    foreach (range; ranges) {
        range = z8_nonnull(range);
        auto number = range.z8_get_attribute("sheet"w);
        auto index = number is null ? 0 : number.to!int;
        if (index < 0 || index >= workbook.sheets.length || (sheet >= 0 && sheet != index))
            throw new Exception("multiple-sheet POI report ranges are not implemented");
        sheet = index;
        auto block = z8_poi_report_block(range.z8_get_attribute("address"w));
        auto text = range.z8_get_attribute("boundaries"w);
        if (text is null) throw new Exception("multiple POI report ranges require explicit boundaries");
        auto boundaries = z8_poi_report_block(text);
        foreach (previous; entries) if (boundaries.firstRow <= previous.boundaries.lastRow &&
            previous.boundaries.firstRow <= boundaries.lastRow && boundaries.firstColumn <= previous.boundaries.lastColumn &&
            previous.boundaries.firstColumn <= boundaries.lastColumn)
            throw new Exception("overlapping POI report range boundaries");
        entries ~= Z8PoiReportEntry(range, block, boundaries);
        if (renameName is null) { renameName = range.z8_get_attribute("sheetName"w); renameSheet = index; }
    }
    auto root = entries[0].boundaries;
    foreach (entry; entries[1 .. $]) {
        if (entry.boundaries.firstRow < root.firstRow) root.firstRow = entry.boundaries.firstRow;
        if (entry.boundaries.lastRow > root.lastRow) root.lastRow = entry.boundaries.lastRow;
        if (entry.boundaries.firstColumn < root.firstColumn) root.firstColumn = entry.boundaries.firstColumn;
        if (entry.boundaries.lastColumn > root.lastColumn) root.lastColumn = entry.boundaries.lastColumn;
    }
    // Java uses a stable list sort, and source open/close hooks can have effects.
    for (size_t i = 1; i < entries.length; ++i) {
        auto entry = entries[i]; auto position = i;
        while (position && z8_poi_report_compare_blocks(entry.boundaries, entries[position - 1].boundaries) < 0) {
            entries[position] = entries[position - 1]; --position;
        }
        entries[position] = entry;
    }
    foreach (entry; entries) {
        int ignoredSheet; wstring ignoredName; Z8PoiCellEdit[] changes; Z8PoiCellCopy[] ignoredCopies; Z8PoiMergedRegion[] ignoredMerges;
        z8_poi_report_apply_simple_range(report, entry.range, workbook, ignoredSheet, ignoredName, changes, ignoredCopies, ignoredMerges);
        edits ~= changes;
    }
    z8_poi_report_apply_root(report, workbook, sheet, root, copies, edits);
}

private ubyte[] z8_poi_report_rename_sheet(const(ubyte)[] input, Z8PoiWorkbook workbook, int index, wstring name) {
    import std.algorithm.searching : canFind, endsWith;
    import std.zip : ZipArchive;
    if (name.length > 31) name = name[0 .. 31];
    if (!name.length || name[0] == '\'' || name[$ - 1] == '\'') throw new Exception("invalid POI report sheet name");
    foreach (wchar c; name)
        if (c == ':' || c == '\\' || c == '/' || c == '?' || c == '*' || c == '[' || c == ']' || c < 0x20)
            throw new Exception("invalid POI report sheet name");
    foreach (i, sheet; workbook.sheets)
        if (i != index && z8_str_compareNoCase(z8_string_or_empty(sheet.name), name) == 0)
            throw new Exception("duplicate POI report sheet name");
    if (workbook.sheets[index].name == name) return input.dup;
    foreach (sheet; workbook.sheets) foreach (row; sheet.rows) foreach (cell; row.cells)
        if (cell.hasFormula) throw new Exception("renaming a POI report sheet with formulas is not implemented");
    auto archive = new ZipArchive(cast(ubyte[])input.dup);
    auto packageRels = z8_poi_read_xml(archive, "_rels/.rels");
    string workbookPath;
    foreach (rel; packageRels.children) if (z8_poi_xml_local(rel.name) == "Relationship"w &&
        rel.attr("TargetMode"w) != "External"w && rel.attr("Type"w).endsWith("/officeDocument"w))
        workbookPath = z8_poi_resolve_path("", rel.attr("Target"w));
    if (!workbookPath.length) throw new Exception("missing XLSX workbook relationship");
    auto member = workbookPath in archive.directory;
    if (member is null) throw new Exception("missing XLSX workbook");
    archive.expand(*member);
    auto original = (*member).expandedData.dup;
    auto xml = z8_poi_xml_decode(original);
    auto root = (new Z8PoiXmlReader(xml)).parseDocument();
    auto definedNames = root.child("definedNames"w);
    if (definedNames !is null && definedNames.children.length)
        throw new Exception("renaming a POI report sheet with defined names is not implemented");
    auto sheets = root.child("sheets"w);
    if (sheets is null) throw new Exception("missing XLSX sheets");
    Z8PoiXmlNode selected; int current;
    foreach (item; sheets.children) if (z8_poi_xml_local(item.name) == "sheet"w) {
        if (current++ == index) { selected = item; break; }
    }
    if (selected is null) throw new Exception("invalid POI report sheet index");
    IZ8PoiProvider provider;
    synchronized (z8_poi_monitor) provider = z8_poi_provider;
    auto writer = cast(IZ8PoiReportWriter)provider;
    if (writer !is null && !selected.name.canFind(':') && z8_poi_report_openxlsx_safe(archive, workbook.sheets.length)) {
        try return writer.renameReportSheet(input, cast(uint)index, name);
        catch (Z8PoiUnsupportedWorkbook) { /* retain compatibility with uncommon workbook layouts */ }
    }
    auto opening = "<"w ~ selected.name;
    foreach (key, value; selected.attributes) {
        if (z8_poi_xml_local(key) == "name"w) value = name;
        opening ~= " "w ~ key ~ "=\""w ~ z8_poi_xml_attribute_escape(value) ~ "\""w;
    }
    opening ~= selected.empty ? "/>"w : ">"w;
    xml = xml[0 .. selected.start] ~ opening ~ xml[selected.startEnd .. $];
    (*member).expandedData = z8_poi_report_encode_xml(xml, original);
    auto output = new ZipArchive(); output.comment = archive.comment;
    foreach (entry; archive.directory.byValue) output.addMember(entry);
    return cast(ubyte[])output.build();
}

private wstring z8_poi_xml_attribute_escape(wstring value) {
    import std.string : replace;
    return value.replace("&"w, "&amp;"w).replace("<"w, "&lt;"w).replace("\""w, "&quot;"w)
        .replace("\r"w, "&#13;"w).replace("\n"w, "&#10;"w).replace("\t"w, "&#9;"w);
}

private int z8_poi_report_column(wstring name) {
    name = z8_string_required(name);
    uint index;
    foreach (c; name) index = index * 26 + cast(uint)(cast(int)c - 'A' + 1);
    auto result = cast(int)(index - 1);
    if (result < 0 || result >= 16384) throw new Exception("invalid POI report column");
    return result + 1; // OOXML uses one-based column indexes.
}

private ubyte[] z8_poi_report_encode_xml(wstring xml, const(ubyte)[] original) {
    bool little, utf16, bom;
    if (original.length >= 2 && original[0] == 0xFF && original[1] == 0xFE) { utf16 = little = bom = true; }
    else if (original.length >= 2 && original[0] == 0xFE && original[1] == 0xFF) { utf16 = bom = true; }
    else if (original.length >= 4 && original[0] == '<' && original[1] == 0 && original[2] != 0 && original[3] == 0) utf16 = little = true;
    else if (original.length >= 4 && original[0] == 0 && original[1] == '<' && original[2] == 0 && original[3] != 0) utf16 = true;
    if (!utf16) return cast(ubyte[])xml.to!string.dup;
    ubyte[] result;
    if (bom) result ~= little ? [cast(ubyte)0xFF, 0xFE] : [cast(ubyte)0xFE, 0xFF];
    foreach (c; xml) {
        if (little) { result ~= cast(ubyte)c; result ~= cast(ubyte)(c >> 8); }
        else { result ~= cast(ubyte)(c >> 8); result ~= cast(ubyte)c; }
    }
    return result;
}

private ubyte[] z8_poi_report_hide_columns(const(ubyte)[] input, Z8IntMap!(Z8Array!wstring) hidden) {
    import std.algorithm.sorting : sort;
    import std.algorithm.searching : endsWith;
    import std.string : lastIndexOf;
    import std.zip : ZipArchive;
    auto archive = new ZipArchive(cast(ubyte[])input.dup);
    auto packageRels = z8_poi_read_xml(archive, "_rels/.rels");
    string workbookPath;
    foreach (rel; packageRels.children) if (z8_poi_xml_local(rel.name) == "Relationship"w &&
        rel.attr("TargetMode"w) != "External"w && rel.attr("Type"w).endsWith("/officeDocument"w))
        workbookPath = z8_poi_resolve_path("", rel.attr("Target"w));
    if (!workbookPath.length) throw new Exception("missing XLSX workbook relationship");
    auto slash = workbookPath.lastIndexOf('/');
    auto base = slash < 0 ? "" : workbookPath[0 .. slash + 1];
    auto filename = slash < 0 ? workbookPath : workbookPath[slash + 1 .. $];
    auto root = z8_poi_read_xml(archive, workbookPath);
    auto relationships = z8_poi_read_xml(archive, base ~ "_rels/" ~ filename ~ ".rels");
    wstring[wstring] targets;
    foreach (rel; relationships.children) if (z8_poi_xml_local(rel.name) == "Relationship"w &&
        rel.attr("TargetMode"w) != "External"w && rel.attr("Type"w).endsWith("/worksheet"w))
        targets[rel.attr("Id"w)] = rel.attr("Target"w);
    string[] sheets;
    auto sheetNodes = root.child("sheets"w);
    if (sheetNodes is null) throw new Exception("missing XLSX sheets");
    foreach (item; sheetNodes.children) if (z8_poi_xml_local(item.name) == "sheet"w) {
        auto target = item.attr("id"w) in targets;
        if (target is null) throw new Exception("missing XLSX sheet relationship");
        sheets ~= z8_poi_resolve_path(base, *target);
    }
    IZ8PoiProvider provider;
    synchronized (z8_poi_monitor) provider = z8_poi_provider;
    auto writer = cast(IZ8PoiReportWriter)provider;
    bool plainSheet = writer !is null && z8_poi_report_openxlsx_safe(archive, sheets.length);
    Z8PoiHiddenColumn[] changes;
    auto keys = hidden.z8_keys();
    foreach (keyIndex; 0 .. keys.z8_size()) {
        auto key = keys.z8_get(keyIndex);
        auto index = cast(int)z8_optional_require(key);
        if (index < 0 || index >= sheets.length) throw new Exception("invalid POI report sheet index");
        bool[int] wanted;
        auto wantedOrder = new Z8JavaMapOrder!(Z8Optional!long)();
        auto columns = z8_nonnull(hidden.z8_get(key));
        foreach (i; 0 .. columns.z8_size()) {
            auto number = z8_poi_report_column(columns.z8_get(i));
            if (number !in wanted) {
                wanted[number] = true; wantedOrder.add(z8_optional_wrap(cast(long)number - 1));
                changes ~= Z8PoiHiddenColumn(cast(uint)index, cast(uint)(number - 1));
            }
        }
        if (!wanted.length) continue;
        auto member = sheets[index] in archive.directory;
        if (member is null) throw new Exception("missing XLSX worksheet");
        archive.expand(*member);
        auto original = (*member).expandedData.dup;
        auto xml = z8_poi_xml_decode(original);
        auto sheet = (new Z8PoiXmlReader(xml)).parseDocument();
        if (z8_poi_xml_local(sheet.name) != "worksheet"w || sheet.empty) throw new Exception("invalid XLSX worksheet");
        Z8PoiXmlNode[] columnGroups;
        foreach (child; sheet.children) if (z8_poi_xml_local(child.name) == "cols"w) columnGroups ~= child;
        if (columnGroups.length || sheet.name.lastIndexOf(':') >= 0) plainSheet = false;
        auto cols = columnGroups.length ? columnGroups[0] : null;
        auto prefixEnd = sheet.name.lastIndexOf(':');
        wstring prefix = prefixEnd < 0 ? ""w : sheet.name[0 .. prefixEnd + 1];
        Z8PoiColumn[] segments, sourceColumns;
        foreach (group; columnGroups) foreach (column; group.children) {
            if (z8_poi_xml_local(column.name) != "col"w || column.children.length)
                throw new Exception("unsupported XLSX column element");
            auto minimum = column.attr("min"w).to!int, maximum = column.attr("max"w).to!int;
            if (minimum < 1 || maximum > 16384 || minimum > maximum) throw new Exception("invalid XLSX column range");
            auto attributes = column.attributes.dup;
            // Moving a column out of a separate <cols> must not lose namespace
            // declarations scoped to that group. Child declarations win.
            if (group !is cols) foreach (name, value; group.attributes)
                if ((name == "xmlns"w || (name.length > 6 && name[0 .. 6] == "xmlns:"w)) && name !in attributes)
                    attributes[name] = value;
            sourceColumns ~= Z8PoiColumn(minimum, maximum, column.name, attributes);
        }
        {
            // POI 3.16's ColumnHelper can retain overlapping ranges after
            // cleaning. Model its insertion/splitting order before flattening
            // them, otherwise a broad range can acquire the wrong width.
            void mergeAttributes(ref Z8PoiColumn target, Z8PoiColumn source) {
                bool collapsed;
                foreach (name, value; source.attributes) {
                    auto local = z8_poi_xml_local(name);
                    if (local == "min"w || local == "max"w) continue;
                    if (local == "collapsed"w) collapsed = true;
                    target.attributes[name] = value;
                }
                foreach (name; target.attributes.keys) if (z8_poi_xml_local(name) == "collapsed"w) target.attributes.remove(name);
                target.attributes["collapsed"w] = collapsed ? "1"w : "0"w;
            }
            Z8PoiColumn clone(Z8PoiColumn source, int first, int last) {
                Z8PoiColumn result = Z8PoiColumn(first, last, source.name, null);
                mergeAttributes(result, source); return result;
            }
            Z8PoiColumn[] clean;
            void addUnique(Z8PoiColumn item) {
                foreach (previous; clean) if (previous.first == item.first && previous.last == item.last) return;
                clean ~= item;
            }
            void insertUnique(Z8PoiColumn item) {
                foreach (previous; clean) if (previous.first == item.first && previous.last == item.last) return;
                clean = [item] ~ clean;
            }
            foreach (column; sourceColumns) {
                Z8PoiColumn[] intersections, remainder;
                foreach (item; clean) {
                    if (column.first <= item.last && column.last >= item.first) intersections ~= item;
                    else remainder ~= item;
                }
                clean = remainder;
                if (!intersections.length) addUnique(clone(column, column.first, column.last));
                else foreach (old; intersections) {
                    auto start = old.first > column.first ? old.first : column.first;
                    auto end = old.last < column.last ? old.last : column.last;
                    auto common = clone(old, start, end); mergeAttributes(common, column); addUnique(common);
                    auto left = old.first < column.first ? old : column;
                    auto leftStart = old.first < column.first ? old.first : column.first;
                    if (leftStart < start) addUnique(clone(left, leftStart, start - 1));
                    auto right = old.last > column.last ? old : column;
                    auto rightEnd = old.last > column.last ? old.last : column.last;
                    if (end < rightEnd) addUnique(clone(right, end + 1, rightEnd));
                }
                sort!((a, b) => a.first == b.first ? a.last < b.last : a.first < b.first)(clean);
            }
            foreach (wantedKey; wantedOrder.ordered()) {
                auto n = cast(int)z8_optional_require(wantedKey) + 1;
                size_t columnIndex;
                while (columnIndex < clean.length && !(clean[columnIndex].first <= n && clean[columnIndex].last >= n)) ++columnIndex;
                if (columnIndex == clean.length) clean ~= Z8PoiColumn(n, n, prefix ~ "col"w, ["hidden"w: "1"w]);
                else {
                    auto current = clean[columnIndex];
                    current.attributes = current.attributes.dup;
                    clean[columnIndex].first = n; clean[columnIndex].last = n;
                    clean[columnIndex].attributes["hidden"w] = "1"w;
                    if (current.first < n) insertUnique(clone(current, current.first, n - 1));
                    if (current.last > n) insertUnique(clone(current, n + 1, current.last));
                }
            }
            auto slots = new Z8PoiColumn[16385];
            bool[16385] occupied;
            foreach (column; clean) foreach (n; column.first .. column.last + 1) if (!occupied[n]) {
                slots[n] = column; occupied[n] = true;
            }
            foreach (n; 1 .. 16385) if (occupied[n]) {
                auto item = slots[n];
                item.first = item.last = n;
                if (segments.length && segments[$ - 1].last + 1 == n && segments[$ - 1].name == item.name &&
                    segments[$ - 1].attributes == item.attributes) segments[$ - 1].last = n;
                else segments ~= item;
            }
        }
        sort!((a, b) => a.first < b.first)(segments);
        wstring content;
        foreach (column; segments) {
            content ~= "<"w ~ column.name ~ " min=\""w ~ column.first.to!wstring ~ "\" max=\""w ~ column.last.to!wstring ~ "\""w;
            foreach (name, value; column.attributes) {
                auto local = z8_poi_xml_local(name);
                // On load POI 3.16 sets collapsed from the attribute's
                // presence, so an absent attribute reproduces false.
                if (local == "min"w || local == "max"w || (local == "collapsed"w && value == "0"w)) continue;
                content ~= " "w ~ name ~ "=\""w ~ z8_poi_xml_attribute_escape(value) ~ "\""w;
            }
            content ~= "/>"w;
        }
        if (cols !is null) {
            auto opening = xml[cols.start .. cols.startEnd];
            if (cols.empty) opening = opening[0 .. $ - 2] ~ ">"w;
            auto updated = xml[0 .. cols.start] ~ opening ~ content ~ "</"w ~ cols.name ~ ">"w;
            auto previous = cols.end;
            foreach (group; columnGroups[1 .. $]) {
                updated ~= xml[previous .. group.start];
                previous = group.end;
            }
            xml = updated ~ xml[previous .. $];
        } else {
            size_t position = sheet.endStart;
            auto data = sheet.child("sheetData"w);
            if (data !is null) position = data.start;
            xml = xml[0 .. position] ~ "<"w ~ prefix ~ "cols>"w ~ content ~ "</"w ~ prefix ~ "cols>"w ~ xml[position .. $];
        }
        (*member).expandedData = z8_poi_report_encode_xml(xml, original);
    }
    if (plainSheet && changes.length) {
        try return writer.hideReportColumns(input, changes);
        catch (Z8PoiUnsupportedWorkbook) { /* use the compatible column editor */ }
    }
    auto output = new ZipArchive(); output.comment = archive.comment;
    foreach (member; archive.directory.byValue) output.addMember(member);
    return cast(ubyte[])output.build();
}
