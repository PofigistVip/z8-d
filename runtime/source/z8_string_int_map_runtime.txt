// String-keyed integer maps retain null keys and null integer values.
class Z8StringIntMap {
    private Z8Optional!long[wstring] entries;
    private bool hasNullKey;
    private Z8Optional!long nullKeyValue;

    private static bool valueEquals(Z8Optional!long left, Z8Optional!long right) {
        return left.present == right.present && (!left.present || left.value == right.value);
    }

    long z8_size() { return cast(long)entries.length + (hasNullKey ? 1 : 0); }
    bool z8_isEmpty() { return entries.length == 0 && !hasNullKey; }
    void z8_clear() { entries.clear(); hasNullKey = false; nullKeyValue = Z8Optional!long.init; }
    bool z8_containsKey(wstring key) { return key is null ? hasNullKey : (key in entries) !is null; }
    bool z8_containsValue(Z8Optional!long value) {
        if (hasNullKey && valueEquals(value, nullKeyValue)) return true;
        foreach (item; entries) if (valueEquals(value, item)) return true;
        return false;
    }
    Z8Optional!long z8_add(wstring key, Z8Optional!long value) {
        if (key is null) {
            auto previous = hasNullKey ? nullKeyValue : Z8Optional!long.init;
            hasNullKey = true; nullKeyValue = value; return previous;
        }
        auto found = key in entries;
        auto previous = found is null ? Z8Optional!long.init : *found;
        entries[key] = value;
        return previous;
    }
    void z8_add(Z8StringIntMap other) {
        z8_nonnull(other);
        if (other is this) return;
        foreach (key, value; other.entries) z8_add(key, value);
        if (other.hasNullKey) z8_add(null, other.nullKeyValue);
    }
    Z8Optional!long z8_get(wstring key) {
        if (key is null) return hasNullKey ? nullKeyValue : Z8Optional!long.init;
        auto found = key in entries;
        return found is null ? Z8Optional!long.init : *found;
    }
    Z8Optional!long z8_remove(wstring key) {
        if (key is null) {
            auto previous = hasNullKey ? nullKeyValue : Z8Optional!long.init;
            hasNullKey = false; nullKeyValue = Z8Optional!long.init; return previous;
        }
        auto found = key in entries;
        if (found is null) return Z8Optional!long.init;
        auto previous = *found;
        entries.remove(key);
        return previous;
    }
    Z8Array!wstring z8_keys() {
        auto result = new Z8Array!wstring();
        foreach (key, value; entries) result.z8_add(key);
        if (hasNullKey) result.z8_add(null);
        return result;
    }
    Z8Array!long z8_values() {
        auto result = new Z8Array!long();
        foreach (value; entries) result.z8_add(value);
        if (hasNullKey) result.z8_add(nullKeyValue);
        return result;
    }
}
Z8StringIntMap z8_map_union(Z8StringIntMap left, Z8StringIntMap right) {
    auto result = new Z8StringIntMap();
    result.z8_add(z8_nonnull(left));
    result.z8_add(z8_nonnull(right));
    return result;
}
void z8_map_add_assign(ref Z8StringIntMap left, lazy Z8StringIntMap previous, lazy Z8StringIntMap right) {
    auto oldValue = previous;
    auto newValue = right;
    left = z8_map_union(oldValue, newValue);
}
