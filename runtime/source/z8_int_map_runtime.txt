// Integer keys use value equality and retain a distinct null key, as in Z8.
class Z8IntMap(V) {
    static if (is(V == long) || is(V == bool) || is(V == Z8Decimal)) alias Element = Z8Optional!V;
    else static if (is(V == wstring)) alias Element = wstring;
    else static if (is(V == Z8Primary) || is(V == Z8Guid)) alias Element = V;
    else static assert(0, "unsupported integer-key map value");

    private Element[long] entries;
    private bool hasNullKey;
    private Element nullKeyValue;

    private static bool valueEquals(Element left, Element right) {
        static if (is(V == long) || is(V == bool) || is(V == Z8Decimal))
            return left.present == right.present && (!left.present || left.value == right.value);
        else static if (is(V == wstring)) return z8_string_equals(left, right);
        else return left is right || (left !is null && left == right);
    }

    long z8_size() { return cast(long)entries.length + (hasNullKey ? 1 : 0); }
    bool z8_isEmpty() { return entries.length == 0 && !hasNullKey; }
    void z8_clear() { entries.clear(); hasNullKey = false; nullKeyValue = Element.init; }
    bool z8_containsKey(Z8Optional!long key) { return key.present ? (key.value in entries) !is null : hasNullKey; }
    bool z8_containsValue(Element value) {
        if (hasNullKey && valueEquals(value, nullKeyValue)) return true;
        foreach (item; entries) if (valueEquals(value, item)) return true;
        return false;
    }
    Element z8_add(Z8Optional!long key, Element value) {
        if (!key.present) {
            auto previous = hasNullKey ? nullKeyValue : Element.init;
            hasNullKey = true; nullKeyValue = value; return previous;
        }
        auto found = key.value in entries;
        auto previous = found is null ? Element.init : *found;
        entries[key.value] = value;
        return previous;
    }
    void z8_add(Z8IntMap!V other) {
        z8_nonnull(other);
        if (other is this) return;
        foreach (key, value; other.entries) z8_add(z8_optional_wrap(key), value);
        if (other.hasNullKey) z8_add(Z8Optional!long.init, other.nullKeyValue);
    }
    Element z8_get(Z8Optional!long key) {
        if (!key.present) return hasNullKey ? nullKeyValue : Element.init;
        auto found = key.value in entries;
        return found is null ? Element.init : *found;
    }
    Element z8_remove(Z8Optional!long key) {
        if (!key.present) {
            auto previous = hasNullKey ? nullKeyValue : Element.init;
            hasNullKey = false; nullKeyValue = Element.init; return previous;
        }
        auto found = key.value in entries;
        if (found is null) return Element.init;
        auto previous = *found;
        entries.remove(key.value);
        return previous;
    }
    Z8Array!long z8_keys() {
        auto result = new Z8Array!long();
        foreach (key, value; entries) result.z8_add(key);
        if (hasNullKey) result.z8_add(Z8Optional!long.init);
        return result;
    }
    Z8Array!V z8_values() {
        auto result = new Z8Array!V();
        foreach (value; entries) result.z8_add(value);
        if (hasNullKey) result.z8_add(nullKeyValue);
        return result;
    }
}
Z8IntMap!V z8_map_union(V)(Z8IntMap!V left, Z8IntMap!V right) {
    auto result = new Z8IntMap!V();
    result.z8_add(z8_nonnull(left));
    result.z8_add(z8_nonnull(right));
    return result;
}
void z8_map_add_assign(V)(ref Z8IntMap!V left, lazy Z8IntMap!V previous, lazy Z8IntMap!V right) {
    auto oldValue = previous;
    auto newValue = right;
    left = z8_map_union(oldValue, newValue);
}
