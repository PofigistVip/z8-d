// GUID keys use the same stored-hash and reference semantics as primary keys.
class Z8GuidMap(V) {
    alias Element = Z8PrimaryKeyMap!V.Element;
    private Z8PrimaryKeyMap!V entries;
    this() { entries = new Z8PrimaryKeyMap!V(); }
    override bool opEquals(Object other) {
        auto right = cast(Z8GuidMap!V)other;
        return right !is null && z8_map_content_equals(this, right);
    }
    long z8_size() { return entries.z8_size(); }
    bool z8_isEmpty() { return entries.z8_isEmpty(); }
    void z8_clear() { entries.z8_clear(); }
    bool z8_containsKey(Z8Guid key) { return entries.z8_containsKey(z8_primary_box(key)); }
    bool z8_containsValue(Element value) { return entries.z8_containsValue(value); }
    Element z8_add(Z8Guid key, Element value) { return entries.z8_add(z8_primary_box(key), value); }
    void z8_add(Z8GuidMap!V other) { entries.z8_add(z8_nonnull(other).entries); }
    Element z8_get(Z8Guid key) { return entries.z8_get(z8_primary_box(key)); }
    Element z8_remove(Z8Guid key) { return entries.z8_remove(z8_primary_box(key)); }
    Z8Array!Z8Guid z8_keys() {
        auto result = new Z8Array!Z8Guid(); auto keys = entries.z8_keys();
        for (long i; i < keys.z8_size(); ++i) result.z8_add(z8_primary_cast!Z8Guid(keys.z8_get(i)));
        return result;
    }
    Z8ValueArray!V z8_values() { return entries.z8_values(); }
}
Z8GuidMap!V z8_map_union(V)(Z8GuidMap!V left, Z8GuidMap!V right) {
    auto result = new Z8GuidMap!V(); result.z8_add(z8_nonnull(left)); result.z8_add(z8_nonnull(right)); return result;
}
void z8_map_add_assign(V)(ref Z8GuidMap!V left, lazy Z8GuidMap!V previous, lazy Z8GuidMap!V right) {
    auto oldValue = previous; auto newValue = right;
    left = z8_map_union(oldValue, newValue);
}
