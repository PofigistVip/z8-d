// SQL values describe an expression tree; operators do not execute SQL.
enum Z8SqlVendor { Oracle, SqlServer, Postgres, H2 }
class Z8SqlPrimary {}
abstract class Z8SqlNumeric : Z8SqlPrimary { abstract wstring z8_format(Z8SqlVendor vendor, bool logical = false); }
Z8SqlString z8_sql_primary_toString(Z8SqlPrimary value) {
    z8_nonnull(value);
    if (auto result = cast(Z8SqlBool)value) return z8_sql_bool_toString(result);
    if (auto result = cast(Z8SqlInt)value) return z8_sql_int_toString(result);
    if (auto result = cast(Z8SqlDecimal)value) return z8_sql_decimal_toString(result);
    if (auto result = cast(Z8SqlString)value) return z8_sql_string_toString(result);
    if (auto result = cast(Z8SqlGuid)value) return z8_sql_guid_toString(result);
    if (auto result = cast(Z8SqlDate)value) return z8_sql_date_toString(result);
    if (auto result = cast(Z8SqlDatespan)value) return z8_sql_datespan_toString(result);
    if (auto result = cast(Z8SqlBinary)value) return z8_sql_binary_toString(result);
    if (auto result = cast(Z8SqlGeometry)value) return z8_sql_geometry_toString(result);
    throw new Exception("unsupported SQL primary conversion");
}
enum Z8SqlBoolKind { constant, group, not, and, or, equal, notEqual, choose, intEqual, intNotEqual, intLess, intGreater, intLessEqual, intGreaterEqual, decimalIsNaN, textEqual, textNotEqual, textLess, textGreater, textLessEqual, textGreaterEqual, textLike, textRegLike, textIsEmpty, guidEqual, guidNotEqual, dateCompare, spanCompare, geometryCompare, geometryPredicate }
class Z8SqlBool : Z8SqlPrimary {
    Z8SqlBoolKind kind;
    bool constant;
    Z8SqlBool left, right, alternative;
    Z8SqlNumeric numericLeft, numericRight;
    Z8SqlString textLeft, textRight;
    Z8SqlString textEscape;
    Z8SqlGuid guidLeft, guidRight;
    Z8SqlDate dateLeft, dateRight;
    string dateOperator;
    Z8SqlDatespan spanLeft, spanRight;
    string spanOperator;
    Z8SqlGeometry geometryLeft, geometryRight;
    string geometryOperator;
    wstring geometryFunction;
    this(Z8SqlBoolKind kind, bool constant = false, Z8SqlBool left = null,
         Z8SqlBool right = null, Z8SqlBool alternative = null) {
        this.kind = kind; this.constant = constant;
        this.left = left; this.right = right; this.alternative = alternative;
    }
}
Z8SqlBool z8_sql_bool_const(bool value) { return new Z8SqlBool(Z8SqlBoolKind.constant, value); }
Z8SqlBool z8_sql_bool_const(Z8Optional!bool value) { return z8_sql_bool_const(z8_optional_require(value)); }
Z8SqlBool z8_sql_bool_group(Z8SqlBool value) {
    return new Z8SqlBool(Z8SqlBoolKind.group, false, z8_nonnull(value));
}
Z8SqlBool z8_sql_bool_not(Z8SqlBool value) {
    return new Z8SqlBool(Z8SqlBoolKind.not, false, z8_nonnull(value));
}
Z8SqlBool z8_sql_bool_binary(string op)(Z8SqlBool left, Z8SqlBool right) {
    z8_nonnull(left);
    static if (op == "&&") enum kind = Z8SqlBoolKind.and;
    else static if (op == "||") enum kind = Z8SqlBoolKind.or;
    else static if (op == "==") enum kind = Z8SqlBoolKind.equal;
    else static if (op == "!=") enum kind = Z8SqlBoolKind.notEqual;
    else static assert(0, "unsupported SQL boolean operator");
    return new Z8SqlBool(kind, false, left, right);
}
Z8SqlBool z8_sql_bool_iif(Z8SqlBool condition, Z8SqlBool yes, Z8SqlBool no) {
    return new Z8SqlBool(Z8SqlBoolKind.choose, false, z8_nonnull(condition), yes, no);
}

private wstring z8_sql_bool_group_format(Z8SqlBool value, bool logical, Z8SqlVendor vendor) {
    z8_nonnull(value);
    auto text = z8_sql_bool_format(value, logical, vendor);
    if ([Z8SqlBoolKind.constant, Z8SqlBoolKind.group, Z8SqlBoolKind.and,
         Z8SqlBoolKind.equal, Z8SqlBoolKind.notEqual, Z8SqlBoolKind.intEqual, Z8SqlBoolKind.intNotEqual,
         Z8SqlBoolKind.intLess, Z8SqlBoolKind.intGreater, Z8SqlBoolKind.intLessEqual, Z8SqlBoolKind.intGreaterEqual,
         Z8SqlBoolKind.textEqual, Z8SqlBoolKind.textNotEqual, Z8SqlBoolKind.textLess, Z8SqlBoolKind.textGreater,
         Z8SqlBoolKind.textLessEqual, Z8SqlBoolKind.textGreaterEqual, Z8SqlBoolKind.guidEqual, Z8SqlBoolKind.guidNotEqual, Z8SqlBoolKind.dateCompare, Z8SqlBoolKind.spanCompare, Z8SqlBoolKind.geometryCompare].canFind(value.kind)) return text;
    return "("w ~ text ~ ")"w;
}
private wstring z8_sql_bool_case(wstring condition, wstring yes, wstring no) {
    return "case  when "w ~ condition ~ " then "w ~ yes ~ " else "w ~ no ~ " end"w;
}
// Boolean constants and these expression nodes have the same formatting in
// all four supported Z8 database vendors. `logical` selects WHERE context.
wstring z8_sql_bool_format(Z8SqlBool value, bool logical = false, Z8SqlVendor vendor = Z8SqlVendor.Postgres) {
    z8_nonnull(value);
    final switch (value.kind) {
        case Z8SqlBoolKind.constant:
            return (value.constant ? "1"w : "0"w) ~ (logical ? "=1"w : ""w);
        case Z8SqlBoolKind.group: return z8_sql_bool_group_format(value.left, logical, vendor);
        case Z8SqlBoolKind.not:
            auto condition = "not("w ~ z8_sql_bool_format(value.left, true, vendor) ~ ")"w;
            return logical ? condition : z8_sql_bool_case(condition, "1"w, "0"w);
        case Z8SqlBoolKind.and:
            auto condition = z8_sql_bool_group_format(value.left, true, vendor) ~ " AND "w ~
                z8_sql_bool_group_format(value.right, true, vendor);
            return logical ? condition : z8_sql_bool_case(condition, "1"w, "0"w);
        case Z8SqlBoolKind.or:
            auto condition = z8_sql_bool_format(value.left, true, vendor) ~ " OR "w ~ z8_sql_bool_format(value.right, true, vendor);
            return logical ? condition : z8_sql_bool_case(condition, "1"w, "0"w);
        case Z8SqlBoolKind.equal: case Z8SqlBoolKind.notEqual:
            auto condition = z8_sql_bool_format(value.left, false, vendor) ~ (value.kind == Z8SqlBoolKind.equal ? "="w : "<>"w) ~ z8_sql_bool_format(value.right, false, vendor);
            return logical ? condition : z8_sql_bool_case(condition, "1"w, "0"w);
        case Z8SqlBoolKind.intEqual: case Z8SqlBoolKind.intNotEqual: case Z8SqlBoolKind.intLess:
        case Z8SqlBoolKind.intGreater: case Z8SqlBoolKind.intLessEqual: case Z8SqlBoolKind.intGreaterEqual:
            auto op = value.kind == Z8SqlBoolKind.intEqual ? "="w : value.kind == Z8SqlBoolKind.intNotEqual ? "<>"w :
                value.kind == Z8SqlBoolKind.intLess ? "<"w : value.kind == Z8SqlBoolKind.intGreater ? ">"w :
                value.kind == Z8SqlBoolKind.intLessEqual ? "<="w : ">="w;
            auto condition = z8_nonnull(value.numericLeft).z8_format(vendor) ~ op ~ z8_nonnull(value.numericRight).z8_format(vendor);
            return logical ? condition : z8_sql_bool_case(condition, "1"w, "0"w);
        case Z8SqlBoolKind.decimalIsNaN:
            if (vendor != Z8SqlVendor.Postgres) throw new Exception("unsupported SQL database vendor");
            return z8_nonnull(value.numericLeft).z8_format(vendor) ~ " = double precision 'NaN'"w;
        case Z8SqlBoolKind.textEqual: case Z8SqlBoolKind.textNotEqual: case Z8SqlBoolKind.textLess:
        case Z8SqlBoolKind.textGreater: case Z8SqlBoolKind.textLessEqual: case Z8SqlBoolKind.textGreaterEqual:
            auto op = value.kind == Z8SqlBoolKind.textEqual ? "="w : value.kind == Z8SqlBoolKind.textNotEqual ? "<>"w :
                value.kind == Z8SqlBoolKind.textLess ? "<"w : value.kind == Z8SqlBoolKind.textGreater ? ">"w :
                value.kind == Z8SqlBoolKind.textLessEqual ? "<="w : ">="w;
            auto condition = z8_sql_string_format(value.textLeft, vendor) ~ op ~ z8_sql_string_format(value.textRight, vendor);
            return logical ? condition : z8_sql_bool_case(condition, "1"w, "0"w);
        case Z8SqlBoolKind.textLike:
            return z8_sql_string_as_text(value.textLeft, vendor) ~ " LIKE "w ~ z8_sql_string_as_text(value.textRight, vendor) ~
                (value.textEscape is null ? ""w : " ESCAPE "w ~ z8_sql_string_as_text(value.textEscape, vendor));
        case Z8SqlBoolKind.guidEqual: case Z8SqlBoolKind.guidNotEqual:
            auto condition = z8_sql_guid_format(value.guidLeft, vendor) ~ (value.kind == Z8SqlBoolKind.guidEqual ? "="w : "<>"w) ~ z8_sql_guid_format(value.guidRight, vendor);
            return logical ? condition : z8_sql_bool_case(condition, "1"w, "0"w);
        case Z8SqlBoolKind.dateCompare:
            auto condition = z8_sql_date_compare_format(value, vendor);
            return logical ? condition : z8_sql_bool_case(condition, "1"w, "0"w);
        case Z8SqlBoolKind.spanCompare:
            auto op = value.spanOperator == "==" ? "="w : value.spanOperator == "!=" ? "<>"w : value.spanOperator.to!wstring;
            auto condition = z8_sql_datespan_format(value.spanLeft, vendor) ~ op ~ z8_sql_datespan_format(value.spanRight, vendor);
            return logical ? condition : z8_sql_bool_case(condition, "1"w, "0"w);
        case Z8SqlBoolKind.geometryCompare:
            auto left = z8_sql_geometry_text(value.geometryLeft, vendor), right = z8_sql_geometry_text(value.geometryRight, vendor);
            auto condition = value.geometryOperator == "&&" ? "st_intersects("w ~ left ~ ", "w ~ right ~ ")"w : left ~ (value.geometryOperator == "==" ? "="w : "<>"w) ~ right;
            return logical ? condition : z8_sql_bool_case(condition, "1"w, "0"w);
        case Z8SqlBoolKind.geometryPredicate: return z8_sql_geometry_function(value.geometryFunction, value.geometryLeft, vendor);
        case Z8SqlBoolKind.textRegLike:
            if (vendor != Z8SqlVendor.Postgres) throw new Exception("unsupported SQL database vendor");
            return z8_sql_string_format(value.textLeft, vendor) ~ " ~ "w ~ z8_sql_string_format(value.textRight, vendor);
        case Z8SqlBoolKind.textIsEmpty:
            auto first = "("w ~ z8_sql_string_format(value.textLeft, vendor) ~ ")"w;
            auto second = "("w ~ z8_sql_string_format(value.textLeft, vendor) ~ ")"w;
            auto empty = vendor == Z8SqlVendor.Oracle || vendor == Z8SqlVendor.SqlServer ? "N''"w : "''"w;
            auto condition = first ~ " is null OR "w ~ second ~ "="w ~ empty;
            return "("w ~ (logical ? condition : z8_sql_bool_case(condition, "1"w, "0"w)) ~ ")"w;
        case Z8SqlBoolKind.choose:
            auto text = z8_sql_bool_case(z8_sql_bool_format(value.left, true, vendor), z8_sql_bool_format(value.right, false, vendor), z8_sql_bool_format(value.alternative, false, vendor));
            return logical ? "("w ~ text ~ ")=1"w : text;
    }
}
