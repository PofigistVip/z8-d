private class Z8PoiXmlNode {
    wstring name, text;
    size_t start, startEnd, endStart, end;
    bool empty;
    wstring[wstring] attributes;
    Z8PoiXmlNode[] children;
    Z8PoiXmlNode child(wstring wanted) {
        foreach (item; children) if (z8_poi_xml_local(item.name) == wanted) return item;
        return null;
    }
    wstring attr(wstring wanted) {
        foreach (key, value; attributes) if (z8_poi_xml_local(key) == wanted) return value;
        return null;
    }
    wstring allText() {
        wstring result = text;
        foreach (item; children) result ~= item.allText();
        return result;
    }
}
private wstring z8_poi_xml_local(wstring name) {
    import std.algorithm.searching : countUntil;
    auto colon = name.countUntil(':'); return colon < 0 ? name : name[colon + 1 .. $];
}
private class Z8PoiXmlReader : Z8NlsReader {
    this(wstring value) { super(value); }
    Z8PoiXmlNode node() {
        auto start = position;
        expect("<"w); auto tag = name(); position -= tag.length + 1;
        bool empty; auto result = new Z8PoiXmlNode(); result.name = tag;
        result.attributes = startTag(tag, empty);
        result.start = start; result.startEnd = position; result.empty = empty;
        if (empty) { result.endStart = position; result.end = position; return result; }
        for (;;) {
            if (starts("</"w)) {
                result.endStart = position;
                position += 2; if (name() != tag) throw new Exception("mismatched XLSX XML element");
                whitespace(); expect(">"w); result.end = position; return result;
            }
            if (starts("<!--"w)) { position += 4; until("-->"w); continue; }
            if (starts("<?"w)) { position += 2; until("?>"w); continue; }
            if (starts("<![CDATA["w)) {
                position += 9; auto begin = position;
                while (position < text.length && !starts("]]>"w)) ++position;
                result.text ~= text[begin .. position]; expect("]]>"w); continue;
            }
            if (position >= text.length) throw new Exception("truncated XLSX XML");
            if (starts("<"w)) { result.children ~= node(); continue; }
            auto begin = position; while (position < text.length && text[position] != '<') ++position;
            result.text ~= decode(text[begin .. position]);
        }
    }
    Z8PoiXmlNode parseDocument() { misc(); auto result = node(); misc(); if (position != text.length) throw new Exception("invalid XLSX XML"); return result; }
}
// XSSFRichTextString.utfDecode performs one pass over UTF-16 code units.
// An escaped underscore such as _x005F_x0041_ must therefore remain _x0041_.
private wstring z8_poi_decode_text(wstring text) {
    wchar[] result;
    size_t i;
    while (i < text.length) {
        if (i + 7 <= text.length && text[i] == '_' && text[i + 1] == 'x' && text[i + 6] == '_') {
            uint code; bool valid = true;
            foreach (j; i + 2 .. i + 6) {
                auto c = text[j];
                int digit = c >= '0' && c <= '9' ? c - '0' : c >= 'a' && c <= 'f' ? c - 'a' + 10 : c >= 'A' && c <= 'F' ? c - 'A' + 10 : -1;
                if (digit < 0) { valid = false; break; }
                code = code * 16 + digit;
            }
            if (valid) { result ~= cast(wchar)code; i += 7; continue; }
        }
        result ~= text[i++];
    }
    return result.length ? result.idup : ""w;
}
private wstring z8_poi_rich_text(Z8PoiXmlNode value) {
    wstring result;
    bool hasRuns;
    foreach (child; value.children) if (z8_poi_xml_local(child.name) == "r"w) {
        hasRuns = true; auto text = child.child("t"w); if (text !is null) result ~= text.allText();
    }
    if (!hasRuns) { auto text = value.child("t"w); if (text !is null) result = text.allText(); }
    return z8_poi_decode_text(result);
}

private wstring z8_poi_column_name(long column) {
    wchar[] result;
    while (column) { --column; result = [cast(wchar)('A' + column % 26)] ~ result; column /= 26; }
    return result.idup;
}
// POI expands C:C to C$1:C$1048576 and 2:2 to $A2:$XFD2.
private int z8_poi_whole_range(wstring text, size_t start, long rowOffset, long columnOffset,
                               out size_t end, out wstring result) {
    import std.uni : isAlpha, isNumber;
    bool identifier(wchar c) { return isAlpha(c) || isNumber(c) || c == '_' || c == '.'; }
    if (start && identifier(text[start - 1])) return 0;
    size_t i = start;
    bool[2] fixed;
    long[2] values;
    bool columns;
    foreach (part; 0 .. 2) {
        if (i < text.length && text[i] == '$') { fixed[part] = true; ++i; }
        auto begin = i;
        bool letters = i < text.length && ((text[i] >= 'A' && text[i] <= 'Z') || (text[i] >= 'a' && text[i] <= 'z'));
        if (part == 0) columns = letters; else if (letters != columns) return 0;
        if (letters) {
            while (i < text.length && ((text[i] >= 'A' && text[i] <= 'Z') || (text[i] >= 'a' && text[i] <= 'z'))) {
                wchar ch = text[i++]; if (ch >= 'a') ch -= 'a' - 'A';
                if (i - begin <= 3) values[part] = values[part] * 26 + ch - 'A' + 1;
            }
            if (i == begin || i - begin > 3 || values[part] > 16384) return 0;
        } else {
            while (i < text.length && text[i] >= '0' && text[i] <= '9') {
                if (i - begin < 7) values[part] = values[part] * 10 + text[i] - '0';
                ++i;
            }
            if (i == begin || i - begin > 7 || values[part] < 1 || values[part] > 1048576) return 0;
        }
        if (part == 0) { if (i >= text.length || text[i] != ':') return 0; ++i; }
    }
    if (i < text.length && (identifier(text[i]) || text[i] == '!' || text[i] == '(')) return 0;
    foreach (part; 0 .. 2) {
        auto limit = columns ? 16384L : 1048576L;
        auto offset = fixed[part] ? 0 : columns ? columnOffset : rowOffset;
        auto shifted = ((values[part] - 1 + offset) & (limit - 1)) + 1;
        if (part) result ~= ':';
        if (columns) {
            if (fixed[part]) result ~= '$'; result ~= z8_poi_column_name(shifted);
            result ~= part == 0 ? "$1"w : "$1048576"w;
        } else {
            result ~= part == 0 ? "$A"w : "$XFD"w;
            if (fixed[part]) result ~= '$'; result ~= shifted.to!wstring;
        }
    }
    end = i; return 1;
}
private bool z8_poi_normalize_formula(wstring source, out wstring result) {
    import std.uni : isAlpha, isNumber;
    bool operand(wchar c) { return isAlpha(c) || isNumber(c) || c == '_' || c == '$' || c == '.' || c == '\''; }
    size_t i;
    while (i < source.length) {
        auto c = source[i];
        if (c == '"' || c == '\'') {
            auto start = i++, quote = c; bool closed;
            while (i < source.length) {
                if (source[i++] != quote) continue;
                if (i < source.length && source[i] == quote) { ++i; continue; }
                closed = true; break;
            }
            if (!closed) return false;
            result ~= source[start .. i]; continue;
        }
        if (c == ' ' || c == '\t' || c == '\r' || c == '\n') {
            while (i < source.length && (source[i] == ' ' || source[i] == '\t' || source[i] == '\r' || source[i] == '\n')) ++i;
            // A space can be the Excel intersection operator; don't turn it
            // into a concatenated identifier without parsing both operands.
            if (result.length && i < source.length && (operand(result[$ - 1]) || result[$ - 1] == ')') && operand(source[i])) return false;
            continue;
        }
        result ~= c; ++i;
    }
    return true;
}
// Translate ordinary A1 references while leaving string literals, quoted sheet
// names, absolute coordinates and function names untouched.
private bool z8_poi_shift_formula(wstring source, long rowOffset, long columnOffset, out wstring result) {
    import std.uni : isAlpha, isNumber;
    wstring formula;
    if (!z8_poi_normalize_formula(source, formula)) return false;
    bool identifier(wchar c) { return isAlpha(c) || isNumber(c) || c == '_' || c == '.'; }
    size_t i;
    while (i < formula.length) {
        auto c = formula[i];
        // External and structured references need the full POI parser.
        if (c == '[') return false;
        if (c == '#') {
            bool recognized;
            foreach (error; ["#NULL!"w, "#DIV/0!"w, "#VALUE!"w, "#REF!"w, "#NAME?"w, "#NUM!"w, "#N/A"w, "#GETTING_DATA"w]) {
                if (i + error.length <= formula.length && formula[i .. i + error.length] == error) {
                    result ~= error; i += error.length; recognized = true; break;
                }
            }
            if (!recognized) return false;
            continue;
        }
        if (c == '"' || c == '\'') {
            auto quote = c, start = i++;
            bool closed;
            while (i < formula.length) {
                if (formula[i++] != quote) continue;
                if (i < formula.length && formula[i] == quote) { ++i; continue; }
                closed = true; break;
            }
            if (!closed) return false;
            // POI 3.16's shared-formula conversion drops sheet qualifiers
            // when translating references (including quoted sheet names).
            if (quote == '\'' && i < formula.length && formula[i] == '!') { ++i; continue; }
            result ~= formula[start .. i]; continue;
        }
        if ((isAlpha(c) || c == '_') && (i == 0 || !identifier(formula[i - 1]))) {
            auto endName = i + 1;
            while (endName < formula.length && identifier(formula[endName])) ++endName;
            if (endName < formula.length && formula[endName] == '!') { i = endName + 1; continue; }
        }
        if (c == '$' || (c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9')) {
            size_t end; wstring range;
            auto matched = z8_poi_whole_range(formula, i, rowOffset, columnOffset, end, range);
            if (matched < 0) return false;
            if (matched > 0) { result ~= range; i = end; continue; }
        }
        if ((c == '$' || (c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z')) &&
            (i == 0 || !identifier(formula[i - 1]))) {
            auto start = i, cursor = i;
            bool fixedColumn = formula[cursor] == '$'; if (fixedColumn) ++cursor;
            auto letters = cursor; long column;
            while (cursor < formula.length && ((formula[cursor] >= 'A' && formula[cursor] <= 'Z') ||
                   (formula[cursor] >= 'a' && formula[cursor] <= 'z'))) {
                wchar ch = formula[cursor++]; if (ch >= 'a') ch -= 'a' - 'A';
                if (cursor - letters <= 3) column = column * 26 + ch - 'A' + 1;
            }
            auto letterCount = cursor - letters;
            bool fixedRow = cursor < formula.length && formula[cursor] == '$'; if (fixedRow) ++cursor;
            auto digits = cursor; long row;
            while (cursor < formula.length && formula[cursor] >= '0' && formula[cursor] <= '9') {
                if (cursor - digits < 7) row = row * 10 + formula[cursor] - '0';
                ++cursor;
            }
            if (letterCount && letterCount <= 3 && column <= 16384 && cursor > digits && cursor - digits <= 7 &&
                row > 0 && row <= 1048576 && (cursor == formula.length ||
                (!identifier(formula[cursor]) && formula[cursor] != '(' && formula[cursor] != '!'))) {
                // SharedFormula in POI 3.16 wraps relative coordinates at the
                // XLSX row and column boundaries.
                auto shiftedColumn = ((column - 1 + (fixedColumn ? 0 : columnOffset)) & 16383L) + 1;
                auto shiftedRow = ((row - 1 + (fixedRow ? 0 : rowOffset)) & 1048575L) + 1;
                if (fixedColumn) result ~= '$'; result ~= z8_poi_column_name(shiftedColumn);
                if (fixedRow) result ~= '$'; result ~= shiftedRow.to!wstring;
                i = cursor; continue;
            }
            i = start;
        }
        result ~= formula[i++];
    }
    return true;
}

private wstring z8_poi_xml_decode(const(ubyte)[] bytes) {
    bool little, utf16;
    size_t start;
    if (bytes.length >= 2 && bytes[0] == 0xFF && bytes[1] == 0xFE) { utf16 = true; little = true; start = 2; }
    else if (bytes.length >= 2 && bytes[0] == 0xFE && bytes[1] == 0xFF) { utf16 = true; start = 2; }
    else if (bytes.length >= 4 && bytes[0] == '<' && bytes[1] == 0 && bytes[2] != 0 && bytes[3] == 0) { utf16 = true; little = true; }
    else if (bytes.length >= 4 && bytes[0] == 0 && bytes[1] == '<' && bytes[2] == 0 && bytes[3] != 0) utf16 = true;
    if (!utf16) return (cast(string)bytes).to!wstring;
    if ((bytes.length - start) % 2) throw new Exception("truncated UTF-16 XLSX XML");
    wchar[] result;
    for (auto i = start; i < bytes.length; i += 2) {
        auto code = cast(uint)(little ? bytes[i] | (bytes[i + 1] << 8) : (bytes[i] << 8) | bytes[i + 1]);
        if (code >= 0xD800 && code <= 0xDBFF) {
            if (i + 3 >= bytes.length) throw new Exception("invalid UTF-16 XLSX XML");
            auto next = cast(uint)(little ? bytes[i + 2] | (bytes[i + 3] << 8) : (bytes[i + 2] << 8) | bytes[i + 3]);
            if (next < 0xDC00 || next > 0xDFFF) throw new Exception("invalid UTF-16 XLSX XML");
            result ~= cast(wchar)code; result ~= cast(wchar)next; i += 2;
        } else {
            if (code >= 0xDC00 && code <= 0xDFFF) throw new Exception("invalid UTF-16 XLSX XML");
            result ~= cast(wchar)code;
        }
    }
    return result.length ? result.idup : ""w;
}
private Z8PoiXmlNode z8_poi_read_xml(ZipArchive archive, string path, bool required = true) {
    auto found = path in archive.directory;
    if (found is null) { if (required) throw new Exception("missing XLSX entry: " ~ path); return null; }
    archive.expand(*found);
    return (new Z8PoiXmlReader(z8_poi_xml_decode(cast(const(ubyte)[])(*found).expandedData))).parseDocument();
}
private string z8_poi_resolve_path(string base, wstring target) {
    import std.string : split;
    auto text = target.to!string;
    if (!text.length || text.canFind('\\') || text.canFind(':')) throw new Exception("invalid XLSX target path");
    auto full = text[0] == '/' ? text[1 .. $] : base ~ text;
    string[] parts;
    foreach (part; full.split("/")) {
        if (!part.length || part == ".") continue;
        if (part == "..") { if (!parts.length) throw new Exception("invalid XLSX target path"); parts = parts[0 .. $ - 1]; }
        else parts ~= part;
    }
    import std.array : join;
    return parts.join("/");
}
private long z8_poi_cell_column(wstring reference, long fallback) {
    if (!reference.length) return fallback;
    long column; size_t i;
    while (i < reference.length && reference[i] >= 'A' && reference[i] <= 'Z') {
        column = column * 26 + reference[i++] - 'A' + 1;
        if (column > 16384) throw new Exception("invalid XLSX cell column");
    }
    if (!i || i == reference.length) throw new Exception("invalid XLSX cell address");
    return column - 1;
}
private Z8PoiWorkbook z8_poi_parse(Z8Binary input) {
    import std.zip : ZipArchive;
    import std.algorithm.searching : endsWith;
    import std.string : lastIndexOf;
    auto bytes = z8_binary_read_all(z8_nonnull(input));
    IZ8PoiProvider provider;
    synchronized (z8_poi_monitor) provider = z8_poi_provider;
    Z8PoiXmlNode root, sharedStrings;
    Z8PoiXmlNode[] worksheets;
    bool usedProvider;
    if (provider !is null) {
        try {
            auto document = provider.open(bytes);
            if (document is null) throw new Exception("XLSX provider returned no document");
            scope(exit) document.close();
            Z8PoiXmlNode read(string xml) {
                try return (new Z8PoiXmlReader(z8_poi_xml_decode(cast(const(ubyte)[])xml))).parseDocument();
                catch (Exception error) throw new Z8PoiUnsupportedWorkbook(error.msg);
            }
            root = read(document.workbookXml());
            auto sharedXml = document.sharedStringsXml();
            if (sharedXml.length) sharedStrings = read(sharedXml);
            foreach (i; 0 .. document.sheetCount()) worksheets ~= read(document.sheetXml(i));
            usedProvider = true;
        } catch (Z8PoiUnsupportedWorkbook) {
            root = null; sharedStrings = null; worksheets = null;
        }
    }
    if (!usedProvider) {
        auto archive = new ZipArchive(bytes);
        auto packageRels = z8_poi_read_xml(archive, "_rels/.rels");
        string workbookPath;
        foreach (rel; packageRels.children) if (z8_poi_xml_local(rel.name) == "Relationship"w &&
            rel.attr("TargetMode"w) != "External"w && rel.attr("Type"w).endsWith("/officeDocument"w))
            workbookPath = z8_poi_resolve_path("", rel.attr("Target"w));
        if (!workbookPath.length) throw new Exception("missing XLSX workbook relationship");
        auto slash = workbookPath.lastIndexOf('/');
        auto base = slash < 0 ? "" : workbookPath[0 .. slash + 1];
        auto filename = slash < 0 ? workbookPath : workbookPath[slash + 1 .. $];
        root = z8_poi_read_xml(archive, workbookPath);
        auto rels = z8_poi_read_xml(archive, base ~ "_rels/" ~ filename ~ ".rels");
        wstring[wstring] targets;
        wstring sharedTarget;
        foreach (rel; rels.children) if (z8_poi_xml_local(rel.name) == "Relationship"w && rel.attr("TargetMode"w) != "External"w) {
            auto kind = rel.attr("Type"w);
            if (kind.endsWith("/worksheet"w)) targets[rel.attr("Id"w)] = rel.attr("Target"w);
            else if (kind.endsWith("/sharedStrings"w)) sharedTarget = rel.attr("Target"w);
        }
        sharedStrings = sharedTarget is null ? null : z8_poi_read_xml(archive, z8_poi_resolve_path(base, sharedTarget));
        auto sheets = root.child("sheets"w);
        if (sheets is null) throw new Exception("missing XLSX sheets");
        foreach (item; sheets.children) if (z8_poi_xml_local(item.name) == "sheet"w) {
            auto target = item.attr("id"w) in targets;
            if (target is null) throw new Exception("missing XLSX sheet relationship");
            worksheets ~= z8_poi_read_xml(archive, z8_poi_resolve_path(base, *target));
        }
    }
    wstring[] strings;
    if (sharedStrings !is null) foreach (entry; sharedStrings.children) if (z8_poi_xml_local(entry.name) == "si"w) strings ~= z8_poi_rich_text(entry);
    auto result = new Z8PoiWorkbook();
    auto properties = root.child("workbookPr"w);
    if (properties !is null) result.date1904 = properties.attr("date1904"w) == "1"w || properties.attr("date1904"w) == "true"w;
    auto sheets = root.child("sheets"w);
    if (sheets is null) throw new Exception("missing XLSX sheets");
    size_t sheetIndex;
    foreach (item; sheets.children) if (z8_poi_xml_local(item.name) == "sheet"w) {
        if (sheetIndex >= worksheets.length) throw new Exception("XLSX provider omitted a sheet");
        auto worksheet = worksheets[sheetIndex++];
        auto data = worksheet.child("sheetData"w), sheet = new Z8PoiSheet();
        sheet.name = item.attr("name"w);
        auto merges = worksheet.child("mergeCells"w);
        sheet.hasMergedRegions = merges !is null && merges.children.length != 0;
        if (merges !is null) foreach (merge; merges.children)
            if (z8_poi_xml_local(merge.name) == "mergeCell"w) sheet.mergedReferences ~= merge.attr("ref"w);
        Z8PoiSharedFormula[wstring] sharedFormulas;
        long nextRow;
        if (data !is null) foreach (rowNode; data.children) if (z8_poi_xml_local(rowNode.name) == "row"w) {
            auto rowNumber = rowNode.attr("r"w);
            long rowIndex = rowNumber is null ? nextRow : rowNumber.to!long - 1;
            if (rowIndex < 0 || rowIndex >= 1048576 || rowIndex in sheet.rows) throw new Exception("invalid XLSX row index");
            nextRow = rowIndex + 1; auto row = new Z8PoiRow(); row.safeExpansionTarget = true;
            foreach (attribute, ignored; rowNode.attributes)
                if (attribute != "r"w && attribute != "spans"w) row.safeExpansionTarget = false;
            long nextColumn;
            foreach (cellNode; rowNode.children) if (z8_poi_xml_local(cellNode.name) == "c"w) {
                auto column = z8_poi_cell_column(cellNode.attr("r"w), nextColumn); nextColumn = column + 1;
                if (column in row.cells) throw new Exception("duplicate XLSX cell");
                auto cell = new Z8PoiCell(); cell.kind = cell.rawKind = cellNode.attr("t"w);
                cell.safeCopy = true;
                foreach (attribute, ignored; cellNode.attributes)
                    if (attribute != "r"w && attribute != "s"w && attribute != "t"w) cell.safeCopy = false;
                foreach (child; cellNode.children) {
                    auto name = z8_poi_xml_local(child.name);
                    if (name != "v"w && !(cell.rawKind == "inlineStr"w && name == "is"w) &&
                        !(name == "f"w && !child.attributes.length)) cell.safeCopy = false;
                }
                auto formula = cellNode.child("f"w); cell.hasFormula = formula !is null;
                if (formula !is null) {
                    cell.formula = formula.allText();
                    if (formula.attr("t"w) == "shared"w) {
                        cell.sharedId = formula.attr("si"w); cell.row = rowIndex; cell.column = column;
                        if (cell.sharedId is null) throw new Exception("missing XLSX shared formula index");
                        if (cell.formula.length) sharedFormulas[cell.sharedId] = Z8PoiSharedFormula(cell.formula, rowIndex, column);
                        else cell.unresolvedSharedFormula = true;
                    }
                }
                auto value = cellNode.child("v"w); if (value !is null) cell.value = value.allText();
                if (cell.kind == "inlineStr"w) { auto inlineValue = cellNode.child("is"w); cell.value = inlineValue is null ? z8_poi_decode_text(cell.value) : z8_poi_rich_text(inlineValue); }
                else if (cell.kind == "s"w) {
                    auto index = cell.value.to!long;
                    if (index < 0 || cast(size_t)index >= strings.length) throw new Exception("invalid XLSX shared string index");
                    cell.value = strings[cast(size_t)index]; cell.kind = "str"w;
                } else if (cell.kind == "str"w) cell.value = z8_poi_decode_text(cell.value);
                row.cells[column] = cell;
            }
            sheet.rows[rowIndex] = row;
        }
        foreach (row; sheet.rows) foreach (cell; row.cells) if (cell.unresolvedSharedFormula) {
            auto master = cell.sharedId in sharedFormulas;
            wstring expanded;
            if (master !is null && z8_poi_shift_formula(master.text, cell.row - master.row, cell.column - master.column, expanded)) {
                cell.formula = expanded; cell.unresolvedSharedFormula = false;
            }
        }
        result.sheets ~= sheet;
    }
    if (sheetIndex != worksheets.length) throw new Exception("XLSX provider returned extra sheets");
    return result;
}
private Z8PoiWorkbook z8_poi_workbook(Z8Object self) {
    self = z8_nonnull(self);
    auto book = cast(Z8PoiWorkbook)self.z8_get_native_state(z8_poi_workbook_name);
    if (book is null) throw new Exception("workbook is not open");
    return book;
}
private void z8_poi_open(Z8Object self, Z8File file) {
    self = z8_nonnull(self); self.z8_set_native_state(z8_poi_workbook_name, null);
    auto input = z8_file_to_binary(z8_nonnull(file));
    scope(exit) z8_binary_close(input);
    self.z8_set_native_state(z8_poi_workbook_name, z8_poi_parse(input));
}
private void z8_poi_close(Z8Object self) { z8_nonnull(self).z8_set_native_state(z8_poi_workbook_name, null); }
private Z8Optional!long z8_poi_sheet_count(Z8Object self) { return z8_optional_wrap(cast(long)z8_poi_workbook(self).sheets.length); }
private Z8Object z8_poi_sheet(Z8Object self, Z8Optional!long index) {
    auto book = z8_poi_workbook(self); auto n = cast(int)z8_optional_require(index);
    if (n < 0 || cast(size_t)n >= book.sheets.length) throw new Exception("XLSX sheet index out of bounds");
    auto result = z8_native_new(z8_poi_sheet_name); result.z8_set_container(self);
    auto state = new Z8PoiSheetRef(); state.value = book.sheets[cast(size_t)n];
    result.z8_set_native_state(z8_poi_sheet_name, state);
    // Cell date interpretation follows the workbook's date window.
    result.z8_set_native_state(z8_poi_workbook_name, book);
    return result;
}
private Z8PoiSheet z8_poi_sheet_state(Z8Object self) {
    auto state = cast(Z8PoiSheetRef)z8_nonnull(self).z8_get_native_state(z8_poi_sheet_name);
    if (state is null || state.value is null) throw new Exception("uninitialized XLSX sheet");
    return state.value;
}
private Z8Optional!long z8_poi_row_count(Z8Object self) { return z8_optional_wrap(cast(long)z8_poi_sheet_state(self).rows.length); }
private Z8Object z8_poi_row(Z8Object self, Z8Optional!long index) {
    auto n = cast(long)cast(int)z8_optional_require(index);
    auto found = n in z8_poi_sheet_state(self).rows;
    auto result = z8_native_new(z8_poi_row_name); result.z8_set_container(self);
    auto state = new Z8PoiRowRef(); if (found !is null) state.value = *found;
    result.z8_set_native_state(z8_poi_row_name, state);
    result.z8_set_native_state(z8_poi_workbook_name, self.z8_get_native_state(z8_poi_workbook_name)); return result;
}
private Z8PoiRow z8_poi_row_state(Z8Object self) {
    auto state = cast(Z8PoiRowRef)z8_nonnull(self).z8_get_native_state(z8_poi_row_name);
    if (state is null || state.value is null) throw new Exception("XLSX row is absent");
    return state.value;
}
private Z8Optional!long z8_poi_cell_count(Z8Object self) { return z8_optional_wrap(cast(long)z8_poi_row_state(self).cells.length); }
private Z8Object z8_poi_cell(Z8Object self, Z8Optional!long index) {
    auto n = cast(long)cast(int)z8_optional_require(index); if (n < 0) throw new Exception("XLSX cell index out of bounds");
    auto found = n in z8_poi_row_state(self).cells; if (found is null) return null;
    auto result = z8_native_new(z8_poi_cell_name); result.z8_set_container(self);
    auto state = new Z8PoiCellRef(); state.value = *found;
    auto book = cast(Z8PoiWorkbook)self.z8_get_native_state(z8_poi_workbook_name);
    state.date1904 = book !is null && book.date1904;
    result.z8_set_native_state(z8_poi_cell_name, state); return result;
}
private Z8PoiCellRef z8_poi_cell_state(Z8Object self) {
    auto state = cast(Z8PoiCellRef)z8_nonnull(self).z8_get_native_state(z8_poi_cell_name);
    if (state is null || state.value is null) throw new Exception("uninitialized XLSX cell"); return state;
}
private double z8_poi_number(Z8PoiCell cell) {
    if (cell.kind == "str"w || cell.kind == "inlineStr"w || cell.kind == "b"w || cell.kind == "e"w)
        throw new Exception("XLSX cell is not numeric");
    return cell.value.length ? z8_literal_parse_double(cell.value.to!string) : 0;
}
private long z8_poi_long(double value) {
    import std.math : isNaN;
    return isNaN(value) ? 0 : value >= cast(double)long.max ? long.max : value <= cast(double)long.min ? long.min : cast(long)value;
}
private wstring z8_poi_string(Z8Object self) {
    auto cell = z8_poi_cell_state(self).value;
    if (cell.hasFormula) {
        if (cell.unresolvedSharedFormula) throw new Exception("shared XLSX formula expansion is unsupported");
        return z8_string_new(cell.formula);
    }
    if (cell.kind == "str"w || cell.kind == "inlineStr"w) return z8_string_new(cell.value);
    if (cell.kind == "b"w) return cell.value == "1"w ? "true"w : "false"w;
    if (cell.kind == "e"w || !cell.value.length) return ""w;
    auto number = z8_poi_number(cell), integer = z8_poi_long(number);
    return z8_string_new(number == cast(double)integer ? integer.to!wstring : z8_java_double_string(number).to!wstring);
}
private Z8Optional!long z8_poi_int(Z8Object self) { return z8_optional_wrap(z8_poi_long(z8_poi_number(z8_poi_cell_state(self).value))); }
private Z8Optional!Z8Decimal z8_poi_decimal(Z8Object self) { return z8_optional_wrap(z8_math_result(z8_poi_number(z8_poi_cell_state(self).value))); }
private Z8Date z8_poi_date(Z8Object self) {
    import std.math : floor, isFinite;
    auto state = z8_poi_cell_state(self); auto number = z8_poi_number(state.value);
    if (!isFinite(number) || number < 0 || number > 2958465) throw new Exception("invalid XLSX date");
    auto days = cast(long)floor(number);
    auto milliseconds = cast(long)floor((number - days) * 86400000.0 + 0.5);
    if (milliseconds == 86400000) { ++days; milliseconds = 0; }
    auto base = z8_date_from_ticks(0);
    z8_date_set(base, state.date1904 ? 1904 : 1899, state.date1904 ? 1 : 12, state.date1904 ? 1 : 31, 0, 0, 0);
    auto day = z8_date_addDay(base, days - (!state.date1904 && days >= 61 ? 1 : 0));
    z8_date_set(day, z8_date_year(day), z8_date_month(day), z8_date_day(day), milliseconds / 3600000,
        milliseconds / 60000 % 60, milliseconds / 1000 % 60, milliseconds % 1000);
    return day;
}
