private enum z8_field_name = "org.zenframework.z8.server.base.table.value.Field"w;
enum Z8FieldAggregation { none, sum, max, min, average, count, array, distinct, concat, bitwiseOr, bitwiseAnd }
// A query adapter owns the cursor and supplies the decoded value for the current
// row. SQL NULL must be represented by the type's Z8 default plus wasNull=true.
interface IZ8FieldCursor {
    bool isClosed();
    bool isGrouped();
    // When asArray is true, return the grouped result as a boxed JSON string.
    Z8Primary read(Z8Object field, bool asArray, out bool wasNull);
}

Z8SqlFieldFormatOptions z8_sql_set_field_options(Z8SqlFieldFormatOptions value) {

}
private interface IZ8FieldAggregationCell {
    Z8FieldAggregation get();
    void put(Z8FieldAggregation value);
}
private class Z8FieldAggregationCell(T) : IZ8FieldAggregationCell {
    T value;
    Z8FieldAggregation get() {
        if (value is T.e_None) return Z8FieldAggregation.none;
        if (value is T.e_Sum) return Z8FieldAggregation.sum;
        if (value is T.e_Max) return Z8FieldAggregation.max;
        if (value is T.e_Min) return Z8FieldAggregation.min;
        if (value is T.e_Average) return Z8FieldAggregation.average;
        if (value is T.e_Count) return Z8FieldAggregation.count;
        if (value is T.e_Array) return Z8FieldAggregation.array;
        if (value is T.e_Distinct) return Z8FieldAggregation.distinct;
        if (value is T.e_Concat) return Z8FieldAggregation.concat;
        if (value is T.e_BitwiseOr) return Z8FieldAggregation.bitwiseOr;
        if (value is T.e_BitwiseAnd) return Z8FieldAggregation.bitwiseAnd;
        throw new Exception("invalid field aggregation");
    }
    void put(Z8FieldAggregation aggregation) {
        final switch (aggregation) {
            case Z8FieldAggregation.none: value = T.e_None; break;
            case Z8FieldAggregation.sum: value = T.e_Sum; break;
            case Z8FieldAggregation.max: value = T.e_Max; break;
            case Z8FieldAggregation.min: value = T.e_Min; break;
            case Z8FieldAggregation.average: value = T.e_Average; break;
            case Z8FieldAggregation.count: value = T.e_Count; break;
            case Z8FieldAggregation.array: value = T.e_Array; break;
            case Z8FieldAggregation.distinct: value = T.e_Distinct; break;
            case Z8FieldAggregation.concat: value = T.e_Concat; break;
            case Z8FieldAggregation.bitwiseOr: value = T.e_BitwiseOr; break;
            case Z8FieldAggregation.bitwiseAnd: value = T.e_BitwiseAnd; break;
        }
    }
}
private interface IZ8FieldSlot {
    Z8Primary get();
    void put(Z8Primary value);
}
private interface IZ8FieldSortCell { wstring text(); void put(wstring direction); }
private class Z8FieldSortCell(T) : IZ8FieldSortCell {
    T value;
    this() { value = T.e_Asc; }
    wstring text() {
        if (value is T.e_Asc) return "asc"w;
        if (value is T.e_Desc) return "desc"w;
        if (value is null) return "null"w;
        throw new Exception("invalid field sort direction");
    }
    void put(wstring direction) {
        if (direction == "asc"w) value = T.e_Asc;
        else if (direction == "desc"w) value = T.e_Desc;
        else throw new Exception("invalid field sort direction");
    }
}

private class Z8FieldOrderCell(T) : IZ8FieldOrderCell if (is(T : Z8Object)) {
    Z8ObjectArray!T value;
    this() { value = new Z8ObjectArray!T(); }
    IZ8ObjectArrayView view() { return value; }
}
private class Z8FieldSlot(T, Base = T) : IZ8FieldSlot {
    T item;
    Z8Primary get() { return z8_primary_box(item); }
    void put(Z8Primary value) { item = z8_primary_cast!Base(value); }
}
class Z8FieldState {
    private Z8Primary boxedValue;
    private IZ8FieldSlot slot;
    private IZ8FieldAggregationCell aggregationCell;
    private IZ8FieldSortCell sortCell;
    private wstring initialSort = "asc"w;
    private IZ8FieldOrderCell orderCell;
    private IZ8FieldOrderCell usedFieldsCell, columnsCell;
    private Z8Object[] nativeDependencies;
    private Z8Object registryNameClassId;
    private bool registryNameIsEntry;
    private Z8FieldAggregation initialAggregation;
    private Z8Primary originalValue;
    Z8SqlPrimary sqlExpression;
    bool resolvingExpression, writingMetadata;
    IZ8FieldCursor cursor;
    bool changed, wasNull = true;
    wstring format;
    Z8Optional!long width, length, precision, scale, srs;
    Z8Optional!bool indexed, unique, trigram, caseInsensitive, totals, html, singleLine;
    Z8Object valueFrom, valueFor, sequencer;
    Z8Fts fts;
    wstring sqlOwnerAlias, sqlColumnName, sqlExpressionAlias;
    this(Z8Primary initial = null, bool wasNull = true) {
        value = initial; this.wasNull = wasNull;
        width = z8_optional_wrap(0L); length = z8_optional_wrap(0L);
        precision = z8_optional_wrap(19L); scale = z8_optional_wrap(4L); srs = z8_optional_wrap(3857L);
        indexed = unique = trigram = caseInsensitive = html = singleLine = z8_optional_wrap(false);
        totals = z8_optional_wrap(true);
    }
    @property Z8Primary value() { return slot is null ? boxedValue : slot.get(); }
    @property void value(Z8Primary item) { if (slot is null) boxedValue = item; else slot.put(item); }
    @property Z8FieldAggregation aggregation() { return aggregationCell is null ? initialAggregation : aggregationCell.get(); }
    @property void aggregation(Z8FieldAggregation value) {
        if (aggregationCell is null) initialAggregation = value; else aggregationCell.put(value);
    }
    private Z8FieldSlot!(T, Base) useSlot(T, Base = T)() {
        auto typed = cast(Z8FieldSlot!(T, Base))slot;
        if (typed !is null) return typed;
        auto previous = value;
        typed = new Z8FieldSlot!(T, Base)(); typed.put(previous);
        slot = typed; boxedValue = null;
        return typed;
    }
}
private auto z8_field_metadata_address(string member)(Z8Object self) {
    return &__traits(getMember, z8_field_state(self), member);
}
private void z8_field_add_native_dependency(Z8Object self, Z8Object dependency) {
    auto state = z8_field_state(self);
    dependency = z8_nonnull(dependency);
    auto view = state.usedFieldsCell is null ? null : state.usedFieldsCell.view();
    if (view is null) state.nativeDependencies ~= dependency;
    else view.z8_erased_data().z8_add(dependency);
}
ref T z8_field_sort_ref(T)(Z8Object self) {
    auto state = z8_field_state(self);
    auto cell = cast(Z8FieldSortCell!T)state.sortCell;
    if (cell is null) {
        if (state.sortCell !is null) throw new Exception("incompatible sort direction enum");
        cell = new Z8FieldSortCell!T(); cell.put(state.initialSort); state.sortCell = cell;
    }
    return cell.value;
}
ref Z8ObjectArray!T z8_field_array_ref(T, string name)(Z8Object self) if (is(T : Z8Object)) {
    auto state = z8_field_state(self);
    IZ8FieldOrderCell* cellSlot;
    static if (name == "arrayOrderBy") cellSlot = &state.orderCell;
    else static if (name == "usedFields") cellSlot = &state.usedFieldsCell;
    else static if (name == "columns") cellSlot = &state.columnsCell;
    else static assert(0, "unknown field array");
    auto cell = cast(Z8FieldOrderCell!T)*cellSlot;
    if (cell is null) {
        if (*cellSlot !is null) throw new Exception("incompatible field array type");
        cell = new Z8FieldOrderCell!T(); *cellSlot = cell;
        static if (name == "usedFields") foreach (dependency; state.nativeDependencies) {
            auto item = cast(T)dependency;
            if (item is null) throw new Exception("incompatible native field dependency");
            cell.value.z8_add(item);
        }
    }
    return cell.value;
}
// Java Field.getUsedFields() preserves the declared order.
Z8Object[] z8_field_used_fields(Z8Object self) {
    auto state = z8_field_state(self);
    Z8Object[] result;
    void add(Z8Object item) {
        foreach (existing; result) if (existing is item) return;
        result ~= item;
    }
    auto used = state.usedFieldsCell is null ? null : state.usedFieldsCell.view();
    if (state.usedFieldsCell is null) {
        foreach (dependency; state.nativeDependencies) add(z8_nonnull(dependency));
    } else if (used !is null) {
        auto items = used.z8_erased_data();
        for (long i; i < items.z8_size(); ++i) add(z8_nonnull(cast(Z8Object)items.z8_get(i)));
    }
    return result;
}
// Java Field.fields() adds the field, usedFields and valueFrom without duplicates.
Z8Object[] z8_field_fields(Z8Object self) {
    auto state = z8_field_state(self);
    Z8Object[] result = [self];
    foreach (item; z8_field_used_fields(self)) {
        bool present; foreach (existing; result) if (existing is item) { present = true; break; }
        if (!present) result ~= item;
    }
    if (state.valueFrom !is null) {
        bool present; foreach (existing; result) if (existing is state.valueFrom) { present = true; break; }
        if (!present) result ~= state.valueFrom;
    }
    return result;
}
// SQL planners also need dependencies of computed fields and ordered arrays.
Z8Object[] z8_field_query_dependencies(Z8Object self) {
    auto result = z8_field_fields(self);
    auto state = z8_field_state(self);
    void add(Z8Object field) {
        if (field is null) return;
        foreach (existing; result) if (existing is field) return;
        result ~= field;
    }
    if (z8_native_is_instance("org.zenframework.z8.server.base.table.value.Expression"w, self)) {
        if (state.resolvingExpression) throw new Exception("recursive SQL field expression");
        state.resolvingExpression = true;
        scope(exit) state.resolvingExpression = false;
        if (state.sqlExpression is null) state.sqlExpression = z8_nonnull(self.z8_field_virtual_expression());
        foreach (field; z8_sql_collect_fields(state.sqlExpression)) add(field);
    }
    auto order = state.orderCell is null ? null : state.orderCell.view();
    if (order !is null) {
        auto fields = order.z8_erased_data();
        for (long i; i < fields.z8_size(); ++i) add(cast(Z8Object)fields.z8_get(i));
    }
    return result;
}
ref T z8_field_aggregation_ref(T)(Z8Object self) {
    auto state = z8_field_state(self);
    auto cell = cast(Z8FieldAggregationCell!T)state.aggregationCell;
    if (cell is null) {
        if (state.aggregationCell !is null) throw new Exception("incompatible aggregation enum");
        cell = new Z8FieldAggregationCell!T(); cell.put(state.initialAggregation); state.aggregationCell = cell;
    }
    return cell.value;
}

// FieldType is a generated BL enum, so its concrete type is supplied by the
// generated Field method while the runtime determines the native field kind.
private wstring z8_field_type_name(Z8Object self) {
    self = z8_nonnull(self);
    enum prefix = "org.zenframework.z8.server.base.table.value."w;
    if (z8_native_is_instance(prefix ~ "AttachmentField"w, self)) return "attachments"w;
    if (z8_native_is_instance(prefix ~ "FileField"w, self) || z8_native_is_instance(prefix ~ "FileExpression"w, self)) return "file"w;
    if (z8_native_is_instance(prefix ~ "DateField"w, self) || z8_native_is_instance(prefix ~ "DateExpression"w, self)) return "date"w;
    if (z8_native_is_instance(prefix ~ "DatetimeField"w, self) || z8_native_is_instance(prefix ~ "DatetimeExpression"w, self)) return "datetime"w;
    if (z8_native_is_instance(prefix ~ "DatespanField"w, self) || z8_native_is_instance(prefix ~ "DatespanExpression"w, self)) return "datespan"w;
    if (z8_native_is_instance(prefix ~ "IntegerField"w, self) || z8_native_is_instance(prefix ~ "IntegerExpression"w, self)) return "int"w;
    if (z8_native_is_instance(prefix ~ "StringField"w, self) || z8_native_is_instance(prefix ~ "StringExpression"w, self) ||
        z8_native_is_instance(prefix ~ "JsonObjectExpression"w, self)) return "string"w;
    if (z8_native_is_instance(prefix ~ "TextField"w, self) || z8_native_is_instance(prefix ~ "TextExpression"w, self)) return "text"w;
    if (z8_native_is_instance(prefix ~ "BoolField"w, self) || z8_native_is_instance(prefix ~ "BoolExpression"w, self)) return "boolean"w;
    if (z8_native_is_instance(prefix ~ "DecimalField"w, self) || z8_native_is_instance(prefix ~ "DecimalExpression"w, self)) return "float"w;
    if (z8_native_is_instance(prefix ~ "GuidField"w, self) || z8_native_is_instance(prefix ~ "GuidExpression"w, self)) return "guid"w;
    if (z8_native_is_instance(prefix ~ "BinaryField"w, self)) return "binary"w;
    if (z8_native_is_instance(prefix ~ "GeometryField"w, self) || z8_native_is_instance(prefix ~ "GeometryExpression"w, self)) return "geometry"w;
    if (z8_native_is_instance(prefix ~ "Expression"w, self)) return "none"w;
    throw new Exception("unknown native Field type");
}

private Z8FieldState z8_field_state(Z8Object self) {
    self = z8_nonnull(self);
    if (!z8_native_is_instance(z8_field_name, self)) throw new Exception("invalid Field object");
    auto state = cast(Z8FieldState)self.z8_get_native_state(z8_field_name);
    if (state is null) { state = new Z8FieldState(); self.z8_set_native_state(z8_field_name, state); }
    return state;
}
void z8_field_attach(Z8Object self, Z8FieldState state) {
    if (!z8_native_is_instance(z8_field_name, self)) throw new Exception("invalid Field object");
    self.z8_set_native_state(z8_field_name, z8_nonnull(state));
}
void z8_field_set(Z8Object self, Z8Primary value) {
    auto state = z8_field_state(self);
    if (!state.changed) { state.originalValue = state.value; state.changed = true; }
    state.value = value;
}

private bool z8_field_is_array(Z8FieldState state) {
    return state.cursor !is null && state.cursor.isGrouped() &&
        (state.aggregation == Z8FieldAggregation.array || state.aggregation == Z8FieldAggregation.distinct);
}
private Z8Primary z8_field_internal_get(Z8Object self) {
    auto state = z8_field_state(self);
    if (state.changed || state.cursor is null || state.cursor.isClosed()) return state.value;
    bool wasNull;
    auto value = state.cursor.read(self, z8_field_is_array(state), wasNull);
    state.wasNull = wasNull;
    return value;
}
private Z8Primary z8_field_current(Z8Object self) {
    auto state = z8_field_state(self);
    if (state.registryNameClassId !is null) {
        // Entries/Requests NameField performs the normal read to maintain
        // wasNull, then resolves the current display name from the registry.
        z8_field_internal_get(self);
        auto classId = z8_primary_cast!wstring(z8_field_current(state.registryNameClassId));
        auto registry = z8_runtime_get_registry();
        if (registry is null) throw new Exception("runtime registry is not configured");
        auto candidates = state.registryNameIsEntry ? registry.entries() : registry.requests();
        foreach (candidate; candidates) if (candidate.classId == classId)
            return z8_primary_box(z8_string_new(candidate.displayName));
        if (state.registryNameIsEntry) return z8_primary_box("<entry removed>"w);
        throw new Exception("request is not registered: " ~ z8_string_or_empty(classId).to!string);
    }
    if (z8_field_is_array(state)) {
        enum prefix = "org.zenframework.z8.server.base.table.value."w;
        bool isString = z8_native_is_instance(prefix ~ "StringField"w, self) ||
            z8_native_is_instance(prefix ~ "TextField"w, self) || z8_native_is_instance(prefix ~ "StringExpression"w, self) ||
            z8_native_is_instance(prefix ~ "TextExpression"w, self) || z8_native_is_instance(prefix ~ "FileExpression"w, self);
        if (!isString) return null;
    }
    return z8_field_internal_get(self);
}
// Mirrors the alias information normally supplied by Query and FormatOptions.
void z8_field_bind_sql(Z8Object self, wstring ownerAlias, wstring columnName, wstring expressionAlias = null) {
    auto state = z8_field_state(self);
    state.sqlOwnerAlias = ownerAlias; state.sqlColumnName = columnName; state.sqlExpressionAlias = expressionAlias;
}
wstring z8_field_sql_format(Z8Object self, Z8SqlVendor vendor, bool booleanLogical = false) {
    import std.string : replace;
    auto state = z8_field_state(self);
    wstring expression;
    if (state.sqlExpressionAlias !is null) expression = state.sqlExpressionAlias;
    else if (z8_native_is_instance("org.zenframework.z8.server.base.table.value.Expression"w, self)) {
        if (state.resolvingExpression) throw new Exception("recursive SQL field expression");
        state.resolvingExpression = true;
        scope(exit) state.resolvingExpression = false;
        auto previous = z8_sql_field_options;
        z8_sql_field_options.aggregation = false;
        scope(exit) z8_sql_field_options = previous;
        if (state.sqlExpression is null) state.sqlExpression = z8_nonnull(self.z8_field_virtual_expression());
        expression = z8_sql_functions_format(state.sqlExpression, vendor);
    } else {
        if (state.sqlOwnerAlias is null) throw new Exception("SQL field owner alias is not configured");
        auto column = state.sqlColumnName is null ? z8_object_name(self) : state.sqlColumnName;
        expression = vendor == Z8SqlVendor.SqlServer ? state.sqlOwnerAlias ~ ".["w ~ column.replace("]"w, "]]"w) ~ "]"w :
            state.sqlOwnerAlias ~ ".\""w ~ column.replace("\""w, "\"\""w) ~ "\""w;
    }
    if (booleanLogical) expression ~= "=1"w;
    if (!z8_sql_field_options.aggregation) return expression;
    auto aggregation = state.aggregation;
    if (aggregation == Z8FieldAggregation.none || aggregation == Z8FieldAggregation.bitwiseOr || aggregation == Z8FieldAggregation.bitwiseAnd)
        return expression;
    auto isGuid = z8_native_is_instance("org.zenframework.z8.server.base.table.value.GuidField"w, self) ||
        z8_native_is_instance("org.zenframework.z8.server.base.table.value.GuidExpression"w, self);
    auto isGeometry = z8_native_is_instance("org.zenframework.z8.server.base.table.value.GeometryField"w, self) ||
        z8_native_is_instance("org.zenframework.z8.server.base.table.value.GeometryExpression"w, self);
    auto isText = z8_native_is_instance("org.zenframework.z8.server.base.table.value.TextField"w, self) ||
        z8_native_is_instance("org.zenframework.z8.server.base.table.value.TextExpression"w, self);
    auto isFile = z8_native_is_instance("org.zenframework.z8.server.base.table.value.FileField"w, self) ||
        z8_native_is_instance("org.zenframework.z8.server.base.table.value.FileExpression"w, self);
    if (aggregation == Z8FieldAggregation.array || aggregation == Z8FieldAggregation.distinct) {
        auto order = state.orderCell is null ? null : state.orderCell.view();
        if (order !is null && order.z8_erased_data().z8_size()) {
            if (isText && !isFile) {
                if (vendor == Z8SqlVendor.Postgres) expression = "CONVERT_FROM("w ~ expression ~ ", 'UTF8')"w;
                else if (vendor == Z8SqlVendor.Oracle) expression = "UTL_RAW.CAST_TO_NVARCHAR2("w ~ expression ~ ")"w;
                else if (vendor == Z8SqlVendor.SqlServer) expression = "Cast("w ~ expression ~ " as nvarchar(max))"w;
            }
            auto baseExpression = expression;
            auto distinct = aggregation == Z8FieldAggregation.distinct;
            if (distinct) {
                if (!z8_sql_field_options.orderBy && vendor == Z8SqlVendor.SqlServer)
                    expression = "DISTINCT '\"' + "w ~ expression ~ " + '\"'"w;
                else expression = "DISTINCT "w ~ expression;
            }
            wstring orderFields;
            auto previous = z8_sql_field_options;
            z8_sql_field_options.aggregation = false;
            scope(exit) z8_sql_field_options = previous;
            auto fields = order.z8_erased_data();
            for (long i; i < fields.z8_size(); ++i) {
                auto field = cast(Z8Object)fields.z8_get(i);
                auto item = z8_field_sql_format(z8_nonnull(field), vendor);
                if (distinct && item != baseExpression) throw new Exception("distinct aggregate order field must match expression");
                if (i) orderFields ~= ", "w;
                auto orderState = z8_field_state(field);
                orderFields ~= item ~ " "w ~ (orderState.sortCell is null ? orderState.initialSort : orderState.sortCell.text());
            }
            auto clause = " ORDER BY "w ~ orderFields;
            if (z8_sql_field_options.orderBy) {
                if (vendor == Z8SqlVendor.Postgres) return "array_agg("w ~ expression ~ clause ~ ")"w;
                return (vendor == Z8SqlVendor.SqlServer ? "STRING_AGG("w : "LISTAGG("w) ~ expression ~ ", ',') WITHIN GROUP (ORDER BY "w ~ orderFields ~ ")"w;
            }
            if (vendor == Z8SqlVendor.Postgres) return "json_agg("w ~ expression ~ clause ~ ")"w;
            if (vendor == Z8SqlVendor.Oracle || vendor == Z8SqlVendor.H2) return "JSON_ARRAYAGG("w ~ expression ~ clause ~ ")"w;
            if (vendor == Z8SqlVendor.SqlServer) {
                if (!distinct) expression = "'\"' + "w ~ expression ~ " + '\"'"w;
                return "'[' + STRING_AGG("w ~ expression ~ ", ',') WITHIN GROUP (ORDER BY "w ~ orderFields ~ ") + ']'"w;
            }
            throw new Exception("unsupported SQL database vendor");
        }
        if (vendor == Z8SqlVendor.Oracle) return "collect("w ~ expression ~ ")"w;
        if (vendor != Z8SqlVendor.Postgres) throw new Exception("unsupported SQL database vendor");
        if (isText && !isFile) expression = "CONVERT_FROM("w ~ expression ~ ", 'UTF8')"w;
        if (aggregation == Z8FieldAggregation.distinct) expression = "DISTINCT "w ~ expression;
        return (z8_sql_field_options.orderBy ? "array_agg("w : "json_agg("w) ~ expression ~ ")"w;
    }
    if (aggregation == Z8FieldAggregation.concat) {
        if (vendor != Z8SqlVendor.Postgres) throw new Exception("unsupported SQL database vendor");
        return "string_agg("w ~ expression ~ ",',')"w;
    }
    if (vendor == Z8SqlVendor.Postgres || vendor == Z8SqlVendor.Oracle) {
        if (isGuid) expression = vendor == Z8SqlVendor.Postgres ? expression ~ "::text"w : "RAWTOHEX("w ~ expression ~ ")"w;
        else if (isText) expression = vendor == Z8SqlVendor.Postgres ? "CONVERT_FROM("w ~ expression ~ ", 'UTF8')"w :
            "UTL_RAW.CAST_TO_NVARCHAR2("w ~ expression ~ ")"w;
    }
    wstring functionName;
    final switch (aggregation) {
        case Z8FieldAggregation.sum: functionName = isGeometry ? "ST_Union"w : "sum"w; break;
        case Z8FieldAggregation.max: functionName = "max"w; break;
        case Z8FieldAggregation.min: functionName = "min"w; break;
        case Z8FieldAggregation.average: functionName = "avg"w; break;
        case Z8FieldAggregation.count: functionName = "count"w; break;
        case Z8FieldAggregation.none: case Z8FieldAggregation.array: case Z8FieldAggregation.distinct:
        case Z8FieldAggregation.concat: case Z8FieldAggregation.bitwiseOr: case Z8FieldAggregation.bitwiseAnd:
            assert(0, "aggregation already handled");
    }
    expression = functionName ~ "("w ~ expression ~ ")"w;
    if (isGuid && vendor == Z8SqlVendor.Postgres) expression ~= "::uuid"w;
    else if (isGuid && vendor == Z8SqlVendor.Oracle) expression = "HEXTORAW("w ~ expression ~ ")"w;
    return expression;
}
void z8_field_reset(Z8Object self) {
    auto state = z8_field_state(self);
    if (state.changed) { state.value = state.originalValue; state.changed = false; }
}
private Z8Primary z8_field_primary(Z8Object self) { return z8_nonnull(self).z8_field_virtual_get(); }
private auto z8_field_cast(T)(Z8Object self) { return z8_primary_cast!T(z8_field_primary(self)); }
private Z8Optional!bool z8_field_changed(Z8Object self) { return z8_optional_wrap(z8_field_state(self).changed); }
private Z8Optional!bool z8_field_was_null(Z8Object self) {
    if (z8_native_is_instance("org.zenframework.z8.server.base.table.value.Expression"w, z8_nonnull(self)))
        return z8_optional_wrap(false);
    return z8_optional_wrap(z8_field_state(self).wasNull);
}
private wstring z8_field_string_value(Z8Object self) {
    return z8_string_new(z8_primary_to_string(z8_nonnull(z8_field_primary(self))));
}
private Z8SqlBool z8_field_is_null(Z8Object self) {
    auto result = new Z8SqlBool(Z8SqlBoolKind.fieldIsNull); result.field = z8_nonnull(self); return result;
}
private Z8SqlBool z8_field_in_vector(Z8Object self, Z8Array!Z8Primary values) {
    auto result = new Z8SqlBool(Z8SqlBoolKind.inVector);
    result.vectorValue = new Z8SqlFieldToken(z8_nonnull(self));
    values = z8_nonnull(values);
    for (long i; i < values.z8_size(); ++i) result.vectorItems ~= new Z8SqlBoxedConstant(values.z8_get(i));
    return result;
}
private Z8SqlBool z8_field_not_in_vector(Z8Object self, Z8Array!Z8Primary values) {
    return z8_sql_bool_not(z8_field_in_vector(self, values));
}
