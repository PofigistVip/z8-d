private Z8ControlState z8_control_state(Z8Object self) {
    self = z8_nonnull(self);
    if (!z8_native_is_instance(z8_control_name, self)) throw new Exception("invalid Control object");
    auto state = cast(Z8ControlState)self.z8_get_native_state(z8_control_name);
    if (state is null) {
        state = new Z8ControlState();
        if (z8_native_is_instance(z8_listbox_name, self)) state.editable = z8_optional_wrap(true);
        self.z8_set_native_state(z8_control_name, state);
    }
    return state;
}

        state.sourceInitialized = true;

    return &state.source;

// Query.fields() expands sections and tabs into the fields needed by a read.
Z8Object[] z8_control_fields(Z8Object control) {
    Z8Object[] result, path;
    void add(Z8Object field) {
        foreach (existing; result) if (existing is field) return;
        result ~= field;
    }
    void visit(Z8Object item) {
        item = z8_nonnull(item);
        if (z8_native_is_instance(z8_field_name, item)) { foreach (field; z8_field_fields(item)) add(field); return; }
        if (z8_native_is_instance(z8_action_name, item) || z8_native_is_instance(z8_action_group_name, item) ||
            z8_native_is_instance(z8_listbox_name, item)) return;
        foreach (parent; path) if (parent is item) throw new Exception("cyclic section controls");
        path ~= item;
        scope(exit) path.length = path.length - 1;
        IZ8FieldOrderCell cell;
        if (z8_native_is_instance(z8_tab_control_name, item)) cell = z8_tab_control_state(item).tabs;
        else if (z8_native_is_instance(z8_section_name, item)) cell = z8_section_state(item).controls;
        else throw new Exception("unsupported control fields");
        if (cell !is null && cell.view() !is null) {
            auto items = cell.view().z8_erased_data();
            for (long i; i < items.z8_size(); ++i) visit(cast(Z8Object)items.z8_get(i));
        }
        if (z8_native_is_instance("org.zenframework.z8.server.base.form.Form"w, item)) {
            auto dependency = z8_control_state(item).dependency;
            if (dependency !is null) add(z8_nonnull(z8_query_primary_key(z8_nonnull(dependency.z8_get_owner()))));
        }
    }
    visit(control);
    return result;
}
Z8Object[] z8_query_form_fields(Z8Object query) {
    Z8Object[] result;
    void add(Z8Object field) { foreach (existing; result) if (existing is field) return; result ~= field; }
    foreach (control; z8_query_array(query, "controls"w))
        foreach (field; z8_control_fields(control)) add(field);
    foreach (name; ["names"w, "columns"w, "extraFields"w])
        foreach (field; z8_query_array(query, name)) add(z8_nonnull(field));
    return result;
}

}

// Writes an override into the current request's open JSON object.
void z8_control_write_custom_meta(Z8Object self, Z8Object query, Z8Object context) {
    if (!z8_native_is_instance(z8_control_name, z8_nonnull(self))) throw new Exception("invalid Control object");

private wstring z8_control_aggregation_name(Z8FieldAggregation value) {
    final switch (value) {
        case Z8FieldAggregation.none: return "none"w;
        case Z8FieldAggregation.sum: return "sum"w;
        case Z8FieldAggregation.min: return "min"w;
        case Z8FieldAggregation.max: return "max"w;
        case Z8FieldAggregation.average: return "average"w;
        case Z8FieldAggregation.count: return "count"w;
        case Z8FieldAggregation.array: return "array"w;
        case Z8FieldAggregation.distinct: return "distinct"w;
        case Z8FieldAggregation.concat: return "concat"w;
        case Z8FieldAggregation.bitwiseOr: return "bitwise_or"w;
        case Z8FieldAggregation.bitwiseAnd: return "bitwise_and"w;
    }
}
// Field.writeMeta for direct fields and authorized Link/Join paths.
void z8_control_write_field_meta(Z8Object field, Z8Object query, Z8Object context, Z8Privileges privileges) {
    field = z8_nonnull(field); query = z8_nonnull(query); privileges = z8_nonnull(privileges);
    if (!z8_native_is_instance(z8_field_name, field)) throw new Exception("metadata control is not a Field");
    auto path = z8_application_query_field_path(query, field, privileges);
    if (!z8_application_query_field_readable(query, field, privileges)) throw new Exception("field metadata access denied");
    auto request = z8_application_require_request();
    auto writer = request.writer;
    if (writer.scopes.length < 2) throw new Exception("metadata response object is not open");
    auto fieldState = z8_field_state(field), control = z8_control_state(field);
    if (fieldState.writingMetadata) return;
    fieldState.writingMetadata = true;
    scope(exit) fieldState.writingMetadata = false;
    bool isLink = z8_native_is_instance("org.zenframework.z8.server.base.table.value.Link"w, field);
    Z8Object linkTarget;
    if (isLink) {
        linkTarget = z8_link_query(field);
        auto parent = field.z8_get_attribute("parentKey"w);
        if (linkTarget is null && parent !is null && z8_str_compareNoCase(parent, "false"w) != 0) linkTarget = field.z8_get_owner();
        if (linkTarget is null) throw new Exception("Link target is not configured");
        if (z8_native_is_instance("org.zenframework.z8.server.base.table.Table"w, linkTarget)) {
            auto linkedKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, linkTarget);
            if (!privileges.getTableAccess(linkedKey).read) throw new Exception("link target access denied");
        }
        z8_monitor_property(writer, "isLink"w, true);
        z8_monitor_property(writer, "owner"w, z8_nonnull(field.z8_get_owner()).z8_id());
    }
    auto type = z8_field_type_id(field);
    if (type == "text"w && fieldState.singleLine.present && fieldState.singleLine.value) type = "string"w;
    z8_monitor_property(writer, "type"w, type);
    z8_monitor_property(writer, "format"w, fieldState.format is null ? ""w : fieldState.format);
    z8_monitor_property(writer, "length"w, fieldState.length.present ? fieldState.length.value : 0L);
    z8_monitor_property(writer, "width"w, fieldState.width.present ? fieldState.width.value : 0L);
    auto aggregation = z8_field_aggregation(field);
    if (aggregation != Z8FieldAggregation.none) z8_monitor_property(writer, "aggregation"w, z8_control_aggregation_name(aggregation));
    bool hasJoin;
    foreach (link; path) if (z8_native_is_instance("org.zenframework.z8.server.base.table.value.Join"w, link)) hasJoin = true;
    if (path.length && !hasJoin) {
        auto first = path[0], last = path[$ - 1], target = z8_nonnull(z8_link_query(first));
        auto primary = z8_nonnull(z8_query_primary_key(target));
        z8_monitor_property(writer, "isCombobox"w, true);
        if (fieldState.columnsCell !is null && fieldState.columnsCell.view() !is null) {
            auto columns = fieldState.columnsCell.view().z8_erased_data();
            Z8Object[] visible;
            for (long i; i < columns.z8_size(); ++i) {
                auto column = z8_nonnull(cast(Z8Object)columns.z8_get(i));
                if (z8_control_metadata_visible(column, query, privileges)) visible ~= column;
            }
            if (visible.length) {
                writer.start('[', "columns"w);
                foreach (column; visible) { writer.start('{'); z8_control_write_meta(column, query, context, privileges); writer.finish('}'); }
                writer.finish(']');
            }
        }
        Z8Object parentKey, lockKey;
        foreach (candidate; z8_query_declared_fields(target)) {
            auto parent = candidate.z8_get_attribute("parentKey"w);
            if (parent !is null && z8_str_compareNoCase(parent, "false"w) != 0) parentKey = candidate;
            if (candidate.z8_get_attribute("lockKey"w) !is null) lockKey = candidate;
        }
        Z8Object[] sort;
        foreach (candidate; z8_query_array(target, "sortFields"w))
            if (z8_application_query_field_readable(query, candidate, privileges)) sort ~= candidate;
        writer.start('{', "query"w);
        z8_monitor_property(writer, "request"w, z8_nonnull(context).z8_class_id());
        z8_monitor_property(writer, "name"w, target.z8_id());
        z8_monitor_property(writer, "primaryKey"w, primary.z8_id());
        z8_monitor_property(writer, "parentKey"w, parentKey is null ? null : parentKey.z8_id());
        z8_monitor_property(writer, "lockKey"w, lockKey is null ? null : lockKey.z8_id());
        z8_application_query_write_order(sort, null); writer.finish('}');
        writer.start('{', "link"w);
        z8_monitor_property(writer, "name"w, first.z8_id());
        z8_monitor_property(writer, "owner"w, last.z8_id());
        z8_monitor_property(writer, "isJoined"w, false);
        z8_monitor_property(writer, "primaryKey"w, primary.z8_id());
        auto parent = first.z8_get_attribute("parentKey"w);
        if (parent !is null && z8_str_compareNoCase(parent, "false"w) != 0) {
            z8_monitor_property(writer, "isParentKey"w, true); writer.start('[', "parentKeys"w); writer.finish(']');
        }
        writer.finish('}');
        z8_application_query_write_order(sort, null);
    } else {
        z8_monitor_property(writer, "isText"w, true);
        if (field is z8_query_primary_key(query)) z8_monitor_property(writer, "isPrimaryKey"w, true);
    }
    if (fieldState.valueFrom !is null) z8_monitor_property(writer, "valueFrom"w, fieldState.valueFrom.z8_id());
    if (fieldState.valueFor !is null) z8_monitor_property(writer, "valueFor"w, fieldState.valueFor.z8_id());
    auto accessField = path.length && !hasJoin ? path[0] : field;
    auto accessOwner = z8_nonnull(accessField.z8_get_owner());
    auto queryKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, accessOwner);
    auto fieldKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, accessField);
    bool readOnly = hasJoin || (control.readOnly.present && control.readOnly.value) ||
        (!path.length && z8_native_is_instance("org.zenframework.z8.server.base.table.value.Expression"w, field)) ||
        !privileges.getFieldAccess(queryKey, fieldKey).write;
    bool required = control.required.present && control.required.value;
    foreach (link; path) {
        auto linkState = z8_control_state(link);
        readOnly = readOnly || (linkState.readOnly.present && linkState.readOnly.value);
        required = required || (linkState.required.present && linkState.required.value);
    }
    auto previousReadOnly = control.readOnly, previousRequired = control.required;
    control.readOnly = z8_optional_wrap(readOnly);
    control.required = z8_optional_wrap(!readOnly && required);
    scope(exit) { control.readOnly = previousReadOnly; control.required = previousRequired; }
    z8_control_write_base_meta(field, query, context, privileges);
    if (isLink) {
        auto primary = z8_nonnull(z8_query_primary_key(linkTarget));
        writer.start('{', "query"w);
        z8_monitor_property(writer, "name"w, linkTarget.z8_id());
        z8_monitor_property(writer, "primaryKey"w, primary.z8_id());
        z8_monitor_property(writer, "text"w, linkTarget.z8_get_attribute("displayName"w));
        z8_monitor_property(writer, "icon"w, linkTarget.z8_get_attribute("icon"w));
        writer.finish('}');
    }
}
private void z8_control_write_base_meta(Z8Object item, Z8Object query, Z8Object context, Z8Privileges privileges) {
    auto writer = z8_application_require_request().writer;
    auto control = z8_control_state(item);
    z8_monitor_property(writer, "name"w, item.z8_id());
    z8_monitor_property(writer, "ui"w, item.z8_get_attribute("ui"w));
    auto title = item.z8_get_attribute("displayName"w);
    if (title is null && z8_native_is_instance(z8_field_name, item)) title = z8_object_name(item);
    if (title !is null && !title.length && z8_native_is_instance(z8_field_name, item)) title = item.z8_id();
    z8_monitor_property(writer, "header"w, title);
    z8_monitor_property(writer, "columnHeader"w, item.z8_get_attribute("columnHeader"w));
    z8_monitor_property(writer, "description"w, item.z8_get_attribute("description"w));
    z8_monitor_property(writer, "icon"w, item.z8_get_attribute("icon"w));
    z8_monitor_property(writer, "height"w, control.height);
    z8_monitor_property(writer, "colSpan"w, control.colSpan);
    z8_monitor_property(writer, "flex"w, control.flex);
    writer.start('{', "label"w); z8_monitor_property(writer, "align"w, item.z8_get_attribute("labelAlign"w)); writer.finish('}');
    z8_monitor_property(writer, "readOnly"w, control.readOnly.present && control.readOnly.value);
    z8_monitor_property(writer, "required"w, control.required.present && control.required.value);
    z8_monitor_property(writer, "editable"w, control.editable);
    z8_monitor_property(writer, "important"w, control.important);
    if (control.sourceInitialized && control.source !is null) {
        auto source = z8_source_state(control.source);
        if (source.query !is null && source.query.z8_class_id() !is null &&
            privileges.getRequestAccess(z8_guid_create_name(source.query.z8_class_id())).execute) {
            writer.start('{', "source"w);
            z8_monitor_property(writer, "request"w, source.query.z8_class_id());
            z8_monitor_property(writer, "text"w, source.query.z8_get_attribute("displayName"w));
            if (source.filterField !is null) {
                auto id = source.filterField.z8_id(), prefix = source.query.z8_id();
                if (prefix.length && id.length > prefix.length && id[0 .. prefix.length] == prefix && id[prefix.length] == '.') id = id[prefix.length + 1 .. $];
                z8_monitor_property(writer, "field"w, id);
            }
            if (source.valueFrom !is null) z8_monitor_property(writer, "link"w, source.valueFrom.z8_id());
            writer.finish('}');
        }
    }
    if (control.dependency !is null) z8_monitor_property(writer, "dependency"w, control.dependency.z8_id());
    if (control.dependsOn !is null) z8_monitor_property(writer, "dependsOn"w, control.dependsOn.z8_id());
    if (control.dependencies !is null && control.dependencies.view() !is null) {
        auto items = control.dependencies.view().z8_erased_data();
        if (items.z8_size()) {
            writer.start('[', "dependencies"w);
            for (long i; i < items.z8_size(); ++i) writer.raw(z8_json_quote(z8_nonnull(cast(Z8Object)items.z8_get(i)).z8_id()));
            writer.finish(']');
        }
    }
    z8_control_write_custom_meta(item, query, context);
}
private bool z8_control_metadata_visible(Z8Object item, Z8Object query, Z8Privileges privileges) {
    item = z8_nonnull(item);
    if (z8_native_is_instance(z8_field_name, item)) {
        z8_application_query_field_path(query, item, privileges);
        return z8_application_query_field_readable(query, item, privileges);
    }
    if (z8_native_is_instance(z8_action_name, item)) return true;
    if (z8_native_is_instance(z8_action_group_name, item)) return true;
    if (z8_native_is_instance(z8_listbox_name, item)) {
        auto target = z8_listbox_state(item).query;
        if (target is null || !z8_native_is_instance("org.zenframework.z8.server.base.table.Table"w, target))
            throw new Exception("Listbox requires a table query");
        auto key = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, target);
        if (!privileges.getTableAccess(key).read) throw new Exception("Listbox table access denied");
        return true;
    }
    if (z8_native_is_instance(z8_tab_control_name, item)) return true;
    if (!z8_native_is_instance(z8_section_name, item)) throw new Exception("unsupported control metadata");
    if (z8_native_is_instance("org.zenframework.z8.server.base.form.Form"w, item)) {
        auto dependency = z8_control_state(item).dependency;
        if (dependency !is null) {
            if (!z8_native_is_instance("org.zenframework.z8.server.base.table.value.Link"w, dependency) ||
                !z8_application_query_field_readable(query, dependency, privileges)) throw new Exception("dependent form link access denied");
            auto target = z8_link_query(dependency);
            if (target is null || !z8_native_is_instance("org.zenframework.z8.server.base.table.Table"w, target))
                throw new Exception("dependent form requires a linked table");
            auto targetKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, target);
            if (!privileges.getTableAccess(targetKey).read) throw new Exception("dependent form table access denied");
        }
    }
    return true;
}
private void z8_control_write_meta_tree(Z8Object item, Z8Object query, Z8Object context, Z8Privileges privileges, ref Z8Object[] path) {
    if (z8_native_is_instance(z8_field_name, item)) { z8_control_write_field_meta(item, query, context, privileges); return; }
    foreach (parent; path) if (parent is item) throw new Exception("cyclic section controls");
    path ~= item;
    scope(exit) path.length = path.length - 1;
    auto writer = z8_application_require_request().writer;
    if (z8_native_is_instance(z8_action_name, item)) { z8_action_write_meta(item, query, context, privileges); return; }
    if (z8_native_is_instance(z8_action_group_name, item)) {
        z8_control_write_base_meta(item, query, context, privileges);
        z8_monitor_property(writer, "isSection"w, true);
        z8_monitor_property(writer, "colCount"w, 3L);
        z8_monitor_property(writer, "isActionGroup"w, true);
        auto actions = z8_action_group_state(item).actions;
        if (actions !is null && actions.view() !is null) {
            auto items = actions.view().z8_erased_data();
            if (items.z8_size()) {
                writer.start('[', "actions"w);
                for (long i; i < items.z8_size(); ++i) {
                    auto action = z8_nonnull(cast(Z8Object)items.z8_get(i));
                    if (!z8_native_is_instance(z8_action_name, action)) throw new Exception("invalid ActionGroup action");
                    writer.start('{'); z8_action_write_meta(action, query, context, privileges); writer.finish('}');
                }
                writer.finish(']');
            }
        }
        return;
    }
    if (z8_native_is_instance(z8_listbox_name, item)) {
        auto state = z8_listbox_state(item);
        auto target = z8_nonnull(state.query);
        auto control = z8_control_state(item), queryState = z8_query_state(target);
        auto savedReadOnly = control.readOnly;
        control.readOnly = z8_optional_wrap((savedReadOnly.present && savedReadOnly.value) || (queryState.readOnly.present && queryState.readOnly.value));
        scope(exit) control.readOnly = savedReadOnly;
        z8_control_write_base_meta(item, target, context, privileges);
        z8_monitor_property(writer, "isListbox"w, true);
        auto key = z8_nonnull(z8_query_primary_key(target));
        auto targetKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, target);
        auto access = privileges.getTableAccess(targetKey);
        writer.start('{', "query"w);
        z8_monitor_property(writer, "request"w, context.z8_class_id());
        z8_monitor_property(writer, "name"w, target.z8_id());
        z8_monitor_property(writer, "primaryKey"w, key.z8_id());
        if (queryState.period !is null && z8_application_query_field_readable(target, queryState.period, privileges))
            z8_monitor_property(writer, "periodKey"w, queryState.period.z8_id());
        writer.start('{', "access"w);
        z8_monitor_property(writer, "read"w, access.read); z8_monitor_property(writer, "write"w, access.write);
        z8_monitor_property(writer, "create"w, access.create); z8_monitor_property(writer, "copy"w, access.copy);
        z8_monitor_property(writer, "destroy"w, access.destroy); writer.finish('}');
        if (!control.readOnly.value) foreach (field; z8_query_declared_fields(target))
            if (field.z8_get_attribute("lockKey"w) !is null) z8_monitor_property(writer, "lockKey"w, field.z8_id());
        foreach (field; z8_query_declared_fields(target)) {
            auto parent = field.z8_get_attribute("parentKey"w);
            if (parent !is null && z8_str_compareNoCase(parent, "false"w) != 0) z8_monitor_property(writer, "parentKey"w, field.z8_id());
        }
        z8_monitor_property(writer, "totals"w, queryState.totals);
        z8_monitor_property(writer, "limit"w, z8_query_limit(target));
        z8_monitor_property(writer, "text"w, target.z8_get_attribute("displayName"w));
        auto columns = z8_listbox_array(item, "columns");
        auto fields = z8_query_form_fields(target);
        foreach (field; columns) { bool found; foreach (previous; fields) if (previous is field) { found = true; break; } if (!found) fields ~= field; }
        if (!columns.length) columns = z8_query_array(target, "columns"w);
        void writeFields(wstring name, Z8Object[] list) {
            Z8Object[] visible; foreach (field; list) if (z8_control_metadata_visible(field, target, privileges)) visible ~= field;
            if (visible.length) {
                writer.start('[', name);
                foreach (field; visible) { writer.start('{'); z8_control_write_meta(field, target, context, privileges); writer.finish('}'); }
                writer.finish(']');
            }
        }
        writeFields("fields"w, fields); writeFields("columns"w, columns);
        auto sort = z8_listbox_array(item, "sortFields");
        if (!sort.length) sort = z8_query_array(target, "sortFields"w);
        Z8Object[] readableSort; foreach (field; sort) if (z8_application_query_field_readable(target, field, privileges)) readableSort ~= field;
        z8_application_query_write_order(readableSort, null);
        auto actions = z8_listbox_array(item, "actions");
        if (actions.length) {
            writer.start('[', "actions"w);
            foreach (action; actions) { writer.start('{'); z8_action_write_meta(action, target, context, privileges); writer.finish('}'); }
            writer.finish(']');
        }
        if (state.link !is null) {
            auto link = state.link;
            if (!z8_native_is_instance("org.zenframework.z8.server.base.table.value.Link"w, link) ||
                !z8_application_query_field_readable(target, link, privileges)) throw new Exception("Listbox link access denied");
            auto linked = z8_link_query(link);
            if (linked !is null && z8_native_is_instance("org.zenframework.z8.server.base.table.Table"w, linked)) {
                auto linkedKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, linked);
                if (!privileges.getTableAccess(linkedKey).read) throw new Exception("Listbox linked table access denied");
            }
            if (linked !is null && z8_native_is_instance("org.zenframework.z8.server.base.table.TreeTable"w, linked))
                throw new Exception("Listbox tree link metadata is not supported yet");
            writer.start('{', "link"w); z8_control_write_field_meta(link, target, context, privileges); writer.finish('}');
        }
        writer.finish('}');
        if (state.selectorInitialized && state.selector !is null) {
            auto selector = z8_selector_state(state.selector);
            auto selectorColumns = z8_selector_array(state.selector, "columns");
            if (selector.link !is null && selectorColumns.length) {
                if (!z8_native_is_instance("org.zenframework.z8.server.base.table.value.Link"w, selector.link) ||
                    !z8_application_query_field_readable(query, selector.link, privileges))
                    throw new Exception("Listbox selector link access denied");
                auto selectorTarget = z8_link_query(selector.link);
                if (selectorTarget !is null && z8_native_is_instance("org.zenframework.z8.server.base.table.Table"w, selectorTarget)) {
                    auto selectorKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, selectorTarget);
                    if (!privileges.getTableAccess(selectorKey).read) throw new Exception("Listbox selector table access denied");
                }
                writer.start('{', "selector"w);
                z8_monitor_property(writer, "isSelector"w, true);
                z8_monitor_property(writer, "header"w, state.selector.z8_get_attribute("displayName"w));
                z8_monitor_property(writer, "icon"w, state.selector.z8_get_attribute("icon"w));
                z8_monitor_property(writer, "copy"w, selector.copy);
                z8_monitor_property(writer, "multiselect"w, selector.multiselect);
                writer.start('{', "link"w); z8_control_write_field_meta(selector.link, query, context, privileges); writer.finish('}');
                Z8Object[] visible;
                foreach (field; selectorColumns) if (z8_control_metadata_visible(field, query, privileges)) visible ~= field;
                if (visible.length) {
                    writer.start('[', "columns"w);
                    foreach (field; visible) { writer.start('{'); z8_control_write_meta(field, query, context, privileges); writer.finish('}'); }
                    writer.finish(']');
                }
                Z8Object[] selectorSort;
                foreach (field; z8_selector_array(state.selector, "sortFields"))
                    if (z8_application_query_field_readable(query, field, privileges)) selectorSort ~= field;
                z8_application_query_write_order(selectorSort, null);
                writer.finish('}');
            }
        }
        return;
    }
    if (z8_native_is_instance(z8_tab_control_name, item)) {
        z8_control_write_base_meta(item, query, context, privileges);
        z8_monitor_property(writer, "isTabControl"w, true);
        auto tabs = z8_tab_control_state(item).tabs;
        if (tabs !is null && tabs.view() !is null) {
            auto items = tabs.view().z8_erased_data();
            if (items.z8_size()) {
                writer.start('[', "tabs"w);
                for (long i; i < items.z8_size(); ++i) {
                    auto tab = z8_nonnull(cast(Z8Object)items.z8_get(i));
                    if (!z8_native_is_instance("org.zenframework.z8.server.base.form.Tab"w, tab)) throw new Exception("invalid TabControl tab");
                    writer.start('{'); z8_control_write_meta_tree(tab, query, context, privileges, path); writer.finish('}');
                }
                writer.finish(']');
            }
        }
        return;
    }
    auto formQuery = query;
    if (z8_native_is_instance("org.zenframework.z8.server.base.form.Form"w, item)) {
        z8_monitor_property(writer, "isForm"w, true);
        auto dependency = z8_control_state(item).dependency;
        if (dependency !is null) {
            formQuery = z8_nonnull(z8_link_query(dependency));
            auto owner = z8_nonnull(dependency.z8_get_owner());
            auto ownerKey = z8_nonnull(z8_query_primary_key(owner));
            auto targetKey = z8_nonnull(z8_query_primary_key(formQuery));
            writer.start('{', "link"w);
            z8_monitor_property(writer, "name"w, dependency.z8_id());
            z8_monitor_property(writer, "primaryKey"w, ownerKey.z8_id());
            z8_monitor_property(writer, "owner"w, owner.z8_id());
            writer.start('{', "query"w); z8_monitor_property(writer, "primaryKey"w, targetKey.z8_id()); writer.finish('}');
            writer.finish('}');
        }
    }
    z8_control_write_base_meta(item, formQuery, context, privileges);
    z8_monitor_property(writer, "isSection"w, true);
    auto section = z8_section_state(item);
    z8_monitor_property(writer, "colCount"w, section.colCount);
    if (section.controls !is null && section.controls.view() !is null) {
        auto items = section.controls.view().z8_erased_data();
        Z8Object[] visible;
        for (long i; i < items.z8_size(); ++i) {
            auto child = z8_nonnull(cast(Z8Object)items.z8_get(i));
            if (z8_control_metadata_visible(child, formQuery, privileges)) visible ~= child;
        }
        if (visible.length) {
            writer.start('[', "controls"w);
            foreach (child; visible) {
                writer.start('{'); z8_control_write_meta_tree(child, formQuery, context, privileges, path); writer.finish('}');
            }
            writer.finish(']');
        }
    }
    if (z8_native_is_instance("org.zenframework.z8.server.base.form.Fieldset"w, item)) z8_monitor_property(writer, "isFieldset"w, true);
    if (z8_native_is_instance("org.zenframework.z8.server.base.form.FieldGroup"w, item)) z8_monitor_property(writer, "isFieldGroup"w, true);
    if (z8_native_is_instance("org.zenframework.z8.server.base.form.Tab"w, item)) z8_monitor_property(writer, "isTab"w, true);
}
private void z8_control_write_meta(Z8Object item, Z8Object query, Z8Object context, Z8Privileges privileges) {
    Z8Object[] path;
    z8_control_write_meta_tree(item, query, context, privileges, path);
}

    z8_native_bind_field(control ~ "f_dependency"w, &z8_control_address!"dependency");

    z8_native_bind_field(z8_selector_name ~ ".f_multiselect"w, &z8_selector_address!"multiselect");
}
