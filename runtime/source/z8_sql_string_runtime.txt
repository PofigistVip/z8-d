enum Z8SqlStringKind { constant, field, textField, group, concat, lower, upper, trimLeft, trimRight, choose, replace, regReplace, substring, reverse, padLeft, padRight, quote, jsonString, jsonInt, fromBool, fromNumeric, fromGuid, fromDate, fromDatespan, fromBinary, fromGeometry, geometryFunction }
class Z8SqlString : Z8SqlPrimary {
    Z8SqlStringKind kind;
    Z8Object field;
    wstring constant;
    Z8SqlString left, right;
    Z8SqlString replacement, flags;
    Z8SqlInt position, length;
    Z8SqlBool condition;
    Z8SqlNumeric numericOperand;
    Z8SqlGuid guidOperand;
    Z8SqlDate dateOperand;
    Z8SqlDatespan spanOperand;
    Z8SqlBinary binaryOperand;
    Z8SqlGeometry geometryOperand;
    wstring geometryFunction;
    this(Z8SqlStringKind kind, wstring constant = null, Z8SqlString left = null,
         Z8SqlString right = null, Z8SqlBool condition = null) {
        this.kind = kind; this.constant = constant; this.left = left; this.right = right; this.condition = condition;
    }
}

Z8SqlString z8_sql_string_unary(string op)(Z8SqlString value) {
    static if (op == "toLower") enum kind = Z8SqlStringKind.lower;
    else static if (op == "toUpper") enum kind = Z8SqlStringKind.upper;
    else static if (op == "trimLeft") enum kind = Z8SqlStringKind.trimLeft;
    else static if (op == "trimRight") enum kind = Z8SqlStringKind.trimRight;
    else static assert(0, "unsupported SQL string method");
    return new Z8SqlString(kind, null, z8_nonnull(value));
}

private wstring z8_sql_fts_vector(Z8SqlString value, Z8Fts config, Z8SqlVendor vendor) {
    auto configuration = z8_nonnull(config).configuration;
    return "to_tsvector("w ~ (configuration is null ? ""w : z8_sql_string_format(z8_sql_string_const(configuration), vendor) ~ ", "w) ~ z8_sql_string_as_text(value, vendor) ~ ")"w;
}
private wstring z8_sql_fts_query(Z8SqlString value, Z8Fts config, Z8SqlVendor vendor) {
    auto configuration = z8_nonnull(config).configuration;
    auto query = z8_nonnull(config.queryType);
    auto prefix = query.id == 1 ? "plain"w : query.id == 2 ? "phrase"w : ""w;
    return prefix ~ "to_tsquery("w ~ (configuration is null ? ""w : z8_sql_string_format(z8_sql_string_const(configuration), vendor) ~ ", "w) ~ z8_sql_string_as_text(value, vendor) ~ ")"w;
}

Z8SqlBool z8_sql_string_compare(string op)(Z8SqlString left, Z8SqlString right) {
    z8_nonnull(left);
    static if (op == "==") enum kind = Z8SqlBoolKind.textEqual;
    else static if (op == "!=") enum kind = Z8SqlBoolKind.textNotEqual;
    else static if (op == "<") enum kind = Z8SqlBoolKind.textLess;
    else static if (op == ">") enum kind = Z8SqlBoolKind.textGreater;
    else static if (op == "<=") enum kind = Z8SqlBoolKind.textLessEqual;
    else static if (op == ">=") enum kind = Z8SqlBoolKind.textGreaterEqual;
    else static assert(0, "unsupported SQL string comparison");
    auto result = new Z8SqlBool(kind);
    result.textLeft = left; result.textRight = right;
    return result;
}
private wstring z8_sql_string_as_text(Z8SqlString value, Z8SqlVendor vendor) {
    auto text = z8_sql_string_format(value, vendor);
    return vendor == Z8SqlVendor.SqlServer ? "Cast("w ~ text ~ " as nvarchar(max))"w : text;
}
// ToString in Z8 examines the SQL token's actual type, which can differ from
// the BL wrapper type (RegLike reports Integer and integer ~ reports Boolean).
private bool z8_sql_bool_is_numeric(Z8SqlBool value) {
    z8_nonnull(value);
    if (value.kind == Z8SqlBoolKind.group) return z8_sql_bool_is_numeric(value.left);
    if (value.kind == Z8SqlBoolKind.choose) return z8_sql_bool_is_numeric(value.right);
    return value.kind == Z8SqlBoolKind.textRegLike;
}
private bool z8_sql_int_is_numeric(Z8SqlInt value) {
    z8_nonnull(value);
    switch (value.kind) {
        case Z8SqlIntKind.bitNot: return false;
        case Z8SqlIntKind.group: case Z8SqlIntKind.negate: case Z8SqlIntKind.abs:
        case Z8SqlIntKind.power: case Z8SqlIntKind.choose: return z8_sql_int_is_numeric(value.left);
        case Z8SqlIntKind.add: case Z8SqlIntKind.subtract: case Z8SqlIntKind.multiply:
        case Z8SqlIntKind.divide: case Z8SqlIntKind.bitAnd: case Z8SqlIntKind.bitOr: case Z8SqlIntKind.bitXor:
            auto left = z8_sql_int_is_numeric(value.left), right = z8_sql_int_is_numeric(value.right);
            if (left != right) throw new Exception("unsupported SQL operand types");
            return left;
        default: return true;
    }
}
private wstring z8_sql_to_string_format(wstring value, bool numeric, Z8SqlVendor vendor) {
    if (vendor == Z8SqlVendor.SqlServer) return "Cast("w ~ value ~ " as nvarchar(max))"w;
    if (vendor == Z8SqlVendor.Postgres && numeric) return "("w ~ value ~ ")::text"w;
    return value;
}
wstring z8_sql_string_format(Z8SqlString value, Z8SqlVendor vendor = Z8SqlVendor.Postgres, bool logical = false) {
    z8_nonnull(value);
    final switch (value.kind) {
        case Z8SqlStringKind.field: return z8_field_sql_format(value.field, vendor);
        case Z8SqlStringKind.textField:
            auto field = z8_field_sql_format(value.field, vendor);
            if (vendor == Z8SqlVendor.Oracle) return "UTL_RAW.CAST_TO_NVARCHAR2("w ~ field ~ ")"w;
            if (vendor == Z8SqlVendor.Postgres) return "CONVERT_FROM("w ~ field ~ ", 'UTF8')"w;
            if (vendor == Z8SqlVendor.SqlServer) return "Cast("w ~ field ~ " as nvarchar(max))"w;
            return field;
        case Z8SqlStringKind.constant:
            wstring escaped;
            for (size_t i = 0; i < value.constant.length; ++i) {
                escaped ~= value.constant[i];
                if (value.constant[i] == '\'') escaped ~= '\'';
            }
            return (vendor == Z8SqlVendor.Oracle || vendor == Z8SqlVendor.SqlServer ? "N'"w : "'"w) ~ escaped ~ "'"w;
        case Z8SqlStringKind.group:
            auto text = z8_sql_string_format(value.left, vendor, logical);
            return text is null ? null : "("w ~ text ~ ")"w;
        case Z8SqlStringKind.concat:
            if (vendor == Z8SqlVendor.H2) throw new Exception("unsupported SQL database vendor");
            auto left = z8_sql_string_format(value.left, vendor), right = z8_sql_string_format(value.right, vendor);
            if (vendor == Z8SqlVendor.SqlServer) return "("w ~ left ~ "+"w ~ right ~ ")"w;
            return "Concat("w ~ left ~ ", "w ~ right ~ ")"w;
        case Z8SqlStringKind.lower: return "LOWER("w ~ z8_sql_string_as_text(value.left, vendor) ~ ")"w;
        case Z8SqlStringKind.upper: return "UPPER("w ~ z8_sql_string_as_text(value.left, vendor) ~ ")"w;
        case Z8SqlStringKind.trimLeft: return "LTRIM("w ~ z8_sql_string_format(value.left, vendor) ~ ")"w;
        case Z8SqlStringKind.trimRight: return "RTRIM("w ~ z8_sql_string_format(value.left, vendor) ~ ")"w;
        case Z8SqlStringKind.choose:
            return z8_sql_bool_case(z8_sql_bool_format(value.condition, true, vendor), z8_sql_string_format(value.left, vendor), z8_sql_string_format(value.right, vendor));
        case Z8SqlStringKind.replace: case Z8SqlStringKind.regReplace:
            if (vendor == Z8SqlVendor.H2) throw new Exception("unsupported SQL database vendor");
            auto name = value.kind == Z8SqlStringKind.replace ? "replace("w : "regexp_replace("w;
            return name ~ z8_sql_string_format(value.left, vendor) ~ ", "w ~ z8_sql_string_format(value.right, vendor) ~ ", "w ~
                z8_sql_string_format(value.replacement, vendor) ~ (value.flags is null ? ""w : ", "w ~ z8_sql_string_format(value.flags, vendor)) ~ ")"w;
        case Z8SqlStringKind.substring:
            if (vendor == Z8SqlVendor.H2) throw new Exception("unsupported SQL database vendor");
            return (vendor == Z8SqlVendor.SqlServer ? "subString("w : "substr("w) ~ z8_sql_string_format(value.left, vendor) ~ ","w ~
                z8_sql_int_format(value.position, vendor) ~ " + 1,"w ~
                (value.length is null ? z8_sql_int_format(z8_sql_string_length(value.left), vendor) : z8_sql_int_format(value.length, vendor)) ~ ")"w;
        case Z8SqlStringKind.reverse: return "REVERSE("w ~ z8_sql_string_format(value.left, vendor) ~ ")"w;
        case Z8SqlStringKind.padLeft: case Z8SqlStringKind.padRight:
            if (vendor == Z8SqlVendor.H2) throw new Exception("unsupported SQL database vendor");
            auto text = z8_sql_string_format(value.left, vendor), length = z8_sql_int_format(value.length, vendor);
            auto pad = value.right is null ? (vendor == Z8SqlVendor.Postgres ? "' '"w : "N' '"w) : z8_sql_string_format(value.right, vendor);
            if (vendor == Z8SqlVendor.Postgres) return (value.kind == Z8SqlStringKind.padLeft ? "lpad("w : "rpad("w) ~ text ~ ", "w ~ length ~ ", "w ~ pad ~ ")"w;
            if (value.kind == Z8SqlStringKind.padLeft) return "RIGHT(REPLICATE("w ~ pad ~ ","w ~ length ~ ")+"w ~ text ~ ","w ~ length ~ ")"w;
            if (vendor == Z8SqlVendor.Oracle) return "RPAD("w ~ text ~ ","w ~ length ~ ","w ~ pad ~ ")"w;
            return "LEFT("w ~ text ~ "+REPLICATE("w ~ pad ~ ","w ~ length ~ "),"w ~ length ~ ")"w;
        case Z8SqlStringKind.quote:
            if (vendor != Z8SqlVendor.Postgres) throw new Exception("unsupported SQL database vendor");
            return "to_json("w ~ z8_sql_string_format(value.left, vendor) ~ ")"w;
        case Z8SqlStringKind.jsonString: case Z8SqlStringKind.jsonInt:
            if (vendor != Z8SqlVendor.Postgres) throw new Exception("unsupported SQL database vendor");
            return "("w ~ z8_sql_string_format(value.left, vendor) ~ ")::json->>"w ~
                (value.kind == Z8SqlStringKind.jsonString ? z8_sql_string_format(value.right, vendor) : z8_sql_int_format(value.position, vendor));
        case Z8SqlStringKind.fromBool:
            auto boolNumeric = z8_sql_bool_is_numeric(value.condition);
            return z8_sql_to_string_format(z8_sql_bool_format(value.condition, false, vendor), boolNumeric, vendor);
        case Z8SqlStringKind.fromNumeric:
            auto intValue = cast(Z8SqlInt) value.numericOperand;
            auto numeric = intValue is null || z8_sql_int_is_numeric(intValue);
            return z8_sql_to_string_format(value.numericOperand.z8_format(vendor), numeric, vendor);
        case Z8SqlStringKind.fromGuid:
            auto guid = z8_sql_guid_format(value.guidOperand, vendor);
            if (vendor == Z8SqlVendor.Oracle) return "RAWTOHEX("w ~ guid ~ ")"w;
            if (vendor == Z8SqlVendor.Postgres) return guid ~ "::text"w;
            if (vendor == Z8SqlVendor.SqlServer) return "Cast("w ~ guid ~ " as nvarchar(max))"w;
            return guid;
        case Z8SqlStringKind.fromDate:
            if (vendor == Z8SqlVendor.Postgres)
                return "TO_CHAR(TO_TIMESTAMP("w ~ z8_sql_date_format(value.dateOperand, vendor) ~ " / 1000), 'DD.MM.YYYY HH24:MI:SS')"w;
            if (vendor == Z8SqlVendor.SqlServer)
                return "Convert(nvarchar, "w ~ z8_sql_date_format(value.dateOperand, vendor) ~ ", 103) + ' ' + Convert(nvarchar(8), "w ~ z8_sql_date_format(value.dateOperand, vendor) ~ ", 114)"w;
            throw new Exception("unsupported SQL database vendor");
        case Z8SqlStringKind.fromDatespan: return z8_sql_datespan_format(value.spanOperand, vendor);
        case Z8SqlStringKind.fromBinary:
            auto binary = z8_sql_binary_format(value.binaryOperand, vendor);
            return vendor == Z8SqlVendor.SqlServer ? "Cast("w ~ binary ~ " as nvarchar(max))"w : binary;
        case Z8SqlStringKind.fromGeometry:
            auto geometry = z8_sql_geometry_format(value.geometryOperand, vendor);
            return vendor == Z8SqlVendor.SqlServer ? "Cast("w ~ (geometry is null ? "null"w : geometry) ~ " as nvarchar(max))"w : geometry;
        case Z8SqlStringKind.geometryFunction: return z8_sql_geometry_function(value.geometryFunction, value.geometryOperand, vendor);
    }
}
