// Core Executable callbacks, shared by job hosts and generated BL classes.
private enum z8_executable_name = "org.zenframework.z8.server.base.Executable"w;
private class Z8ExecutableState { Z8ApplicationSession defaultSession; }
private Z8ExecutableState z8_executable_state(Z8Object self) {
    self = z8_nonnull(self);
    if (!z8_native_is_instance(z8_executable_name, self)) throw new Exception("invalid Executable object");
    auto state = cast(Z8ExecutableState)self.z8_get_native_state(z8_executable_name);
    if (state is null) { state = new Z8ExecutableState(); self.z8_set_native_state(z8_executable_name, state); }
    return state;
}
private void z8_executable_switch_user(Z8Object self, Z8Guid id) {
    auto request = z8_application_require_request();
    auto original = z8_nonnull(z8_executable_state(self).defaultSession);
    if (!z8_nonnull(original.user).builtinAdministrator) throw new Exception("job user switch access denied");
    id = z8_nonnull(id);
    if (z8_nonnull(z8_nonnull(request.session).user).id == id) return;
    auto resolver = request.jobUserResolver;
    if (resolver is null) resolver = cast(IZ8JobUserResolver)z8_connection_current_backend();
    if (resolver is null) throw new Exception("job user resolver is not configured");
    auto user = z8_nonnull(resolver.resolveJobUser(id));
    if (user.id is null || user.id != id || user.privileges is null) throw new Exception("invalid resolved job user");
    request.session = new Z8ApplicationSession(""w, user, original.system);
}
private void z8_executable_reset_user(Z8Object self) {
    auto request = z8_application_require_request();
    request.session = z8_nonnull(z8_executable_state(self).defaultSession);
}
private void z8_executable_default_execute(Z8Object self) {
    z8_nonnull(self).z8_executable_virtual_execute_parameters();
}
private void z8_executable_default_parameters(Z8Object self, Z8ObjectArray!Z8Object parameters) { z8_nonnull(self); }
void z8_schema_generator_configure(Z8Object self) { z8_action_state(self).useTransaction = z8_optional_wrap(false); }
void z8_schema_generator_require_admin(Z8Object self) {
    z8_nonnull(self);
    auto request = z8_application_require_request();
    if (!z8_nonnull(z8_nonnull(request.session).user).administrator)
        throw new Exception("You must be a member of Administrators security group to perform this action.");
}
void z8_schema_generator_execute(Z8Object self) {
    z8_schema_generator_require_admin(self);
    auto request = z8_application_require_request();
    auto backend = cast(IZ8SchemaReconciler)z8_connection_current_backend();
    if (backend is null) throw new Exception("schema reconciliation is not configured");
    auto scheduler = cast(IZ8SchedulerSuspension)request.schedulerControl;
    if (request.schedulerControl !is null && scheduler is null) throw new Exception("scheduler cannot be suspended");
    if (scheduler !is null) scheduler.suspend();
    scope(exit) if (scheduler !is null) scheduler.resume();
    request.setEventsLevel(Z8EventsLevel.system);
    scope(exit) request.restoreEventsLevel();
    self.z8_schema_generator_virtual_before_start();
    backend.reconcileSchema();
    self.z8_schema_generator_virtual_after_finish();
}
private IZ8SchemaOperations z8_schema_operations(Z8Object self) {
    z8_nonnull(self);
    auto backend = cast(IZ8SchemaOperations)z8_connection_current_backend();
    if (backend is null) throw new Exception("schema operations are not configured");
    return backend;
}
private Z8Optional!bool z8_schema_table_exists(Z8Object self, wstring table) {
    return z8_optional_wrap(z8_schema_operations(self).tableExists(z8_string_required(table)));
}
private Z8Optional!bool z8_schema_field_exists(Z8Object self, wstring table, wstring field) {
    return z8_optional_wrap(z8_schema_operations(self).fieldExists(z8_string_required(table), z8_string_required(field)));
}
private void z8_schema_rename_table(Z8Object self, wstring table, wstring replacement) {
    z8_schema_operations(self).renameTable(z8_string_required(table), z8_string_required(replacement));
}
private void z8_schema_rename_field(Z8Object self, wstring table, wstring field, wstring replacement) {
    z8_schema_operations(self).renameField(z8_string_required(table), z8_string_required(field), z8_string_required(replacement));
}
private void z8_schema_drop_table(Z8Object self, wstring table) {
    z8_schema_operations(self).dropTable(z8_string_required(table));
}
private void z8_executable_progress(Z8Object self, Z8Optional!long percent) {
    z8_nonnull(self);
    auto monitor = cast(IZ8JobProgress)z8_application_require_request().monitor;
    if (monitor is null) throw new Exception("job monitor does not support progress");
    monitor.setWorked(cast(int)z8_optional_require(percent));
}
private void z8_executable_message(Z8MonitorLevel level)(Z8Object self, wstring text) {
    z8_nonnull(self);
    auto monitor = z8_nonnull(z8_application_require_request().monitor);
    monitor.message(level, z8_string_required(text));
}
// Run a BL callback without choosing a scheduler or transaction policy.
void z8_executable_execute(Z8Object self) {
    if (!z8_native_is_instance(z8_executable_name, z8_nonnull(self))) throw new Exception("invalid Executable object");
    self.z8_executable_virtual_execute();
}
// Run with Executable.run's transaction and error-reporting behavior.
bool z8_executable_run(Z8Object self) {
    auto request = z8_application_require_request();
    auto session = z8_nonnull(request.session);
    auto originalUser = session.user;
    auto state = z8_executable_state(self);
    auto previousSession = state.defaultSession;
    state.defaultSession = session;
    scope(exit) { request.session = session; session.user = originalUser; state.defaultSession = previousSession; }
    IZ8DatabaseConnection connection;
    bool begun;
    try {
        if (z8_optional_require(z8_action_state(self).useTransaction)) {
            connection = z8_connection_current_backend();
            if (connection is null) throw new Exception("no current database connection");
            connection.beginTransaction(); begun = true;
        }
        z8_executable_execute(self);
        if (connection !is null) { connection.commit(); begun = false; }
        return true;
    } catch (Throwable error) {
        if (begun) {
            try { connection.rollback(); }
            catch (Throwable rollbackError) { error = rollbackError; }
        }
        z8_nonnull(request.monitor).message(Z8MonitorLevel.error, error.msg.to!wstring);
        return false;
    }
}
void z8_bind_builtin_executable() {
    z8_native_bind(z8_executable_name ~ ".m_execute"w, &z8_executable_default_execute);
    z8_native_bind(z8_executable_name ~ ".m_execute"w, &z8_executable_default_parameters);
    enum schema = "org.zenframework.z8.server.db.generator.SchemaGenerator.m_"w;
    z8_native_bind(schema ~ "tableExists"w, &z8_schema_table_exists);
    z8_native_bind(schema ~ "fieldExists"w, &z8_schema_field_exists);
    z8_native_bind(schema ~ "renameTable"w, &z8_schema_rename_table);
    z8_native_bind(schema ~ "renameField"w, &z8_schema_rename_field);
    z8_native_bind(schema ~ "dropTable"w, &z8_schema_drop_table);
    z8_native_bind(schema ~ "beforeStart"w, &z8_query_default_event);
    z8_native_bind(schema ~ "afterFinish"w, &z8_query_default_event);
    z8_native_bind(z8_executable_name ~ ".m_switchUser"w, &z8_executable_switch_user);
    z8_native_bind(z8_executable_name ~ ".m_resetUser"w, &z8_executable_reset_user);
    z8_native_bind(z8_executable_name ~ ".m_progress"w, &z8_executable_progress);
    z8_native_bind(z8_executable_name ~ ".m_info"w, &z8_executable_message!(Z8MonitorLevel.info));
    z8_native_bind(z8_executable_name ~ ".m_warning"w, &z8_executable_message!(Z8MonitorLevel.warning));
    z8_native_bind(z8_executable_name ~ ".m_error"w, &z8_executable_message!(Z8MonitorLevel.error));
}
shared static this() { z8_bind_builtin_executable(); }
