// State and value operations of base.entity.Entity (Entity.java).
private enum z8_entity_name = "org.zenframework.z8.server.base.entity.Entity"w;
private __gshared wstring z8_entity_id_key = "recordId"w, z8_entity_origin_key = "originId"w;
private wstring* z8_entity_id_key_address() { return &z8_entity_id_key; }
private wstring* z8_entity_origin_key_address() { return &z8_entity_origin_key; }
private class Z8EntityState {
    Z8PrimaryMap defaults, changes;
    bool initialized, persistent, deleted;
    Z8Object registry;
    this() { defaults = new Z8PrimaryMap(); changes = new Z8PrimaryMap(); }
}
private Z8EntityState z8_entity_state(Z8Object self) {
    self = z8_nonnull(self);
    if (!z8_native_is_instance(z8_entity_name, self)) throw new Exception("invalid Entity object");
    auto state = cast(Z8EntityState)self.z8_get_native_state(z8_entity_name);
    if (state is null) { state = new Z8EntityState(); self.z8_set_native_state(z8_entity_name, state); }
    return state;
}
private Z8Primary z8_entity_get(Z8Object self, wstring key) {
    auto state = z8_entity_state(self);
    return state.changes.z8_containsKey(key) ? state.changes.z8_get(key) : state.defaults.z8_get(key);
}
private Z8Object z8_entity_init(Z8Object self, wstring key, Z8Primary value) {
    auto state = z8_entity_state(self); state.defaults.z8_add(key, value); state.initialized = true; return self;
}
private Z8Object z8_entity_init_map(Z8Object self, Z8PrimaryMap values) {
    auto state = z8_entity_state(self); state.defaults.z8_add(z8_nonnull(values)); state.initialized = true; return self;
}
private Z8Object z8_entity_set(Z8Object self, wstring key, Z8Primary value) {
    auto state = z8_entity_state(self); state.initialized = true;
    if (key == z8_entity_id_key) { state.defaults.z8_add(key, value); return self; }
    auto current = state.changes.z8_get(key);
    if (current is null) current = state.defaults.z8_get(key);
    if (!z8_primary_collection_equals(value, current)) {
        state.changes.z8_add(key, value); self.z8_entity_virtual_on_change(key);
    }
    return self;
}
private Z8Object z8_entity_set_map(Z8Object self, Z8PrimaryMap values) {
    auto keys = z8_nonnull(values).z8_keys();
    for (long i; i < keys.z8_size(); ++i) { auto key = keys.z8_get(i); z8_entity_set(self, key, values.z8_get(key)); }
    return self;
}
private Z8Guid z8_entity_guid(Z8Object self, wstring key) {
    auto value = z8_entity_get(self, key);
    if (value is null) return null;
    return value.kind == Z8PrimaryKind.guid ? value.guidValue : z8_guid_parse(z8_primary_to_string(value));
}
private Z8Guid z8_entity_get_id(Z8Object self) { return z8_entity_guid(self, z8_entity_id_key); }
private Z8Guid z8_entity_get_origin_id(Z8Object self) { return z8_entity_guid(self, z8_entity_origin_key); }
private Z8Object z8_entity_set_id(Z8Object self, Z8Guid id) { return z8_entity_set(self, z8_entity_id_key, z8_primary_box(id)); }
private Z8Object z8_entity_set_origin_id(Z8Object self, Z8Guid id) { return z8_entity_set(self, z8_entity_origin_key, z8_primary_box(id)); }
private Z8Object z8_entity_set_json_object(Z8Object self, wstring key, Z8Object value) {
    if (value is null) return z8_entity_set(self, key, null);
    return z8_entity_set(self, key, z8_primary_box(z8_json_object_to_string(value)));
}
private Z8Object z8_entity_set_json_array(Z8Object self, wstring key, Z8Object value) {
    return value is null ? z8_entity_set(self, key, null) : z8_entity_set(self, key, z8_primary_box(z8_json_array_string(value)));
}
private Z8Object z8_entity_set_entity(Z8Object self, wstring key, Z8Object value) {
    return z8_entity_set(self, key, z8_primary_box(value is null ? z8_guid_zero() : z8_entity_get_id(value)));
}
private Z8Optional!bool z8_entity_is_null(Z8Object self) {
    auto id = z8_entity_get_id(self); return z8_optional_wrap(id is null || id == z8_guid_zero());
}
private Z8Optional!bool z8_entity_initialized(Z8Object self) { return z8_optional_wrap(z8_entity_state(self).initialized); }
private Z8Optional!bool z8_entity_persistent(Z8Object self) { return z8_optional_wrap(z8_entity_state(self).persistent); }
private Z8Optional!bool z8_entity_deleted(Z8Object self) { return z8_optional_wrap(z8_entity_state(self).deleted); }
private Z8Optional!bool z8_entity_changed(Z8Object self) { return z8_optional_wrap(!z8_entity_state(self).changes.z8_isEmpty()); }
private Z8Optional!bool z8_entity_changed_key(Z8Object self, wstring key) { return z8_optional_wrap(z8_entity_state(self).changes.z8_containsKey(key)); }
private Z8Optional!bool z8_entity_has(Z8Object self, wstring key) {
    auto state = z8_entity_state(self); return z8_optional_wrap(state.changes.z8_containsKey(key) || state.defaults.z8_containsKey(key));
}
private Z8Object z8_entity_create_id(Z8Object self, Z8Guid id) {
    if (!z8_optional_require(z8_entity_is_null(self))) throw new Exception("Entity is already created");
    z8_entity_set_id(self, id); self.z8_entity_virtual_on_create(); return self;
}
private Z8Object z8_entity_create(Z8Object self) { return z8_entity_create_id(self, z8_guid_create_random()); }
private Z8Object z8_entity_delete(Z8Object self) { z8_entity_state(self).deleted = true; self.z8_entity_virtual_on_delete(); return self; }
private Z8Object z8_entity_touch(Z8Object self, wstring key) { z8_entity_state(self); self.z8_entity_virtual_on_change(key); return self; }
private Z8Object z8_entity_touch_many(Z8Object self, Z8Array!wstring keys) {
    keys = z8_nonnull(keys); for (long i; i < keys.z8_size(); ++i) z8_entity_touch(self, keys.z8_get(i)); return self;

private Z8Optional!bool z8_entity_equal(Z8Object self, Z8Object other) {
    self = z8_nonnull(self);
    if (other is null || self.classinfo !is other.classinfo) return z8_optional_wrap(false);
    auto a = z8_entity_get_id(self), b = z8_entity_get_id(other);
    return z8_optional_wrap(a is b || (a !is null && b !is null && a == b));
}
private Z8Optional!bool z8_entity_not_equal(Z8Object self, Z8Object other) { return z8_optional_wrap(!z8_optional_require(z8_entity_equal(self, other))); }
Z8Object z8_entity_copy_default(Z8Object self) {
    auto state = z8_entity_state(self);
    auto copy = z8_object_new_instance(self.z8_class_id());
    copy.z8_set_container(z8_object_container(self));
    z8_entity_set_map(copy, state.defaults); z8_entity_set_map(copy, state.changes);
    z8_entity_set_id(copy, null); z8_entity_state(copy).initialized = state.initialized;
    z8_entity_set_origin_id(copy, z8_entity_get_id(self));
    return copy;
}
private Z8Object z8_entity_copy(Z8Object self, Z8Optional!bool withEntityId) {
    auto copy = z8_nonnull(self).z8_entity_virtual_copy();
    if (z8_optional_require(withEntityId)) z8_entity_set_id(copy, z8_entity_get_id(self));
    return copy;
}

private Z8Object z8_entity_find(Z8ObjectArray!Z8Object entities, Z8Guid id) {
    auto values = z8_nonnull(entities).z8_erased_data();
    for (long i; i < values.z8_size(); ++i) {
        auto entity = cast(Z8Object)values.z8_get(i); auto candidate = z8_entity_get_id(entity);
        if (candidate is id || (candidate !is null && id !is null && candidate == id)) return entity;
    }
    return null;
}
