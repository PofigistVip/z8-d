// A primary key retains its concrete scalar kind and Java HashMap hash.
class Z8PrimaryKeyMap(V) {
    static if (is(V == long) || is(V == bool) || is(V == Z8Decimal)) alias Element = Z8Optional!V;
    else static if (is(V == wstring) || is(V == class)) alias Element = V;
    else static assert(0, "unsupported primary-key map value");
    enum sharePrimary = is(V == long) || is(V == bool) || is(V == Z8Decimal) || is(V == wstring) ||
        is(V == Z8Guid) || is(V == Z8Binary) || is(V == Z8Datespan) || is(V == Z8Date) ||
        is(V == Z8Geometry) || is(V == Z8File) || is(V == Object);
    static if (sharePrimary) private Z8PrimaryKeyMap!Z8Primary primaryData;
    private struct Entry { Z8Primary key; Element value; ulong token; }
    private Entry[][uint] buckets;
    private size_t count;
    private Z8JavaMapOrder!ulong order;
    private uint[ulong] tokenHashes;
    private ulong nextToken;
    private Z8JavaMapOrder!ulong ordering() { if (order is null) order = new Z8JavaMapOrder!ulong(); return order; }
    private Entry* orderedEntry(ulong token) {
        auto hash = token in tokenHashes;
        if (hash is null) throw new Exception("map ordering is inconsistent");
        auto bucket = *hash in buckets;
        if (bucket !is null) foreach (ref entry; *bucket) if (entry.token == token) return &entry;
        throw new Exception("map ordering is inconsistent");
    }

    private static uint keyHash(Z8Primary key) {
        auto h = key is null ? 0u : cast(uint)z8_primary_hash(key);
        return h ^ (h >> 16);
    }
    private static bool keyEquals(Z8Primary left, Z8Primary right) {
        // HashMap compares the lookup key against the key already stored.
        return z8_primary_collection_equals(right, left);
    }
    private static bool valueEquals(Element left, Element right) {
        static if (is(V == long) || is(V == bool) || is(V == Z8Decimal))
            return left.present == right.present && (!left.present || left.value == right.value);
        else static if (is(V == wstring)) return z8_string_equals(left, right);
        else static if (is(V == Z8Primary)) return z8_primary_collection_equals(left, right);
        else return left is right || (left !is null && right !is null && left == right);
    }
    override bool opEquals(Object other) {
        auto right = cast(Z8PrimaryKeyMap!V)other;
        import std.meta : AliasSeq;
        static foreach (ScalarKey; AliasSeq!(long, bool, Z8Decimal)) {{
            auto scalarKeys = cast(Z8ScalarKeyMap!(ScalarKey, V))other;
            if (scalarKeys !is null) return this.opEquals(z8_primary_key_map_view(scalarKeys));
        }}
        static if (is(V == wstring)) {
            auto stringKeys = cast(Z8StringMap)other;
            if (stringKeys !is null) return this.opEquals(z8_primary_key_map_view(stringKeys));
        } else static if (is(V == Z8Primary)) {
            auto stringKeys = cast(Z8PrimaryMap)other;
            if (stringKeys !is null) return this.opEquals(z8_primary_key_map_view(stringKeys));
        } else {
            auto stringKeys = cast(Z8StringValueMap!V)other;
            if (stringKeys !is null) return this.opEquals(z8_primary_key_map_view(stringKeys));
        }
        auto guids = cast(Z8GuidMap!V)other;
        if (guids !is null) return this.opEquals(guids.entries);
        auto keyCollection = cast(IZ8CollectionView)other;
        if (keyCollection !is null) {
            auto objectKeys = cast(Z8ReferenceKeyMap!(Z8Object, V))keyCollection.z8_erased_storage();
            if (objectKeys !is null) return this.opEquals(objectKeys.entries);
        }
        static foreach (K; AliasSeq!(Z8Date, Z8Datespan, Z8Binary, Z8Geometry, Z8File, Z8Object)) {{
            auto references = cast(Z8ReferenceKeyMap!(K, V))other;
            if (references !is null) return this.opEquals(references.entries);
        }}
        static if (sharePrimary) {
            if (right !is null && (primaryData !is null || right.primaryData !is null))
                return z8_primary_map_view(this).opEquals(z8_primary_map_view(right));
            auto primary = cast(Z8PrimaryKeyMap!Z8Primary)other;
            if (primary !is null) return z8_primary_map_view(this).opEquals(primary);
        }
        static if (is(V == Z8Primary)) {
            auto collection = cast(IZ8CollectionView)other;
            if (collection !is null) {
                auto objects = cast(Z8PrimaryKeyMap!Object)collection.z8_erased_storage();
                if (objects !is null) return this.opEquals(z8_primary_map_view(objects));
            }
            static foreach (U; AliasSeq!(long, bool, Z8Decimal, wstring, Z8Guid, Z8Binary, Z8Datespan, Z8Date, Z8Geometry, Z8File)) {{
                auto typed = cast(Z8PrimaryKeyMap!U)other;
                if (typed !is null) return this.opEquals(z8_primary_map_view(typed));
            }}
        }
        return right !is null && z8_map_content_equals(this, right);
    }
    long z8_size() { static if (sharePrimary) if (primaryData !is null) return primaryData.z8_size(); return cast(long)count; }
    bool z8_isEmpty() { static if (sharePrimary) if (primaryData !is null) return primaryData.z8_isEmpty(); return count == 0; }
    void z8_clear() { static if (sharePrimary) if (primaryData !is null) { primaryData.z8_clear(); return; } buckets.clear(); tokenHashes.clear(); count = 0; nextToken = 0; if (order !is null) order.clear(); }
    bool z8_containsKey(Z8Primary key) {
        static if (sharePrimary) if (primaryData !is null) return primaryData.z8_containsKey(key);
        auto bucket = keyHash(key) in buckets;
        if (bucket !is null) foreach (entry; *bucket) if (keyEquals(key, entry.key)) return true;
        return false;
    }
    bool z8_containsValue(Element value) {
        static if (sharePrimary) if (primaryData !is null) return primaryData.z8_containsValue(z8_array_primary_box(value));
        foreach (bucket; buckets) foreach (entry; bucket) if (valueEquals(value, entry.value)) return true;
        return false;
    }
    Element z8_add(Z8Primary key, Element value) {
        static if (sharePrimary) if (primaryData !is null) return z8_array_primary_cast!V(primaryData.z8_add(key, z8_array_primary_box(value)));
        auto hash = keyHash(key);
        auto bucket = hash in buckets;
        if (bucket !is null) foreach (ref entry; *bucket) if (keyEquals(key, entry.key)) {
            auto previous = entry.value; entry.value = value; return previous;
        }
        if (nextToken == ulong.max) throw new Exception("map entry limit reached");
        auto token = ++nextToken;
        ordering().addWithHash(token, hash); tokenHashes[token] = hash;
        buckets[hash] ~= Entry(key, value, token);
        ++count;
        return Element.init;
    }
    void z8_add_discard(Z8Primary key, Element value) {
        static if (sharePrimary) if (primaryData !is null) { primaryData.z8_add(key, z8_array_primary_box(value)); return; }
        z8_add(key, value);
    }
    void z8_remove_discard(Z8Primary key) {
        static if (sharePrimary) if (primaryData !is null) { primaryData.z8_remove(key); return; }
        z8_remove(key);
    }
    void z8_add(Z8PrimaryKeyMap!V other) {
        z8_nonnull(other);
        static if (sharePrimary) if (primaryData !is null || other.primaryData !is null) {
            z8_primary_map_view(this).z8_add(z8_primary_map_view(other)); return;
        }
        if (other is this) return;
        ordering().preparePutAll(other.count);
        if (other.order !is null) foreach (token; other.order.ordered()) {
            auto entry = other.orderedEntry(token); z8_add(entry.key, entry.value);
        }
    }
    Element z8_get(Z8Primary key) {
        static if (sharePrimary) if (primaryData !is null) return z8_array_primary_cast!V(primaryData.z8_get(key));
        auto bucket = keyHash(key) in buckets;
        if (bucket !is null) foreach (entry; *bucket) if (keyEquals(key, entry.key)) return entry.value;
        return Element.init;
    }
    Element z8_remove(Z8Primary key) {
        static if (sharePrimary) if (primaryData !is null) return z8_array_primary_cast!V(primaryData.z8_remove(key));
        auto hash = keyHash(key);
        auto bucket = hash in buckets;
        if (bucket !is null) foreach (i, entry; *bucket) if (keyEquals(key, entry.key)) {
            auto previous = entry.value;
            ordering().remove(entry.token); tokenHashes.remove(entry.token);
            *bucket = (*bucket)[0 .. i] ~ (*bucket)[i + 1 .. $];
            if (!(*bucket).length) buckets.remove(hash);
            --count;
            return previous;
        }
        return Element.init;
    }
    Z8Array!Z8Primary z8_keys() {
        static if (sharePrimary) if (primaryData !is null) return primaryData.z8_keys();
        auto result = new Z8Array!Z8Primary();
        if (order !is null) foreach (token; order.ordered()) result.z8_add(orderedEntry(token).key);
        return result;
    }
    Z8Array!Z8Primary z8_insertion_keys() {
        static if (sharePrimary) if (primaryData !is null) return primaryData.z8_insertion_keys();
        auto result = new Z8Array!Z8Primary(); bool hasNull;
        if (order !is null) foreach (entry; order.keys) {
            auto key = orderedEntry(entry.key).key;
            if (key is null) hasNull = true; else result.z8_add(key);
        }
        if (hasNull) result.z8_add(null);
        return result;
    }
    Z8ValueArray!V z8_values() {
        static if (sharePrimary) if (primaryData !is null) {
            auto result = new Z8Array!V(); result.primaryData = primaryData.z8_values(); return result;
        }
        auto result = new Z8ValueArray!V();
        if (order !is null) foreach (token; order.ordered()) result.z8_add(orderedEntry(token).value);
        return result;
    }
}
Z8PrimaryKeyMap!V z8_map_union(V)(Z8PrimaryKeyMap!V left, Z8PrimaryKeyMap!V right) {
    auto result = new Z8PrimaryKeyMap!V();
    result.z8_add(z8_nonnull(left)); result.z8_add(z8_nonnull(right));
    return result;
}
void z8_map_add_assign(V)(ref Z8PrimaryKeyMap!V left, lazy Z8PrimaryKeyMap!V previous, lazy Z8PrimaryKeyMap!V right) {
    auto oldValue = previous; auto newValue = right;
    left = z8_map_union(oldValue, newValue);
}

Z8PrimaryKeyMap!Z8Primary z8_primary_map_view(V)(Z8PrimaryKeyMap!V values) if (Z8PrimaryKeyMap!V.sharePrimary) {
    if (values is null) return null;
    if (values.primaryData is null) {
        auto result = new Z8PrimaryKeyMap!Z8Primary();
        foreach (hash, bucket; values.buckets) foreach (entry; bucket)
            result.buckets[hash] ~= typeof(result).Entry(entry.key, z8_array_primary_box(entry.value), entry.token);
        result.count = values.count; result.order = values.order;
        result.tokenHashes = values.tokenHashes; result.nextToken = values.nextToken;
        values.buckets = null; values.order = null; values.tokenHashes = null; values.primaryData = result;
    }
    return values.primaryData;
}
