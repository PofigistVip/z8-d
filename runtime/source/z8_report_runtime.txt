// Scalar properties and query metadata for base.form.report.Report.
private enum z8_report_name = "org.zenframework.z8.server.base.form.report.Report"w;
private __gshared wstring z8_report_birt = "birt"w, z8_report_poi = "poi"w,
    z8_report_pdf = "pdf"w, z8_report_excel = "xls"w, z8_report_word = "doc"w,
    z8_report_wordx = "docx"w, z8_report_html = "html"w, z8_report_powerpoint = "ppt"w;
private class Z8ReportState {
    wstring engine = "birt"w, name, templateName, format;
    IZ8FieldOrderCell ranges;
    Z8IntMap!(Z8Array!wstring) hiddenColumns;
    Z8Object[wstring] sources;
}
private Z8ReportState z8_report_state(Z8Object self) {
    self = z8_nonnull(self);
    if (!z8_native_is_instance(z8_report_name, self)) throw new Exception("invalid Report object");
    auto state = cast(Z8ReportState)self.z8_get_native_state(z8_report_name);
    if (state is null) { state = new Z8ReportState(); self.z8_set_native_state(z8_report_name, state); }
    return state;
}
private auto z8_report_address(string member)(Z8Object self) { return &__traits(getMember, z8_report_state(self), member); }
ref Z8ObjectArray!T z8_report_ranges_ref(T)(Z8Object self) if (is(T : Z8Object)) {
    auto state = z8_report_state(self); auto cell = cast(Z8FieldOrderCell!T)state.ranges;
    if (cell is null) {
        if (state.ranges !is null) throw new Exception("incompatible report ranges array");
        cell = new Z8FieldOrderCell!T(); state.ranges = cell;
    }
    return cell.value;
}
private Z8IntMap!(Z8Array!wstring)* z8_report_hidden_columns(Z8Object self) {
    auto state = z8_report_state(self);
    if (state.hiddenColumns is null) state.hiddenColumns = new Z8IntMap!(Z8Array!wstring)();
    return &state.hiddenColumns;
}
void z8_report_register_data_source(Z8Object self, Z8Object source) {
    auto object = z8_report_source_object(source);
    z8_report_state(self).sources[object.z8_index()] = source;
}
private Z8Object z8_report_get_data_source(Z8Object self, wstring index) {
    auto found = z8_string_required(index) in z8_report_state(self).sources;
    if (found is null) throw new Exception("report data source is not registered");
    return z8_nonnull(*found);
}
private Z8Object z8_report_get_data_source_object(Z8Object self, Z8Object source) {
    return z8_report_get_data_source(self, z8_nonnull(source).z8_index());
}
private wstring* z8_report_constant(string name)() {
    static if (name == "BIRT") return &z8_report_birt;
    else static if (name == "POI") return &z8_report_poi;
    else static if (name == "Pdf") return &z8_report_pdf;
    else static if (name == "Excel") return &z8_report_excel;
    else static if (name == "Word") return &z8_report_word;
    else static if (name == "WordX") return &z8_report_wordx;
    else static if (name == "Html") return &z8_report_html;
    else static if (name == "Powerpoint") return &z8_report_powerpoint;
    else static assert(0, "unknown Report constant");
}
private void z8_report_write_meta(Z8Object report, Z8JsonWriterState writer) {
    auto state = z8_report_state(report);
    writer.start('{');
    z8_monitor_property(writer, "request"w, report.z8_class_id());
    z8_monitor_property(writer, "text"w, report.z8_get_attribute("displayName"w));
    z8_monitor_property(writer, "description"w, report.z8_get_attribute("description"w));
    z8_monitor_property(writer, "icon"w, report.z8_get_attribute("icon"w));
    z8_monitor_property(writer, "url"w, report.z8_get_attribute("url"w));
    z8_monitor_property(writer, "id"w, report.z8_id());
    z8_monitor_property(writer, "format"w, state.format is null ? z8_report_pdf : state.format);
    writer.finish('}');
}
private void z8_report_default_prepare(Z8Object self, Z8Guid record) { z8_nonnull(self); }
// PoiReport can preserve the template and update column visibility.
// Range expansion requires the full spreadsheet engine.
private Z8File z8_report_poi_run(Z8Object report, Z8ReportState state) {
    import std.algorithm.searching : endsWith;
    import std.file : copy, mkdirRecurse, read, write;
    import std.path : buildPath, dirName;
    import std.string : replace, toLower;
    auto templateName = z8_string_required(state.templateName).to!string;
    auto templatePath = templateName.toLower().endsWith(".xlsx") ? templateName : templateName ~ ".xlsx";
    auto source = z8_resources_load(("reports/" ~ templatePath).to!wstring);
    auto backend = z8_connection_current_backend();
    if (backend is null) throw new Exception("report requires a database connection");
    backend.beginTransaction();
    scope(exit) backend.rollback();
    auto input = z8_file_to_binary(source);
    scope(exit) z8_binary_close(input);
    auto workbook = z8_poi_parse(input); // Fail on invalid templates before returning a file.
    int renameSheet; wstring renameName; Z8PoiCellEdit[] edits; Z8PoiCellCopy[] copies; Z8PoiMergedRegion[] addedMerges; Z8PoiCellRemoval[] removals;
    ubyte[] renamedBytes; bool hasRenames;
    if (state.ranges !is null) {
        auto ranges = state.ranges.view().z8_erased_data();
        Z8Object[] selected;
        for (long i; i < ranges.z8_size(); ++i) selected ~= z8_nonnull(cast(Z8Object)ranges.z8_get(i));
        // PoiReport.multiplySheets applies the first requested name for each
        // sheet before opening any range source.
        wstring[] sheetNames = new wstring[workbook.sheets.length];
        bool[] named = new bool[workbook.sheets.length];
        bool[] used = new bool[workbook.sheets.length]; size_t usedCount;
        foreach (range; selected) {
            auto number = range.z8_get_attribute("sheet"w);
            auto index = number is null ? 0 : number.to!int;
            if (index < 0 || index >= workbook.sheets.length) throw new Exception("invalid POI report sheet");
            if (!used[index]) { used[index] = true; ++usedCount; }
            auto requested = range.z8_get_attribute("sheetName"w);
            if (requested !is null && !named[index]) { sheetNames[index] = requested; named[index] = true; hasRenames = true; }
        }
        if (usedCount > 1) foreach (index, selectedSheet; used)
            if (selectedSheet && index >= 16) throw new Exception("POI report sheet ordering is not implemented");
        if (hasRenames) {
            renamedBytes = cast(ubyte[])read(z8_file_system_path(source));
            foreach (index, requested; sheetNames) if (named[index]) {
                renamedBytes = z8_poi_report_rename_sheet(renamedBytes, workbook, cast(int)index, requested);
                workbook.sheets[index].name = requested.length > 31 ? requested[0 .. 31] : requested;
            }
        }
        z8_poi_report_apply_ranges(report, selected, workbook, renameSheet, renameName, edits, copies, addedMerges, removals);
    }
    auto name = state.name is null ? templateName : state.name.to!string;
    if (state.name is null && name.toLower().endsWith(".xlsx")) name = name[0 .. $ - 5];
    name = name.replace("/", "-").replace("\\", "-").replace(":", "-").replace("\n", " ");
    if (name.endsWith(".")) name = name[0 .. $ - 1];
    auto day = z8_date_toString(z8_date_now(), "yyyy.MM.dd"w).to!string;
    auto target = buildPath(z8_file_working_root, "reports", day, z8_guid_to_string(z8_guid_create_random()).to!string, name ~ ".xlsx");
    mkdirRecurse(dirName(target));
    if (!hasRenames && !edits.length && !copies.length && !addedMerges.length && !removals.length && (state.hiddenColumns is null || !state.hiddenColumns.z8_size())) copy(z8_file_system_path(source), target);
    else {
        auto bytes = hasRenames ? renamedBytes : cast(ubyte[])read(z8_file_system_path(source));
        if (edits.length || copies.length || addedMerges.length || removals.length) {
            IZ8PoiProvider provider;
            synchronized (z8_poi_monitor) provider = z8_poi_provider;
            auto writer = cast(IZ8PoiReportWriter)provider;
            if (writer is null) throw new Exception("POI report expressions require the OpenXLSX provider");
            import std.zip : ZipArchive;
            if (!z8_poi_report_openxlsx_safe(new ZipArchive(bytes.dup), workbook.sheets.length, true))
                throw new Exception("POI report expressions are unsupported for this workbook layout");
            if (copies.length) {
                auto expander = cast(IZ8PoiReportExpander)provider;
                if (expander is null) throw new Exception("POI report expansion requires OpenXLSX support");
                bytes = expander.writeExpandedReport(bytes, copies, edits);
            } else if (edits.length) bytes = writer.writeReportCells(bytes, edits);
            if (removals.length) {
                auto remover = cast(IZ8PoiReportCellRemover)provider;
                if (remover is null) throw new Exception("POI report sparse expansion requires cell removal support");
                bytes = remover.removeReportCells(bytes, removals);
            }
            if (addedMerges.length) {
                auto mergeWriter = cast(IZ8PoiReportMergeWriter)provider;
                if (mergeWriter is null) throw new Exception("POI report merged expansion requires OpenXLSX support");
                bytes = mergeWriter.addReportMerges(bytes, addedMerges);
            }
        }
        if (state.hiddenColumns !is null && state.hiddenColumns.z8_size()) bytes = z8_poi_report_hide_columns(bytes, state.hiddenColumns);
        write(target, bytes);
    }
    return z8_file_from_disk(target);
}
private Z8File z8_report_default_execute(Z8Object self, Z8Guid record) {
    auto state = z8_report_state(self);
    if (state.engine == z8_report_poi) return z8_report_poi_run(self, state);
    throw new Exception("report engine is not implemented: " ~ z8_string_or_empty(state.engine).to!string);
}
private Z8File z8_report_run(Z8Object self, Z8Guid record) {
    self = z8_nonnull(self);
    self.z8_report_virtual_prepare(record);
    return self.z8_report_virtual_execute(record);
}
void z8_bind_builtin_report() {
    z8_native_bind_field(z8_report_name ~ ".f_engine"w, &z8_report_address!"engine");
    z8_native_bind_field(z8_report_name ~ ".f_name"w, &z8_report_address!"name");
    z8_native_bind_field(z8_report_name ~ ".f_template"w, &z8_report_address!"templateName");
    z8_native_bind_field(z8_report_name ~ ".f_format"w, &z8_report_address!"format");
    z8_native_bind_field(z8_report_name ~ ".f_hiddenColumns"w, &z8_report_hidden_columns);
    static foreach (name; ["BIRT", "POI", "Pdf", "Excel", "Word", "WordX", "Html", "Powerpoint"])
        z8_native_bind_field(z8_report_name ~ ".f_"w ~ name.to!wstring, &z8_report_constant!name);
    z8_native_bind(z8_report_name ~ ".m_prepare"w, &z8_report_default_prepare);
    z8_native_bind(z8_report_name ~ ".m_execute"w, &z8_report_default_execute);
    z8_native_bind(z8_report_name ~ ".m_run"w, &z8_report_run);
    z8_native_bind(z8_report_name ~ ".m_getDataSource"w, &z8_report_get_data_source);
    z8_native_bind(z8_report_name ~ ".m_getDataSource"w, &z8_report_get_data_source_object);
}
shared static this() { z8_bind_builtin_report(); }
