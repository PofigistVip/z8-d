// ZIP operations for files and binary values.
private ubyte[] z8_zip_path(string source) {
    import std.file : isDir, dirEntries, SpanMode, read;
    import std.path : baseName;
    import std.zip : ZipArchive, ArchiveMember, CompressionMethod;
    auto archive = new ZipArchive();
    void add(string path, string name) {
        if (isDir(path)) {
            foreach (child; dirEntries(path, SpanMode.shallow)) add(child.name, name ~ "/" ~ baseName(child.name));
        } else {
            auto entry = new ArchiveMember();
            entry.name = name; entry.flags = 0x800;
            entry.expandedData = cast(ubyte[])read(path);
            entry.compressionMethod = CompressionMethod.deflate;
            archive.addMember(entry);
        }
    }
    if (isDir(source)) {
        foreach (child; dirEntries(source, SpanMode.shallow)) add(child.name, baseName(child.name));
    } else add(source, baseName(source));
    return cast(ubyte[])archive.build();
}

private void z8_zip_extract(ubyte[] bytes, string directory, Z8Encoding encoding) {
    import std.file : mkdirRecurse, exists, isSymlink, write;
    import std.path : buildPath, dirName, isAbsolute;
    import std.string : replace, split;
    import std.zip : ZipArchive;
    z8_nonnull(encoding);
    auto archive = new ZipArchive(bytes);
    foreach (name, entry; archive.directory) {
        auto entryEncoding = (entry.flags & 0x800) ? z8_encoding_by_id(4) : z8_nonnull(encoding);
        auto normalized = z8_encoding_decode(cast(const(ubyte)[])name, entryEncoding).to!string.replace("\\", "/");
        if (isAbsolute(normalized) || normalized.canFind(':')) throw new Exception("invalid archive path");
        auto path = directory;
        auto parts = normalized.split("/");
        foreach (part; parts) {
            if (part == "..") throw new Exception("invalid archive path");
            if (!part.length || part == ".") continue;
            path = buildPath(path, part);
            if (exists(path) && isSymlink(path)) throw new Exception("archive path is a symbolic link");
        }
        if (normalized.length && normalized[$ - 1] == '/') mkdirRecurse(path);
        else {
            auto parent = dirName(path);
            if (parent.length) mkdirRecurse(parent);
            archive.expand(entry);
            write(path, entry.expandedData);
        }
    }
}

Z8File z8_file_zip(Z8File destination, Z8File source) {
    import std.file : mkdirRecurse, write;
    import std.path : dirName;
    auto bytes = z8_zip_path(z8_file_system_path(source));
    auto path = z8_file_system_path(destination);
    auto parent = dirName(path);
    if (parent.length) mkdirRecurse(parent);
    write(path, bytes);
    return destination;
}
Z8File z8_file_unzip(Z8File archive, Z8File directory) {
    return z8_file_unzip(archive, directory, z8_encoding_by_id(4));
}
Z8File z8_file_unzip(Z8File archive, Z8File directory, Z8Encoding encoding) {
    // A request upload (or a database-backed file) can have content without a
    // local path. Use the file's input stream just as file.java does.
    z8_zip_extract(z8_binary_read_all(z8_file_to_binary(archive)), z8_file_system_path(directory), encoding);
    return archive;
}

private Z8Binary z8_binary_zip_result(ubyte[] data) {
    auto temporary = z8_file_createTempFile(""w, ""w);
    z8_file_write_bytes(temporary, data, false);
    return z8_file_to_binary(temporary);
}
Z8Binary z8_binary_zip(Z8File source) { return z8_binary_zip_result(z8_zip_path(z8_file_system_path(source))); }
Z8Binary z8_binary_zip(Z8Array!Z8Binary values) {
    import std.zip : ZipArchive, ArchiveMember, CompressionMethod;
    z8_nonnull(values);
    auto archive = new ZipArchive();
    foreach (i; 0 .. values.z8_size()) {
        auto value = z8_nonnull(values.z8_get(i));
        auto entry = new ArchiveMember();
        entry.name = z8_string_required(value.name).to!string;
        if (entry.name in archive.directory) throw new Exception("duplicate archive entry");
        entry.flags = 0x800;
        entry.expandedData = z8_binary_read_all(value);
        entry.compressionMethod = CompressionMethod.deflate;
        archive.addMember(entry);
    }
    return z8_binary_zip_result(cast(ubyte[])archive.build());
}
void z8_binary_unzip(Z8Binary archive, Z8File directory) {
    z8_binary_unzip(archive, directory, z8_encoding_by_id(4));
}
void z8_binary_unzip(Z8Binary archive, Z8File directory, Z8Encoding encoding) {
    z8_nonnull(archive);
    z8_zip_extract(z8_binary_read_all(archive), z8_file_system_path(directory), encoding);
}
