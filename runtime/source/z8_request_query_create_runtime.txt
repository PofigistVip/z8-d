// Transactional subset of CreateAction for direct table records.
void z8_application_dispatch_query_create(wstring className) {
    z8_application_dispatch_query_create_target(className, null);
}
private void z8_application_dispatch_query_create_target(wstring className, Z8Object query) {
    auto request = z8_application_require_request();
    auto session = z8_nonnull(request.session);
    auto params = z8_nonnull(request.parameters);
    if (request.writer.scopes.length != 2) throw new Exception("response object is not open");
    if (params.z8_get("action"w) != "create"w) throw new Exception("unsupported query action");
    if (!z8_application_may_execute(session, className, "create"w)) throw new Exception("request access denied");
    auto keys = params.z8_keys();
    for (long i; i < keys.z8_size(); ++i) {
        auto name = keys.z8_get(i);
        if (name is null) throw new Exception("invalid query create parameter");
    }
    if (query is null) query = z8_object_new_instance_native(z8_string_required(className));
    if (!z8_native_is_instance("org.zenframework.z8.server.base.table.Table"w, query))
        throw new Exception("query create requires a Table");
    auto privileges = z8_nonnull(session.user).privileges;
    if (privileges is null) throw new Exception("query create access denied");
    auto context = query;
    query = z8_application_action_query(context, privileges, params.z8_get("query"w), params.z8_get("link"w));
    auto queryKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, query);
    if (!privileges.getTableAccess(queryKey).create) throw new Exception("query create access denied");
    auto key = z8_query_primary_key(query);
    if (key is null) throw new Exception("query has no primary key");
    auto selected = z8_application_query_fields(query, privileges, params);
    z8_application_query_context_values(query, privileges, params, false, context);
    auto json = params.z8_get("data"w);
    if (json is null || !json.length) throw new Exception("query create requires records");
    auto records = z8_json_parse(json.to!string);
    if (records.type != JSONType.array || !records.array.length) throw new Exception("query create requires records");
    auto connection = z8_connection_current_backend();
    if (connection is null) throw new Exception("no current database connection");
    Z8Guid[] ids;
    connection.beginTransaction();
    scope(failure) connection.rollback();
    foreach (record; records.array) {
        if (record.type != JSONType.object) throw new Exception("invalid query create record");
        Z8Guid id;
        auto member = key.z8_id().to!string in record.object;
        if (member !is null && member.type != JSONType.null_) {
            if (member.type != JSONType.string) throw new Exception("invalid query create record id");
            if (member.str.length) id = z8_guid_parse(member.str.to!wstring);
        }
        if (id is null || id == z8_guid_zero()) id = z8_guid_create_random();
        z8_query_on_new(query);
        foreach (name, value; record.object) {
            auto field = z8_query_try_find_field_id(query, name.to!wstring);
            if (field is null) continue;
            if (field is key) continue;
            if (field.z8_get_owner() !is query || z8_native_is_instance("org.zenframework.z8.server.base.table.value.Expression"w, field))
                throw new Exception("linked and computed fields are not supported by query create");
            z8_application_query_set_value(field, value, true);
        }
        z8_field_set(key, z8_primary_box(id));
        if (z8_query_user_events_enabled()) query.z8_query_virtual_onCreateAction(id);
        z8_query_insert_id(query, id);
        ids ~= id;
    }
    connection.commit();
    z8_application_query_write_affected(query, selected, ids, request.writer, privileges, params, context);
}
