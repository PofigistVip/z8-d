private import std.stdio : File;
class Z8Binary {
    ubyte[] data;
    size_t position;
    private Z8BinaryFileInput fileInput;
    wstring name = ""w;
    this(ubyte[] data = null) { this.data = data; }
}
private class Z8BinaryFileInput {
    import std.stdio : File;
    File stream;
    bool closed;
    this(string path) { stream = File(path, "rb"); }
    void close() { if (!closed) { closed = true; stream.close(); } }
}
Z8Binary z8_binary_from_file(string path) {
    auto result = new Z8Binary();
    result.fileInput = new Z8BinaryFileInput(path);
    return result;
}
ubyte[] z8_binary_read_chunk(Z8Binary value, ubyte[] buffer) {
    z8_nonnull(value);
    if (value.fileInput !is null) {
        if (value.fileInput.closed) throw new Exception("binary stream is closed");
        return value.fileInput.stream.rawRead(buffer);
    }
    auto remaining = value.data.length - value.position;
    auto count = remaining < buffer.length ? remaining : buffer.length;
    buffer[0 .. count] = value.data[value.position .. value.position + count];
    value.position += count;
    return buffer[0 .. count];
}
private void z8_binary_finish_read(Z8Binary value) {
    if (value.fileInput !is null) value.fileInput.close();
}
// HTTP hosts can stream binary content without loading a file into memory.
void z8_binary_close(Z8Binary value) { z8_binary_finish_read(z8_nonnull(value)); }
ulong z8_binary_remaining(Z8Binary value) {
    z8_nonnull(value);
    if (value.fileInput !is null) {
        if (value.fileInput.closed) throw new Exception("binary stream is closed");
        auto size = value.fileInput.stream.size, position = value.fileInput.stream.tell;
        return size > position ? size - position : 0;
    }
    return value.data.length - value.position;
}
private ubyte[] z8_binary_read_all(Z8Binary value) {
    z8_nonnull(value);
    scope(exit) z8_binary_finish_read(value);
    ubyte[] result;
    ubyte[16384] buffer;
    for (;;) {
        auto chunk = z8_binary_read_chunk(value, buffer[]);
        if (!chunk.length) break;
        result ~= chunk;
    }
    return result;
}
private void z8_binary_copy_to(Z8Binary value, ref File output) {
    z8_nonnull(value);
    scope(exit) z8_binary_finish_read(value);
    ubyte[16384] buffer;
    for (;;) {
        auto chunk = z8_binary_read_chunk(value, buffer[]);
        if (!chunk.length) break;
        output.rawWrite(chunk);
    }
}
Z8Binary z8_binary_from_string(wstring value) {
    return new Z8Binary(z8_encoding_encode(value, z8_encoding_by_id(0)));
}

Z8Binary z8_binary_literal(string text) {
    import std.string : split, strip;
    ubyte[] bytes;
    foreach (part; text.split(",")) {
        auto value = part.strip();
        auto number = value.length > 2 && (value[0..2] == "0x" || value[0..2] == "0X") ?
            value[2..$].to!uint(16) : value.to!uint;
        if (number > 255) throw new Exception("invalid binary literal");
        bytes ~= cast(ubyte)number;
    }
    return new Z8Binary(bytes);
}

Z8Binary z8_binary_set_name(Z8Binary value, wstring name) { z8_nonnull(value).name = name; return value; }
wstring z8_binary_get_name(Z8Binary value) { return z8_nonnull(value).name; }
wstring z8_binary_to_string(Z8Binary value) { return z8_string_required(z8_binary_get_name(value)); }
long z8_binary_length(Z8Binary value) {
    z8_nonnull(value);
    if (value.fileInput !is null) {
        if (value.fileInput.closed) throw new Exception("binary stream is closed");
        auto size = value.fileInput.stream.size, position = value.fileInput.stream.tell;
        auto remaining = size > position ? size - position : 0;
        return remaining > int.max ? int.max : cast(long)remaining;
    }
    return cast(long)(value.data.length - value.position);
}
long z8_binary_hash(Z8Binary value) { return cast(long)cast(int)cast(uint)z8_nonnull(value).toHash(); }

private size_t z8_binary_name_start(wstring name) {
    size_t start;
    foreach (i, wchar c; name) if (c == '/' || c == '\\') start = i + 1;
    return start;
}
private size_t z8_binary_extension_start(wstring name, size_t start) {
    auto dot = name.length;
    foreach (i, wchar c; name) if (i >= start && c == '.') dot = i;
    return dot;
}
wstring z8_binary_base_name(Z8Binary value) {
    auto name = z8_binary_to_string(value);
    auto start = z8_binary_name_start(name);
    auto dot = z8_binary_extension_start(name, start);
    return name[start .. dot];
}
wstring z8_binary_extension(Z8Binary value) {
    import std.uni : toLower;
    import std.utf : encode;
    auto name = z8_binary_to_string(value);
    auto dot = z8_binary_extension_start(name, z8_binary_name_start(name));
    if (dot == name.length) return ""w;
    wchar[] lower;
    foreach (dchar c; name[dot + 1 .. $]) encode(lower, toLower(c));
    return lower.length ? lower.idup : ""w;
}
