// ServerConfig values used by native Application methods. Extensions are kept
// in their original categories and returned in Java's order, after the PDF type.
class Z8ApplicationConfig {
    bool multitenant;
    int uploadMaxMiB = 5, downloadMaxMiB = 1;
    wstring[] textExtensions = ["txt"w, "xml"w];
    wstring[] imageExtensions = ["tif"w, "tiff"w, "jpg"w, "jpeg"w, "gif"w, "png"w, "bmp"w];
    wstring[] emailExtensions = ["eml"w, "mime"w];
    wstring[] officeExtensions = ["doc"w, "docx"w, "rtf"w, "xls"w, "xlsx"w, "ppt"w, "pptx"w,
        "odt"w, "odp"w, "ods"w, "odf"w, "odg"w, "wpd"w, "sxw"w, "sxi"w, "sxc"w, "sxd"w, "stw"w, "vsd"w];
    Z8ApplicationConfig copy() {
        auto result = new Z8ApplicationConfig();
        result.multitenant = multitenant; result.uploadMaxMiB = uploadMaxMiB; result.downloadMaxMiB = downloadMaxMiB;
        result.textExtensions = textExtensions.dup; result.imageExtensions = imageExtensions.dup;
        result.emailExtensions = emailExtensions.dup; result.officeExtensions = officeExtensions.dup;
        return result;
    }
}
private __gshared Z8ApplicationConfig z8_application_config;
private __gshared Object z8_application_config_lock;
shared static this() {
    z8_application_config_lock = new Object();
    z8_application_config = new Z8ApplicationConfig();
}
void z8_application_configure(Z8ApplicationConfig config) {
    auto snapshot = z8_nonnull(config).copy();
    synchronized (z8_application_config_lock) z8_application_config = snapshot;
}
Z8ApplicationConfig z8_application_configuration() {
    synchronized (z8_application_config_lock) return z8_application_config.copy();
}
// The embedding server chooses how messages, log events and files are stored.
interface IZ8ApplicationMonitor {
    void message(Z8MonitorLevel level, wstring text);
    void log(Z8MonitorLevel level, wstring text);
    void print(Z8File value);
}
struct Z8MonitorEntry {
    Z8MonitorLevel level;
    wstring text, source;
    Z8Date time;
}
// Request-local monitor; a host can persist logs and publish its captured
// messages and files using the surrounding application's response protocol.
class Z8RequestMonitor : IZ8ApplicationMonitor {
    private Z8MonitorEntry[] messageEntries, logEntries;
    private Z8File[] printedFiles;
    private bool logHasErrors;
    wstring source;
    this(wstring source = null) { this.source = source; }
    void message(Z8MonitorLevel level, wstring text) {
        messageEntries ~= Z8MonitorEntry(level, text, source, z8_date_now());
    }
    void log(Z8MonitorLevel level, wstring text) {
        logEntries ~= Z8MonitorEntry(level, text, source, z8_date_now());
        if (level == Z8MonitorLevel.error) logHasErrors = true;
    }
    void print(Z8File value) { printedFiles ~= value; }
    Z8MonitorEntry[] messages() { return messageEntries.dup; }
    Z8MonitorEntry[] logs() { return logEntries.dup; }
    Z8File[] files() { return printedFiles.dup; }
    bool hasErrors() { return logHasErrors; }
    wstring infoJson(wstring server = null) { return z8_monitor_info_json(messageEntries, printedFiles, server); }
}
private void z8_monitor_property(T)(Z8JsonWriterState writer, wstring key, T value) {
    writer.property(key);
    z8_json_writer_primary(writer, z8_primary_box(value));
}
// Matches JsonWriter.writeInfo(messages, files, server) from the original Z8.
wstring z8_monitor_info_json(Z8MonitorEntry[] messages, Z8File[] files, wstring server = null) {
    auto writer = new Z8JsonWriterState();
    writer.start('{');
    writer.start('[', "messages"w);
    foreach (entry; messages) {
        writer.start('{');
        z8_monitor_property(writer, "text"w, entry.text);
        z8_monitor_property(writer, "source"w, entry.source);
        z8_monitor_property(writer, "time"w, entry.time);
        wstring level;
        final switch (entry.level) {
            case Z8MonitorLevel.info: level = "info"w; break;
            case Z8MonitorLevel.warning: level = "warning"w; break;
            case Z8MonitorLevel.error: level = "error"w; break;
        }
        z8_monitor_property(writer, "type"w, level);
        writer.finish('}');
    }
    writer.finish(']');
    writer.start('[', "files"w);
    foreach (file; files) {
        z8_nonnull(file);
        writer.start('{');
        z8_monitor_property(writer, "name"w, file.name);
        z8_monitor_property(writer, "time"w, file.time);
        z8_monitor_property(writer, "size"w, file.size);
        z8_monitor_property(writer, "path"w, file.path);
        z8_monitor_property(writer, "id"w, file.id);
        z8_monitor_property(writer, "user"w, file.user);
        z8_monitor_property(writer, "author"w, file.author);
        writer.start('{', "details"w);
        auto details = z8_nonnull(file.details);
        if (details.hasNullKey) throw new Exception("null file detail key");
        foreach (key; details.insertionKeys) z8_monitor_property(writer, key, details.entries[key]);
        writer.finish('}'); writer.finish('}');
    }
    writer.finish(']');
    if (files.length) z8_monitor_property(writer, "server"w, server);
    writer.finish('}');
    return writer.stream;
}

// The host supplies an authenticated identity and the database's system user.
// As in Session.java, restoreUser returns to setUser's identity, not a stack.
class Z8ApplicationSession {
    wstring id;
    Z8UserData user, system;
    private Z8UserData originalUser;
    long lastAccessTime;
    this(wstring id, Z8UserData user, Z8UserData system) {
        this.id = id; this.system = system; setUser(user);
    }
    void setUser(Z8UserData value) { user = value; originalUser = value; touch(); }
    void switchUser(Z8UserData value) { user = value; }
    void restoreUser() { user = originalUser; }
    void touch() { lastAccessTime = z8_optional_require(z8_application_current_time()); }
}
// Event levels have the same ordering and nested semantics as IRequest.java.
enum Z8EventsLevel { all, user, system, none }
// The host supplies a request for the current thread; response wrappers share
// the request's writer, as ApplicationServer.getRequest() does in Z8.
class Z8ApplicationRequest {
    Z8StringMap parameters;
    Z8Array!Z8File files;
    IZ8ApplicationMonitor monitor;
    Z8ApplicationSession session;
    private Z8EventsLevel[] eventLevels;
    void setEventsLevel(Z8EventsLevel level) { eventLevels ~= level; }
    void restoreEventsLevel() { if (eventLevels.length) eventLevels.length--; }
    Z8EventsLevel eventsLevel() { return eventLevels.length ? eventLevels[$ - 1] : Z8EventsLevel.all; }
    bool eventsEnabled(Z8EventsLevel level) { return eventsLevel() <= level; }
    private Z8JsonWriterState writer;
    private Z8Binary binaryResponse;
    private bool hasBinaryResponse;
    this() {
        parameters = new Z8StringMap();
        files = new Z8Array!Z8File();
        writer = new Z8JsonWriterState();
    }
}
// The server can obtain the accumulated response without a generated BL type.
wstring z8_application_response_text(Z8ApplicationRequest request) {
    return z8_nonnull(request).writer.stream;
}
void z8_application_set_response_binary(Z8ApplicationRequest request, Z8Binary content) {
    request = z8_nonnull(request);
    request.binaryResponse = content; request.hasBinaryResponse = true;
}
bool z8_application_has_binary_response(Z8ApplicationRequest request) { return z8_nonnull(request).hasBinaryResponse; }
Z8Binary z8_application_response_binary(Z8ApplicationRequest request) { return z8_nonnull(request).binaryResponse; }
// Host-side equivalents of RequestTarget.processRequest and its error path.
// BL handlers write properties into the open response object.
void z8_application_start_response(Z8ApplicationRequest request, wstring requestId, bool success = true) {
    auto writer = z8_nonnull(request).writer;
    if (writer.stream.length || writer.scopes.length != 1) throw new Exception("response already started");
    writer.start('{');
    z8_monitor_property(writer, "request"w, requestId);
    z8_monitor_property(writer, "success"w, success);
}
private void z8_application_response_info(Z8JsonWriterState writer, Z8RequestMonitor monitor, wstring server) {
    z8_monitor_property(writer, "server"w, server);
    writer.property("info"w);
    writer.raw(z8_nonnull(monitor).infoJson(server));
}
void z8_application_finish_response(Z8ApplicationRequest request, Z8RequestMonitor monitor, wstring server = null) {
    auto writer = z8_nonnull(request).writer;
    if (writer.scopes.length != 2) throw new Exception("unbalanced response writer");
    z8_application_response_info(writer, monitor, server);
    writer.finish('}');
}
wstring z8_application_error_response(wstring requestId, Z8RequestMonitor monitor, wstring server = null) {
    auto request = new Z8ApplicationRequest();
    z8_application_start_response(request, requestId, false);
    request.writer.start('[', "data"w); request.writer.finish(']');
    z8_application_finish_response(request, monitor, server);
    return request.writer.stream;
}
// Dispatch ordinary OBJECT responses; Query and job requests use other adapters.
void z8_application_dispatch_object(wstring className,
        bool delegate(Z8ApplicationSession session, wstring className, wstring action) mayExecute) {
    z8_application_dispatch_object_target(className, mayExecute, null);
}
private void z8_application_dispatch_object_target(wstring className,
        bool delegate(Z8ApplicationSession session, wstring className, wstring action) mayExecute, Z8Object target) {
    auto request = z8_application_require_request();
    auto session = z8_nonnull(request.session);
    className = z8_string_required(className);
    auto writer = request.writer;
    if (writer.scopes.length != 2) throw new Exception("response object is not open");
    auto action = request.parameters.z8_get("action"w);
    if (mayExecute is null || !mayExecute(session, className, action)) throw new Exception("request access denied");
    if (target is null) target = z8_object_new_instance_native(className);
    if (z8_native_is_instance("org.zenframework.z8.server.base.query.Query"w, target) ||
        z8_native_is_instance("org.zenframework.z8.server.base.Executable"w, target))
        throw new Exception("request target requires a specialized adapter");
    auto monitor = cast(Z8RequestMonitor)request.monitor;
    if (monitor !is null) monitor.source = target.z8_get_attribute("displayName"w);
    if (action == "content"w) {
        z8_application_set_response_binary(request, target.z8_object_process_content(request.parameters, request.files));
        return;
    }
    auto data = target.z8_object_get_data(request.parameters);
    writer.property("data"w);
    if (data is null) writer.raw("null"w);
    else {
        if (!z8_native_is_instance(z8_json_array_name, data)) throw new Exception("invalid object response data");
        writer.raw(z8_json_array_render_state(z8_json_array_state(data)));
    }
    z8_monitor_property(writer, "ui"w, target.z8_get_attribute("ui"w));
}
// The original RequestDispatcher checks request classIdKey before creating
// the target. A missing session or privilege snapshot denies access.
bool z8_application_may_execute(Z8ApplicationSession session, wstring className, wstring action) {
    if (session is null || session.user is null || session.user.privileges is null || className is null || !className.length) return false;
    return session.user.privileges.getRequestAccess(z8_guid_create_name(className)).execute;
}
void z8_application_dispatch_authorized_object(wstring className) {
    z8_application_dispatch_object(className, (Z8ApplicationSession session, wstring name, wstring action) {
        return z8_application_may_execute(session, name, action);
    });
}
// Dispatch a request using the session's snapshot without constructing the BL
// class twice. The host still owns the active database connection.
void z8_application_dispatch_request() {
    auto request = z8_application_require_request();
    auto className = z8_string_required(request.parameters.z8_get("request"w));
    auto action = request.parameters.z8_get("action"w);
    if (request.writer.scopes.length != 2) throw new Exception("response object is not open");
    if (!z8_application_may_execute(request.session, className, action)) throw new Exception("request access denied");
    auto target = z8_object_new_instance_native(className);
    if (z8_native_is_instance("org.zenframework.z8.server.base.query.Query"w, target)) {
        if (action == "destroy"w) z8_application_dispatch_query_destroy_target(className, target);
        else if (action == "update"w) z8_application_dispatch_query_update_target(className, target);
        else if (action == "create"w) z8_application_dispatch_query_create_target(className, target);
        else if (action == "copy"w) z8_application_dispatch_query_copy_target(className, target);
        else z8_application_dispatch_query_read_target(className, target);
    }
    else z8_application_dispatch_object_target(className,
        (Z8ApplicationSession session, wstring name, wstring selectedAction) {
            return z8_application_may_execute(session, name, selectedAction);
        }, target);
}
private Z8ApplicationRequest z8_application_current_request;
Z8ApplicationRequest z8_application_request() { return z8_application_current_request; }
// Returns the previous context so nested host operations can restore it.
Z8ApplicationRequest z8_application_set_request(Z8ApplicationRequest value) {
    auto previous = z8_application_current_request;
    z8_application_current_request = value;
    return previous;
}
private Z8ApplicationRequest z8_application_require_request() {
    auto value = z8_application_current_request;
    if (value is null) throw new Exception("no active application request");
    return value;
}
private Z8Optional!bool z8_application_is_request() { return z8_optional_wrap(z8_application_current_request !is null); }
private Z8ApplicationSession z8_application_session() { return z8_nonnull(z8_application_require_request().session); }
private Z8Object z8_application_user() { return z8_user_wrap(z8_application_session().user); }
private Z8Object z8_application_system() { return z8_user_wrap(z8_application_session().system); }
private Z8Guid z8_application_user_id() { return z8_nonnull(z8_application_session().user).id; }
private Z8Guid z8_application_system_id() { return z8_nonnull(z8_application_session().system).id; }
private void z8_application_switch_user(Z8Object user) { z8_application_session().switchUser(z8_user_data(user)); }
private void z8_application_restore_user() { z8_application_session().restoreUser(); }
private Z8Optional!bool z8_application_is_multitenant() { return z8_optional_wrap(z8_application_configuration().multitenant); }
private Z8Optional!long z8_application_max_upload() {
    return z8_optional_wrap(cast(long)cast(int)(z8_application_configuration().uploadMaxMiB * 1_048_576));
}
private Z8Optional!long z8_application_max_download() {
    return z8_optional_wrap(cast(long)cast(int)(z8_application_configuration().downloadMaxMiB * 1_048_576));
}
private Z8Array!wstring z8_application_supported_file_types() {
    auto config = z8_application_configuration();
    auto result = new Z8Array!wstring();
    result.z8_add("pdf"w);
    foreach (extensions; [config.textExtensions, config.imageExtensions, config.emailExtensions, config.officeExtensions])
        foreach (extension; extensions) result.z8_add(z8_string_new(extension));
    return result;
}
private Z8Optional!long z8_application_current_time() {
    return z8_optional_wrap((Clock.currTime(UTC()) - SysTime.fromUnixTime(0, UTC())).total!"msecs");
}
private Z8StringMap z8_application_request_parameters() { return z8_nonnull(z8_application_require_request().parameters); }
private Z8Array!Z8File z8_application_request_files() {
    auto source = z8_nonnull(z8_application_require_request().files);
    auto result = new Z8Array!Z8File(); result.z8_addAll(source); return result;
}
private Z8Object z8_application_response_writer() {
    auto request = z8_application_require_request();
    auto wrapper = z8_native_new(z8_json_writer_name);
    z8_nonnull(wrapper).z8_set_native_state(z8_json_writer_name, request.writer);
    return wrapper;
}
private void z8_application_message(Z8MonitorLevel level)(wstring text) {
    auto monitor = z8_application_require_request().monitor;
    if (monitor !is null) monitor.message(level, z8_string_required(text));
}
private void z8_application_log(Z8MonitorLevel level)(wstring text) {
    auto monitor = z8_application_require_request().monitor;
    z8_nonnull(monitor).log(level, z8_string_required(text));
}
private void z8_application_print(Z8File value) {
    auto monitor = z8_application_require_request().monitor;
    z8_nonnull(monitor).print(value);
}
private void z8_application_redirect(wstring location) { throw new Z8Redirect(z8_string_required(location)); }
void z8_bind_builtin_application_request() {
    enum prefix = "org.zenframework.z8.server.base.application.Application."w;
    z8_native_bind(prefix ~ "m_isRequest"w, &z8_application_is_request);
    z8_native_bind(prefix ~ "m_user"w, &z8_application_user);
    z8_native_bind(prefix ~ "m_system"w, &z8_application_system);
    z8_native_bind(prefix ~ "m_userId"w, &z8_application_user_id);
    z8_native_bind(prefix ~ "m_systemId"w, &z8_application_system_id);
    z8_native_bind(prefix ~ "m_switchUser"w, &z8_application_switch_user);
    z8_native_bind(prefix ~ "m_restoreUser"w, &z8_application_restore_user);
    z8_native_bind(prefix ~ "m_isMultitenant"w, &z8_application_is_multitenant);
    z8_native_bind(prefix ~ "m_maxUploadSize"w, &z8_application_max_upload);
    z8_native_bind(prefix ~ "m_maxDownloadSize"w, &z8_application_max_download);
    z8_native_bind(prefix ~ "m_supportedFileTypes"w, &z8_application_supported_file_types);
    z8_native_bind(prefix ~ "m_currentTimeMillis"w, &z8_application_current_time);
    z8_native_bind(prefix ~ "m_requestParameters"w, &z8_application_request_parameters);
    z8_native_bind(prefix ~ "m_requestFiles"w, &z8_application_request_files);
    z8_native_bind(prefix ~ "m_responseWriter"w, &z8_application_response_writer);
    z8_native_bind(prefix ~ "m_info"w, &z8_application_message!(Z8MonitorLevel.info));
    z8_native_bind(prefix ~ "m_warning"w, &z8_application_message!(Z8MonitorLevel.warning));
    z8_native_bind(prefix ~ "m_error"w, &z8_application_message!(Z8MonitorLevel.error));
    z8_native_bind(prefix ~ "m_logInfo"w, &z8_application_log!(Z8MonitorLevel.info));
    z8_native_bind(prefix ~ "m_logWarning"w, &z8_application_log!(Z8MonitorLevel.warning));
    z8_native_bind(prefix ~ "m_logError"w, &z8_application_log!(Z8MonitorLevel.error));
    z8_native_bind(prefix ~ "m_print"w, &z8_application_print);
    z8_native_bind(prefix ~ "m_redirect"w, &z8_application_redirect);
}
