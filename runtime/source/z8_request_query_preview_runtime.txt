// PreviewAction uses the existing converter and a PDF merger for multiple files.
private void z8_application_dispatch_query_preview_target(wstring className, Z8Object context) {
    import std.file : exists, mkdir, rmdirRecurse, tempDir;
    import std.path : absolutePath, buildPath;
    import std.process : pipeProcess, wait, Redirect, Config;
    import std.uuid : randomUUID;
    auto request = z8_application_require_request();
    auto session = z8_nonnull(request.session);
    if (!z8_application_may_execute(session, className, "preview"w)) throw new Exception("request access denied");
    auto privileges = z8_nonnull(z8_nonnull(session.user).privileges);
    auto params = request.parameters;
    auto query = z8_application_action_query(context, privileges, params.z8_get("query"w), params.z8_get("link"w));
    auto queryKey = z8_native_call!Z8Guid("org.zenframework.z8.server.runtime.OBJECT.m_key"w, query);
    if (!privileges.getTableAccess(queryKey).read) throw new Exception("preview access denied");
    auto fieldId = params.z8_get("field"w);
    if (fieldId is null || !fieldId.length) throw new Exception("preview field is required");
    auto field = z8_query_find_field_id(query, fieldId);
    if (!z8_application_query_field_readable(query, field, privileges)) throw new Exception("preview field access denied");
    if (!["file"w, "attachments"w, "string"w, "text"w].canFind(z8_field_type_id(field)))
        throw new Exception("preview requires a file field");
    auto record = params.z8_get("recordId"w);
    if (record is null || !record.length) throw new Exception("preview record ID is required");
    auto id = z8_guid_parse(record);
    if (id is null || id == z8_guid_zero()) throw new Exception("preview record ID is required");
    Z8Array!Z8File files;
    {
        auto saved = new Z8QuerySavedState(query, [field]); scope(exit) saved.restore();
        auto fields = new Z8ObjectArray!Z8Object(); fields.z8_add(field);
        if (!z8_optional_require(z8_query_read_record(query, id, fields))) throw new Exception("preview record was not found");
        files = z8_file_parse(z8_string_or_empty(z8_primary_cast!wstring(z8_field_primary(field))));
    }
    if (!files.z8_size()) throw new Exception("preview attachment was not found");
    if (files.z8_size() > 1000) throw new Exception("too many preview attachments");
    string temporary;
    scope(exit) if (temporary.length && exists(temporary)) rmdirRecurse(temporary);
    void ensureTemporary() {
        if (temporary.length) return;
        temporary = buildPath(absolutePath(tempDir()), "z8-preview-" ~ randomUUID().toString());
        mkdir(temporary);
    }
    string[] inputs;
    wstring[] unsupported;
    foreach (i; 0 .. files.z8_size()) {
        auto metadata = z8_nonnull(files.z8_get(i));
        auto name = z8_string_required(metadata.name);
        auto extension = z8_file_converter_extension(name);
        if (!z8_optional_require(z8_file_converter_convertible(extension))) {
            auto values = new Z8Array!Z8Primary(); values.z8_add(z8_primary_box(name));
            unsupported ~= z8_string_format(z8_string_new("$Preview.unsupportedFileFormat$"w), values);
            continue;
        }
        auto file = z8_system_files_get_id(metadata.id);
        if (file is null) throw new Exception("preview attachment was not found");
        file.name = name;
        file = z8_system_files_get_file(file);
        if (!z8_optional_require(z8_file_converter_pdf(extension)) ||
                params.z8_containsKey("background"w) || params.z8_containsKey("stamps"w)) {
            ensureTemporary();
            auto target = z8_file_from_path(buildPath(temporary, "attachment-" ~ i.to!string ~ ".pdf").to!wstring);
            file = z8_file_converter_files(file, target, params);
        }
        if (z8_attachment_pdf_page_count(file) < 1) throw new Exception("preview PDF is invalid");
        inputs ~= z8_file_system_path(file);
    }
    if (unsupported.length) {
        ensureTemporary();
        auto commentPath = buildPath(temporary, "unsupported.pdf");
        ubyte[] font;
        synchronized (z8_pdf_font_lock) font = z8_pdf_font_data;
        if (font !is null) {
            import std.file : write;
            import std.utf : toUTF16;
            wstring[] lines;
            foreach (message; unsupported) {
                wstring line;
                foreach (dchar ch; message) {
                    if (line.length >= 70) { lines ~= line; line = null; }
                    line ~= toUTF16([ch]);
                }
                lines ~= line;
            }
            wstring json = "["w;
            foreach (i, line; lines) {
                if (i) json ~= ","w;
                json ~= "{\"comment\":"w ~ z8_json_quote(line) ~ "}"w;
            }
            json ~= "]"w;
            auto rows = z8_json_parse(json.to!string);
            write(commentPath, z8_export_pdf(["comment"w], [""w], ["string"w], rows, null, [100], [0]));
        } else {
            import std.array : join;
            auto text = z8_binary_from_string("\uFEFF"w ~ unsupported.join(" "w));
            scope(exit) z8_binary_close(text);
            z8_file_converter_office_convert(text, "txt"w, commentPath, null);
        }
        inputs ~= commentPath;
    }
    string output;
    if (inputs.length == 1) output = inputs[0];
    else {
        ensureTemporary();
        output = buildPath(temporary, "preview.pdf");
        auto configured = z8_server_config_value("pdf.merge.executable"w);
        auto executable = configured is null || !configured.length ? "pdfunite" : configured.to!string;
        enum size_t maxOutput = 1024 * 1024;
        auto process = pipeProcess([executable] ~ inputs ~ [output],
            Redirect.stdin | Redirect.stdout | Redirect.stderrToStdout, null, Config.suppressConsole);
        process.stdin.close();
        import std.array : appender;
        import std.algorithm.comparison : min;
        auto captured = appender!string();
        foreach (ubyte[] chunk; process.stdout.byChunk(4096)) {
            auto remaining = maxOutput - captured.data.length;
            captured.put(chunk[0 .. min(remaining, chunk.length)]);
        }
        auto status = wait(process.pid);
        if (status != 0 || captured.data.length >= maxOutput || !exists(output))
            throw new Exception("PDF preview merge failed: " ~ captured.data);
    }
    auto input = z8_file_to_binary(z8_file_from_path(output.to!wstring));
    ubyte[] bytes;
    { scope(exit) z8_binary_close(input); bytes = z8_binary_snapshot(input, z8_temporary_archives_bytes_limit); }
    z8_monitor_property(request.writer, "source"w, z8_temporary_archive_store(bytes, "preview.pdf"w));
}
