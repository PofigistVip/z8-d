// Views of object collections retain shared storage across BL inheritance casts.
template Z8ValueArray(T) {
    static if (is(T : Z8Object)) alias Z8ValueArray = Z8ObjectArray!T;
    else static if (is(T : IZ8CollectionView)) alias Z8ValueArray = Z8NestedObjectArray!T;
    else alias Z8ValueArray = Z8Array!T;
}

// Collection views expose the original backing store, including maps in arrays.
interface IZ8CollectionView {
    Object z8_erased_storage();
}
// All typed views expose their shared storage, including nested dimensions.
interface IZ8ObjectArrayView : IZ8CollectionView {
    Z8Array!Object z8_erased_data();
}

T z8_object_collection_view(T)(Object value) if (is(T : IZ8CollectionView)) {
    if (value is null) return null;
    auto exact = cast(T)value;
    if (exact !is null) return exact;
    auto view = cast(IZ8CollectionView)value;
    if (view is null) throw new Exception("invalid collection cast");
    return T.z8_from_storage(view.z8_erased_storage());
}

private ref T z8_object_array_slot(T)(ref Object item) {
    if (item !is null && cast(T)item is null) throw new Exception("invalid object array element cast");
    return *cast(T*)&item;
}
class Z8ObjectArray(T) : IZ8ObjectArrayView if (is(T : Z8Object)) {
    private Z8Array!Object data;
    this() { data = new Z8Array!Object(); }
    this(T[] values) { this(); foreach (value; values) data.z8_add(value); }
    private this(Z8Array!Object sharedData) { data = sharedData; }
    static Z8ObjectArray!T z8_from_data(Z8Array!Object data) { return new Z8ObjectArray!T(data); }
    static Z8ObjectArray!T z8_from_storage(Object storage) {
        auto data = cast(Z8Array!Object)storage;
        if (data is null) throw new Exception("invalid array storage");
        return z8_from_data(data);
    }
    Z8Array!Object z8_erased_data() { return data; }
    Object z8_erased_storage() { return data; }
    @property size_t z8_version() { return data.z8_version; }
    override bool opEquals(Object other) {
        if (this is other) return true;
        auto right = cast(IZ8ObjectArrayView)other;
        return right !is null && data.opEquals(right.z8_erased_data());
    }
    long z8_size() { return data.z8_size(); }
    bool z8_isEmpty() { return data.z8_isEmpty(); }
    void z8_clear() { data.z8_clear(); }
    ref T z8_at(long index) { return z8_object_array_slot!T(data.z8_at(index)); }
    T z8_get(long index) { return z8_at(index); }
    T z8_first() { return z8_isEmpty() ? null : z8_get(0); }
    T z8_last() { return z8_isEmpty() ? null : z8_get(z8_size() - 1); }
    Z8ObjectArray!T z8_add(T value) { data.z8_add(value); return this; }
    Z8ObjectArray!T z8_add(long index, T value) { data.z8_add(index, value); return this; }

    T z8_removeAt(long index) {
        auto value = data.z8_removeAt(index);
        auto result = cast(T)value;
        if (value !is null && result is null) throw new Exception("invalid object array element cast");
        return result;
    }
    bool z8_remove(T value) { return data.z8_remove(value); }
    Z8ObjectArray!T z8_removeAll(T value) { data.z8_removeAll(value); return this; }

    // Java RCollection.sort compares CLASS wrappers, which are not Comparable.
    Z8ObjectArray!T z8_sort() {
        if (z8_size() > 1) throw new Exception("object array elements are not comparable");
        ++data.z8_version;
        return this;
    }
    Z8ObjectArray!T z8_unique() { return new Z8ObjectArray!T(data.z8_unique()); }
    wstring z8_join(wstring separator = ""w) {
        wstring result;
        auto expectedVersion = z8_version;
        for (long i = 0; i != z8_size(); ++i) {
            if (z8_version != expectedVersion) throw new Exception("array modified during iteration");
            if (i) result ~= z8_string_required(separator);
            auto text = z8_nonnull(z8_get(i)).z8_object_string_impl();
            result ~= text is null ? "null"w : text;
        }
        return result;
    }
    void z8_set(long index, T value) { data.z8_set(index, value); }
}
Z8ObjectArray!To z8_object_array_view(To, From)(Z8ObjectArray!From value)
if (is(To : Z8Object) && is(From : Z8Object)) {
    if (value is null) return null;
    return new Z8ObjectArray!To(value.data);
}

    auto result = new Z8ObjectArray!T();

}

    auto oldValue = previous; auto newValue = right;
    left = z8_array_union(oldValue, newValue);
}
