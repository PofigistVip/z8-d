long z8_str_compare(wstring a, wstring b) {
    b = z8_string_required(b);
    auto length = a.length < b.length ? a.length : b.length;
    foreach (i; 0 .. length) if (a[i] != b[i]) return cast(long)a[i] - cast(long)b[i];
    return cast(long)a.length - cast(long)b.length;
}
long z8_str_compareNoCase(wstring a, wstring b) {
    import std.uni : toUpper, toLower;
    b = z8_string_required(b);
    auto length = a.length < b.length ? a.length : b.length;
    foreach (i; 0 .. length) {
        if (a[i] == b[i]) continue;
        auto upperA = toUpper(cast(dchar)a[i]), upperB = toUpper(cast(dchar)b[i]);
        if (upperA == upperB) continue;
        auto lowerA = toLower(upperA), lowerB = toLower(upperB);
        if (lowerA != lowerB) return cast(long)lowerA - cast(long)lowerB;
    }
    return cast(long)a.length - cast(long)b.length;
}
wstring z8_str_toLowerCase(wstring value) {
    import std.uni : toLower;
    auto result = toLower(value);
    return result.length ? result.idup : ""w;
}
wstring z8_str_toUpperCase(wstring value) {
    import std.uni : toUpper;
    auto result = toUpper(value);
    return result.length ? result.idup : ""w;
}
wstring z8_str_firstToLowerCase(wstring value) {
    if (!value.length || (value[0] >= 0xD800 && value[0] <= 0xDFFF)) return value.length ? value : ""w;
    return z8_str_toLowerCase(value[0 .. 1]) ~ value[1 .. $];
}
wstring z8_str_firstToUpperCase(wstring value) {
    if (!value.length || (value[0] >= 0xD800 && value[0] <= 0xDFFF)) return value.length ? value : ""w;
    return z8_str_toUpperCase(value[0 .. 1]) ~ value[1 .. $];
}
wstring z8_string_replicate(wstring value, long count) {
    value = z8_string_required(value);
    auto n = cast(int)count;
    if (cast(int)(cast(long)value.length * n) < 0) throw new Exception("invalid string capacity");
    wstring result = ""w;
    for (int i = 0; i < n; ++i) result ~= value;
    return result;
}
wstring z8_string_concat(Z8Array!wstring values, wstring delimiter) {
    z8_nonnull(values);
    wstring result = ""w;
    foreach (i; 0 .. values.z8_size()) {
        if (i) result ~= z8_string_required(delimiter);
        result ~= z8_string_required(values.z8_get(i));
    }
    if (values.z8_size()) z8_string_required(delimiter);
    return result;
}

bool z8_str_startsWith(wstring value, wstring prefix, long offset = 0) {
    prefix = z8_string_required(prefix);
    auto start = cast(int)offset;
    if (start < 0 || cast(size_t)start > value.length || prefix.length > value.length - cast(size_t)start) return false;
    return value[cast(size_t)start .. cast(size_t)start + prefix.length] == prefix;
}
bool z8_str_endsWith(wstring value, wstring suffix) {
    suffix = z8_string_required(suffix);
    return suffix.length <= value.length && value[value.length - suffix.length .. $] == suffix;
}
long z8_str_indexOf(wstring value, wstring needle, long offset = 0) {
    needle = z8_string_required(needle);
    auto raw = cast(int)offset;
    size_t start = raw < 0 ? 0 : cast(size_t)raw;
    if (start > value.length) start = value.length;
    if (needle.length == 0) return cast(long)start;
    if (needle.length > value.length) return -1;
    for (size_t i = start; i <= value.length - needle.length; ++i)
        if (value[i .. i + needle.length] == needle) return cast(long)i;
    return -1;
}
long z8_str_lastIndexOf(wstring value, wstring needle, long offset = int.max) {
    needle = z8_string_required(needle);
    long start = cast(int)offset;
    if (start < 0 || needle.length > value.length) return -1;
    auto last = cast(long)(value.length - needle.length);
    if (start > last) start = last;
    for (long i = start; i >= 0; --i)
        if (value[cast(size_t)i .. cast(size_t)i + needle.length] == needle) return i;
    return -1;
}
wstring z8_str_substring(wstring value, long start) {
    auto i = cast(int)start;
    if (i < 0 || cast(size_t)i > value.length) throw new Exception("string index out of bounds");
    return value[cast(size_t)i .. $];
}
wstring z8_str_substring(wstring value, long start, long count) {
    auto i = cast(int)start;
    auto end = cast(int)(cast(long)i + cast(int)count);
    if (end >= cast(long)value.length) end = cast(int)value.length;
    if (i < 0 || end < i || cast(size_t)end > value.length) throw new Exception("string index out of bounds");
    return value[cast(size_t)i .. cast(size_t)end];
}
wstring z8_str_charAt(wstring value, long index) {
    auto i = cast(int)index;
    if (i < 0 || cast(size_t)i >= value.length) throw new Exception("string index out of bounds");
    return value[cast(size_t)i .. cast(size_t)i + 1];
}
wstring z8_str_trimLeft(wstring value) {
    size_t start;
    while (start < value.length && value[start] <= ' ') ++start;
    return value[start .. $];
}
wstring z8_str_trimRight(wstring value) {
    auto end = value.length;
    while (end > 0 && value[end - 1] <= ' ') --end;
    return value[0 .. end];
}
wstring z8_str_trim(wstring value) { return z8_str_trimRight(z8_str_trimLeft(value)); }
wstring z8_str_left(wstring value, long count) { return z8_str_substring(value, 0, count); }
wstring z8_str_right(wstring value, long count) {
    auto start = cast(int)(cast(long)cast(int)value.length - cast(int)count);
    if (start < 0) start = 0;
    return z8_str_substring(value, start);
}
wstring z8_str_insert(wstring value, long index, wstring what) {

    auto i = cast(int)index;
    if (i < 0 || cast(size_t)i > value.length) throw new Exception("string index out of bounds");
    return value[0 .. cast(size_t)i] ~ what ~ value[cast(size_t)i .. $];
}
wstring z8_str_replace(wstring value, long index, long count, wstring replacement) {
    replacement = z8_string_required(replacement);
    auto i = cast(int)index;
    auto end = cast(int)(cast(long)i + cast(int)count);
    if (i < 0 || end < 0 || cast(size_t)i > value.length || cast(size_t)end > value.length)
        throw new Exception("string index out of bounds");
    return value[0 .. cast(size_t)i] ~ replacement ~ value[cast(size_t)end .. $];
}
wstring z8_str_padding(wstring value, long length, wstring padding, bool left) {
    padding = z8_string_required(padding);
    auto requested = cast(int)length;
    if (requested <= 0 || cast(size_t)requested <= value.length) return value;
    if (!padding.length) padding = " "w;
    auto count = cast(size_t)requested - value.length;
    wchar[] fill = new wchar[count];
    foreach (i; 0 .. count) fill[i] = padding[i % padding.length];
    return left ? fill.idup ~ value : value ~ fill.idup;
}

wstring z8_str_padRight(wstring value, long length, wstring padding = " "w) { return z8_str_padding(value, length, padding, false); }
bool z8_bool_parse(wstring value) {
    import std.uni : toLower;
    auto text = value.to!string.toLower;
    if (text == "" || text == "0" || text == "false" || text == "no" || text == "n" || text == "bool.false") return false;
    if (text == "1" || text == "true" || text == "yes" || text == "y" || text == "bool.true") return true;
    throw new Exception("invalid bool");
}

wstring z8_int_format(ulong value, int radix, bool negative = false) {
    immutable wchar[] digits = "0123456789abcdefghijklmnopqrstuvwxyz"w;
    wchar[] reversed;
    do { reversed ~= digits[cast(size_t)(value % radix)]; value /= radix; } while (value);
    if (negative) reversed ~= '-';
    wchar[] result = new wchar[reversed.length];
    foreach (i; 0 .. reversed.length) result[i] = reversed[$ - 1 - i];
    return result.idup;
}
wstring z8_int_toString(long value, long base = 10) {
    auto radix = cast(int)base;
    if (radix < 2 || radix > 36) radix = 10;
    auto magnitude = value < 0 ? 0UL - cast(ulong)value : cast(ulong)value;
    return z8_int_format(magnitude, radix, value < 0);
}
wstring z8_int_toHexString(long value) { return z8_int_format(cast(ulong)value, 16); }
wstring z8_int_toBinaryString(long value) { return z8_int_format(cast(ulong)value, 2); }
long z8_int_parse(wstring value, long radix = 10) {
    auto base = cast(int)radix;
    if (base < 2 || base > 36 || value.length == 0) throw new Exception("invalid integer");
    if (value[$-1] == 'L' || value[$-1] == 'l') value = value[0 .. $-1];
    if (!value.length) throw new Exception("invalid integer");
    size_t i; bool negative;
    if (value[0] == '-' || value[0] == '+') { negative = value[0] == '-'; ++i; }
    if (i == value.length) throw new Exception("invalid integer");
    ulong magnitude, limit = cast(ulong)long.max + (negative ? 1UL : 0UL);
    for (; i < value.length; ++i) {
        auto c = value[i];
        int digit = c >= '0' && c <= '9' ? c - '0' : c >= 'a' && c <= 'z' ? c - 'a' + 10 : c >= 'A' && c <= 'Z' ? c - 'A' + 10 : -1;
        if (digit < 0 || digit >= base || magnitude > (limit - cast(ulong)digit) / cast(ulong)base)
            throw new Exception("invalid integer");
        magnitude = magnitude * cast(ulong)base + cast(ulong)digit;
    }
    if (negative && magnitude == cast(ulong)long.max + 1UL) return long.min;
    return negative ? -cast(long)magnitude : cast(long)magnitude;
}
