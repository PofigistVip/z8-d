private enum z8_query_name = "org.zenframework.z8.server.base.query.Query"w;

enum Z8QueryReadLock { none, update, share, updateNoWait, shareNoWait }
struct Z8QueryReadRequest {
    Z8Object[] fields, sortFields, groupFields;
    Z8SqlBool where, having;
    // The executor must combine these query filters with where/having.
    Z8SqlBool queryWhere, queryHaving, queryScope;
    int start, limit = -1;
    Z8Guid recordId;
    bool byRecordId, disableGrouping;
    Z8QueryReadLock readLock;
}
interface IZ8QueryCursor : IZ8FieldCursor {
    Z8Object[] fields();
    bool next();
    void close();
}

    IZ8QueryCursor read(Z8Object query, Z8QueryReadRequest request);

// Writer operations receive the affected fields; values are their defaults,

    long update(Z8Object query, Z8Guid id, Z8Object[] fields);
    long destroy(Z8Object query, Z8Guid id);
}
private interface IZ8QueryLockCell { Z8QueryReadLock get(); }
private class Z8QueryLockCell(T) : IZ8QueryLockCell {
    T value;
    this() { value = T.e_None; }
    Z8QueryReadLock get() {
        if (value is T.e_None) return Z8QueryReadLock.none;
        if (value is T.e_Update) return Z8QueryReadLock.update;
        if (value is T.e_Share) return Z8QueryReadLock.share;
        if (value is T.e_UpdateNoWait) return Z8QueryReadLock.updateNoWait;
        if (value is T.e_ShareNoWait) return Z8QueryReadLock.shareNoWait;
        throw new Exception("invalid query read lock");
    }
}

private Z8QueryState z8_query_state(Z8Object self) {
    self = z8_nonnull(self);
    if (!z8_native_is_instance(z8_query_name, self)) throw new Exception("invalid Query object");
    auto state = cast(Z8QueryState)self.z8_get_native_state(z8_query_name);
    if (state is null) { state = new Z8QueryState(); self.z8_set_native_state(z8_query_name, state); }
    return state;
}
private auto z8_query_metadata_address(string member)(Z8Object self) {
    return &__traits(getMember, z8_query_state(self), member);
}
ref Z8ObjectArray!T z8_query_array_ref(T, string name)(Z8Object self) if (is(T : Z8Object)) {
    auto state = z8_query_state(self);
    auto key = name.to!wstring;
    auto found = key in state.arrays;
    auto cell = found is null ? null : cast(Z8FieldOrderCell!T)*found;
    if (cell is null) {
        if (found !is null) throw new Exception("incompatible query array type");
        cell = new Z8FieldOrderCell!T(); state.arrays[key] = cell;
    }
    return cell.value;
}
ref T z8_query_lock_ref(T)(Z8Object self) {
    auto state = z8_query_state(self);
    auto cell = cast(Z8QueryLockCell!T)state.lockCell;
    if (cell is null) {
        if (state.lockCell !is null) throw new Exception("incompatible query read lock enum");
        cell = new Z8QueryLockCell!T(); state.lockCell = cell;
    }
    return cell.value;
}
Z8QueryReadLock z8_query_read_lock(Z8Object self) {
    auto cell = z8_query_state(self).lockCell;
    return cell is null ? Z8QueryReadLock.none : cell.get();
}
// Return a snapshot of a query collection for a database or UI adapter.
Z8Object[] z8_query_array(Z8Object self, wstring name) {
    switch (name) {
        case "controls"w: case "columns"w: case "names"w: case "quickFilters"w: case "filterFields"w:
        case "extraFields"w: case "sortFields"w: case "groupFields"w: case "actions"w: case "reports"w:
        case "aggregateBy"w: case "groupBy"w: break;
        default: throw new Exception("unknown query collection");
    }
    auto state = z8_query_state(self);
    auto entry = name in state.arrays;
    if (entry is null) return [];
    auto cell = cast(IZ8FieldOrderCell)*entry;
    if (cell is null) throw new Exception("invalid query collection");
    auto view = cell.view();
    if (view is null) return [];
    auto values = view.z8_erased_data();
    Z8Object[] result;
    for (long i; i < values.z8_size(); ++i) result ~= cast(Z8Object)values.z8_get(i);
    return result;
}

private Z8Object[] z8_query_read_fields(Z8ObjectArray!Z8Object value) {
    if (value is null) return null;
    Z8Object[] result;
    for (long i; i < value.z8_size(); ++i) {
        auto field = z8_nonnull(value.z8_get(i));
        if (!z8_native_is_instance(z8_field_name, field)) throw new Exception("invalid query field");
        result ~= field;
    }
    return result;
}
private void z8_query_close(Z8Object self) {
    auto state = z8_query_state(self);
    auto cursor = state.cursor;
    if (cursor is null) return;
    state.cursor = null;
    auto fields = state.cursorFields; state.cursorFields = null;
    foreach (field; fields) {
        auto fieldState = z8_field_state(field);
        if (fieldState.cursor is cursor) fieldState.cursor = null;
    }
    cursor.close();
}
// Temporary record lookup and counting detach the active cursor and changed
// field values, then restore them even when a callback or executor fails.
private class Z8QuerySavedState {
    private struct FieldState { Z8Object field; IZ8FieldCursor cursor; Z8Primary changedValue; bool changed; }
    private Z8Object query;
    private IZ8QueryCursor cursor;
    private Z8Object[] cursorFields;
    private FieldState[] fields;
    this(Z8Object query) {
        this.query = query;
        auto state = z8_query_state(query);
        cursor = state.cursor; cursorFields = state.cursorFields;
        auto all = z8_query_declared_fields(query);
        foreach (field; cursorFields) {
            bool found;
            foreach (existing; all) if (existing is field) { found = true; break; }
            if (!found) all ~= field;
        }
        foreach (field; all) {
            auto fieldState = z8_field_state(field);
            auto changed = fieldState.changed;
            auto value = changed ? z8_field_primary(field) : null;
            fields ~= FieldState(field, fieldState.cursor, value, changed);
            if (changed) z8_field_reset(field);
            fieldState.cursor = null;
        }
        state.cursor = null; state.cursorFields = null;
    }
    void restore() {
        z8_query_close(query);
        auto state = z8_query_state(query);
        state.cursor = cursor; state.cursorFields = cursorFields;
        foreach (entry; fields) {
            z8_field_state(entry.field).cursor = entry.cursor;
            if (entry.changed) z8_field_set(entry.field, entry.changedValue);
        }
    }
}
private void z8_query_prepare_read(Z8Object self, ref Z8QueryReadRequest request) {
    if (z8_query_user_events_enabled()) self.z8_query_virtual_beforeRead();
    if (!request.byRecordId) {
        request.queryWhere = z8_query_where(self); request.queryHaving = z8_query_having(self);
        request.queryScope = z8_query_scope(self);
    }
}
private IZ8QueryCursor z8_query_execute_read(Z8Object self, Z8QueryReadRequest request) {
    auto executor = z8_query_executor(self);
    if (executor is null) throw new Exception("query executor is not configured");
    IZ8QueryCursor cursor;
    scope(failure) if (cursor !is null) cursor.close();
    try { cursor = executor.read(self, request); }
    finally { if (z8_query_user_events_enabled()) self.z8_query_virtual_afterRead(); }
    if (cursor is null) throw new Exception("query executor returned a null cursor");
    return cursor;
}
private void z8_query_open(Z8Object self, Z8QueryReadRequest request, bool useQueryLimit = true) {
    auto state = z8_query_state(self);
    if (z8_query_executor(self) is null) throw new Exception("query executor is not configured");
    z8_query_prepare_read(self, request);
    z8_query_close(self);
    if (useQueryLimit && request.limit < 0 && state.limit.present) request.limit = cast(int)state.limit.value;
    request.readLock = z8_query_read_lock(self);
    auto cursor = z8_query_execute_read(self, request);
    scope(failure) cursor.close();
    auto fields = cursor.fields().dup;
    Z8FieldState[] fieldStates;
    foreach (field; fields) {
        if (field is null || !z8_native_is_instance(z8_field_name, field)) throw new Exception("query cursor returned an invalid field");
        fieldStates ~= z8_field_state(field);
    }
    foreach (fieldState; fieldStates) fieldState.cursor = cursor;
    state.cursor = cursor; state.cursorFields = fields;
}
private Z8Optional!bool z8_query_next(Z8Object self) {
    auto cursor = z8_query_state(self).cursor;
    if (cursor is null) throw new Exception("Query.read must be called before Query.next");
    auto hasNext = cursor.next();
    if (!hasNext) z8_query_close(self);
    return z8_optional_wrap(hasNext);
}
private Z8Optional!long z8_query_count_where(Z8Object self, Z8SqlBool where) {
    auto state = z8_query_state(self);
    auto counter = cast(IZ8QueryCounter)z8_query_executor(self);
    if (counter is null) throw new Exception("query executor does not support count");
    auto saved = new Z8QuerySavedState(self);
    scope(exit) saved.restore();
    Z8QueryReadRequest request;
    request.where = where; z8_query_prepare_read(self, request);
    long result;
    try { result = counter.count(self, request); }
    finally { if (z8_query_user_events_enabled()) self.z8_query_virtual_afterRead(); }
    return z8_optional_wrap(cast(long)cast(int)result);
}
private Z8Optional!long z8_query_count(Z8Object self) { return z8_query_count_where(self, null); }

private bool z8_query_user_events_enabled() {
    auto request = z8_application_request();
    return request is null || request.eventsEnabled(Z8EventsLevel.user);
}
private void z8_query_on_new(Z8Object self) {
    if (z8_query_user_events_enabled()) {
        self.z8_query_virtual_onNew();
        foreach (field; z8_query_declared_fields(self, true)) {
            if (z8_field_is_changed(field)) continue;
            auto value = field.z8_field_virtual_default();
            if (value !is null && !z8_primary_collection_equals(value, z8_field_default_value(field)))
                z8_field_set(field, value);
        }
    }
    if (z8_native_is_instance(z8_users_name, self)) z8_users_on_new(self);
}
private Z8Guid z8_query_insert_id(Z8Object self, Z8Guid id) {
    auto writer = z8_query_writer(self);
    auto fields = z8_query_declared_fields(self, true);
    auto key = z8_query_primary_key(self);
    if (key !is null && !z8_field_is_changed(key)) z8_field_set(key, z8_primary_box(id));
    z8_table_before_write(self, true);
    if (z8_query_user_events_enabled()) self.z8_query_virtual_beforeCreate(id);
    if (z8_query_user_events_enabled() && z8_native_is_instance(z8_users_name, self)) z8_users_before_write(self, id, true);
    writer.insert(self, fields);
    if (z8_query_user_events_enabled()) self.z8_query_virtual_afterCreate(id);
    auto result = key is null ? id : z8_primary_cast!Z8Guid(z8_field_default_value(key));
    foreach (field; fields) z8_field_reset(field);
    return result;
}
private Z8Guid z8_query_create_id(Z8Object self, Z8Guid id) {
    z8_query_on_new(self);
    return z8_query_insert_id(self, id);
}

private long z8_query_update_id_run(Z8Object self, Z8Guid id, bool resetChanged) {
    auto writer = z8_query_writer(self);
    if (id !is null && id == z8_guid_zero()) return 0;
    if (id !is null) {
        z8_table_before_write(self, false);
        if (z8_query_user_events_enabled() && z8_native_is_instance(z8_role_table_access_name, self)) z8_role_table_access_before_update(self, id);
        if (z8_query_user_events_enabled() && z8_native_is_instance(z8_role_field_access_name, self)) z8_role_field_access_before_update(self);
        if (z8_query_user_events_enabled()) self.z8_query_virtual_beforeUpdate(id);
        if (z8_query_user_events_enabled() && z8_native_is_instance(z8_users_name, self)) z8_users_before_write(self, id, false);
    }
    Z8Object[] changed;
    foreach (field; z8_query_declared_fields(self, true)) if (z8_field_is_changed(field)) changed ~= field;
    auto result = changed.length ? writer.update(self, id, changed) : 0L;
    if (id !is null && z8_query_user_events_enabled()) self.z8_query_virtual_afterUpdate(id);
    if (resetChanged) foreach (field; changed) z8_field_reset(field);
    return result;
}
private Z8Optional!long z8_query_update_id(Z8Object self, Z8Guid id) {
    return z8_optional_wrap(cast(long)cast(int)z8_query_update_id_run(self, id, true));
}
private Z8Optional!long z8_query_destroy_id(Z8Object self, Z8Guid id) {
    auto writer = z8_query_writer(self);
    if (id !is null && id == z8_guid_zero()) return z8_optional_wrap(0L);
    if (z8_query_user_events_enabled()) self.z8_query_virtual_beforeDestroy(id);
    if (z8_native_is_instance(z8_users_name, self)) z8_users_before_destroy(id);
    if (z8_native_is_instance(z8_user_roles_name, self)) z8_user_roles_before_destroy(id);
    if (z8_query_user_events_enabled() && z8_native_is_instance(z8_roles_name, self)) z8_roles_before_destroy(id);
    auto result = writer.destroy(self, id);
    if (z8_query_user_events_enabled()) self.z8_query_virtual_afterDestroy(id);
    return z8_optional_wrap(cast(long)cast(int)result);
}

private Z8Primary[] z8_query_copy_values(Z8Object self, Z8Object[] fields, Z8Guid id, out bool found) {
    auto saved = new Z8QuerySavedState(self);
    scope(exit) saved.restore();
    Z8QueryReadRequest request;
    request.fields = fields; request.recordId = id; request.byRecordId = id !is null;
    z8_query_open(self, request, false);
    found = z8_optional_require(z8_query_next(self));
    Z8Primary[] values;
    if (found) foreach (field; fields) values ~= z8_field_primary(field);
    return values;
}
private Z8Guid z8_query_copy(Z8Object self, Z8Guid id) {
    auto fields = z8_query_declared_fields(self, true);
    bool[] changed;
    foreach (field; fields) changed ~= z8_field_is_changed(field);
    z8_query_on_new(self);
    bool found;
    auto values = z8_query_copy_values(self, fields, id, found);
    if (found) foreach (index, field; fields) {
        auto unique = z8_field_state(field).unique;
        if (!(unique.present && unique.value) && !changed[index]) z8_field_set(field, values[index]);
    }
    auto key = z8_nonnull(z8_query_primary_key(self));
    auto newId = z8_primary_cast!Z8Guid(z8_field_primary(key));
    if (newId is null || newId == id || newId == z8_guid_zero()) {
        newId = z8_guid_create_random(); z8_field_set(key, z8_primary_box(newId));
    }
    if (z8_query_user_events_enabled()) self.z8_query_virtual_onCopy(id);
    return z8_query_insert_id(self, newId);
}
private Z8Optional!long z8_query_update_where(Z8Object self, Z8SqlBool where) {
    auto key = z8_nonnull(z8_query_primary_key(self));
    z8_query_read_request(self, z8_query_key_fields(self), null, where, Z8Optional!long.init, Z8Optional!long.init);
    int result;
    while (z8_optional_require(z8_query_next(self))) {
        auto id = z8_primary_cast!Z8Guid(z8_field_primary(key));
        result += cast(int)z8_query_update_id_run(self, id, false);
    }
    z8_query_close(self);
    foreach (field; z8_query_declared_fields(self, true))
        if (z8_field_is_changed(field)) z8_field_reset(field);
    return z8_optional_wrap(cast(long)result);
}
private Z8Optional!long z8_query_destroy_where(Z8Object self, Z8SqlBool where) {
    auto key = z8_nonnull(z8_query_primary_key(self));
    Z8QueryReadRequest request;
    request.fields = [key]; request.where = where; request.disableGrouping = true;
    z8_query_open(self, request);
    int result;
    while (z8_optional_require(z8_query_next(self))) {
        auto id = z8_primary_cast!Z8Guid(z8_field_primary(key));
        result += cast(int)z8_optional_require(z8_query_destroy_id(self, id));
    }
    z8_query_close(self);
    return z8_optional_wrap(cast(long)result);
}

private void z8_query_group_request(Z8Object self, Z8ObjectArray!Z8Object groups, Z8ObjectArray!Z8Object fields,
                                    Z8ObjectArray!Z8Object sort, Z8SqlBool where, Z8SqlBool having,
                                    Z8Optional!long start, Z8Optional!long limit) {

private Z8ObjectArray!Z8Object z8_query_data_fields(Z8Object self, bool primaryOnly = false) {
    self = z8_nonnull(self);
    if (!z8_native_is_instance(z8_query_name, self)) throw new Exception("invalid Query object");
    auto result = new Z8ObjectArray!Z8Object();
    foreach (member; z8_object_members(self)) {
        if (z8_native_is_instance(z8_field_name, member) &&
            (!primaryOnly || !z8_native_is_instance(z8_expression_name, member)) && !result.z8_contains(member)) result.z8_add(member);
    }
    return result;
}

Z8Object[] z8_query_declared_fields(Z8Object self, bool primaryOnly = false) {
    auto fields = z8_query_data_fields(self, primaryOnly);

    for (long i; i < fields.z8_size(); ++i) result ~= fields.z8_get(i);

}
private Z8Object z8_query_find_direct(string kind)(Z8Object self, wstring value) {
    value = z8_string_required(value);
    auto fields = z8_query_data_fields(self);
    for (long i; i < fields.z8_size(); ++i) {
        auto field = fields.z8_get(i);
        static if (kind == "index") auto actual = z8_object_index(field);
        else static if (kind == "id") auto actual = z8_object_id(field);
        else static if (kind == "name") auto actual = z8_object_name(field);
        else static assert(0, "unknown field lookup");
        if (actual == value) return field;
    }
    throw new Exception("query field not found");
}
private Z8Object z8_query_find_field_id(Z8Object self, wstring id) {
    import std.array : split;
    self = z8_nonnull(self); id = z8_string_required(id);
    auto own = self.z8_id();
    if (id.length < own.length || id[0 .. own.length] != own) throw new Exception("query field not found");
    auto path = id[own.length .. $];
    if (own.length && path.length) {
        if (path[0] != '.') throw new Exception("query field not found");
        path = path[1 .. $];
    }
    auto current = self;
    auto parts = path.split("."w);
    foreach (i, part; parts) {
        if (!part.length) {
            if (i + 1 == parts.length) continue;
            throw new Exception("query field not found");
        }
        Z8Object found;
        foreach (member; z8_object_members(current)) if (member.z8_index() == part) { found = member; break; }
        if (found is null) throw new Exception("query field not found");
        current = found;
    }
    if (!z8_native_is_instance(z8_field_name, current)) throw new Exception("query field not found");
    return current;
}
