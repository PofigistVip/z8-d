private final class Z8MoscowTimeZone : TimeZone {
    private enum long epoch = 621355968000000000L, perSecond = 10_000_000L;
    private immutable(TimeZone) earlier;
    this(immutable(TimeZone) earlier) immutable {
        super("Europe/Moscow", "MSK", "MSD");
        this.earlier = earlier;
    }
    @property override bool hasDST() @safe const nothrow { return true; }
    private Z8ZoneTransition state(long stdTime) @safe const nothrow {
        auto seconds = (stdTime - epoch) / perSecond;
        Z8ZoneTransition result = Z8ZoneTransition(0, 10800, false);
        foreach (transition; z8_moscow_transitions) {
            if (transition.unixSeconds > seconds) break;
            result = transition;
        }
        return result;
    }
    override bool dstInEffect(long stdTime) @safe const scope nothrow {
        return stdTime < epoch ? earlier.dstInEffect(stdTime) : state(stdTime).daylight;
    }
    override long utcToTZ(long stdTime) @safe const scope nothrow {
        return stdTime < epoch ? earlier.utcToTZ(stdTime) : stdTime + cast(long)state(stdTime).offsetSeconds * perSecond;
    }
    override long tzToUTC(long local) @safe const scope nothrow {
        if (local < epoch + 10800L * perSecond) return earlier.tzToUTC(local);
        // Java chooses the later instant for ambiguous local times.
        foreach (offset; [7200, 10800, 14400]) {
            auto utc = local - cast(long)offset * perSecond;
            if (utc >= epoch && state(utc).offsetSeconds == offset) return utc;
        }
        int previous = 10800;
        foreach (transition; z8_moscow_transitions) {
            auto utc = epoch + transition.unixSeconds * perSecond;
            if (transition.offsetSeconds > previous && local >= utc + cast(long)previous * perSecond &&
                local < utc + cast(long)transition.offsetSeconds * perSecond)
                return local - cast(long)previous * perSecond;
            previous = transition.offsetSeconds;
        }
        return local - 10800L * perSecond;
    }
}

    }
