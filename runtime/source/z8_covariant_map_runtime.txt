            else {

                if (value !is null && result is null) throw new Exception("invalid map value cast");
                return result;
            }
        } else return value;
    }

            auto primary = cast(Z8GuidMap!Z8Primary)other;
            if (primary !is null) return z8_primary_map_view(data).opEquals(primary);
        }

                   is(M == Z8ReferenceKeyMap!(K, Object)) && is(V : Z8Object)) {
            auto primary = cast(Z8ReferenceKeyMap!(K, Z8Primary))other;
            if (primary !is null) return z8_primary_map_view(data).opEquals(primary);
        }
        static if (keyObject && is(M == Z8ReferenceKeyMap!(Z8Object, U), U)) {{
            auto primaryKeys = cast(Z8PrimaryKeyMap!U)other;
            if (primaryKeys !is null) return data.entries.opEquals(primaryKeys);

                    auto primary = cast(Z8ReferenceKeyMap!(Z8Object, Z8Primary))collection.z8_erased_storage();
                    if (primary !is null) return z8_primary_map_view(data).opEquals(primary);
                }
                static if (valueObject && is(U == Object)) {
                    auto objectPrimaryKeys = cast(Z8PrimaryKeyMap!Object)collection.z8_erased_storage();
                    if (objectPrimaryKeys !is null) return data.entries.opEquals(objectPrimaryKeys);
                }
            }
        }}
        static if (keyObject && is(M == Z8ReferenceKeyMap!(Z8Object, Z8Primary))) {
            auto collection = cast(IZ8CollectionView)other;
            if (collection !is null) {
                import std.meta : AliasSeq;
                static foreach (Candidate; AliasSeq!(long, bool, Z8Decimal, wstring, Z8Guid, Z8Binary, Z8Datespan, Z8Date, Z8Geometry, Z8File, Object)) {{

            }
        }
        static if (is(K == class)) {
            static if (is(M == Z8ObjectKeyMap!(K, Object)) && is(V : Z8Object)) {
                auto primary = cast(Z8ObjectKeyMap!(K, Z8Primary))other;
                if (primary !is null) return z8_primary_map_view(data).opEquals(primary);
            }
        }
        static if (valueObject && is(K == wstring) && is(M == Z8StringValueMap!Object)) {

                auto stringPrimaryKeys = cast(Z8PrimaryKeyMap!Object)collection.z8_erased_storage();

            }

            auto collection = cast(IZ8CollectionView)other;
            if (collection !is null) {

                auto collection = cast(IZ8CollectionView)other;
                if (collection !is null) {
                    auto referencePrimaryKeys = cast(Z8PrimaryKeyMap!Object)collection.z8_erased_storage();
                    if (referencePrimaryKeys !is null) return data.entries.opEquals(referencePrimaryKeys);
                }
            }
        }
        static if (is(K == Z8Primary) && is(M == Z8PrimaryKeyMap!Object) && valueObject) {
            auto collection = cast(IZ8CollectionView)other;
            if (collection !is null) {
                import std.meta : AliasSeq;

                    if (scalarKeys !is null) return data.opEquals(z8_primary_key_map_view(scalarKeys));
                }}
                auto stringKeys = cast(Z8StringValueMap!Object)collection.z8_erased_storage();
                if (stringKeys !is null) return data.opEquals(z8_primary_key_map_view(stringKeys));
                auto guids = cast(Z8GuidMap!Object)collection.z8_erased_storage();
                if (guids !is null) return data.opEquals(guids.entries);
                static foreach (KeyType; AliasSeq!(Z8Date, Z8Datespan, Z8Binary, Z8Geometry, Z8File, Z8Object)) {{
                    auto references = cast(Z8ReferenceKeyMap!(KeyType, Object))collection.z8_erased_storage();
                    if (references !is null) return data.opEquals(references.entries);
                }}
            }
        }
        auto view = cast(IZ8CollectionView)other;
        auto right = view is null ? null : cast(M)view.z8_erased_storage();
        return right !is null && data.opEquals(right);
    }
    long z8_size() { return data.z8_size(); }

    bool z8_containsKey(Key key) { return data.z8_containsKey(key); }

    Element z8_get(Key key) { return checkedValue(data.z8_get(key)); }
    Element z8_add(Key key, Element value) { return checkedValue(data.z8_add(key, value)); }
    void z8_add_discard(Key key, Element value) {
        static if (__traits(compiles, data.z8_add_discard(key, value))) data.z8_add_discard(key, value);
        else data.z8_add(key, value);
    }
    void z8_remove_discard(Key key) {
        static if (__traits(compiles, data.z8_remove_discard(key))) data.z8_remove_discard(key);
        else data.z8_remove(key);
    }
    void z8_add(Z8CovariantMap!(K, V, M, keyObject, valueObject) other) { data.z8_add(z8_nonnull(other).data); }
    Element z8_remove(Key key) { return checkedValue(data.z8_remove(key)); }
    Z8ValueArray!K z8_keys() {
        static if (keyObject) return z8_object_array_view!K(data.z8_keys());
        else return data.z8_keys();
    }
    Z8ValueArray!V z8_values() {
        static if (valueObject && is(V : IZ8CollectionView)) return Z8NestedObjectArray!V.z8_from_data(data.z8_values());
        else static if (valueObject) return new Z8ObjectArray!V(data.z8_values());
        else return data.z8_values();
    }
}
