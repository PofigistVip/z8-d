// Request-scoped Entity registry; persistence is handled by EntityTable.
private enum z8_entity_registry_name = "org.zenframework.z8.server.base.entity.Registry"w;
private class Z8EntityRegistryState {
    Z8GuidMap!Z8Object entities;
    Z8Object[] pending;
    Z8Object[wstring] tables;
    Z8ApplicationRequest request;
    bool checked = true, logEnabled;
    size_t changeCount;
    this() { entities = new Z8GuidMap!Z8Object(); }
}
private Z8EntityRegistryState z8_entity_registry_state(Z8Object self) {
    self = z8_nonnull(self);
    if (!z8_native_is_instance(z8_entity_registry_name, self)) throw new Exception("invalid Entity Registry");
    auto state = cast(Z8EntityRegistryState)self.z8_get_native_state(z8_entity_registry_name);
    if (state is null) { state = new Z8EntityRegistryState(); self.z8_set_native_state(z8_entity_registry_name, state); }
    return state;
}
private Z8Object z8_entity_registry_local;
private Z8Object z8_entity_registry_reset(Z8Object self) {
    auto state = z8_entity_registry_state(self);
    state.pending = null; state.tables = null; state.entities.z8_clear(); ++state.changeCount; state.checked = true; return self;
}
private Z8Object z8_entity_get_table(Z8Object entity) {
    auto className = z8_nonnull(entity).z8_get_attribute("entityTable"w);
    auto prototype = z8_object_new_instance_native(className);
    auto registry = z8_runtime_get_registry();
    if (registry is null) throw new Exception("runtime registry is not configured");
    auto table = registry.newTableByName(z8_object_name(prototype));
    if (table is null || !z8_native_is_instance(z8_entity_table_name, table))
        throw new Exception("entity table is not registered");
    return table;
}
private Z8Object z8_entity_registry_table(Z8EntityRegistryState state, Z8Object entity) {
    auto className = z8_nonnull(entity).z8_get_attribute("entityTable"w);
    if (auto table = className in state.tables) return *table;
    auto table = z8_entity_get_table(entity); state.tables[className] = table; return table;
}
private void z8_entity_registry_log(Z8EntityRegistryState state, wstring text) {
    if (state.logEnabled) z8_nonnull(z8_application_require_request().monitor).log(Z8MonitorLevel.info, text);
}
private Z8Object z8_entity_registry_flush(Z8Object self) {
    auto state = z8_entity_registry_state(self);
    z8_entity_registry_log(state, ">>> REGISTRY TRANSACTION BEGIN >>>"w);
    z8_entity_registry_check(state);
    auto keys = state.entities.z8_keys(); auto expectedCount = state.changeCount;
    auto connection = z8_connection_current_backend();
    if (connection is null) throw new Exception("no current database connection");
    for (int phase; phase < 3; ++phase) {
        for (long i; i < keys.z8_size(); ++i) {
            if (state.changeCount != expectedCount) throw new Exception("entity registry modified during iteration");
            auto id = keys.z8_get(i); auto entity = state.entities.z8_get(id);
            auto values = z8_entity_state(entity);
            if (phase == 0 && !values.persistent && !values.deleted) {
                values.persistent = true;
                auto table = z8_entity_registry_table(state, entity);
                z8_entity_registry_log(state, "REGISTRY CREATE: "w ~ z8_object_name(table));
                z8_query_create_id(table, id);
            } else if (phase == 1 && z8_optional_require(z8_entity_changed(entity))) {
                auto table = z8_entity_registry_table(state, entity);
                z8_entity_registry_log(state, "REGISTRY UPDATE: "w ~ z8_object_name(table));
                z8_entity_table_update(table, entity);
            } else if (phase == 2 && values.persistent && values.deleted) {
                auto table = z8_entity_registry_table(state, entity);
                z8_entity_registry_log(state, "REGISTRY DESTROY: "w ~ z8_object_name(table));
                z8_entity_table_destroy(table, entity);
            }
        }
        connection.flush();
    }
    z8_entity_registry_log(state, "<<< REGISTRY TRANSACTION END <<<"w);
    return self;
}
private Z8Object z8_entity_registry_get_current() {
    if (z8_entity_registry_local is null) z8_entity_registry_local = z8_native_new(z8_entity_registry_name);
    auto state = z8_entity_registry_state(z8_entity_registry_local);
    auto request = z8_application_request();
    if (state.request !is request) { state.request = request; z8_entity_registry_reset(z8_entity_registry_local); }
    return z8_entity_registry_local;
}
private Z8Object z8_entity_registry_log_enabled(Z8Object self, Z8Optional!bool enabled) {
    z8_entity_registry_state(self).logEnabled = z8_optional_require(enabled); return self;
}
private Z8Optional!bool z8_entity_registry_is_log_enabled(Z8Object self) { return z8_optional_wrap(z8_entity_registry_state(self).logEnabled); }
private void z8_entity_registry_check(Z8EntityRegistryState state) {
    if (state.checked) return;
    state.checked = true;
    size_t write;
    foreach (entity; state.pending) {
        auto id = z8_entity_get_id(entity);
        if (id !is null && id != z8_guid_zero()) {
            if (!state.entities.z8_containsKey(id)) ++state.changeCount;
            state.entities.z8_add(id, entity);
        }
        else state.pending[write++] = entity;
    }
    state.pending.length = write;
}
private Z8Object z8_entity_registry_find(Z8Object self, Z8Guid id) {
    auto state = z8_entity_registry_state(self);
    if (id is null || id == z8_guid_zero()) return null;
    z8_entity_registry_check(state); return state.entities.z8_get(id);
}
private void z8_entity_registry_add(Z8Object self, Z8Object entity) {
    auto state = z8_entity_registry_state(self);
    auto id = z8_entity_get_id(entity);
    if (id is null || id == z8_guid_zero()) { state.pending ~= entity; state.checked = false; }
    else if (!state.entities.z8_containsKey(id)) { state.entities.z8_add(id, entity); ++state.changeCount; }
    else throw new Exception("duplicate entity");
}
private void z8_entity_registry_remove(Z8Object self, Z8Object entity) {
    auto state = z8_entity_registry_state(self);
    foreach (i, candidate; state.pending) if (z8_optional_require(z8_entity_equal(entity, candidate))) {
        state.pending = state.pending[0 .. i] ~ state.pending[i + 1 .. $]; break;
    }
    auto id = z8_entity_get_id(entity);
    if (state.entities.z8_containsKey(id)) ++state.changeCount;
    state.entities.z8_remove(id);
}
private Z8Object z8_entity_register_to(Z8Object self, Z8Object registry) {
    auto state = z8_entity_state(self);
    if (state.registry !is null) z8_entity_registry_remove(state.registry, self);
    if (registry !is null) z8_entity_registry_add(registry, self);
    state.registry = registry; self.z8_entity_virtual_on_register(); return self;
}
private Z8Object z8_entity_register(Z8Object self) { return z8_entity_register_to(self, z8_entity_registry_get_current()); }
private Z8Object z8_entity_register_explicit(Z8Object self, Z8Object registry) { return z8_entity_register_to(self, z8_nonnull(registry)); }
private Z8Object z8_entity_unregister(Z8Object self) { return z8_entity_register_to(self, null); }
private Z8Optional!bool z8_entity_is_registered(Z8Object self) { return z8_optional_wrap(z8_entity_state(self).registry !is null); }
private Z8Object z8_entity_get_registry(Z8Object self) { return z8_nonnull(z8_entity_state(self).registry); }
private Z8Object z8_entity_related(Z8Object self, Z8Guid id) { return z8_entity_registry_find(z8_entity_get_registry(self), id); }
private Z8Object z8_entity_related_key(Z8Object self, wstring key) { return z8_entity_related(self, z8_entity_guid(self, key)); }
private Z8Object z8_entity_origin(Z8Object self) { return z8_entity_related(self, z8_entity_get_origin_id(self)); }
private void z8_entity_registry_visit_filtered(Z8Object self, Z8Object visitor, Z8Object filter) {
    if (visitor is null) return;
    visitor = z8_nonnull(visitor);
    if (!z8_native_is_instance(z8_entity_visitor_name, visitor)) throw new Exception("invalid Entity Visitor");
    auto state = z8_entity_registry_state(self); z8_entity_registry_check(state);
    auto keys = state.entities.z8_keys();
    auto expectedCount = state.changeCount;
    for (long i; i < keys.z8_size(); ++i) {
        if (state.changeCount != expectedCount) throw new Exception("entity registry modified during iteration");
        auto entity = state.entities.z8_get(keys.z8_get(i));
        if (filter is null || z8_entity_filter_accept(filter, entity)) visitor.z8_entity_visitor_virtual_visit(entity);
    }
}
private void z8_entity_registry_visit(Z8Object self, Z8Object visitor) { z8_entity_registry_visit_filtered(self, visitor, null); }
M z8_entity_registry_find_typed(M)(Z8Object self, Z8Object filter) {
    auto result = new M();
    if (filter is null) return result;
    auto state = z8_entity_registry_state(self); z8_entity_registry_check(state);
    auto keys = state.entities.z8_keys();
    auto expectedCount = state.changeCount;
    for (long i; i < keys.z8_size(); ++i) {
        if (state.changeCount != expectedCount) throw new Exception("entity registry modified during iteration");
        auto id = keys.z8_get(i); auto entity = state.entities.z8_get(id);
        if (z8_entity_filter_accept(filter, entity)) {
            auto typed = cast(M.Element)entity;
            if (entity !is null && typed is null) throw new Exception("invalid Entity type");
            result.z8_add(id, typed);
        }
    }
    return result;
}
private Z8Object z8_entity_registry_find_first(Z8Object self, Z8Object filter) {
    if (filter is null) return null;
    auto state = z8_entity_registry_state(self); z8_entity_registry_check(state);
    auto keys = state.entities.z8_keys();
    auto expectedCount = state.changeCount;
    for (long i; i < keys.z8_size(); ++i) {
        if (state.changeCount != expectedCount) throw new Exception("entity registry modified during iteration");
        auto entity = state.entities.z8_get(keys.z8_get(i));
        if (z8_entity_filter_accept(filter, entity)) return entity;
    }
    return null;
}
private Z8Object z8_entity_registry_find_copy(Z8Object self, Z8Guid origin) {
    if (origin is null) return null;
    auto state = z8_entity_registry_state(self); z8_entity_registry_check(state);
    auto keys = state.entities.z8_keys();
    auto expectedCount = state.changeCount;
    for (long i; i < keys.z8_size(); ++i) {
        if (state.changeCount != expectedCount) throw new Exception("entity registry modified during iteration");
        auto entity = state.entities.z8_get(keys.z8_get(i));
        auto id = z8_entity_get_origin_id(entity);
        if (id !is null && id == origin) return entity;
    }
    return null;
}
private void z8_entity_visit(Z8Object self, Z8Object visitor) { z8_entity_registry_visit(z8_entity_get_registry(self), visitor); }
private void z8_entity_visit_filtered(Z8Object self, Z8Object visitor, Z8Object filter) { z8_entity_registry_visit_filtered(z8_entity_get_registry(self), visitor, filter); }
private Z8Object z8_entity_find_first(Z8Object self, Z8Object filter) { return z8_entity_registry_find_first(z8_entity_get_registry(self), filter); }
private Z8Object z8_entity_find_copy(Z8Object self, Z8Guid id) { return z8_entity_registry_find_copy(z8_entity_get_registry(self), id); }
private Z8ObjectArray!Z8Object z8_entity_find_all(Z8Object self, Z8Object filter) {
    auto result = new Z8ObjectArray!Z8Object();
    if (filter is null) return result;
    auto state = z8_entity_registry_state(z8_entity_get_registry(self)); z8_entity_registry_check(state);
    auto keys = state.entities.z8_keys();
    auto expectedCount = state.changeCount;
    for (long i; i < keys.z8_size(); ++i) {
        if (state.changeCount != expectedCount) throw new Exception("entity registry modified during iteration");
        auto entity = state.entities.z8_get(keys.z8_get(i));
        if (z8_entity_filter_accept(filter, entity)) result.z8_add(entity);
    }
    return result;
}

}
