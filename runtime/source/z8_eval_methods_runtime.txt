// Built-in expression functions and methods (Z8 Calculator / function).
private Z8Primary z8_eval_string_method(wstring value, wstring name, Z8EvalValue[] arguments) {
    long intArg(size_t i) {
        if (i >= arguments.length || arguments[i].primary is null || arguments[i].primary.kind != Z8PrimaryKind.integer)
            throw new Exception("invalid Eval method argument");
        return arguments[i].primary.integerValue;
    }
    wstring textArg(size_t i) {
        if (i >= arguments.length || arguments[i].primary is null || arguments[i].primary.kind != Z8PrimaryKind.text)
            throw new Exception("invalid Eval method argument");
        return arguments[i].primary.textValue;
    }
    if (!arguments.length) {
        if (name == "length"w) return z8_primary_box(cast(long)value.length);
        if (name == "isEmpty"w) return z8_primary_box(value.length == 0);
        if (name == "toString"w || name == "get"w) return z8_primary_box(z8_string_new(value));
        if (name == "trim"w) return z8_primary_box(z8_string_new(z8_str_trim(value)));
        if (name == "trimLeft"w) return z8_primary_box(z8_string_new(z8_str_trimLeft(value)));
        if (name == "trimRight"w) return z8_primary_box(z8_string_new(z8_str_trimRight(value)));
        if (name == "toUpperCase"w) return z8_primary_box(z8_string_new(z8_str_toUpperCase(value)));
        if (name == "toLowerCase"w) return z8_primary_box(z8_string_new(z8_str_toLowerCase(value)));
        if (name == "firstToUpperCase"w) return z8_primary_box(z8_str_firstToUpperCase(value));
        if (name == "firstToLowerCase"w) return z8_primary_box(z8_str_firstToLowerCase(value));
        if (name == "file"w) return z8_primary_box(z8_file_from_path(value));
        if (name == "binary"w) return z8_primary_box(new Z8Binary(z8_encoding_encode(value, z8_encoding_by_id(0))));
    } else if (arguments.length == 1) {
        if (name == "compare"w) return z8_primary_box(z8_str_compare(value, textArg(0)));
        if (name == "compareNoCase"w) return z8_primary_box(z8_str_compareNoCase(value, textArg(0)));
        if (name == "indexOf"w) return z8_primary_box(z8_str_indexOf(value, textArg(0)));
        if (name == "lastIndexOf"w) return z8_primary_box(z8_str_lastIndexOf(value, textArg(0)));
        if (name == "startsWith"w) return z8_primary_box(z8_str_startsWith(value, textArg(0)));
        if (name == "endsWith"w) return z8_primary_box(z8_str_endsWith(value, textArg(0)));
        if (name == "matches"w) return z8_primary_box(z8_regex_matches(value, textArg(0)));
        if (name == "charAt"w) return z8_primary_box(z8_string_new(z8_str_charAt(value, intArg(0))));
        if (name == "substring"w) return z8_primary_box(z8_string_new(z8_str_substring(value, intArg(0))));
        if (name == "left"w) return z8_primary_box(z8_string_new(z8_str_left(value, intArg(0))));
        if (name == "right"w) return z8_primary_box(z8_string_new(z8_str_right(value, intArg(0))));
        if (name == "padLeft"w) return z8_primary_box(z8_string_new(z8_str_padLeft(value, intArg(0))));
        if (name == "padRight"w) return z8_primary_box(z8_string_new(z8_str_padRight(value, intArg(0))));
    } else if (arguments.length == 2) {
        if (name == "indexOf"w) return z8_primary_box(z8_str_indexOf(value, textArg(0), intArg(1)));
        if (name == "lastIndexOf"w) return z8_primary_box(z8_str_lastIndexOf(value, textArg(0), intArg(1)));
        if (name == "startsWith"w) return z8_primary_box(z8_str_startsWith(value, textArg(0), intArg(1)));
        if (name == "substring"w) return z8_primary_box(z8_string_new(z8_str_substring(value, intArg(0), intArg(1))));
        if (name == "padLeft"w) return z8_primary_box(z8_string_new(z8_str_padLeft(value, intArg(0), textArg(1))));
        if (name == "padRight"w) return z8_primary_box(z8_string_new(z8_str_padRight(value, intArg(0), textArg(1))));
        if (name == "insert"w) return z8_primary_box(z8_string_new(z8_str_insert(value, intArg(0), textArg(1))));
        if (name == "replaceFirst"w) return z8_primary_box(z8_string_new(z8_regex_replaceFirst(value, textArg(0), textArg(1))));
        if (name == "replaceAll"w) return z8_primary_box(z8_string_new(z8_regex_replaceAll(value, textArg(0), textArg(1))));
    } else if (arguments.length == 3 && name == "replace"w)
        return z8_primary_box(z8_string_new(z8_str_replace(value, intArg(0), intArg(1), textArg(2))));
    throw new Exception("unsupported Eval string method");
}
private Z8Primary z8_eval_number_method(Z8Primary value, wstring name, Z8EvalValue[] arguments) {
    long intArg(size_t i) {
        if (i >= arguments.length || arguments[i].primary is null || arguments[i].primary.kind != Z8PrimaryKind.integer)
            throw new Exception("invalid Eval numeric argument");
        return arguments[i].primary.integerValue;
    }
    bool integral = value.kind == Z8PrimaryKind.integer;
    auto number = integral ? z8_decimal_from_int(value.integerValue) : value.decimalValue;
    if (!arguments.length) {
        if (name == "toString"w) return z8_primary_box(z8_string_new(z8_primary_to_string(value)));
        if (name == "get"w) return integral || !number.nan ? value : null;
        if (integral) {
            auto n = value.integerValue;
            if (name == "getInt"w) return z8_primary_box(cast(long)cast(int)n);
            if (name == "decimal"w) return z8_primary_box(z8_decimal_from_int(n));
            if (name == "date"w) return z8_primary_box(z8_date_from_ticks(n));
            if (name == "abs"w) return z8_primary_box(z8_int_abs(n));
            if (name == "signum"w) return z8_primary_box(z8_int_signum(n));
            if (name == "not"w) return z8_primary_box(z8_int_not(n));
            if (name == "toHexString"w) return z8_primary_box(z8_string_new(z8_int_toHexString(n)));
            if (name == "toBinaryString"w) return z8_primary_box(z8_string_new(z8_int_toBinaryString(n)));
        } else {
            if (name == "getDouble"w) return z8_primary_box(z8_math_result(number.z8_as_double()));
            if (name == "isNaN"w) return z8_primary_box(number.nan);
            if (name == "precision"w) return z8_primary_box(number.nan ? 0L : cast(long)z8_precision(number.coefficient));
            if (name == "scale"w) return z8_primary_box(number.nan ? 0L : cast(long)number.scale);
            if (name == "signum"w) return z8_primary_box(number.z8_signum());
            if (name == "ceil"w) return z8_primary_box(number.z8_ceil());
            if (name == "floor"w) return z8_primary_box(number.z8_floor());
            if (name == "round"w) return z8_primary_box(number.z8_round_int());
        }
        static foreach (method; ["abs", "sin", "cos", "tan", "asin", "acos", "atan", "sinh", "cosh", "tanh", "toRadians", "toDegrees", "exp", "ln", "log10", "sqrt", "cbrt"]) {
            if (name == method.to!wstring) return z8_primary_box(z8_decimal_math!method(number));
        }
    } else if (arguments.length == 1) {
        if (integral) {
            auto n = value.integerValue;
            if (name == "and"w) return z8_primary_box(z8_int_and(n, intArg(0)));
            if (name == "or"w) return z8_primary_box(z8_int_or(n, intArg(0)));
            if (name == "xor"w) return z8_primary_box(z8_int_xor(n, intArg(0)));
            if (name == "leftShift"w) return z8_primary_box(z8_int_leftShift(n, intArg(0)));
            if (name == "rightShift"w) return z8_primary_box(z8_int_rightShift(n, intArg(0)));
            if (name == "toString"w) return z8_primary_box(z8_string_new(z8_int_toString(n, intArg(0))));
        } else if (name == "round"w) return z8_primary_box(number.z8_round(intArg(0)));
        if (name == "power"w) {
            auto exponent = arguments[0].primary;
            if (exponent is null || (exponent.kind != Z8PrimaryKind.decimal && (integral || exponent.kind != Z8PrimaryKind.integer)))
                throw new Exception("invalid Eval power argument");
            auto power = exponent.kind == Z8PrimaryKind.integer ? z8_decimal_from_int(exponent.integerValue) : exponent.decimalValue;
            return z8_primary_box(integral ? z8_int_power(value.integerValue, power) : z8_decimal_math!"power"(number, power));
        }
    } else if (!integral && arguments.length == 2 && name == "round"w)
        return z8_primary_box(number.z8_round(intArg(0), intArg(1)));
    throw new Exception("unsupported Eval numeric method");
}
private Z8Primary z8_eval_temporal_method(Z8Primary value, wstring name, Z8EvalValue[] arguments) {
    bool isDate = value.kind == Z8PrimaryKind.date;
    long intArg(size_t i) {
        if (i >= arguments.length || arguments[i].primary is null || arguments[i].primary.kind != Z8PrimaryKind.integer)
            throw new Exception("invalid Eval temporal argument");
        return arguments[i].primary.integerValue;
    }
    if (!arguments.length) {
        if (name == "toString"w) return z8_primary_box(z8_string_new(z8_primary_to_string(value)));
        if (isDate) {
            auto date = value.dateValue;
            static foreach (method; ["ticks", "year", "quarter", "weekOfYear", "month", "day", "dayOfWeek", "dayOfYear", "daysInMonth", "hours", "minutes", "seconds", "milliseconds", "zoneOffset"]) {
                if (name == method.to!wstring) return z8_primary_box(mixin("z8_date_" ~ method ~ "(date)"));
            }
            static foreach (method; ["truncYear", "truncQuarter", "truncMonth", "truncWeek", "truncDay", "truncHour", "truncMinute", "truncSecond"]) {
                if (name == method.to!wstring) return z8_primary_box(mixin("z8_date_" ~ method ~ "(date)"));
            }
            if (name == "toIsoString"w) return z8_primary_box(z8_string_new(z8_date_toIsoString(date)));
        } else {
            auto span = value.datespanValue;
            static foreach (method; ["days", "hours", "minutes", "seconds", "milliseconds", "totalHours", "totalMinutes", "totalSeconds"]) {
                if (name == method.to!wstring) return z8_primary_box(mixin("z8_datespan_" ~ method ~ "(span)"));
            }
            static foreach (method; ["truncDay", "truncHour", "truncMinute"]) {
                if (name == method.to!wstring) { mixin("z8_datespan_" ~ method ~ "(span);"); return null; }
            }
        }
    } else if (arguments.length == 1) {
        auto arg = arguments[0].primary;
        if (arg is null) throw new Exception("invalid Eval temporal argument");
        if (name == "toString"w && arg.kind == Z8PrimaryKind.text)
            return z8_primary_box(z8_string_new(isDate ? z8_date_toString(value.dateValue, arg.textValue) : z8_datespan_to_string(value.datespanValue, arg.textValue)));
        if (isDate && arg.kind == Z8PrimaryKind.integer) {
            auto date = value.dateValue;
            auto count = arg.integerValue;
            if (name == "setZoneOffset"w) { z8_date_setZoneOffset(date, count); return null; }
            static foreach (method; ["addYear", "addQuarter", "addMonth", "addDay", "addHour", "addMinute", "addSecond", "addMillisecond"]) {
                if (name == method.to!wstring) return z8_primary_box(mixin("z8_date_" ~ method ~ "(date, count)"));
            }
        } else if (!isDate && arg.kind == Z8PrimaryKind.integer) {
            auto span = value.datespanValue;
            auto count = arg.integerValue;
            static foreach (method; ["set", "setTotalHours", "setTotalMinutes", "setTotalSeconds"]) {
                if (name == method.to!wstring) { mixin("z8_datespan_" ~ method ~ "(span, count);"); return null; }
            }
        }
    } else if (isDate) {
        auto date = value.dateValue;
        if (arguments.length == 3 && name == "setDate"w) { z8_date_setDate(date, intArg(0), intArg(1), intArg(2)); return null; }
        if ((arguments.length == 3 || arguments.length == 4) && name == "setTime"w) {
            z8_date_setTime(date, intArg(0), intArg(1), intArg(2), arguments.length == 4 ? intArg(3) : 0); return null;
        }
        if ((arguments.length == 6 || arguments.length == 7) && name == "set"w) {
            z8_date_set(date, intArg(0), intArg(1), intArg(2), intArg(3), intArg(4), intArg(5), arguments.length == 7 ? intArg(6) : 0); return null;
        }
    } else if ((arguments.length == 4 || arguments.length == 5) && name == "set"w) {
        z8_datespan_set(value.datespanValue, intArg(0), intArg(1), intArg(2), intArg(3), arguments.length == 5 ? intArg(4) : 0); return null;
    }
    throw new Exception("unsupported Eval temporal method");
}
private Z8Primary z8_eval_type_method(Z8EvalContext context, wstring name, Z8EvalValue[] arguments) {
    Z8Primary arg(size_t i, Z8PrimaryKind kind) {
        if (i >= arguments.length || arguments[i].primary is null || arguments[i].primary.kind != kind)
            throw new Exception("invalid Eval type method argument");
        return arguments[i].primary;
    }
    if (!arguments.length) {
        if (context == Z8EvalContext.guidType && name == "create"w) return z8_primary_box(z8_guid_create_random());
        if (context == Z8EvalContext.dateType) {
            if (name == "now"w) return z8_primary_box(z8_date_now());
            if (name == "today"w) return z8_primary_box(z8_date_today());
            if (name == "yesterday"w) return z8_primary_box(z8_date_yesterday());
            if (name == "tomorrow"w) return z8_primary_box(z8_date_tomorrow());
        }
    } else if (arguments.length == 1) {
        if (name == "parse"w) {
            auto text = arg(0, Z8PrimaryKind.text).textValue;
            if (context == Z8EvalContext.booleanType) return z8_primary_box(z8_bool_parse(text));
            if (context == Z8EvalContext.integerType) return z8_primary_box(z8_int_parse(text));
            if (context == Z8EvalContext.decimalType) return z8_primary_box(z8_decimal_parse(text));
            if (context == Z8EvalContext.guidType) return z8_primary_box(z8_guid_parse(text));
            if (context == Z8EvalContext.dateType) return z8_primary_box(z8_date_parse(text));
            if (context == Z8EvalContext.datespanType) return z8_primary_box(z8_datespan_parse(text));
        }
        if (context == Z8EvalContext.dateType && name == "parseIsoString"w)
            return z8_primary_box(z8_date_parseIsoString(arg(0, Z8PrimaryKind.text).textValue));
        if (context == Z8EvalContext.guidType && name == "create"w) {
            auto value = arguments[0].primary;
            if (value !is null && value.kind == Z8PrimaryKind.text) return z8_primary_box(z8_guid_create_name(value.textValue));
            return z8_primary_box(z8_guid_create(arg(0, Z8PrimaryKind.integer).integerValue));
        }
    } else if (arguments.length == 2) {
        if (context == Z8EvalContext.integerType && name == "parse"w)
            return z8_primary_box(z8_int_parse(arg(0, Z8PrimaryKind.text).textValue, arg(1, Z8PrimaryKind.integer).integerValue));
        if (context == Z8EvalContext.dateType && name == "parse"w)
            return z8_primary_box(z8_date_parse(arg(0, Z8PrimaryKind.text).textValue, arg(1, Z8PrimaryKind.text).textValue));
        if (context == Z8EvalContext.dateType && name == "isEqualDate"w)
            return z8_primary_box(z8_date_isEqualDate(arg(0, Z8PrimaryKind.date).dateValue, arg(1, Z8PrimaryKind.date).dateValue));
        if (context == Z8EvalContext.guidType && name == "create"w)
            return z8_primary_box(z8_guid_create(arg(0, Z8PrimaryKind.integer).integerValue, arg(1, Z8PrimaryKind.integer).integerValue));
        if (context == Z8EvalContext.stringType && name == "replicate"w)
            return z8_primary_box(z8_string_new(z8_string_replicate(arg(0, Z8PrimaryKind.text).textValue, arg(1, Z8PrimaryKind.integer).integerValue)));
    }
    throw new Exception("unsupported Eval type method");
}
private Z8EvalValue z8_eval_method(Z8EvalValue object, wstring name, Z8EvalValue[] arguments) {
    if (object.reportObject !is null) throw new Exception("POI report object context member is not implemented");
    foreach (argument; arguments) if (argument.reportObject !is null)
        throw new Exception("POI report object context member is not implemented");
    bool evaluated = object.evaluated;
    foreach (argument; arguments) evaluated = evaluated && argument.evaluated;
    if (!evaluated) {
        wstring call = z8_eval_expression_repr(object) ~ "."w ~ name ~ "("w;
        foreach (i, argument; arguments) { if (i) call ~= ", "w; call ~= z8_eval_value_text(argument); }
        return Z8EvalValue(false, null, null, call ~ ")"w);
    }
    if (object.context == Z8EvalContext.json && (name == "min"w || name == "max"w))
        return z8_eval_json_aggregate(name, arguments);
    if (object.context != Z8EvalContext.none && object.context != Z8EvalContext.json)
        return Z8EvalValue(true, z8_eval_type_method(object.context, name, arguments), null, null);
    if (object.collection !is null) {
        auto array = object.collection;
        if (!arguments.length) {
            if (name == "size"w) return Z8EvalValue(true, z8_primary_box(array.z8_size()), null, null);
            if (name == "isEmpty"w) return Z8EvalValue(true, z8_primary_box(array.z8_isEmpty()), null, null);
            if (name == "first"w || name == "last"w)
                return Z8EvalValue(true, array.z8_size() ? array.z8_get(name == "first"w ? 0 : array.z8_size() - 1) : null, null, null);
            if (name == "toString"w) return Z8EvalValue(true, z8_primary_box(z8_eval_value_text(object)), null, null);
            if (name == "clear"w) { array.z8_clear(); return Z8EvalValue(true); }
            if (name == "reverse"w) { array.z8_reverse(); return object; }
            if (name == "sort"w) { array.z8_sort(); return object; }
            if (name == "unique"w) { object.collection = array.z8_unique(); return object; }
            if (name == "hashCode"w) {
                uint hash = 1;
                for (long i; i < array.z8_size(); ++i) {
                    auto item = array.z8_get(i);
                    hash = 31u * hash + (item is null ? 0u : cast(uint)z8_primary_hash(item));
                }
                return Z8EvalValue(true, z8_primary_box(cast(long)cast(int)hash), null, null);
            }
        }
        if (name == "get"w && arguments.length == 1 && arguments[0].primary !is null && arguments[0].primary.kind == Z8PrimaryKind.integer)
            return Z8EvalValue(true, array.z8_get(cast(int)arguments[0].primary.integerValue), null, null);
        if (name == "removeAt"w && arguments.length == 1 && arguments[0].primary !is null && arguments[0].primary.kind == Z8PrimaryKind.integer)
            return Z8EvalValue(true, array.z8_removeAt(arguments[0].primary.integerValue), null, null);
        if ((name == "remove"w || name == "removeAll"w) && arguments.length == 1 && arguments[0].collection !is null) {
            array.z8_removeAll(arguments[0].collection);
            return name == "remove"w ? Z8EvalValue(true) : object;
        }
        if (name == "join"w && arguments.length <= 1) {
            wstring separator = ""w, joined;
            if (arguments.length) {
                if (arguments[0].primary is null || arguments[0].primary.kind != Z8PrimaryKind.text) throw new Exception("invalid Eval join separator");
                separator = arguments[0].primary.textValue;
            }
            for (long i; i < array.z8_size(); ++i) {
                if (i) joined ~= separator;
                joined ~= z8_primary_to_string(z8_nonnull(array.z8_get(i)));
            }
            return Z8EvalValue(true, z8_primary_box(z8_string_new(joined)), null, null);
        }
        throw new Exception("unsupported Eval collection method");
    }
    if (object.primary !is null && object.primary.kind == Z8PrimaryKind.guid && name == "components"w && !arguments.length) {
        auto result = Z8EvalValue(true);
        result.collection = new Z8Array!Z8Primary([z8_primary_box(z8_guid_major(object.primary.guidValue)), z8_primary_box(z8_guid_minor(object.primary.guidValue))]);
        return result;
    }
    if (object.primary !is null && object.primary.kind == Z8PrimaryKind.text && name == "split"w && (arguments.length == 1 || arguments.length == 2)) {
        if (arguments[0].primary is null || arguments[0].primary.kind != Z8PrimaryKind.text ||
            (arguments.length == 2 && (arguments[1].primary is null || arguments[1].primary.kind != Z8PrimaryKind.integer)))
            throw new Exception("invalid Eval split arguments");
        auto strings = z8_regex_split(object.primary.textValue, arguments[0].primary.textValue,
            arguments.length == 2 ? arguments[1].primary.integerValue : 0);
        auto result = Z8EvalValue(true);
        result.collection = new Z8Array!Z8Primary();
        for (long i; i < strings.z8_size(); ++i) result.collection.z8_add(z8_primary_box(strings.z8_get(i)));
        return result;
    }
    if (object.primary !is null && !arguments.length) {
        if (name == "hashCode"w) return Z8EvalValue(true, z8_primary_box(z8_primary_hash(object.primary)), null, null);
        if (name == "toString"w) return Z8EvalValue(true, z8_primary_box(z8_string_new(z8_primary_to_string(object.primary))), null, null);
        if (name == "toNumber"w && object.primary.kind == Z8PrimaryKind.boolean)
            return Z8EvalValue(true, z8_primary_box(z8_string_new(object.primary.booleanValue ? "1"w : "0"w)), null, null);
        if (object.primary.kind == Z8PrimaryKind.guid && (name == "major"w || name == "minor"w))
            return Z8EvalValue(true, z8_primary_box(name == "major"w ? z8_guid_major(object.primary.guidValue) : z8_guid_minor(object.primary.guidValue)), null, null);
        if (name == "get"w) {
            auto value = object.primary;
            if (value.kind == Z8PrimaryKind.boolean || value.kind == Z8PrimaryKind.guid) return Z8EvalValue(true, value, null, null);
            if (value.kind == Z8PrimaryKind.datespan)
                return Z8EvalValue(true, z8_primary_box(z8_datespan_milliseconds(value.datespanValue)), null, null);
            if (value.kind == Z8PrimaryKind.date) {
                auto source = z8_nonnull(value.dateValue);
                auto copy = new Z8Date(source.value);
                copy.explicitOffset = source.explicitOffset; copy.fullFirstWeek = source.fullFirstWeek;
                return Z8EvalValue(true, z8_primary_box(copy), null, null);
            }
        }
    }
    if (object.primary !is null) {
        Z8EvalContext type;
        final switch (object.primary.kind) {
            case Z8PrimaryKind.boolean: type = Z8EvalContext.booleanType; break;
            case Z8PrimaryKind.integer: type = Z8EvalContext.integerType; break;
            case Z8PrimaryKind.decimal: type = Z8EvalContext.decimalType; break;
            case Z8PrimaryKind.text: type = Z8EvalContext.stringType; break;
            case Z8PrimaryKind.guid: type = Z8EvalContext.guidType; break;
            case Z8PrimaryKind.date: type = Z8EvalContext.dateType; break;
            case Z8PrimaryKind.datespan: type = Z8EvalContext.datespanType; break;
            case Z8PrimaryKind.binary: case Z8PrimaryKind.geometry: case Z8PrimaryKind.file: case Z8PrimaryKind.user: break;
        }
        if ((name == "parse"w && type != Z8EvalContext.stringType && type != Z8EvalContext.none) ||
            (name == "parseIsoString"w && type == Z8EvalContext.dateType) ||
            (name == "create"w && type == Z8EvalContext.guidType) ||
            (name == "replicate"w && type == Z8EvalContext.stringType) ||
            (["now"w, "today"w, "yesterday"w, "tomorrow"w, "isEqualDate"w].canFind(name) && type == Z8EvalContext.dateType))
            return Z8EvalValue(true, z8_eval_type_method(type, name, arguments), null, null);
    }
    if (object.primary !is null && object.primary.kind == Z8PrimaryKind.text)
        return Z8EvalValue(true, z8_eval_string_method(object.primary.textValue, name, arguments), null, null);
    if (object.primary !is null && object.primary.kind == Z8PrimaryKind.binary) {
        auto binary = object.primary.binaryValue;
        if (!arguments.length) {
            if (name == "length"w) return Z8EvalValue(true, z8_primary_box(z8_binary_length(binary)), null, null);
            if (name == "getName"w) return Z8EvalValue(true, z8_primary_box(z8_string_new(z8_binary_get_name(binary))), null, null);
            if (name == "getBaseName"w) return Z8EvalValue(true, z8_primary_box(z8_string_new(z8_binary_base_name(binary))), null, null);
            if (name == "getExtension"w) return Z8EvalValue(true, z8_primary_box(z8_string_new(z8_binary_extension(binary))), null, null);
        }
        if (name == "setName"w && arguments.length == 1 && arguments[0].primary !is null && arguments[0].primary.kind == Z8PrimaryKind.text)
            return Z8EvalValue(true, z8_primary_box(z8_binary_set_name(binary, arguments[0].primary.textValue)), null, null);
    }
    if (object.primary !is null && object.primary.kind == Z8PrimaryKind.file && !arguments.length) {
        auto file = object.primary.fileValue;
        if (name == "fileName"w) return Z8EvalValue(true, z8_primary_box(z8_string_new(z8_file_fileName(file))), null, null);
        if (name == "baseName"w) return Z8EvalValue(true, z8_primary_box(z8_string_new(z8_file_baseName_instance(file))), null, null);
        if (name == "extension"w) return Z8EvalValue(true, z8_primary_box(z8_string_new(z8_file_extension_instance(file))), null, null);
        if (name == "folder"w) return Z8EvalValue(true, z8_primary_box(z8_string_new(z8_file_folder(file))), null, null);
        if (name == "string"w) return Z8EvalValue(true, z8_primary_box(z8_string_new(z8_file_path(file))), null, null);
        if (name == "getPath"w) {
            import std.string : replace;
            return Z8EvalValue(true, z8_primary_box(z8_string_new(z8_file_path(file).replace("\\"w, "/"w))), null, null);
        }
        if (name == "parent"w) return Z8EvalValue(true, z8_primary_box(z8_file_parent(file)), null, null);
        if (name == "lastModified"w) return Z8EvalValue(true, z8_primary_box(z8_file_lastModified(file)), null, null);
        if (name == "isEmpty"w) return Z8EvalValue(true, z8_primary_box(z8_file_isEmpty(file)), null, null);
        if (name == "exists"w) return Z8EvalValue(true, z8_primary_box(z8_file_exists(file)), null, null);
        if (name == "isFolder"w) return Z8EvalValue(true, z8_primary_box(z8_file_isFolder(file)), null, null);
    }
    if (object.primary !is null && (object.primary.kind == Z8PrimaryKind.integer || object.primary.kind == Z8PrimaryKind.decimal))
        return Z8EvalValue(true, z8_eval_number_method(object.primary, name, arguments), null, null);
    if (object.primary !is null && (object.primary.kind == Z8PrimaryKind.date || object.primary.kind == Z8PrimaryKind.datespan))
        return Z8EvalValue(true, z8_eval_temporal_method(object.primary, name, arguments), null, null);
    throw new Exception("unsupported Eval method");
}
private wstring z8_eval_format_decimal(Z8Decimal number, wstring pattern) {
    import std.math : signbit;
    wstring result; size_t implicitArgument, lastArgument;
    for (size_t i; i < pattern.length; ++i) {
        if (pattern[i] != '%') { result ~= pattern[i]; continue; }
        if (++i == pattern.length) throw new Exception("incomplete Eval format");
        if (pattern[i] == '%') { result ~= '%'; continue; }
        if (pattern[i] == 'n') { version (Windows) result ~= "\r\n"w; else result ~= "\n"w; continue; }
        size_t start = i, argument;
        while (i < pattern.length && pattern[i] >= '0' && pattern[i] <= '9') argument = argument * 10 + pattern[i++] - '0';
        bool explicitArgument = i > start && i < pattern.length && pattern[i] == '$';
        if (explicitArgument) ++i; else { i = start; argument = ++implicitArgument; }
        bool left, plus, space, zero, group, parens, alternate, reuse;
        while (i < pattern.length) {
            auto flag = pattern[i];
            if (flag == '-') { if (left) throw new Exception("duplicate Eval format flag"); left = true; }
            else if (flag == '+') { if (plus) throw new Exception("duplicate Eval format flag"); plus = true; }
            else if (flag == ' ') { if (space) throw new Exception("duplicate Eval format flag"); space = true; }
            else if (flag == '0') { if (zero) throw new Exception("duplicate Eval format flag"); zero = true; }
            else if (flag == ',') { if (group) throw new Exception("duplicate Eval format flag"); group = true; }
            else if (flag == '(') { if (parens) throw new Exception("duplicate Eval format flag"); parens = true; }
            else if (flag == '#') { if (alternate) throw new Exception("duplicate Eval format flag"); alternate = true; }
            else if (flag == '<') { if (reuse) throw new Exception("duplicate Eval format flag"); reuse = true; }
            else break;
            ++i;
        }
        if (reuse) { argument = lastArgument; if (!explicitArgument) --implicitArgument; }
        size_t width;
        while (i < pattern.length && pattern[i] >= '0' && pattern[i] <= '9') {
            width = width * 10 + pattern[i++] - '0';
            if (width > 1_000_000) throw new Exception("Eval format width too large");
        }
        size_t precision = 6;
        if (i < pattern.length && pattern[i] == '.') {
            precision = 0; ++i; auto digits = i;
            while (i < pattern.length && pattern[i] >= '0' && pattern[i] <= '9') {
                precision = precision * 10 + pattern[i++] - '0';
                if (precision > 1000) throw new Exception("Eval format precision too large");
            }
            if (i == digits) throw new Exception("missing Eval format precision");
        }
        if (argument != 1 || i >= pattern.length || pattern[i] != 'f' || (left && zero) || (plus && space) || ((left || zero) && !width))
            throw new Exception("unsupported Eval decimal format");
        lastArgument = argument;
        auto value = number.z8_as_double();
        auto rounded = z8_math_result(value).z8_round(cast(long)precision, 4);
        auto magnitude = rounded.z8_string();
        bool negative = signbit(value) != 0;
        if (magnitude.length && magnitude[0] == '-') magnitude = magnitude[1 .. $];
        auto body = z8_message_number(magnitude, precision, group);
        if (!precision && alternate) body ~= ',';
        wstring prefix = negative ? (parens ? "("w : "-"w) : plus ? "+"w : space ? " "w : ""w;
        wstring suffix = negative && parens ? ")"w : ""w;
        auto length = prefix.length + body.length + suffix.length;
        if (width > length) {
            auto padding = new wchar[width - length]; padding[] = zero ? '0' : ' ';
            if (left) suffix ~= cast(wstring)padding;
            else if (zero) prefix ~= cast(wstring)padding;
            else prefix = cast(wstring)padding ~ prefix;
        }
        result ~= prefix ~ body ~ suffix;
    }
    return result;
}
private Z8EvalValue z8_eval_function(wstring name, Z8EvalValue[] arguments) {
    foreach (argument; arguments) if (argument.reportObject !is null)
        throw new Exception("POI report object context member is not implemented");
    bool evaluated = true;
    foreach (argument; arguments) evaluated = evaluated && argument.evaluated;
    if (!evaluated) {
        wstring call = name ~ "("w;
        foreach (i, argument; arguments) { if (i) call ~= ", "w; call ~= z8_eval_value_text(argument); }
        return Z8EvalValue(false, null, null, call ~ ")"w);
    }
    if (name != "format"w || arguments.length != 2 || arguments[0].json !is null || arguments[0].collection !is null || arguments[0].context != Z8EvalContext.none ||
        arguments[1].json !is null || arguments[1].collection !is null || arguments[1].context != Z8EvalContext.none ||
        (arguments[1].primary !is null && arguments[1].primary.kind != Z8PrimaryKind.text))
        throw new Exception("unsupported Eval function or arguments");
    auto value = arguments[0].primary;
    auto pattern = arguments[1].primary is null ? null : arguments[1].primary.textValue;
    wstring formatted;
    if (value is null) formatted = "<null>"w;
    else if (value.kind == Z8PrimaryKind.date) formatted = z8_date_toString(value.dateValue, pattern is null ? "dd.MM.yyy"w : pattern);
    else if (value.kind == Z8PrimaryKind.decimal) formatted = pattern is null ? z8_eval_formatted_text(arguments[0]) : z8_eval_format_decimal(value.decimalValue, pattern);
    else formatted = z8_primary_to_string(value);
    return Z8EvalValue(true, z8_primary_box(z8_string_new(formatted)), null, null);
}
