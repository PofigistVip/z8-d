// Native adapter for org.zenframework.z8.server.base.json.parser.JsonPath.
private enum z8_json_path_name = "org.zenframework.z8.server.base.json.parser.JsonPath"w;
private struct Z8JsonPathPart {
    bool ordinal;
    int index;
    wstring name;
}
private class Z8JsonPathState {
    wstring text;
    Z8JsonPathPart[] parts;
}
private bool z8_json_path_word(wchar c) {
    return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') || c == '_';
}
private Z8JsonPathState z8_json_path_parse(wstring source) {
    source = z8_string_required(source);
    while (source.length && source[0] <= 0x20) source = source[1 .. $];
    while (source.length && source[$ - 1] <= 0x20) source = source[0 .. $ - 1];
    auto result = new Z8JsonPathState();
    result.text = source;
    size_t i;
    while (i < source.length) {
        if ((i == 0 && z8_json_path_word(source[i])) ||
            (source[i] == '/' && i + 1 < source.length && z8_json_path_word(source[i + 1]))) {
            if (source[i] == '/') ++i;
            auto start = i;
            while (i < source.length && z8_json_path_word(source[i])) ++i;
            Z8JsonPathPart part; part.name = source[start .. i]; result.parts ~= part;
        } else if (source[i] == '[') {
            auto start = i + 1, end = start;
            while (end < source.length && source[end] >= '0' && source[end] <= '9') ++end;
            if (end > start && end < source.length && source[end] == ']') {
                auto number = z8_int_parse(source[start .. end]);
                if (number > int.max) throw new Exception("invalid JSON path index");
                Z8JsonPathPart part; part.ordinal = true; part.index = cast(int)number; result.parts ~= part;
                i = end + 1;
            } else ++i;
        } else ++i;
    }
    return result;
}
private Z8JsonPathState z8_json_path_state(Z8Object self) {
    self = z8_nonnull(self);
    if (!z8_native_is_instance(z8_json_path_name, self)) throw new Exception("invalid native JSON path");
    auto state = cast(Z8JsonPathState)self.z8_get_native_state(z8_json_path_name);
    if (state is null) throw new Exception("uninitialized JSON path");
    return state;
}
private Z8Object z8_json_path_wrap(Z8JsonPathState state) {
    auto result = z8_native_new(z8_json_path_name);
    z8_nonnull(result).z8_set_native_state(z8_json_path_name, state);
    return result;
}
private Z8Object z8_json_path_new(wstring source) { return z8_json_path_wrap(z8_json_path_parse(source)); }
private Z8Object z8_json_path_assign(Z8Object self, wstring source) {
    self = z8_nonnull(self);
    if (!z8_native_is_instance(z8_json_path_name, self)) throw new Exception("invalid native JSON path");
    self.z8_set_native_state(z8_json_path_name, z8_json_path_parse(source));
    return self;
}
private Z8Optional!long z8_json_path_length(Z8Object self) {
    return z8_optional_wrap(cast(long)z8_json_path_state(self).parts.length);
}
private wstring z8_json_path_part_name(Z8Object self, Z8Optional!long index) {
    auto parts = z8_json_path_state(self).parts;
    auto position = cast(int)z8_optional_require(index);
    if (position < 0 || cast(size_t)position >= parts.length || parts[position].ordinal)
        throw new Exception("invalid JSON path name");
    return z8_string_new(parts[position].name);
}
private Z8Object z8_json_path_parent(Z8Object self) {
    auto parts = z8_json_path_state(self).parts;
    if (!parts.length) throw new Exception("JSON path has no parent");
    auto parent = new Z8JsonPathState();
    parent.parts = parts[0 .. $ - 1].dup;
    parent.text = " "w[0 .. 0];
    foreach (part; parent.parts)
        parent.text ~= part.ordinal ? "["w ~ part.index.to!wstring ~ "]"w : "/"w ~ part.name;
    return z8_json_path_wrap(parent);
}
private wstring z8_json_path_string(Z8Object self) { return z8_string_new(z8_json_path_state(self).text); }
private struct Z8JsonPathValue {
    JSONValue value;
    Object child;
    bool decimal;
    Z8Decimal number;
    bool missing = true;
}
private Z8JsonPathValue z8_json_path_evaluate(Z8Object self, Z8Object root) {
    auto path = z8_json_path_state(self);
    Z8JsonPathValue result;
    if (path.text.length == 0) {
        if (root !is null) throw new Exception("unsupported JSON path root value");
        return result;
    }
    if (root is null) return result;
    Object current = z8_json_state_from_object(root);
    foreach (part; path.parts) {
        if (part.ordinal) {
            auto array = cast(Z8JsonArrayState)current;
            if (array is null) return result;
            auto position = cast(size_t)part.index;
            if (position >= array.value.array.length) throw new Exception("JSON path index out of bounds");
            result.value = array.value.array[position];
            auto decimalValue = position in array.decimals;
            result.decimal = decimalValue !is null;
            if (result.decimal) result.number = *decimalValue;
            if (result.value.type == JSONType.array || result.value.type == JSONType.object) {
                auto child = position in array.children;
                if (child is null) { array.children[position] = z8_json_new_state(result.value); child = position in array.children; }
                current = *child;
            } else current = null;
        } else {
            auto object = cast(Z8JsonObjectState)current;
            if (object is null) return result;
            auto key = part.name.to!string;
            auto value = key in object.value.object;
            if (value is null) return result;
            result.value = *value;
            auto decimalValue = key in object.decimals;
            result.decimal = decimalValue !is null;
            if (result.decimal) result.number = *decimalValue;
            if (result.value.type == JSONType.array || result.value.type == JSONType.object) {
                auto child = key in object.children;
                if (child is null) { object.children[key] = z8_json_new_state(result.value); child = key in object.children; }
                current = *child;
            } else current = null;
        }
        if (result.value.type == JSONType.null_) return Z8JsonPathValue.init;
    }
    result.child = current;
    result.missing = false;
    return result;
}
private wstring z8_json_path_get_string(Z8Object self, Z8Object root, wstring fallback = null) {
    auto result = z8_json_path_evaluate(self, root);
    if (result.missing) return fallback;
    if (result.decimal || result.value.type != JSONType.string) throw new Exception("JSON path value is not a string");
    return z8_string_new(z8_json_value_text(result.value));
}
private Z8Optional!long z8_json_path_get_int(Z8Object self, Z8Object root, Z8Optional!long fallback = Z8Optional!long.init) {
    auto result = z8_json_path_evaluate(self, root);
    if (result.missing) return fallback;
    if (result.decimal || (result.value.type != JSONType.integer && result.value.type != JSONType.uinteger))
        throw new Exception("JSON path value is not an integer");
    return z8_optional_wrap(result.value.type == JSONType.integer ? result.value.integer : cast(long)result.value.uinteger);
}
private Z8Optional!Z8Decimal z8_json_path_get_decimal(Z8Object self, Z8Object root, Z8Optional!Z8Decimal fallback = Z8Optional!Z8Decimal.init) {
    auto result = z8_json_path_evaluate(self, root);
    if (result.missing) return fallback;
    if (result.decimal) return z8_optional_wrap(result.number);
    if (result.value.type != JSONType.float_) throw new Exception("JSON path value is not a decimal");
    return z8_optional_wrap(z8_math_result(result.value.floating));
}
private Z8Optional!bool z8_json_path_get_bool(Z8Object self, Z8Object root, Z8Optional!bool fallback = Z8Optional!bool.init) {
    auto result = z8_json_path_evaluate(self, root);
    if (result.missing) return fallback;
    if (result.decimal || (result.value.type != JSONType.true_ && result.value.type != JSONType.false_))
        throw new Exception("JSON path value is not a boolean");
    return z8_optional_wrap(result.value.type == JSONType.true_);
}
private Z8Object z8_json_path_get_object(Z8Object self, Z8Object root, Z8Object fallback = null) {
    auto result = z8_json_path_evaluate(self, root);
    if (result.missing) {
        if (fallback !is null) throw new Exception("unsupported JSON path default object");
        return null;
    }
    if (result.child is null) throw new Exception("JSON path value is not an object");
    return z8_json_wrap_state(result.child);
}
private Z8Guid z8_json_path_get_guid(Z8Object self, Z8Object root, Z8Guid fallback = null) {
    auto result = z8_json_path_evaluate(self, root);
    if (result.missing) return fallback;
    throw new Exception("JSON path value is not a guid");
}
private Z8Date z8_json_path_get_date(Z8Object self, Z8Object root, Z8Date fallback = null) {
    auto result = z8_json_path_evaluate(self, root);
    if (result.missing) return fallback;
    throw new Exception("JSON path value is not a date");
}
private wstring z8_json_path_string_value(Z8Object self, Z8Object root) { return z8_json_path_get_string(self, root); }
private Z8Optional!long z8_json_path_int_value(Z8Object self, Z8Object root) { return z8_json_path_get_int(self, root); }
private Z8Optional!Z8Decimal z8_json_path_decimal_value(Z8Object self, Z8Object root) { return z8_json_path_get_decimal(self, root); }
private Z8Optional!bool z8_json_path_bool_value(Z8Object self, Z8Object root) { return z8_json_path_get_bool(self, root); }
private Z8Object z8_json_path_object_value(Z8Object self, Z8Object root) { return z8_json_path_get_object(self, root); }
private Z8Guid z8_json_path_guid_value(Z8Object self, Z8Object root) { return z8_json_path_get_guid(self, root); }
private Z8Date z8_json_path_date_value(Z8Object self, Z8Object root) { return z8_json_path_get_date(self, root); }
