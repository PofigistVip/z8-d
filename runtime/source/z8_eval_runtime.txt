// Eval context and native BL adapter.
private enum z8_eval_name = "org.zenframework.z8.server.base.eval.Eval"w;
private struct Z8EvalVariable { Z8Primary primary; Z8Object json; Z8StringMap requestParameters; }
private class Z8EvalState {
    Z8EvalVariable[wstring] variables;
    Z8Object reportContext;
    Z8Object reportSource;
    Z8JavaMapOrder!wstring order;
    wstring[] errors;
    this() { order = new Z8JavaMapOrder!wstring(); }
}
private enum Z8EvalContext { none, json, booleanType, integerType, decimalType, stringType, guidType, dateType, datespanType }
private struct Z8EvalValue {
    bool evaluated; Z8Primary primary; Z8Object json; wstring expression; Z8EvalContext context;
    Z8StringMap requestParameters;
    Z8Object reportObject;
    Z8Array!Z8Primary collection;
}
private Z8EvalState z8_eval_state(Z8Object self) {
    self = z8_nonnull(self);
    if (!z8_native_is_instance(z8_eval_name, self)) throw new Exception("invalid Eval object");
    auto state = cast(Z8EvalState)self.z8_get_native_state(z8_eval_name);
    if (state is null) { state = new Z8EvalState(); self.z8_set_native_state(z8_eval_name, state); }
    return state;
}
private Z8Object z8_eval_set_primary(Z8Object self, wstring name, Z8Primary value) {
    auto state = z8_eval_state(self); name = z8_string_required(name);
    if ((name in state.variables) is null) state.order.add(name);
    state.variables[name] = Z8EvalVariable(value, null); return self;
}
private Z8Object z8_eval_set_json(Z8Object self, wstring name, Z8Object value) {
    auto state = z8_eval_state(self); name = z8_string_required(name); z8_nonnull(value);
    if (!z8_native_is_instance(z8_json_object_name, value) && !z8_native_is_instance(z8_json_array_name, value))
        throw new Exception("invalid Eval JSON value");
    if ((name in state.variables) is null) state.order.add(name);
    // Java stores the underlying JSON collection, so replacing the wrapper's
    // collection later must not replace the value held in the Eval context.
    Object jsonState = z8_native_is_instance(z8_json_object_name, value) ?
        cast(Object)z8_json_object_state(value) : cast(Object)z8_json_array_state(value);
    state.variables[name] = Z8EvalVariable(null, z8_json_wrap_state(jsonState)); return self;
}
private Z8Object z8_eval_clear(Z8Object self) {
    auto state = z8_eval_state(self); state.variables.clear(); state.order.clear(); state.errors = null; return self;
}
private Z8Array!wstring z8_eval_errors(Z8Object self) { return new Z8Array!wstring(z8_eval_state(self).errors.dup); }
private wstring z8_eval_value_text(Z8EvalValue value) {
    if (!value.evaluated) return value.expression;
    if (value.reportObject !is null) throw new Exception("POI report object context member is not implemented");
    if (value.collection !is null) {
        wstring result = "["w;
        for (long i; i < value.collection.z8_size(); ++i) {
            if (i) result ~= ", "w;
            auto item = value.collection.z8_get(i);
            result ~= item is null ? "null"w : z8_primary_to_string(item);
        }
        return result ~ "]"w;
    }
    if (value.context != Z8EvalContext.none && value.context != Z8EvalContext.json) {
        wstring typeName;
        final switch (value.context) {
            case Z8EvalContext.booleanType: typeName = "bool"w; break;
            case Z8EvalContext.integerType: typeName = "integer"w; break;
            case Z8EvalContext.decimalType: typeName = "decimal"w; break;
            case Z8EvalContext.stringType: typeName = "string"w; break;
            case Z8EvalContext.guidType: typeName = "guid"w; break;
            case Z8EvalContext.dateType: typeName = "date"w; break;
            case Z8EvalContext.datespanType: typeName = "datespan"w; break;
            case Z8EvalContext.none: case Z8EvalContext.json: break;
        }
        return "class org.zenframework.z8.server.types."w ~ typeName;
    }
    if (value.json !is null) return z8_native_is_instance(z8_json_object_name, value.json) ?
        z8_json_object_to_string(value.json) : z8_json_array_string(value.json);
    return value.primary is null ? "null"w : z8_primary_to_string(value.primary);
}
private wstring z8_eval_formatted_text(Z8EvalValue value) {
    if (value.primary is null && value.json is null && value.collection is null) return "<null>"w;
    if (value.primary !is null && value.primary.kind == Z8PrimaryKind.date)
        return z8_date_toString(value.primary.dateValue, "dd.MM.yyy"w);
    if (value.primary !is null && value.primary.kind == Z8PrimaryKind.decimal) {
        auto number = value.primary.decimalValue;
        if (number.nan) return "NaN"w;
        auto rounded = z8_math_result(number.z8_as_double()).z8_round(2, 4);
        auto formatted = z8_message_number(rounded.z8_string(), 2);
        if (number.coefficient < 0 && rounded.coefficient == 0) formatted = "-"w ~ formatted;
        return (number.coefficient < 0 ? ""w : " "w) ~ formatted;
    }
    return z8_eval_value_text(value);
}
private wstring z8_eval_context_info(Z8Object self) {
    auto state = z8_eval_state(self); wstring result = "variables: {"w;
    foreach (i, name; state.order.ordered()) {
        if (i) result ~= ", "w;
        auto value = state.variables[name];
        result ~= name ~ "="w ~ z8_eval_value_text(Z8EvalValue(true, value.primary, value.json, null));
    }
    return result ~ "}, functions: {}"w;
}
private Z8EvalValue z8_eval_expression(Z8EvalState state, wstring text) { return (new Z8EvalParser(state, text)).parse(); }
private Z8Optional!bool z8_eval_boolean(Z8Object self, wstring expression) {
    auto value = z8_eval_expression(z8_eval_state(self), expression);
    if (!value.evaluated) throw new Exception("Eval expression is not evaluated");
    if (value.primary is null || value.primary.kind != Z8PrimaryKind.boolean) throw new Exception("Eval boolean result expected");
    return z8_optional_wrap(value.primary.booleanValue);
}
private wstring z8_eval_text_from_state(Z8EvalState state, wstring source, bool requireResolved = false, bool reportSyntax = false) {
    import std.algorithm.searching : canFind;
    import std.string : indexOf;
    if (source is null) return "null"w;
    auto first = source.indexOf("${"w);
    if (first < 0) return z8_string_new(source);
    auto end = source[cast(size_t)first + 2 .. $].indexOf('}');
    if (end < 0) throw new Exception("unterminated Eval text expression");
    auto start = cast(size_t)first, last = start + 2 + cast(size_t)end;
    auto single = first == 0 && last + 1 == source.length;
    Z8EvalValue value; bool failed;
    void evaluatePart() {
        failed = false;
        try {
            auto expression = source[start + 2 .. last];
            if (reportSyntax && expression.canFind('"')) throw new Exception("POI report expression syntax is not implemented");
            value = z8_eval_expression(state, expression);
            if (requireResolved && (!value.evaluated || value.context != Z8EvalContext.none ||
                value.json !is null || value.requestParameters !is null || value.reportObject !is null))
                throw new Exception("POI report expression context is not implemented");
        }
        catch (Exception error) {
            if (single || requireResolved) throw error;
            state.errors ~= error.msg.to!wstring; failed = true;
        }
    }
    evaluatePart();
    if (single) return z8_string_new(z8_eval_value_text(value));
    wstring result = source[0 .. cast(size_t)first];
    for (;;) {
        result ~= failed ? source[start .. last + 1] : value.evaluated ?
            z8_eval_formatted_text(value) : "${"w ~ value.expression ~ "}"w;
        auto remaining = source[last + 1 .. $];
        auto next = remaining.indexOf("${"w);
        if (next < 0) return z8_string_new(result ~ remaining);
        result ~= remaining[0 .. cast(size_t)next];
        start = last + 1 + cast(size_t)next;
        end = source[start + 2 .. $].indexOf('}');
        if (end < 0) throw new Exception("unterminated Eval text expression");
        last = start + 2 + cast(size_t)end;
        evaluatePart();
    }
}
private wstring z8_eval_text(Z8Object self, wstring source) {
    return z8_eval_text_from_state(z8_eval_state(self), source);
}
void z8_bind_builtin_eval() {
    enum p = z8_eval_name ~ ".m_"w;
    z8_native_bind(p ~ "setVariable"w, &z8_eval_set_primary); z8_native_bind(p ~ "setVariable"w, &z8_eval_set_json);
    z8_native_bind(p ~ "clear"w, &z8_eval_clear); z8_native_bind(p ~ "getErrors"w, &z8_eval_errors);
    z8_native_bind(p ~ "getContextInfo"w, &z8_eval_context_info);
    z8_native_bind(p ~ "evaluateBoolean"w, &z8_eval_boolean); z8_native_bind(p ~ "evaluateText"w, &z8_eval_text);
}
shared static this() { z8_bind_builtin_eval(); }
