// Native form Action state and metadata.
private enum z8_action_name = "org.zenframework.z8.server.base.form.action.Action"w;
private interface IZ8ActionTypeCell { wstring text(); }
private class Z8ActionTypeCell(T) : IZ8ActionTypeCell {
    T value;
    this() { value = T.e_Default; }
    wstring text() {
        if (value is T.e_Default) return "default"w;
        if (value is T.e_Danger) return "danger"w;
        if (value is T.e_Primary) return "primary"w;
        if (value is T.e_Success) return "success"w;
        throw new Exception("invalid action type");
    }
}
private class Z8ActionState {
    IZ8FieldOrderCell parameters;
    IZ8ActionTypeCell type;
    Z8Optional!bool useTransaction = z8_optional_wrap(true);
}
private Z8ActionState z8_action_state(Z8Object self) {
    self = z8_nonnull(self);
    if (!z8_native_is_instance(z8_action_name, self)) throw new Exception("invalid Action object");
    auto state = cast(Z8ActionState)self.z8_get_native_state(z8_action_name);
    if (state is null) { state = new Z8ActionState(); self.z8_set_native_state(z8_action_name, state); }
    return state;
}
ref Z8ObjectArray!T z8_action_parameters_ref(T)(Z8Object self) if (is(T : Z8Object)) {
    auto state = z8_action_state(self);
    auto cell = cast(Z8FieldOrderCell!T)state.parameters;
    if (cell is null) {
        if (state.parameters !is null) throw new Exception("incompatible action parameters array");
        cell = new Z8FieldOrderCell!T(); state.parameters = cell;
    }
    return cell.value;
}
ref T z8_action_type_ref(T)(Z8Object self) {
    auto state = z8_action_state(self);
    auto cell = cast(Z8ActionTypeCell!T)state.type;
    if (cell is null) {
        if (state.type !is null) throw new Exception("incompatible action type enum");
        cell = new Z8ActionTypeCell!T(); state.type = cell;
    }
    return cell.value;
}
private Z8Optional!bool* z8_action_use_transaction(Z8Object self) { return &z8_action_state(self).useTransaction; }
private void z8_action_default_execute_one(Z8Object self, Z8Guid record, Z8Object context, Z8Array!Z8Guid selected, Z8Object query) { z8_nonnull(self); }
private void z8_action_default_execute_many(Z8Object self, Z8Array!Z8Guid records, Z8Object context, Z8Array!Z8Guid selected, Z8Object query) {
    z8_nonnull(self); z8_nonnull(records);
    auto expectedVersion = records.z8_version;
    for (long i; i < records.z8_size(); ++i) {
        if (expectedVersion != records.z8_version) throw new Exception("array modified during action execution");
        self.z8_action_virtual_execute_one(records.z8_get(i), context, selected, query);
    }
}
private void z8_action_restart_scheduler(Z8Object self, Z8Array!Z8Guid records, Z8Object context, Z8Array!Z8Guid selected, Z8Object query) {
    z8_nonnull(self);
    auto control = z8_application_require_request().schedulerControl;
    if (control is null) throw new Exception("scheduler is not configured");
    control.restart();
}
private void z8_action_write_meta(Z8Object item, Z8Object query, Z8Object context, Z8Privileges privileges) {
    auto state = z8_action_state(item);
    z8_control_write_base_meta(item, query, context, privileges);
    auto writer = z8_application_require_request().writer;
    z8_monitor_property(writer, "isAction"w, true);
    z8_monitor_property(writer, "request"w, z8_nonnull(context).z8_class_id());
    z8_monitor_property(writer, "type"w, state.type is null ? "default"w : state.type.text());
    writer.start('[', "parameters"w);
    if (state.parameters !is null && state.parameters.view() !is null) {
        auto parameters = state.parameters.view().z8_erased_data();
        for (long i; i < parameters.z8_size(); ++i) z8_parameter_write_meta(cast(Z8Object)parameters.z8_get(i), writer);
    }
    writer.finish(']');
}
void z8_bind_builtin_action() {
    z8_native_bind_field(z8_action_name ~ ".f_useTransaction"w, &z8_action_use_transaction);
    z8_native_bind(z8_action_name ~ ".m_execute"w, &z8_action_default_execute_one);
    z8_native_bind(z8_action_name ~ ".m_execute"w, &z8_action_default_execute_many);
    z8_native_bind("org.zenframework.z8.server.base.table.system.view.RestartSchedulerAction.m_execute"w, &z8_action_restart_scheduler);
}
shared static this() { z8_bind_builtin_action(); }
