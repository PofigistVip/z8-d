private long z8_str_compare(wstring a, wstring b) {
    auto length = a.length < b.length ? a.length : b.length;
    foreach (i; 0 .. length) if (a[i] != b[i]) return cast(long)a[i] - cast(long)b[i];
    return cast(long)a.length - cast(long)b.length;
}
private bool z8_str_startsWith(wstring value, wstring prefix, long offset = 0) {
    auto start = cast(int)offset;
    if (start < 0 || cast(size_t)start > value.length || prefix.length > value.length - cast(size_t)start) return false;
    return value[cast(size_t)start .. cast(size_t)start + prefix.length] == prefix;
}
private bool z8_str_endsWith(wstring value, wstring suffix) {
    return suffix.length <= value.length && value[value.length - suffix.length .. $] == suffix;
}
private long z8_str_indexOf(wstring value, wstring needle, long offset = 0) {
    auto raw = cast(int)offset;
    size_t start = raw < 0 ? 0 : cast(size_t)raw;
    if (start > value.length) start = value.length;
    if (needle.length == 0) return cast(long)start;
    if (needle.length > value.length) return -1;
    for (size_t i = start; i <= value.length - needle.length; ++i)
        if (value[i .. i + needle.length] == needle) return cast(long)i;
    return -1;
}
private long z8_str_lastIndexOf(wstring value, wstring needle, long offset = int.max) {
    long start = cast(int)offset;
    if (start < 0 || needle.length > value.length) return -1;
    auto last = cast(long)(value.length - needle.length);
    if (start > last) start = last;
    for (long i = start; i >= 0; --i)
        if (value[cast(size_t)i .. cast(size_t)i + needle.length] == needle) return i;
    return -1;
}
private wstring z8_str_substring(wstring value, long start) {
    auto i = cast(int)start;
    if (i < 0 || cast(size_t)i > value.length) throw new Exception("string index out of bounds");
    return value[cast(size_t)i .. $];
}
private wstring z8_str_substring(wstring value, long start, long count) {
    auto i = cast(int)start;
    auto end = cast(int)(cast(long)i + cast(int)count);
    if (end >= cast(long)value.length) end = cast(int)value.length;
    if (i < 0 || end < i || cast(size_t)end > value.length) throw new Exception("string index out of bounds");
    return value[cast(size_t)i .. cast(size_t)end];
}
private wstring z8_str_charAt(wstring value, long index) {
    auto i = cast(int)index;
    if (i < 0 || cast(size_t)i >= value.length) throw new Exception("string index out of bounds");
    return value[cast(size_t)i .. cast(size_t)i + 1];
}
private long z8_int_abs(long value) { return value == long.min ? value : value < 0 ? -value : value; }
private long z8_int_signum(long value) { return value < 0 ? -1 : value > 0 ? 1 : 0; }
private long z8_int_parse(wstring value, long radix = 10) {
    auto base = cast(int)radix;
    if (base < 2 || base > 36 || value.length == 0) throw new Exception("invalid integer");
    if (value[$-1] == 'L' || value[$-1] == 'l') value = value[0 .. $-1];
    if (!value.length) throw new Exception("invalid integer");
    size_t i; bool negative;
    if (value[0] == '-' || value[0] == '+') { negative = value[0] == '-'; ++i; }
    if (i == value.length) throw new Exception("invalid integer");
    ulong magnitude, limit = cast(ulong)long.max + (negative ? 1UL : 0UL);
    for (; i < value.length; ++i) {
        auto c = value[i];
        int digit = c >= '0' && c <= '9' ? c - '0' : c >= 'a' && c <= 'z' ? c - 'a' + 10 : c >= 'A' && c <= 'Z' ? c - 'A' + 10 : -1;
        if (digit < 0 || digit >= base || magnitude > (limit - cast(ulong)digit) / cast(ulong)base)
            throw new Exception("invalid integer");
        magnitude = magnitude * cast(ulong)base + cast(ulong)digit;
    }
    if (negative && magnitude == cast(ulong)long.max + 1UL) return long.min;
    return negative ? -cast(long)magnitude : cast(long)magnitude;
}
